장치마다 판, 확장, 기능, 한도가 다릅니다. 그래서 시작할 때 한 번 물어 모아 두고, 그 답으로 어느 길로 그릴지를 정합니다. 아래는 lavapipe에 물은 결과입니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp
#include "vkx_device.hpp"   // 「물리 장치와 논리 장치」
#include <algorithm>
#include <cstddef>
#include <set>
#include <string>

using Eds3 = VkPhysicalDeviceExtendedDynamicState3FeaturesEXT;

// 한 번 물어서 모아 둔다. 그리는 코드는 장치 대신 이것을 본다
struct Caps {
    uint32_t device = 0, usable = 0;     // 장치의 판, 쓸 수 있는 판
    std::set<std::string> ext;           // 장치가 아는 확장
    VkPhysicalDeviceFeatures core{};
    VkPhysicalDeviceMeshShaderFeaturesEXT mesh{
        .sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_MESH_SHADER_FEATURES_EXT};
    Eds3 eds3{.sType =
        VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_EXTENDED_DYNAMIC_STATE_3_FEATURES_EXT};
    VkPhysicalDeviceLimits limits{};
    bool has(const char* name) const { return ext.contains(name); }
};

Caps ask(VkPhysicalDevice gpu, uint32_t written) {   // written: 인스턴스에 적은 판
    Caps c;
    VkPhysicalDeviceProperties p;
    vkGetPhysicalDeviceProperties(gpu, &p);
    c.device = p.apiVersion;
    c.usable = std::min(written, p.apiVersion);      // 둘 중 낮은 쪽까지만 쓴다
    c.limits = p.limits;
    uint32_t n = 0;
    vkEnumerateDeviceExtensionProperties(gpu, nullptr, &n, nullptr);
    std::vector<VkExtensionProperties> all(n);
    vkEnumerateDeviceExtensionProperties(gpu, nullptr, &n, all.data());
    for (auto& e : all) c.ext.insert(e.extensionName);

    // 확장의 기능 구조체는 그 확장을 아는 장치에만 엮는다. 안 엮으면 0으로 남는다
    VkPhysicalDeviceFeatures2 f{
        .sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_FEATURES_2};
    if (c.has("VK_EXT_mesh_shader")) f.pNext = &c.mesh;
    if (c.has("VK_EXT_extended_dynamic_state3")) {
        c.eds3.pNext = f.pNext;
        f.pNext = &c.eds3;
    }
    vkGetPhysicalDeviceFeatures2(gpu, &f);
    c.core = f.features;
    c.mesh.pNext = c.eds3.pNext = nullptr;
    return c;
}

// 판이 오르며 코어로 올라온 기능: 그 판부터는 코어, 아래에서는 확장으로 찾는다
struct Promoted { const char* what; uint32_t since; const char* ext; };
const Promoted promoted[] = {
    {"동적 렌더링", VK_API_VERSION_1_3, "VK_KHR_dynamic_rendering"},
    {"synchronization2", VK_API_VERSION_1_3, "VK_KHR_synchronization2"},
    {"타임라인 세마포어", VK_API_VERSION_1_2, "VK_KHR_timeline_semaphore"}};

void versions(const char* title, const Caps& c) {
    std::printf("[%s] 장치의 판 %u.%u.%u, 쓸 수 있는 판 %u.%u\n ", title,
                VK_API_VERSION_MAJOR(c.device), VK_API_VERSION_MINOR(c.device),
                VK_API_VERSION_PATCH(c.device), VK_API_VERSION_MAJOR(c.usable),
                VK_API_VERSION_MINOR(c.usable));
    for (auto& p : promoted)
        std::printf(" %s %s |", p.what,
                    c.usable >= p.since ? "코어" : c.has(p.ext) ? "확장" : "없음");
    std::puts("");
}

// 갈림은 시작할 때 한 번 정하고, 그리는 코드는 정해진 길만 따른다
struct Plan { const char* cull; const char* texture; const char* polygon; };
Plan plan(const Caps& c) {
    return {c.mesh.meshShader ? "메시 셰이더" : "컴퓨트 셰이더",
            c.core.textureCompressionBC ? "BC"
                : c.core.textureCompressionASTC_LDR ? "ASTC"
                : c.core.textureCompressionETC2 ? "ETC2" : "압축 없음",
            c.eds3.extendedDynamicState3PolygonMode ? "동적 상태"
                                                    : "파이프라인 둘"};
}

int count_on(const Eds3& f) {   // sType, pNext 뒤로 VkBool32 31개가 늘어서 있다
    VkBool32 bits[31];
    std::memcpy(bits, reinterpret_cast<const char*>(&f) +
                offsetof(Eds3, extendedDynamicState3TessellationDomainOrigin),
                sizeof bits);
    return int(std::count(bits, bits + 31, VkBool32{VK_TRUE}));
}

// 큐 하나짜리 장치를 만들어 보고 결과만 받는다. ext와 chain이 켜려는 확장과 기능
const char* attempt(VkPhysicalDevice gpu, std::vector<const char*> ext,
                    const void* chain) {
    float priority = 1.0f;
    VkDeviceQueueCreateInfo q{.sType = VK_STRUCTURE_TYPE_DEVICE_QUEUE_CREATE_INFO,
                              .queueCount = 1, .pQueuePriorities = &priority};
    VkDeviceCreateInfo ci{.sType = VK_STRUCTURE_TYPE_DEVICE_CREATE_INFO,
                          .pNext = chain,
                          .queueCreateInfoCount = 1, .pQueueCreateInfos = &q,
                          .enabledExtensionCount = uint32_t(ext.size()),
                          .ppEnabledExtensionNames = ext.data()};
    VkDevice dev = VK_NULL_HANDLE;
    VkResult r = vkCreateDevice(gpu, &ci, nullptr, &dev);
    if (r == VK_SUCCESS) vkDestroyDevice(dev, nullptr);
    return r == VK_SUCCESS ? "VK_SUCCESS"
         : r == VK_ERROR_FEATURE_NOT_PRESENT ? "VK_ERROR_FEATURE_NOT_PRESENT"
         : r == VK_ERROR_EXTENSION_NOT_PRESENT ? "VK_ERROR_EXTENSION_NOT_PRESENT"
         : "그 밖의 오류";
}

const char* yn(VkBool32 b) { return b ? "예" : "아니오"; }

int main() {
    vkx::Instance inst = vkx::make_instance();       // 1.3을 적은 인스턴스
    uint32_t one = 1;
    VkPhysicalDevice gpu = VK_NULL_HANDLE;
    vkEnumeratePhysicalDevices(inst.handle, &one, &gpu);   // 첫 장치만 받는다
    Caps caps = ask(gpu, VK_API_VERSION_1_3);
    versions("1.3을 적은 인스턴스", caps);
    versions("1.1을 적었다면", ask(gpu, VK_API_VERSION_1_1));

    std::printf("기능\n  메시 셰이더: 확장 %s, meshShader %s\n",
                yn(caps.has("VK_EXT_mesh_shader")), yn(caps.mesh.meshShader));
    std::printf("  깊이 경계 시험: depthBounds %s\n", yn(caps.core.depthBounds));
    std::printf("  동적 상태 3: 31개 중 %d개 (다각형 모드 %s, 블렌딩 식 %s, "
                "보수적 래스터화 %s)\n", count_on(caps.eds3),
                yn(caps.eds3.extendedDynamicState3PolygonMode),
                yn(caps.eds3.extendedDynamicState3ColorBlendEquation),
                yn(caps.eds3.extendedDynamicState3ConservativeRasterizationMode));

    Caps hidden = caps;                  // 대체 경로를 돌려 보려고 일부러 가린다
    hidden.mesh.meshShader = hidden.core.textureCompressionBC = VK_FALSE;
    hidden.eds3 = {};
    auto show = [](const char* who, Plan p) {
        std::printf("  %s: 덩어리 컬링 %s, 텍스처 %s, 다각형 모드 %s\n",
                    who, p.cull, p.texture, p.polygon);
    };
    std::puts("갈림");
    show("이 장치", plan(caps));
    show("가린 장치", plan(hidden));

    // 어느 렌더러의 필요를 이 장치의 한도, 명세가 보장하는 값과 견준다
    struct Need { const char* what; uint32_t need, here, sure; };
    const VkPhysicalDeviceLimits& l = caps.limits;
    const Need needs[] = {
        {"푸시 상수 바이트", 200, l.maxPushConstantsSize, 128},
        {"함께 붙이는 세트", 3, l.maxBoundDescriptorSets, 4},
        {"살아 있는 할당", 5000, l.maxMemoryAllocationCount, 4096}};
    std::puts("한도");
    for (auto& n : needs)
        std::printf("  %s: 필요 %u, 이 장치 %u, 보장 %u -> %s\n", n.what, n.need,
                    n.here, n.sure,
                    n.need <= n.sure ? "어디서나 됨"
                    : n.need <= n.here ? "이 장치에서만 됨" : "안 됨");

    std::puts("묻지 않고 켜면");
    VkPhysicalDeviceFeatures2 want{
        .sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_FEATURES_2};
    want.features.depthBounds = VK_TRUE;
    std::printf("  depthBounds: %s\n", attempt(gpu, {}, &want));
    std::printf("  VK_KHR_fragment_shading_rate: %s\n",
                attempt(gpu, {"VK_KHR_fragment_shading_rate"}, nullptr));
    Eds3 some{.sType = caps.eds3.sType};
    some.extendedDynamicState3ConservativeRasterizationMode = VK_TRUE;
    std::printf("  동적 상태 3의 보수적 래스터화: %s\n",
                attempt(gpu, {"VK_EXT_extended_dynamic_state3"}, &some));
    std::printf("  동적 상태 3을 물은 대로: %s\n",
                attempt(gpu, {"VK_EXT_extended_dynamic_state3"}, &caps.eds3));
    vkx::destroy(inst);
}
```

```text
  동적 렌더링 코어 | synchronization2 코어 | 타임라인 세마포어 코어 |
[1.1을 적었다면] 장치의 판 1.4.318, 쓸 수 있는 판 1.1
  동적 렌더링 확장 | synchronization2 확장 | 타임라인 세마포어 확장 |
기능

  깊이 경계 시험: depthBounds 아니오
  동적 상태 3: 31개 중 16개 (다각형 모드 예, 블렌딩 식 예, 보수적 래스터화 아니오)
갈림

  가린 장치: 덩어리 컬링 컴퓨트 셰이더, 텍스처 압축 없음, 다각형 모드 파이프라인 둘

  함께 붙이는 세트: 필요 3, 이 장치 8, 보장 4 -> 어디서나 됨
  살아 있는 할당: 필요 5000, 이 장치 4294967295, 보장 4096 -> 이 장치에서만 됨

  VK_KHR_fragment_shading_rate: VK_ERROR_EXTENSION_NOT_PRESENT
```

- **쓸 수 있는 판**: 장치가 1.4여도 인스턴스의 `apiVersion`에 적은 판까지만 씁니다([「인스턴스와 확장」](#/read/cpp-vulkan/setup-instance)). 판이 오르며 코어가 된 기능은 그 판부터 확장 이름 없이 쓰고, 아래 판에서는 확장으로 찾습니다. 확장으로 쓸 때는 장치를 만들 때 이름을 적고, 함수는 `KHR`이 붙은 원래 이름으로 찾습니다
- **기능 비트**: 확장을 지원해도 그 확장의 기능을 다 지원하는 것은 아니어서 비트를 따로 묻습니다. 메시 셰이더는 확장과 `meshShader`가 모두 있어야 하고, `VK_EXT_extended_dynamic_state3`은 상태 31개에 비트가 하나씩 있어 lavapipe도 16개만 지원합니다([「동적 상태」](#/read/cpp-vulkan/ps-dynamic)). 코어도 같아서 1.2의 `vkCmdDrawIndirectCount`는 `drawIndirectCount` 비트를 켜야 씁니다
- **갈림**: `plan`처럼 길을 한 번 정해 두면 장치를 묻는 조건문이 그리는 코드에 흩어지지 않습니다. 개발 기계는 대개 다 지원해서 대체 경로가 돌지 않으므로, 답을 일부러 가려서 돌려 봅니다. 메시 셰이더가 없으면 컴퓨트 셰이더로 컬링하고([「인스턴싱과 간접 그리기」](#/read/cpp-vulkan/tech-instancing)), 포맷은 하나씩 따로 묻습니다([「압축 포맷」](#/read/cpp-vulkan/tex-compressed))
- **한도**: 명세는 한도마다 보장하는 값을 정해 두었고, 여기서는 셋 다 그보다 큽니다. 200바이트 푸시 상수는 여기서 되지만 128바이트만 주는 장치에서는 안 됩니다. 검증 레이어는 지금 장치의 한도로 검사하므로 이 차이를 알려 주지 않습니다. 보장값 안에서 짜거나 한도를 읽어 나눕니다
- **묻지 않고 켜면**: 지원하지 않는 기능을 켜면 `VK_ERROR_FEATURE_NOT_PRESENT`, 없는 확장을 적으면 `VK_ERROR_EXTENSION_NOT_PRESENT`로 장치를 만들지 못합니다. 확장이 있어도 없는 비트 하나가 섞이면 실패합니다. 물어서 받은 구조체를 그대로 넘기면 됩니다([「물리 장치와 논리 장치」](#/read/cpp-vulkan/setup-device))
- **켜지 않고 쓰면**: 지원하는 기능도 켜야 쓸 수 있습니다. lavapipe는 켜지 않은 비등방 필터를 그냥 받아 주었고, 검증 레이어만 VUID-VkSamplerCreateInfo-anisotropyEnable-01070으로 알렸습니다. 드라이버가 받아 주던 것은 다른 장치에서 깨질 수 있습니다

## 프로파일

물을 것이 많아지면 목표를 묶음으로 정합니다. **벌칸 프로파일**은 판, 확장, 기능, 한도, 포맷의 요구를 이름 붙여 묶은 JSON 정의입니다. 크로노스의 로드맵 프로파일과 구글의 안드로이드 기준 프로파일([「안드로이드」](#/read/cpp-vulkan/port-android))이 있습니다.

- **라이브러리**: SDK의 `vulkan_profiles.hpp`에서 `vpGetPhysicalDeviceProfileSupport`는 장치가 프로파일을 채우는지 답하고, `vpCreateDevice`는 프로파일의 확장과 기능을 켠 장치를 만듭니다
- **프로파일 레이어**: `VK_LAYER_KHRONOS_profiles`는 좋은 장치가 프로파일만큼만 지원한다고 답하게 합니다. 넘어선 사용을 막지는 않으므로 검증 레이어와 함께 켭니다
- **통계**: gpuinfo.org는 기기 보고서를 모아, 확장과 기능을 지원하는 기기의 비율을 보여 줍니다

<Note>
숫자는 lavapipe의 답이고 장치마다 다릅니다. "1.1을 적었다면"은 같은 장치에 판만 바꿔 계산한 것입니다. 실제로 1.1을 적은 인스턴스에서 lavapipe는 `vkCmdBeginRendering`을 찾아 주지 않았고(포인터를 주는 구현도 있을 수 있지만 부르면 안 됩니다), `VK_KHR_dynamic_rendering`과 그것이 기대는 `VK_KHR_depth_stencil_resolve`, `VK_KHR_create_renderpass2`를 켠 장치에서 `vkCmdBeginRenderingKHR`을 찾아 주었습니다. 기대는 확장을 빼면 검증 레이어가 VUID-vkCreateDevice-ppEnabledExtensionNames-01387로 알립니다.
</Note>

## 정리

- 쓸 수 있는 판은 인스턴스에 적은 판과 장치의 판 중 낮은 쪽이고, 코어로 올라온 기능은 그 아래 판에서 확장으로 찾는다
- 확장, 기능 비트, 한도, 포맷을 시작할 때 한 번 묻고 갈림을 정하며, 대체 경로는 답을 가려서 시험한다
- 명세가 보장하는 한도 안에서 짜고, 프로파일과 프로파일 레이어로 목표 장치를 정해 개발 기계에서 확인한다
