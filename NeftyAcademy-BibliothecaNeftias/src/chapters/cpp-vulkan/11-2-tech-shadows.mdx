그림자 맵은 두 번 그리는 방법입니다. 먼저 빛의 자리에서 장면의 깊이만 그려 두고, 다음에 카메라에서 그리면서 프래그먼트마다 빛이 본 깊이를 셈해 그 맵과 견줍니다. 맵에 적힌 것보다 멀면 빛과 나 사이에 무엇인가 있다는 뜻이므로 그늘입니다. 아래 프로그램은 바닥에 벽 하나를 세우고 비스듬히 빛을 비춘 뒤, 위에서 내려다본 바닥을 찍습니다(`.`은 빛, `#`은 그늘, `|`는 벽이 선 자리).

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_pipeline.hpp,vkx_draw.hpp
#include <cmath>
#include <string>
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"

// 두 패스가 함께 쓰는 정점 셰이더: view로 화면 위치를, light로 빛이 본 위치를 낸다
const char* VS = R"(#version 450
layout(push_constant) uniform Push { mat4 view; mat4 light; };
layout(location = 0) in vec3 position;
layout(location = 0) out vec3 from_light;
void main() {
    gl_Position = view * vec4(position, 1.0);
    from_light = (light * vec4(position, 1.0)).xyz;
})";
const char* SHADOW_FS = "#version 450\nvoid main() {}";   // 깊이만 남긴다
const char* MAIN_FS = R"(#version 450
layout(set = 0, binding = 0) uniform sampler2DShadow shadow_map;
layout(location = 0) in vec3 from_light;
layout(location = 0) out float lit;
void main() {
    // 비교 샘플러: 셋째 좌표(내 깊이)가 맵의 깊이보다 크지 않으면 1(빛을 받는다)
    lit = texture(shadow_map, vec3(from_light.xy * 0.5 + 0.5, from_light.z));
})";

struct Vec3 { float x, y, z; };
struct Mat4 { float m[16]; };   // GLSL처럼 열 우선

// 직교 투영: right·up 축의 ±half를 화면에, forward 축의 ±4를 깊이 0~1에 편다
Mat4 ortho(Vec3 right, float half_x, Vec3 up, float half_y, Vec3 forward) {
    return {{right.x / half_x, up.x / half_y, forward.x / 8, 0,
             right.y / half_x, up.y / half_y, forward.y / 8, 0,
             right.z / half_x, up.z / half_y, forward.z / 8, 0,
             0, 0, 0.5f, 1}};
}

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const VkExtent2D SCREEN{32, 8}, MAP{32, 32};
    vkx::Image target = vkx::make_image(
        dev, SCREEN, VK_FORMAT_R8_UNORM,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Image map = vkx::make_image(
        dev, MAP, VK_FORMAT_D32_SFLOAT,
        VK_IMAGE_USAGE_DEPTH_STENCIL_ATTACHMENT_BIT | VK_IMAGE_USAGE_SAMPLED_BIT);
    vkx::Buffer pixels = vkx::make_buffer(dev, SCREEN.width * SCREEN.height,
                                          VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);

    // 바닥(y = 0)과, 바닥에 선 벽(x = 0, 높이 1). 삼각형 넷
    const Vec3 scene[] = {
        {-4, 0, -2}, {4, 0, -2}, {4, 0, 2},  {-4, 0, -2}, {4, 0, 2}, {-4, 0, 2},
        {0, 0, -1},  {0, 1, -1}, {0, 1, 1},  {0, 0, -1},  {0, 1, 1}, {0, 0, 1}};
    vkx::Buffer vertices = vkx::make_buffer(
        dev, sizeof scene, VK_BUFFER_USAGE_VERTEX_BUFFER_BIT, true);
    std::memcpy(vertices.mapped, scene, sizeof scene);
    VkVertexInputBindingDescription binding{0, sizeof(Vec3)};
    VkVertexInputAttributeDescription attribute{
        0, 0, VK_FORMAT_R32G32B32_SFLOAT, 0};
    VkPipelineVertexInputStateCreateInfo input{
        .sType = VK_STRUCTURE_TYPE_PIPELINE_VERTEX_INPUT_STATE_CREATE_INFO,
        .vertexBindingDescriptionCount = 1, .pVertexBindingDescriptions = &binding,
        .vertexAttributeDescriptionCount = 1,
        .pVertexAttributeDescriptions = &attribute};

    // 맵 밖은 깊이 1(가장 멂)로 읽혀 빛을 받는 것으로 친다
    VkSamplerCreateInfo sci{
        .sType = VK_STRUCTURE_TYPE_SAMPLER_CREATE_INFO,
        .addressModeU = VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_BORDER,
        .addressModeV = VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_BORDER,
        .compareEnable = VK_TRUE, .compareOp = VK_COMPARE_OP_LESS_OR_EQUAL,
        .borderColor = VK_BORDER_COLOR_FLOAT_OPAQUE_WHITE};
    VkSampler sampler;
    vkx::check(vkCreateSampler(dev.handle, &sci, nullptr, &sampler), "샘플러");
    const auto SAMPLED = VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER;
    vkx::Bindings set = vkx::make_bindings(dev, {SAMPLED});
    vkx::bind_image(dev, set, 0, SAMPLED, map.view,
                    VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL, sampler);
    VkPipelineLayout layout = vkx::make_layout(dev, {set.layout}, 2 * sizeof(Mat4));

    auto vs = vkx::make_shader(dev, vkx::compile(VS, shaderc_vertex_shader));
    auto shadow_fs =
        vkx::make_shader(dev, vkx::compile(SHADOW_FS, shaderc_fragment_shader));
    auto main_fs =
        vkx::make_shader(dev, vkx::compile(MAIN_FS, shaderc_fragment_shader));
    VkPipeline shadow_pass = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = shadow_fs, .layout = layout,
              .depth = map.format, .vertex_input = &input,
              .dynamic = {VK_DYNAMIC_STATE_DEPTH_BIAS_ENABLE,
                          VK_DYNAMIC_STATE_DEPTH_BIAS}});
    VkPipeline main_pass = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = main_fs, .layout = layout,
              .color = target.format, .vertex_input = &input});

    // 빛은 (2, -1, 0) 쪽으로 비스듬히 내리쬔다. 카메라는 바로 위에서 내려다본다
    const float s = 1 / std::sqrt(5.0f);
    const Mat4 light = ortho({0, 0, 1}, 2, {s, 2 * s, 0}, 2, {2 * s, -s, 0});
    const Mat4 camera = ortho({1, 0, 0}, 4, {0, 0, 1}, 2, {0, -1, 0});

    struct Case { const char* what; float bias, slope; };
    const Case cases[] = {{"바이어스 없음", 0, 0},
                          {"받는 쪽 깊이에서 0.02를 뺌", 0.02f, 0},
                          {"받는 쪽 깊이에서 0.12를 뺌", 0.12f, 0},
                          {"맵을 그릴 때 기울기 바이어스 1.5", 0, 1.5f}};
    for (const Case& k : cases) {
        VkCommandBuffer cmd = vkx::begin(cmds);
        const VkDeviceSize zero = 0;
        vkCmdBindVertexBuffers(cmd, 0, 1, &vertices.handle, &zero);

        // 1) 빛의 자리에서 깊이만 그린다
        vkx::transition(cmd, map, VK_IMAGE_LAYOUT_UNDEFINED,
                        VK_IMAGE_LAYOUT_DEPTH_ATTACHMENT_OPTIMAL,
                        VK_PIPELINE_STAGE_2_FRAGMENT_SHADER_BIT, VK_ACCESS_2_NONE,
                        VK_PIPELINE_STAGE_2_EARLY_FRAGMENT_TESTS_BIT,
                        VK_ACCESS_2_DEPTH_STENCIL_ATTACHMENT_WRITE_BIT);
        VkRenderingAttachmentInfo depth{
            .sType = VK_STRUCTURE_TYPE_RENDERING_ATTACHMENT_INFO,
            .imageView = map.view,
            .imageLayout = VK_IMAGE_LAYOUT_DEPTH_ATTACHMENT_OPTIMAL,
            .loadOp = VK_ATTACHMENT_LOAD_OP_CLEAR,
            .storeOp = VK_ATTACHMENT_STORE_OP_STORE,
            .clearValue = {.depthStencil = {1.0f, 0}}};
        VkRenderingInfo info{.sType = VK_STRUCTURE_TYPE_RENDERING_INFO,
                             .renderArea = {{0, 0}, MAP}, .layerCount = 1,
                             .pDepthAttachment = &depth};
        vkCmdBeginRendering(cmd, &info);
        VkViewport viewport{0, 0, float(MAP.width), float(MAP.height), 0, 1};
        VkRect2D scissor{{0, 0}, MAP};
        vkCmdSetViewport(cmd, 0, 1, &viewport);
        vkCmdSetScissor(cmd, 0, 1, &scissor);
        vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, shadow_pass);
        vkCmdSetDepthBiasEnable(cmd, k.slope != 0);
        vkCmdSetDepthBias(cmd, 0, 0, k.slope);   // 상수, 한계, 기울기에 곱할 값
        const Mat4 both[2] = {light, light};
        vkCmdPushConstants(cmd, layout, VK_SHADER_STAGE_ALL, 0, sizeof both, both);
        vkCmdDraw(cmd, 12, 1, 0, 0);
        vkCmdEndRendering(cmd);
        vkx::transition(cmd, map, VK_IMAGE_LAYOUT_DEPTH_ATTACHMENT_OPTIMAL,
                        VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL,
                        VK_PIPELINE_STAGE_2_EARLY_FRAGMENT_TESTS_BIT |
                            VK_PIPELINE_STAGE_2_LATE_FRAGMENT_TESTS_BIT,
                        VK_ACCESS_2_DEPTH_STENCIL_ATTACHMENT_WRITE_BIT,
                        VK_PIPELINE_STAGE_2_FRAGMENT_SHADER_BIT,
                        VK_ACCESS_2_SHADER_SAMPLED_READ_BIT);

        // 2) 카메라에서 그리며 맵과 견준다. 받는 쪽 바이어스는 행렬의 깊이에서 뺀다
        Mat4 biased = light;
        biased.m[14] -= k.bias;
        const Mat4 views[2] = {camera, biased};
        vkx::begin_drawing(cmd, target);
        vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, main_pass);
        vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, layout, 0, 1,
                                &set.set, 0, nullptr);
        vkCmdPushConstants(cmd, layout, VK_SHADER_STAGE_ALL, 0, sizeof views,
                           views);
        vkCmdDraw(cmd, 12, 1, 0, 0);
        vkx::end_drawing(cmd, target, pixels);
        vkx::submit_and_wait(dev, cmds);

        std::printf("%s\n", k.what);
        const auto* px = static_cast<const unsigned char*>(pixels.mapped);
        for (uint32_t y = 0; y < SCREEN.height; ++y) {
            std::string row;
            for (uint32_t x = 0; x < SCREEN.width; ++x) {
                if (x == 16) row += y >= 2 && y < 6 ? '|' : ' ';   // 벽이 선 자리
                row += px[y * SCREEN.width + x] ? '.' : '#';
            }
            std::puts(row.c_str());
        }
    }

    vkDestroyPipeline(dev.handle, shadow_pass, nullptr);
    vkDestroyPipeline(dev.handle, main_pass, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    for (VkShaderModule m : {vs, shadow_fs, main_fs})
        vkDestroyShaderModule(dev.handle, m, nullptr);
    vkx::destroy(dev, set);
    vkDestroySampler(dev.handle, sampler, nullptr);
    for (vkx::Buffer* b : {&vertices, &pixels}) vkx::destroy(dev, *b);
    for (vkx::Image* i : {&target, &map}) vkx::destroy(dev, *i);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **두 패스**: 깊이 맵은 색 첨부물 없이 깊이만 그리고, 다 그린 뒤 읽는 레이아웃으로 바꿉니다. 배리어의 앞쪽은 깊이 시험 단계의 쓰기입니다([「파이프라인 배리어」](#/read/cpp-vulkan/vsync-barriers))
- **비교 샘플러**: `compareEnable`을 켠 샘플러를 `sampler2DShadow`로 읽으면 깊이 값이 아니라 견준 결과(0이나 1)가 나옵니다. 맵 밖은 테두리 색을 흰색(깊이 1)으로 두어 빛을 받는 것으로 칩니다
- **여드름**: 바이어스가 없으면 빛을 받는 바닥에 줄무늬가 생깁니다. 맵의 텍셀 하나가 바닥의 넓은 조각을 깊이 하나로 대신하므로, 그 조각의 먼 쪽 절반은 자기 자신보다 멀다고 나옵니다. 빛이 비스듬할수록 심합니다
- **떠 보이는 그림자**: 받는 쪽 깊이에서 0.02를 빼면 줄무늬가 사라지지만 그림자가 벽에서 한 칸 떨어지고, 0.12를 빼면 세 칸 떨어집니다. 벽 밑동에서는 벽과 바닥의 깊이 차이가 바이어스보다 작기 때문입니다
- **기울기 바이어스**: `vkCmdSetDepthBias`의 셋째 값은 삼각형의 깊이 기울기에 곱해져, 비스듬한 면일수록 맵에 더 깊게 적힙니다. 바닥은 필요한 만큼 밀리고 빛을 마주 보는 벽은 거의 그대로여서, 그림자가 벽에 붙은 채로 줄무늬가 사라졌습니다

## 더 다듬기

- 비교 샘플러의 필터를 `LINEAR`로 두면(깊이 포맷이 선형 필터를 지원해야 합니다) 둘레 네 텍셀의 비교 결과가 섞여 가장자리가 부드러워집니다(PCF). 셰이더에서 둘레를 여러 번 읽어 평균하면 더 넓게 흐려집니다
- 바이어스의 첫째 값(상수)은 깊이 포맷이 가를 수 있는 가장 작은 차이의 배수입니다. `D32_SFLOAT`에서는 그 단위가 삼각형의 깊이에 따라 달라지므로 기울기 쪽을 먼저 맞춥니다
- 빛의 투영이 덮는 범위와 깊이 범위를 장면에 바싹 맞출수록 텍셀 하나가 맡는 넓이가 줄어 두 문제가 함께 줄어듭니다. 넓은 야외는 카메라에서 가까운 곳부터 맵을 여러 장 나눠 씁니다(캐스케이드)

<Note>
받는 쪽에서 빼는 값을 푸시 상수에 따로 두지 않고 행렬의 깊이 항에서 뺐습니다. 행렬 둘로 푸시 상수 128바이트가 꽉 차기 때문입니다([「푸시 상수」](#/read/cpp-vulkan/desc-push)). 점 광원은 원근 투영으로, 사방을 비추면 큐브 맵 여섯 면에 깊이를 그립니다.
</Note>

## 정리

- 빛의 자리에서 깊이만 그린 맵과 프래그먼트의 깊이를 비교 샘플러로 견줘 그늘을 가린다
- 바이어스가 없으면 줄무늬가, 지나치면 물체에서 떨어진 그림자가 생기므로 기울기에 비례하는 바이어스로 맞춘다
- 맵의 범위를 장면에 바싹 맞추고, 가장자리는 비교 결과를 섞어 부드럽게 한다
