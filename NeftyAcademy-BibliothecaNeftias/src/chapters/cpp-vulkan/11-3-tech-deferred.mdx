빛이 많은 장면을 그리는 대로 조명하면(순방향), 나중에 다른 물체에 가려질 프래그먼트에도 조명을 다 셈합니다. 지연 렌더링은 일을 둘로 나눕니다. 먼저 물체를 그리면서 조명의 재료(바탕색, 법선, 깊이)만 화면 크기의 이미지들에 적어 둡니다. 이 이미지 묶음이 G-버퍼입니다. 다음에 화면을 덮는 삼각형 하나를 그리며 픽셀마다 재료를 읽어 조명을 한 번만 셈합니다. 아래 프로그램은 겹친 사각형 다섯을 두 방식으로 그려 프래그먼트 셰이더가 돈 횟수를 셉니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_pipeline.hpp,vkx_draw.hpp,vkx_texture.hpp
#include <algorithm>
#include <cstdlib>
#include <string>
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"
#include "vkx_texture.hpp"

// 사각형 다섯을 먼 것부터 겹쳐 그린다. 세계 좌표를 클립 좌표와 같게 두었다
const char* VS = R"(#version 450
layout(location = 0) out vec3 position;
layout(location = 1) out vec3 normal;
layout(location = 2) out float albedo;
void main() {
    float i = float(gl_InstanceIndex);
    vec2 corner = vec2(gl_VertexIndex % 2, (gl_VertexIndex + 1) / 3 % 2);
    position = vec3(-1.0 + 0.1 * i + 1.6 * corner, 0.9 - 0.15 * i);
    normal = vec3(0.2 * (i - 2.0), 0.2, -1.0);
    albedo = 0.2 + 0.2 * i;
    gl_Position = vec4(position, 1.0);
})";
// 비싼 부분: 빛 넷을 돌며 더한다. 두 방식이 같은 함수를 쓴다
const std::string SHADE = R"(#version 450
const vec3 LIGHTS[4] = vec3[](vec3(-1, -1, -0.5), vec3(1, -1, -0.5),
                              vec3(-1, 1, -0.5), vec3(1, 1, -0.5));
float shade(vec3 P, vec3 N, float albedo) {
    float sum = 0.0;
    for (int k = 0; k < 4; ++k) {
        vec3 to = LIGHTS[k] - P;
        sum += max(dot(normalize(N), normalize(to)), 0.0) / (1.0 + dot(to, to));
    }
    return albedo * sum;
})";
const std::string FORWARD_FS = SHADE + R"(
layout(location = 0) in vec3 position;
layout(location = 1) in vec3 normal;
layout(location = 2) in float albedo;
layout(location = 0) out float color;
void main() { color = shade(position, normal, albedo); })";
// 지연 1단계: 셈하지 않고 재료만 적는다. 출력이 둘이다
const char* GBUFFER_FS = R"(#version 450
layout(location = 1) in vec3 normal;
layout(location = 2) in float albedo;
layout(location = 0) out float g_albedo;
layout(location = 1) out vec4 g_normal;
void main() { g_albedo = albedo; g_normal = vec4(normal, 0.0); })";
// 지연 2단계: 화면의 픽셀마다 적어 둔 재료를 읽어 한 번만 셈한다
const std::string LIGHTING_FS = SHADE + R"(
layout(set = 0, binding = 0) uniform sampler2D g_albedo;
layout(set = 0, binding = 1) uniform sampler2D g_normal;
layout(set = 0, binding = 2) uniform sampler2D g_depth;
layout(location = 0) out float color;
void main() {
    ivec2 at = ivec2(gl_FragCoord.xy);
    float depth = texelFetch(g_depth, at, 0).r;
    vec2 xy = gl_FragCoord.xy / vec2(textureSize(g_depth, 0)) * 2.0 - 1.0;
    color = depth == 1.0 ? 0.0   // 아무것도 그려지지 않은 픽셀
          : shade(vec3(xy, depth), texelFetch(g_normal, at, 0).xyz,
                  texelFetch(g_albedo, at, 0).r);
})";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const VkExtent2D SIZE{24, 12};
    const auto DRAWN = VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT;
    const auto READ = VK_IMAGE_USAGE_SAMPLED_BIT;
    const auto COPIED = VK_IMAGE_USAGE_TRANSFER_SRC_BIT;
    const auto GRAY = VK_FORMAT_R8_UNORM;
    vkx::Image forward = vkx::make_image(dev, SIZE, GRAY, DRAWN | COPIED);
    vkx::Image lit = vkx::make_image(dev, SIZE, GRAY, DRAWN | COPIED);
    // G-버퍼: 바탕색, 법선, 깊이
    vkx::Image albedo = vkx::make_image(dev, SIZE, GRAY, DRAWN | READ);
    vkx::Image normal =
        vkx::make_image(dev, SIZE, VK_FORMAT_R16G16B16A16_SFLOAT, DRAWN | READ);
    vkx::Image depth = vkx::make_image(
        dev, SIZE, VK_FORMAT_D32_SFLOAT,
        VK_IMAGE_USAGE_DEPTH_STENCIL_ATTACHMENT_BIT | READ);
    vkx::Image* gbuffer[] = {&albedo, &normal, &depth};
    const uint32_t COUNT = SIZE.width * SIZE.height;
    const auto FILLED = VK_BUFFER_USAGE_TRANSFER_DST_BIT;
    vkx::Buffer a = vkx::make_buffer(dev, COUNT, FILLED, true);   // 순방향의 그림
    vkx::Buffer b = vkx::make_buffer(dev, COUNT, FILLED, true);   // 지연의 그림

    VkSampler sampler = vkx::make_sampler(dev, VK_FILTER_NEAREST,
                                          VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE);
    const auto SAMPLED = VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER;
    vkx::Bindings set = vkx::make_bindings(dev, {SAMPLED, SAMPLED, SAMPLED});
    for (uint32_t i = 0; i < 3; ++i)
        vkx::bind_image(dev, set, i, SAMPLED, gbuffer[i]->view,
                        VK_IMAGE_LAYOUT_READ_ONLY_OPTIMAL, sampler);
    VkPipelineLayout layout = vkx::make_layout(dev, {set.layout});

    auto shader = [&](const char* glsl, shaderc_shader_kind kind) {
        return vkx::make_shader(dev, vkx::compile(glsl, kind));
    };
    VkShaderModule modules[] = {
        shader(VS, shaderc_vertex_shader),
        shader(FORWARD_FS.c_str(), shaderc_fragment_shader),
        shader(GBUFFER_FS, shaderc_fragment_shader),
        shader(vkx::FULLSCREEN_VS, shaderc_vertex_shader),
        shader(LIGHTING_FS.c_str(), shaderc_fragment_shader)};
    VkPipeline forward_pass = vkx::make_pipeline(
        dev, {.vertex = modules[0], .fragment = modules[1], .layout = layout,
              .color = forward.format, .depth = depth.format});
    VkPipeline gbuffer_pass = vkx::make_pipeline(   // 색 첨부물이 둘
        dev, {.vertex = modules[0], .fragment = modules[2], .layout = layout,
              .color = albedo.format, .more = {normal.format},
              .depth = depth.format});
    VkPipeline lighting_pass = vkx::make_pipeline(
        dev, {.vertex = modules[3], .fragment = modules[4], .layout = layout,
              .color = lit.format});

    VkQueryPoolCreateInfo qci{
        .sType = VK_STRUCTURE_TYPE_QUERY_POOL_CREATE_INFO,
        .queryType = VK_QUERY_TYPE_PIPELINE_STATISTICS, .queryCount = 3,
        .pipelineStatistics =
            VK_QUERY_PIPELINE_STATISTIC_FRAGMENT_SHADER_INVOCATIONS_BIT};
    VkQueryPool stats;
    vkx::check(vkCreateQueryPool(dev.handle, &qci, nullptr, &stats), "쿼리 풀");

    VkCommandBuffer cmd = vkx::begin(cmds);
    vkCmdResetQueryPool(cmd, stats, 0, 3);
    auto draw = [&](VkPipeline pipeline, uint32_t query, uint32_t vertices,
                    uint32_t instances) {
        vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
        vkCmdBeginQuery(cmd, stats, query, 0);   // 프래그먼트 셰이더 실행을 센다
        vkCmdDraw(cmd, vertices, instances, 0, 0);
        vkCmdEndQuery(cmd, stats, query);
    };

    // 순방향: 그리는 프래그먼트마다 조명을 셈한다
    vkx::begin_drawing(cmd, forward, {}, &depth);
    draw(forward_pass, 0, 6, 5);
    vkx::end_drawing(cmd, forward, a);

    // 지연 1단계: 첨부물 셋에 재료를 적는다
    const auto STAGES = VK_PIPELINE_STAGE_2_COLOR_ATTACHMENT_OUTPUT_BIT |
                        VK_PIPELINE_STAGE_2_EARLY_FRAGMENT_TESTS_BIT |
                        VK_PIPELINE_STAGE_2_LATE_FRAGMENT_TESTS_BIT;
    const auto WRITES = VK_ACCESS_2_COLOR_ATTACHMENT_WRITE_BIT |
                        VK_ACCESS_2_DEPTH_STENCIL_ATTACHMENT_WRITE_BIT;
    VkRenderingAttachmentInfo to[3];
    for (int i = 0; i < 3; ++i) {
        vkx::transition(cmd, *gbuffer[i], VK_IMAGE_LAYOUT_UNDEFINED,
                        VK_IMAGE_LAYOUT_ATTACHMENT_OPTIMAL, STAGES, WRITES, STAGES,
                        WRITES);
        to[i] = {.sType = VK_STRUCTURE_TYPE_RENDERING_ATTACHMENT_INFO,
                 .imageView = gbuffer[i]->view,
                 .imageLayout = VK_IMAGE_LAYOUT_ATTACHMENT_OPTIMAL,
                 .loadOp = VK_ATTACHMENT_LOAD_OP_CLEAR,
                 .storeOp = VK_ATTACHMENT_STORE_OP_STORE};
    }
    to[2].clearValue.depthStencil = {1.0f, 0};
    VkRenderingInfo info{.sType = VK_STRUCTURE_TYPE_RENDERING_INFO,
                         .renderArea = {{0, 0}, SIZE}, .layerCount = 1,
                         .colorAttachmentCount = 2, .pColorAttachments = to,
                         .pDepthAttachment = &to[2]};
    vkCmdBeginRendering(cmd, &info);   // 뷰포트와 시저는 앞에서 정한 것이 남아 있다
    draw(gbuffer_pass, 1, 6, 5);
    vkCmdEndRendering(cmd);
    for (vkx::Image* im : gbuffer)
        vkx::transition(cmd, *im, VK_IMAGE_LAYOUT_ATTACHMENT_OPTIMAL,
                        VK_IMAGE_LAYOUT_READ_ONLY_OPTIMAL, STAGES, WRITES,
                        VK_PIPELINE_STAGE_2_FRAGMENT_SHADER_BIT,
                        VK_ACCESS_2_SHADER_SAMPLED_READ_BIT);

    // 지연 2단계: 화면을 덮는 삼각형 하나로 픽셀마다 조명을 셈한다
    vkx::begin_drawing(cmd, lit);
    vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, layout, 0, 1,
                            &set.set, 0, nullptr);
    draw(lighting_pass, 2, 3, 1);
    vkx::end_drawing(cmd, lit, b);
    vkx::submit_and_wait(dev, cmds);

    uint64_t shaded[3];
    vkGetQueryPoolResults(dev.handle, stats, 0, 3, sizeof shaded, shaded, 8,
                          VK_QUERY_RESULT_64_BIT | VK_QUERY_RESULT_WAIT_BIT);
    const auto* pa = static_cast<const unsigned char*>(a.mapped);
    const auto* pb = static_cast<const unsigned char*>(b.mapped);
    int worst = 0;
    for (uint32_t i = 0; i < COUNT; ++i)
        worst = std::max(worst, std::abs(pa[i] - pb[i]));
    for (uint32_t y = 0; y < SIZE.height; ++y) {
        std::string row;
        for (uint32_t x = 0; x < SIZE.width; ++x)
            row += " .:-=+*#%@"[pb[y * SIZE.width + x] * 10 / 256];
        row.erase(row.find_last_not_of(' ') + 1);
        std::puts(row.c_str());
    }
    std::printf("순방향: 조명 셰이더 %u번\n", unsigned(shaded[0]));
    std::printf("지연:   재료만 적는 셰이더 %u번 + 조명 셰이더 %u번(화면 %u픽셀)\n",
                unsigned(shaded[1]), unsigned(shaded[2]), COUNT);
    std::printf("두 그림에서 가장 크게 다른 픽셀의 차이: %d/255\n", worst);

    vkDestroyQueryPool(dev.handle, stats, nullptr);
    for (VkPipeline p : {forward_pass, gbuffer_pass, lighting_pass})
        vkDestroyPipeline(dev.handle, p, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    for (VkShaderModule m : modules) vkDestroyShaderModule(dev.handle, m, nullptr);
    vkx::destroy(dev, set);
    vkDestroySampler(dev.handle, sampler, nullptr);
    for (vkx::Buffer* buf : {&a, &b}) vkx::destroy(dev, *buf);
    for (vkx::Image* im : {&forward, &lit, &albedo, &normal, &depth})
        vkx::destroy(dev, *im);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}

   ................
 :--------------------
 :--+*++++++++++++++++++
 :--++++++++++++++++++++
 :--=+++++++++++++++++==
 :--=+++++==============
 :--=+++++=======+++====
 :--=++++++===+++++++++=
 :--=+++++++++++++++++++
 :--++++++++++++++++++++
  --++++++++++++++******
     ++++====+++++******
순방향: 조명 셰이더 922번
지연:   재료만 적는 셰이더 922번 + 조명 셰이더 288번(화면 288픽셀)
두 그림에서 가장 크게 다른 픽셀의 차이: 0/255
```

- **여러 첨부물에 한꺼번에**: `VkRenderingInfo`의 `colorAttachmentCount`를 2로 하고, 파이프라인에 포맷 둘을 알리고(도우미의 `more`), 셰이더의 `out`을 location 0과 1에 둡니다. 세 곳의 개수와 포맷이 서로 맞아야 합니다
- **셈 횟수**: 조명 셰이더는 순방향에서 922번, 지연에서 288번 돌았습니다. 지연의 첫 단계도 922번 돌지만 값을 적기만 합니다. 겹침과 빛이 많을수록 차이가 벌어집니다
- **위치는 깊이에서**: 위치는 적어 두지 않고 픽셀 좌표와 깊이로 되살립니다. 여기서는 세계 좌표를 클립 좌표와 같게 두어 그대로 썼고, 보통은 투영의 역행렬을 곱합니다
- **같은 그림**: 두 방식의 결과는 한 단계도 다르지 않았습니다. 재료가 같고 식이 같기 때문입니다
- **읽기 전에 바꾸기**: G-버퍼는 다 그린 뒤 `READ_ONLY_OPTIMAL`로 바꿔 읽습니다. 색과 깊이를 같은 줄로 다루려고 단계와 접근을 합쳐 적었습니다([「파이프라인 배리어」](#/read/cpp-vulkan/vsync-barriers))

## 치르는 값

- **메모리와 대역폭**: 픽셀마다 이미지 여러 장을 쓰고 다시 읽습니다. Khronos의 샘플은 색 첨부물 셋에 픽셀당 96비트와 깊이 32비트를 씁니다
- **타일 기반 GPU**: 패스를 둘로 나누면 G-버퍼를 메모리에 내보냈다가 다시 읽어야 합니다. 한 렌더링 안에서 같은 픽셀의 값을 입력 첨부물로 읽게 하면(서브패스, 동적 렌더링에서는 `dynamicRenderingLocalRead`) G-버퍼가 타일 메모리를 떠나지 않습니다. 같은 샘플은 Mali-G76에서 처리한 타일 수가 55%쯤 줄었다고 적습니다
- **투명한 물체**: G-버퍼에는 픽셀마다 표면 하나만 남습니다. 반투명한 것은 조명이 끝난 뒤 순방향으로 따로 그립니다
- **순방향 쪽의 길**: 깊이만 먼저 그려 두면 이른 깊이 시험이 가려진 프래그먼트를 버려, 순방향도 픽셀마다 한 번에 가깝게 셈합니다([「깊이 버퍼」](#/read/cpp-vulkan/tex-depth))

<Note>
조명 패스는 아무것도 그려지지 않은 픽셀에서도, 빛이 닿지 않는 픽셀에서도 모든 빛을 돕니다. 빛이 수백 개가 되면 화면을 타일로 나눠 타일마다 닿는 빛의 목록을 컴퓨트 셰이더로 만들어 두고, 조명 셰이더가 그 목록만 돌게 합니다.
</Note>

## 정리

- 지연 렌더링은 재료를 G-버퍼에 적는 단계와 픽셀마다 조명을 한 번 셈하는 단계로 나뉜다
- 색 첨부물이 여럿이면 렌더링 정보, 파이프라인의 포맷, 셰이더 출력의 개수가 맞아야 한다
- 메모리와 대역폭을 더 쓰고 투명한 물체를 따로 그려야 하며, 타일 기반 GPU에서는 한 렌더링 안에서 읽게 짠다
