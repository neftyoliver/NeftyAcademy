파이프라인은 상태를 굳혀 만든 객체라서, 뒷면을 버릴지 말지 하나만 달라도 파이프라인이 하나 더 필요합니다. 셰이더 조합에 이런 상태 조합이 곱해지면 파이프라인 수가 금방 불어납니다. **동적 상태**는 파이프라인을 만들 때 "이 상태는 굳히지 않는다"고 알려 두고, 그릴 때 커맨드 버퍼의 명령으로 정하는 상태입니다. [「깊이 버퍼」](#/read/cpp-vulkan/tex-depth)에서 깊이 비교를 그렇게 바꿨습니다. 아래 프로그램은 파이프라인 하나로 컬링, 앞면의 방향, 토폴로지를 바꿔 가며 다섯 줄을 그립니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_pipeline.hpp,vkx_draw.hpp
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"

// 정점 여섯 개를 지그재그로 놓았다. 셋씩 끊으면(LIST) 삼각형 둘이 되는데,
// 화면에서 볼 때 왼쪽 것은 시계 방향, 오른쪽 것은 반시계 방향으로 돈다
const char* VS = R"(#version 450
layout(location = 0) flat out float shade;
const vec2 pos[6] = vec2[](vec2(-0.9, 1), vec2(-0.9, -1), vec2(0, 1),
                           vec2(0, -1), vec2(0.9, 1), vec2(0.9, -1));
void main() {
    gl_Position = vec4(pos[gl_VertexIndex], 0.0, 1.0);
    shade = gl_VertexIndex < 3 ? 0.4 : 0.95;
})";

const char* FS = R"(#version 450
layout(location = 0) flat in float shade;
layout(location = 0) out float value;
void main() { value = shade; })";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);

    struct Row { const char* label; VkCullModeFlags cull; VkFrontFace front;
                 VkPrimitiveTopology topology; };
    const auto CCW = VK_FRONT_FACE_COUNTER_CLOCKWISE, CW = VK_FRONT_FACE_CLOCKWISE;
    const auto LIST = VK_PRIMITIVE_TOPOLOGY_TRIANGLE_LIST;
    const Row rows[] = {
        {"cull NONE", VK_CULL_MODE_NONE, CCW, LIST},
        {"cull BACK", VK_CULL_MODE_BACK_BIT, CCW, LIST},
        {"cull FRONT", VK_CULL_MODE_FRONT_BIT, CCW, LIST},
        {"cull BACK, front CW", VK_CULL_MODE_BACK_BIT, CW, LIST},
        {"cull NONE, STRIP", VK_CULL_MODE_NONE, CCW,
         VK_PRIMITIVE_TOPOLOGY_TRIANGLE_STRIP}};
    const uint32_t W = 48, H = std::size(rows);
    vkx::Image target = vkx::make_image(
        dev, {W, H}, VK_FORMAT_R32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels =
        vkx::make_buffer(dev, W * H * 4, VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    VkPipelineLayoutCreateInfo lci{
        .sType = VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};
    VkPipelineLayout layout;
    vkCreatePipelineLayout(dev.handle, &lci, nullptr, &layout);
    auto vs = vkx::make_shader(dev, vkx::compile(VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    // 파이프라인은 하나. 세 가지 상태를 굳히지 않고 그릴 때 정하겠다고 알린다
    VkPipeline pipeline = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = fs, .layout = layout,
              .color = target.format,
              .dynamic = {VK_DYNAMIC_STATE_CULL_MODE, VK_DYNAMIC_STATE_FRONT_FACE,
                          VK_DYNAMIC_STATE_PRIMITIVE_TOPOLOGY}});

    VkCommandBuffer cmd = vkx::begin(cmds);
    vkx::begin_drawing(cmd, target, {{-1.0f}});
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
    for (uint32_t y = 0; y < H; ++y) {
        VkViewport line{0.0f, float(y), float(W), 1.0f, 0.0f, 1.0f};
        vkCmdSetViewport(cmd, 0, 1, &line);
        vkCmdSetCullMode(cmd, rows[y].cull);             // 1.3에서 코어가 된 명령들
        vkCmdSetFrontFace(cmd, rows[y].front);
        vkCmdSetPrimitiveTopology(cmd, rows[y].topology);
        vkCmdDraw(cmd, 6, 1, 0, 0);
    }
    vkx::end_drawing(cmd, target, pixels);
    vkx::submit_and_wait(dev, cmds);

    auto* v = static_cast<const float*>(pixels.mapped);
    const char* ramp = ".:-=+*#%@";
    for (uint32_t y = 0; y < H; ++y) {
        std::printf("%-20s|", rows[y].label);
        for (uint32_t x = 0; x < W; ++x) {
            float s = v[y * W + x];
            std::putchar(s < 0 ? ' ' : ramp[int(s * 8.999f)]);
        }
        std::puts("|");
    }
    std::printf("파이프라인 1개로 설정 %u가지를 그림\n", H);

    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    vkx::destroy(dev, pixels);
    vkx::destroy(dev, target);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
cull NONE           |  ===========                      @@@@@@@@@@@  |
cull BACK           |                                   @@@@@@@@@@@  |
cull FRONT          |  ===========                                   |
cull BACK, front CW |  ===========                                   |
cull NONE, STRIP    |  =================================@@@@@@@@@@@  |
파이프라인 1개로 설정 5가지를 그림
```

- **컬링과 앞면**: 삼각형이 화면에서 도는 방향으로 앞뒤를 가립니다. 반시계 방향을 앞면으로 두면 `BACK`은 시계 방향인 왼쪽 삼각형을 버리고, 앞면을 시계 방향으로 바꾸면 반대가 됩니다. 닫힌 물체의 안쪽 면은 보이지 않으므로 뒷면을 버려 래스터화할 양을 반쯤으로 줄입니다
- **토폴로지**: 같은 정점 여섯 개를 `TRIANGLE_STRIP`으로 읽으면 이웃한 세 정점마다 삼각형이 되어 띠 전체가 채워집니다. 동적으로 바꿀 수 있는 것은 같은 부류 안에서뿐이어서(삼각형 목록·띠·부채), 파이프라인에는 여전히 부류를 정하는 토폴로지를 적습니다
- **모두 정하고 그린다**: 동적이라고 알린 상태는 그리기 전에 반드시 값을 넣어야 합니다. `vkCmdSetCullMode`를 빼면 검증 레이어가 VUID-vkCmdDraw-None-07840으로 알립니다. 넣은 값은 커맨드 버퍼에 남아 다음 그리기에도 쓰이지만, 그 상태를 굳혀 만든 파이프라인을 붙이면 무효가 되므로 다시 넣어야 합니다
- **파이프라인 수**: 컬링 셋, 앞면 둘, 토폴로지 둘을 모두 굳혀 만들면 조합이 12개까지 나옵니다. 여기에 깊이 비교와 깊이 쓰기까지 곱하면 수십 개가 되는데, 동적 상태로 두면 셰이더 조합마다 하나면 됩니다

## 어떤 상태가 동적인가

| 들어온 때 | 동적으로 둘 수 있는 상태 |
| --- | --- |
| 1.0 | 뷰포트, 시저, 선 굵기, 깊이 바이어스, 블렌드 상수, 깊이 경계, 스텐실의 마스크와 기준값 |
| 1.3 | 컬링, 앞면, 토폴로지, 깊이 시험·쓰기·비교, 스텐실 시험과 연산, 깊이 바이어스 켜기, 프리미티브 재시작, 래스터화 끄기 |
| 확장 | 다각형 모드, 블렌딩 켜기와 식, 색 쓰기 마스크, 샘플 수(`VK_EXT_extended_dynamic_state3`), 정점 입력(`VK_EXT_vertex_input_dynamic_state`) |

1.3의 것들은 1.3 장치라면 기능을 따로 켜지 않아도 쓸 수 있습니다. 확장의 것들은 기능 비트를 확인하고 쓰는데, `VK_EXT_extended_dynamic_state3`은 상태마다 비트가 따로 있어 장치가 일부만 지원할 수 있습니다([「장치 기능에 따라 가르기」](#/read/cpp-vulkan/port-features)).

<Note>
동적 상태를 끝까지 밀고 간 것이 `VK_EXT_shader_object`입니다. 파이프라인 객체 없이 셰이더를 단계별로 따로 만들어 붙이고, 모든 상태를 명령으로 정합니다. 상태 조합이 많은 엔진에서 파이프라인을 미리 만들어 두는 부담을 없애려는 확장이고, 드라이버가 지원하지 않을 때 대신 구현해 주는 레이어(`VK_LAYER_KHRONOS_shader_object`)도 있습니다.
</Note>

## 정리

- 동적 상태는 파이프라인에 굳히지 않고 그릴 때 명령으로 정하는 상태이고, 파이프라인 수를 셰이더 조합 수로 줄인다
- 1.3에서 컬링, 앞면, 토폴로지, 깊이·스텐실 설정이 코어의 동적 상태가 됐다
- 동적이라고 알린 상태는 그리기 전에 모두 값을 넣어야 하고, 그 상태를 굳힌 파이프라인을 붙이면 다시 넣는다
