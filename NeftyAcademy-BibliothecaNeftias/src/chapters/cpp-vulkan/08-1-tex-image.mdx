텍스처 하나를 셰이더에서 읽으려면 객체 셋이 필요합니다. **이미지**는 텍셀을 담은 메모리이고(차원, 포맷, 크기, 밉 수, 층 수, 쓰임새), **뷰**는 그 이미지의 어느 부분을 어떤 종류로 볼지 정하며, **샘플러**는 좌표로 텍셀을 고르고 섞는 방법입니다. 지금까지는 이미지를 렌더 타깃으로만 썼는데([「렌더 패스와 동적 렌더링」](#/read/cpp-vulkan/tri-renderpass)), 이 쪽의 `vkx_texture.hpp`는 CPU의 픽셀을 올려 텍스처로 만들고 샘플러를 만듭니다.

```cpp vk file=vkx_texture.hpp with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp
#pragma once
#include "vkx_cmd.hpp"
#include "vkx_resource.hpp"

namespace vkx {

// CPU의 픽셀을 스테이징 버퍼를 거쳐 올려, 셰이더가 읽을 텍스처로 만든다.
// mips가 1보다 크면 밉 사슬의 자리만 잡고 0번 수준만 채운다
inline Image make_texture(const Device& d, Commands& c, VkExtent2D extent,
                          VkFormat format, const void* pixels, VkDeviceSize bytes,
                          uint32_t mips = 1) {
    Image im = make_image(d, extent, format,
                          VK_IMAGE_USAGE_SAMPLED_BIT |
                              VK_IMAGE_USAGE_TRANSFER_DST_BIT |
                              VK_IMAGE_USAGE_TRANSFER_SRC_BIT, mips);
    Buffer staging = make_buffer(d, bytes, VK_BUFFER_USAGE_TRANSFER_SRC_BIT, true);
    std::memcpy(staging.mapped, pixels, bytes);
    const auto FILLING = VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL;
    const auto COPY = VK_PIPELINE_STAGE_2_COPY_BIT;
    VkCommandBuffer cmd = begin(c);
    transition(cmd, im, VK_IMAGE_LAYOUT_UNDEFINED, FILLING,
               VK_PIPELINE_STAGE_2_NONE, VK_ACCESS_2_NONE, COPY,
               VK_ACCESS_2_TRANSFER_WRITE_BIT);
    VkBufferImageCopy region{   // 버퍼의 처음부터, 밉 0 전체로
        .imageSubresource = {im.aspect, 0, 0, 1},
        .imageExtent = {extent.width, extent.height, 1}};
    vkCmdCopyBufferToImage(cmd, staging.handle, im.handle, FILLING, 1, &region);
    transition(cmd, im, FILLING, VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL, COPY,
               VK_ACCESS_2_TRANSFER_WRITE_BIT, VK_PIPELINE_STAGE_2_ALL_COMMANDS_BIT,
               VK_ACCESS_2_SHADER_SAMPLED_READ_BIT);
    submit_and_wait(d, c);
    destroy(d, staging);
    return im;
}

// 확대·축소 필터와 주소 모드를 같은 값으로 채운 샘플러. 밉은 있는 만큼 쓴다
inline VkSampler make_sampler(
        const Device& d, VkFilter filter, VkSamplerAddressMode address,
        VkSamplerMipmapMode mipmap = VK_SAMPLER_MIPMAP_MODE_NEAREST) {
    VkSamplerCreateInfo ci{
        .sType = VK_STRUCTURE_TYPE_SAMPLER_CREATE_INFO, .magFilter = filter,
        .minFilter = filter, .mipmapMode = mipmap, .addressModeU = address,
        .addressModeV = address, .addressModeW = address,
        .maxLod = VK_LOD_CLAMP_NONE};
    VkSampler s;
    check(vkCreateSampler(d.handle, &ci, nullptr, &s), "샘플러");
    return s;
}

}  // namespace vkx
```

올리는 길은 [「스테이징」](#/read/cpp-vulkan/buf-staging)과 같고, 복사 앞뒤에 레이아웃 전환이 붙습니다([「레이아웃 전환」](#/read/cpp-vulkan/tex-layout)). 아래 프로그램은 텍셀 네 개짜리 텍스처를 좌표 -1부터 2까지 읽어, 샘플러와 뷰에 따라 무엇이 나오는지 한 줄씩 찍습니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_pipeline.hpp,vkx_draw.hpp,vkx_texture.hpp
#include "vkx_bind.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"
#include "vkx_texture.hpp"

const char* VS = R"(#version 450
layout(location = 0) out float u;
void main() {       // 화면을 덮는 삼각형. u는 왼쪽 끝 -1에서 오른쪽 끝 2까지
    vec2 p = vec2((gl_VertexIndex << 1) & 2, gl_VertexIndex & 2);
    gl_Position = vec4(p * 2.0 - 1.0, 0.0, 1.0);
    u = p.x * 3.0 - 1.0;
})";

const char* FS = R"(#version 450
layout(binding = 0) uniform sampler2D tex;
layout(location = 0) in float u;
layout(location = 0) out float value;
void main() { value = texture(tex, vec2(u, 0.5)).r; })";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);

    // 4×1 텍스처. R은 밝아지고 G는 어두워진다(바이트 순서 R, G, B, A)
    const uint8_t shade[4] = {43, 99, 156, 213};
    uint8_t texels[4][4];
    for (int k = 0; k < 4; ++k) {
        texels[k][0] = shade[k], texels[k][1] = shade[3 - k];
        texels[k][2] = 0, texels[k][3] = 255;
    }
    vkx::Image tex = vkx::make_texture(dev, cmds, {4, 1}, VK_FORMAT_R8G8B8A8_UNORM,
                                       texels, sizeof texels);
    // 같은 이미지를 달리 보는 뷰: 셰이더가 R을 읽으면 G가 나온다
    VkImageViewCreateInfo vci{
        .sType = VK_STRUCTURE_TYPE_IMAGE_VIEW_CREATE_INFO, .image = tex.handle,
        .viewType = VK_IMAGE_VIEW_TYPE_2D, .format = tex.format,
        .components = {.r = VK_COMPONENT_SWIZZLE_G},
        .subresourceRange = {VK_IMAGE_ASPECT_COLOR_BIT, 0, 1, 0, 1}};
    VkImageView swapped;
    vkCreateImageView(dev.handle, &vci, nullptr, &swapped);

    struct Row { const char* label; VkFilter filter; VkSamplerAddressMode address;
                 VkImageView view; };
    const auto NEAR = VK_FILTER_NEAREST, LINEAR = VK_FILTER_LINEAR;
    const Row rows[] = {
        {"NEAREST REPEAT", NEAR, VK_SAMPLER_ADDRESS_MODE_REPEAT, tex.view},
        {"NEAREST MIRRORED_REPEAT", NEAR,
         VK_SAMPLER_ADDRESS_MODE_MIRRORED_REPEAT, tex.view},
        {"NEAREST CLAMP_TO_EDGE", NEAR, VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE,
         tex.view},
        {"NEAREST CLAMP_TO_BORDER", NEAR, VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_BORDER,
         tex.view},
        {"LINEAR  REPEAT", LINEAR, VK_SAMPLER_ADDRESS_MODE_REPEAT, tex.view},
        {"LINEAR  CLAMP_TO_EDGE", LINEAR, VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE,
         tex.view},
        {"NEAREST REPEAT, view R=G", NEAR, VK_SAMPLER_ADDRESS_MODE_REPEAT,
         swapped}};
    const uint32_t W = 48, H = std::size(rows);

    vkx::Image target = vkx::make_image(
        dev, {W, H}, VK_FORMAT_R32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels =
        vkx::make_buffer(dev, W * H * 4, VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    const auto TEXTURE = VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER;
    std::vector<vkx::Bindings> binds;       // 줄마다 세트 하나: 뷰와 샘플러 한 쌍
    std::vector<VkSampler> samplers;
    for (const Row& r : rows) {
        samplers.push_back(vkx::make_sampler(dev, r.filter, r.address));
        binds.push_back(vkx::make_bindings(dev, {TEXTURE}));
        vkx::bind_image(dev, binds.back(), 0, TEXTURE, r.view,
                        VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL, samplers.back());
    }
    VkPipelineLayout layout = vkx::make_layout(dev, {binds[0].layout});
    auto vs = vkx::make_shader(dev, vkx::compile(VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipeline pipeline = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = fs, .layout = layout,
              .color = target.format});

    VkCommandBuffer cmd = vkx::begin(cmds);
    vkx::begin_drawing(cmd, target, {{-1.0f}});
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
    for (uint32_t y = 0; y < H; ++y) {      // 줄 y에는 y번 세트로 그린다
        VkViewport line{0.0f, float(y), float(W), 1.0f, 0.0f, 1.0f};
        vkCmdSetViewport(cmd, 0, 1, &line);
        vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, layout, 0, 1,
                                &binds[y].set, 0, nullptr);
        vkCmdDraw(cmd, 3, 1, 0, 0);
    }
    vkx::end_drawing(cmd, target, pixels);
    vkx::submit_and_wait(dev, cmds);

    auto* v = static_cast<const float*>(pixels.mapped);
    const char* ramp = ".:-=+*#%@";
    std::printf("%-25s|-1%14s0%15s1%15s2\n", "u", "", "", "");
    for (uint32_t y = 0; y < H; ++y) {
        std::printf("%-25s|", rows[y].label);
        for (uint32_t x = 0; x < W; ++x)
            std::putchar(ramp[int(v[y * W + x] * 8.999f)]);
        std::puts("|");
    }

    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    for (vkx::Bindings& b : binds) vkx::destroy(dev, b);
    for (VkSampler s : samplers) vkDestroySampler(dev.handle, s, nullptr);
    vkDestroyImageView(dev.handle, swapped, nullptr);
    for (vkx::Image* i : {&tex, &target}) vkx::destroy(dev, *i);
    vkx::destroy(dev, pixels);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
u                        |-1              0               1               2
NEAREST REPEAT           |::::====****%%%%::::====****%%%%::::====****%%%%|
NEAREST MIRRORED_REPEAT  |%%%%****====::::::::====****%%%%%%%%****====::::|
NEAREST CLAMP_TO_EDGE    |::::::::::::::::::::====****%%%%%%%%%%%%%%%%%%%%|
NEAREST CLAMP_TO_BORDER  |................::::====****%%%%................|
LINEAR  REPEAT           |=-:--==++**##%#*=-:--==++**##%#*=-:--==++**##%#*|
LINEAR  CLAMP_TO_EDGE    |:::::::::::::::::::--==++**##%%%%%%%%%%%%%%%%%%%|
NEAREST REPEAT, view R=G |%%%%****====::::%%%%****====::::%%%%****====::::|
```

- **좌표**: `texture`의 좌표는 텍스처 전체가 0에서 1입니다. 텍셀 네 개가 0~1을 4등분하므로 `NEAREST`에서는 텍셀마다 네 글자씩 나왔습니다. 텍셀의 가운데는 0.125, 0.375처럼 칸의 중앙이고, 정수 좌표로 텍셀 하나를 그대로 읽을 때는 `texelFetch`를 씁니다
- **주소 모드**: 0~1 밖의 좌표를 어떻게 다룰지입니다. `REPEAT`는 되풀이하고, `MIRRORED_REPEAT`는 뒤집어 가며 되풀이하고, `CLAMP_TO_EDGE`는 끝 텍셀을 늘이고, `CLAMP_TO_BORDER`는 테두리 색(기본은 투명한 검정)을 돌려줍니다
- **필터**: `LINEAR`는 좌표를 둘러싼 텍셀들을 거리에 따라 섞습니다. `REPEAT`에서는 끝 텍셀과 첫 텍셀도 섞이므로(되풀이 경계의 `#*=-`), 타일처럼 이어 붙이지 않는 텍스처에 쓰면 가장자리에 반대쪽 색이 배어 나옵니다. 그럴 때는 `CLAMP_TO_EDGE`를 씁니다
- **뷰**: 마지막 줄은 같은 이미지, 같은 셰이더인데 뷰의 `components`에서 R 자리에 G를 내주게 했습니다. 뷰는 밉과 층의 범위, 종류(2D, 배열, 큐브), 성분의 순서를 정하는 창이어서, 이미지 하나에 뷰를 여럿 둘 수 있습니다. 큐브 맵은 층 6개짜리 이미지를 `VK_IMAGE_VIEW_TYPE_CUBE`로 본 것입니다
- **샘플러는 따로**: 샘플러는 이미지와 무관한 객체라 몇 개만 만들어 모든 텍스처에 돌려 씁니다. 동시에 만들 수 있는 수(`maxSamplerAllocationCount`)는 4000개만 보장됩니다. 비등방 필터(`anisotropyEnable`)는 `samplerAnisotropy` 기능이 있어야 하고, 밉을 고르는 설정은 [「밉맵」](#/read/cpp-vulkan/tex-mipmaps)에서 봅니다

<Note>
색을 담은 텍스처는 `VK_FORMAT_R8G8B8A8_SRGB`처럼 SRGB 포맷으로 만들어야 합니다. 그래야 읽을 때 GPU가 선형 값으로 바꿔 주고, 필터도 선형 값끼리 섞습니다. 법선이나 거칠기처럼 색이 아닌 데이터는 UNORM으로 둡니다. 둘을 뒤바꾸면 그림이 전체적으로 어둡거나 뿌옇게 나옵니다. 어떤 포맷을 샘플링하고 선형 필터로 읽을 수 있는지는 `vkGetPhysicalDeviceFormatProperties`의 `optimalTilingFeatures`로 확인합니다.
</Note>

## 정리

- 이미지는 텍셀을 담고, 뷰는 그 일부를 보는 창이고, 샘플러는 좌표로 읽는 방법이며, 셋은 따로 만들어 조합한다
- 텍스처는 스테이징 버퍼에서 `vkCmdCopyBufferToImage`로 올리고, 읽기 전에 SHADER_READ_ONLY 레이아웃으로 바꾼다
- 주소 모드는 0~1 밖의 좌표를, 필터는 텍셀 사이의 값을 정하고, 색 텍스처는 SRGB 포맷으로 만든다
