export const terms = [
  'rayQueryEXT', 'rayQueryInitializeEXT', 'rayQueryProceedEXT',

레이 쿼리는 광선 쏘기를 셰이더 안의 함수 호출로 만든 것입니다. 파이프라인의 종류를 바꾸지 않고 컴퓨트 셰이더나 프래그먼트 셰이더가 가속 구조에 직접 묻고, 답을 그 자리에서 받습니다. 히트 셰이더도 바인딩 테이블도 없습니다. 아래 프로그램은 앞 쪽의 장면([「레이 트레이싱 파이프라인」](#/read/cpp-vulkan/rt-pipeline))에서 가운데 세모만 불투명하지 않게 바꿔 두 가지로 그립니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_compute.hpp,vkx_pipeline.hpp,vkx_draw.hpp,vkx_accel.hpp,vkx_scene.hpp
#include <string>
#include "vkx_compute.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"
#include "vkx_scene.hpp"

// 두 셰이더가 함께 쓰는 앞부분: 가장 가까운 교차의 인스턴스 번호와 거리를 찾는다
const std::string QUERY = R"(#version 460
#extension GL_EXT_ray_query : require
layout(set = 0, binding = 0) uniform accelerationStructureEXT scene;
const vec3 FORWARD = vec3(0, 0, 1), LIGHT = normalize(vec3(-0.6, -0.2, -1.0));

bool closest(vec3 origin, vec3 direction, float tmin, uint flags,
             out uint index, out float t) {
    rayQueryEXT q;
    rayQueryInitializeEXT(q, scene, flags, 0xff, origin, tmin, direction, 100.0);
    while (rayQueryProceedEXT(q)) {   // 불투명하지 않은 후보를 만날 때마다 멈춘다
        if (rayQueryGetIntersectionTypeEXT(q, false) !=
            gl_RayQueryCandidateIntersectionTriangleEXT) continue;
        vec2 b = rayQueryGetIntersectionBarycentricsEXT(q, false);
        if (min(min(b.x, b.y), 1.0 - b.x - b.y) < 0.2)   // 테두리만 맞은 것으로
            rayQueryConfirmIntersectionEXT(q);
    }
    if (rayQueryGetIntersectionTypeEXT(q, true) ==
        gl_RayQueryCommittedIntersectionNoneEXT) return false;
    index = rayQueryGetIntersectionInstanceCustomIndexEXT(q, true);
    t = rayQueryGetIntersectionTEXT(q, true);
    return true;
}
// 벽 위의 한 점에서 빛 쪽이 막혔으면 4, 빛을 받으면 5
uint shade(vec3 at) {
    uint index;
    float t;
    return closest(at, LIGHT, 0.01, gl_RayFlagsTerminateOnFirstHitEXT, index, t)
               ? 4u : 5u;
}
)";
// 컴퓨트 셰이더: 픽셀마다 광선을 쏘아 장면 전체를 그린다
const std::string COMPUTE = R"(
layout(local_size_x = 8, local_size_y = 4) in;
layout(set = 0, binding = 1, r32ui) uniform writeonly uimage2D target;
void main() {
    ivec2 p = ivec2(gl_GlobalInvocationID.xy), size = imageSize(target);
    if (p.x >= size.x || p.y >= size.y) return;
    vec2 uv = (vec2(p) + 0.5) / vec2(size);
    vec3 origin = vec3(uv.x * 8.0 - 4.0, uv.y * 2.0 - 1.0, -5.0);
    uint index, value = 0u;
    float t;
    if (closest(origin, FORWARD, 0.0, 0u, index, t))
        value = index == 3u ? shade(origin + FORWARD * t) : 1u + index;
    imageStore(target, p, uvec4(value));
})";
// 프래그먼트 셰이더: 벽은 래스터화로 그리고, 그림자만 광선으로 묻는다
const std::string FRAGMENT = R"(
layout(location = 0) out uint color;
void main() {
    vec3 at = vec3(gl_FragCoord.x / 6.0 - 4.0, gl_FragCoord.y / 6.0 - 1.0, 2.0);
    if (abs(at.x) > 3.5 || abs(at.y) > 0.9) discard;   // 벽 바깥
    color = shade(at);
})";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_rt_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    // 가운데 세모만 불투명하지 않게: 그 삼각형과의 교차는 셰이더가 정한다
    vkx::Scene scene = vkx::make_scene(
        dev, cmds, VK_GEOMETRY_INSTANCE_FORCE_NO_OPAQUE_BIT_KHR);
    const VkExtent2D SIZE{48, 12};
    vkx::Image target = vkx::make_image(
        dev, SIZE, VK_FORMAT_R32_UINT,
        VK_IMAGE_USAGE_STORAGE_BIT | VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT |
            VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels = vkx::make_buffer(dev, SIZE.width * SIZE.height * 4,
                                          VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    const auto IMAGE = VK_DESCRIPTOR_TYPE_STORAGE_IMAGE;
    const auto GENERAL = VK_IMAGE_LAYOUT_GENERAL;
    vkx::Bindings set = vkx::make_bindings(
        dev, {VK_DESCRIPTOR_TYPE_ACCELERATION_STRUCTURE_KHR, IMAGE});
    vkx::bind_accel(dev, set, 0, scene.top);
    vkx::bind_image(dev, set, 1, IMAGE, target.view, GENERAL);
    VkPipelineLayout layout = vkx::make_layout(dev, {set.layout});
    auto shader = [&](const std::string& glsl, shaderc_shader_kind kind) {
        return vkx::make_shader(dev, vkx::compile(glsl.c_str(), kind));
    };
    VkShaderModule modules[] = {
        shader(QUERY + COMPUTE, shaderc_compute_shader),
        shader(vkx::FULLSCREEN_VS, shaderc_vertex_shader),
        shader(QUERY + FRAGMENT, shaderc_fragment_shader)};
    VkPipeline trace = vkx::make_compute(dev, modules[0], layout);
    VkPipeline draw = vkx::make_pipeline(
        dev, {.vertex = modules[1], .fragment = modules[2], .layout = layout,
              .color = target.format});

    for (int way = 0; way < 2; ++way) {
        VkCommandBuffer cmd = vkx::begin(cmds);
        const auto POINT = way == 0 ? VK_PIPELINE_BIND_POINT_COMPUTE
                                    : VK_PIPELINE_BIND_POINT_GRAPHICS;
        vkCmdBindDescriptorSets(cmd, POINT, layout, 0, 1, &set.set, 0, nullptr);
        if (way == 0) {   // 컴퓨트: 스토리지 이미지에 쓰고 복사해 온다
            const auto STAGE = VK_PIPELINE_STAGE_2_COMPUTE_SHADER_BIT;
            const auto WRITE = VK_ACCESS_2_SHADER_STORAGE_WRITE_BIT;
            vkx::transition(cmd, target, VK_IMAGE_LAYOUT_UNDEFINED, GENERAL,
                            VK_PIPELINE_STAGE_2_NONE, VK_ACCESS_2_NONE, STAGE,
                            WRITE);
            vkCmdBindPipeline(cmd, POINT, trace);
            vkCmdDispatch(cmd, (SIZE.width + 7) / 8, (SIZE.height + 3) / 4, 1);
            vkx::transition(cmd, target, GENERAL,
                            VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL, STAGE, WRITE,
                            VK_PIPELINE_STAGE_2_COPY_BIT,
                            VK_ACCESS_2_TRANSFER_READ_BIT);
            vkx::copy_to_buffer(cmd, target, pixels);
        } else {          // 래스터화: 화면을 덮는 삼각형 하나를 그린다
            vkx::begin_drawing(cmd, target);
            vkCmdBindPipeline(cmd, POINT, draw);
            vkCmdDraw(cmd, 3, 1, 0, 0);
            vkx::end_drawing(cmd, target, pixels);
        }
        vkx::submit_and_wait(dev, cmds);

        std::puts(way == 0 ? "컴퓨트 셰이더에서: 장면 전체"
                           : "프래그먼트 셰이더에서: 벽과 그림자만");
        const auto* px = static_cast<const uint32_t*>(pixels.mapped);
        for (uint32_t y = 0; y < SIZE.height; ++y) {
            std::string row;   // 0 빈 곳, 1~3 세모, 4 그늘진 벽, 5 빛 받는 벽
            for (uint32_t x = 0; x < SIZE.width; ++x)
                row += ".123#-"[px[y * SIZE.width + x]];
            std::puts(row.c_str());
        }
    }

    vkDestroyPipeline(dev.handle, trace, nullptr);
    vkDestroyPipeline(dev.handle, draw, nullptr);
    for (VkShaderModule m : modules) vkDestroyShaderModule(dev.handle, m, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkx::destroy(dev, set);
    vkx::destroy(dev, pixels);
    vkx::destroy(dev, target);
    vkx::destroy(dev, scene);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **쿼리 객체**: `rayQueryInitializeEXT`로 광선을 정하고 `rayQueryProceedEXT`가 거짓을 돌려줄 때까지 돌린 뒤, 확정된 교차가 있는지와 그 인스턴스 번호, 거리를 물어봅니다. 인자는 `traceRayEXT`와 닮았지만 셰이더를 고르던 것들(표의 오프셋, 미스 번호, 페이로드)이 없습니다
- **후보와 확정**: 불투명한 삼각형과의 교차는 구현이 바로 확정하므로 반복문의 몸통이 한 번도 돌지 않습니다. 불투명하지 않은 기하에서는 후보마다 멈추고, 셰이더가 `rayQueryConfirmIntersectionEXT`를 불러야만 맞은 것이 됩니다. 여기서는 무게중심 좌표로 테두리만 받아들여 세모 가운데를 뚫었습니다. 그림자 광선도 같은 함수를 지나므로 그림자에도 구멍이 났습니다. 파이프라인에서는 any-hit 셰이더가 하던 일입니다. 삼각형이 아닌 상자 기하의 후보에는 `rayQueryGenerateIntersectionEXT`로 거리를 알려 줍니다
- **앞 쪽과 같은 그림**: 가운데 세모를 불투명하게 되돌리면, 컴퓨트 쪽 그림은 레이 트레이싱 파이프라인이 그린 것과 글자 하나 다르지 않습니다
- **프래그먼트 셰이더에서**: 벽은 화면을 덮는 삼각형을 래스터화해 그리고, 프래그먼트마다 그림자 광선 하나만 쏘았습니다. 세모는 그리지도 않았는데 그림자가 생겼습니다. 실제로는 G-버퍼에서 위치를 얻어 이렇게 묻습니다([「지연 렌더링」](#/read/cpp-vulkan/tech-deferred))
- **그림자 맵과 견주면**: 빛에서 본 깊이 맵이 없으니 그 해상도에서 오는 계단도 없고, 바이어스는 시작점을 조금 띄우는 것으로 끝납니다([「그림자」](#/read/cpp-vulkan/tech-shadows)). 대신 가속 구조를 짓고 물체가 움직일 때마다 고쳐야 하며, 픽셀마다 탐색 비용을 냅니다

## 어느 쪽을 쓸까

레이 트레이싱 파이프라인은 물체마다 다른 셰이더를 표에서 골라 부르고 그 셰이더가 광선을 다시 쏠 수 있어서, 반사처럼 맞은 곳의 재질을 계산해야 하는 일에 맞습니다. 레이 쿼리는 다른 셰이더를 부르지 못하고 결과만 돌려주므로, 그림자나 가려짐처럼 맞았는지와 거리만 알면 되는 일을 기존 셰이더에 끼워 넣기에 좋습니다. 두 기능은 지원 여부를 따로 알리므로(`rayQuery`, `rayTracingPipeline`) 장치에 묻고 고릅니다. Arm은 자사 문서를 쓸 당시의 Mali 기기가 레이 쿼리만 지원한다고 적었습니다.

플래그와 마스크로 탐색을 줄입니다. 무엇이든 맞기만 하면 되는 그림자 광선에는 `gl_RayFlagsTerminateOnFirstHitEXT`를, 모든 기하를 불투명으로 쳐도 되는 광선에는 `gl_RayFlagsOpaqueEXT`를 줍니다. 인스턴스의 8비트 마스크와 광선의 마스크가 한 비트도 겹치지 않으면 그 인스턴스는 건너뛰므로, 그림자를 드리우지 않을 물체를 비트로 가릅니다.

<Note>
장면이 모두 불투명하다고 보고 `rayQueryProceedEXT`를 반복문 없이 한 번만 부르는 코드가 흔합니다. 불투명하지 않은 기하가 하나라도 섞이면 탐색이 첫 후보에서 멈춘 채로 끝나서, 이 예제에서는 가운데 세모와 그 뒤의 벽이 함께 사라졌습니다. 그렇게 쓸 광선에는 `gl_RayFlagsOpaqueEXT`를 주어 후보에서 멈추지 않게 합니다.
</Note>

## 정리

- 레이 쿼리는 어느 셰이더 단계에서나 가속 구조에 직접 광선을 쏘고, 결과를 그 셰이더가 바로 받는다
- 불투명한 기하는 저절로 확정되고, 불투명하지 않은 기하의 후보는 셰이더가 확정해야 맞은 것이 된다
- 재질마다 셰이더를 고르고 다시 쏘는 일에는 레이 트레이싱 파이프라인이, 가려짐만 묻는 일에는 레이 쿼리가 맞는다
