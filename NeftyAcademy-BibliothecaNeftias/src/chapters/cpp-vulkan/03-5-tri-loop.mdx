화면에 그리는 프로그램은 매 프레임 같은 일을 되풀이합니다. 스왑체인에서 이미지를 받고, 그 이미지에 그리는 명령을 기록해 내고, 다 그려지면 화면으로 내보냅니다. 이 쪽은 지금까지의 조각을 모아 그 고리를 만듭니다. CPU가 GPU를 매번 기다리지 않도록 프레임 두 개를 겹쳐 진행하고, 창 크기가 바뀌면 스왑체인을 다시 만듭니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_pipeline.hpp,vkx_resource.hpp,vkx_swapchain.hpp
#include "vkx_cmd.hpp"
#include "vkx_pipeline.hpp"
#include "vkx_resource.hpp"
#include "vkx_swapchain.hpp"

const char* VS = R"(#version 450
layout(location = 0) out float shade;
const vec2 pos[3] = vec2[](vec2(0.0, -0.9), vec2(0.9, 0.8), vec2(-0.9, 0.8));
const float tone[3] = float[](0.0, 1.0, 0.5);
void main() {
    gl_Position = vec4(pos[gl_VertexIndex], 0.0, 1.0);
    shade = tone[gl_VertexIndex];
})";

const char* FS = R"(#version 450
layout(location = 0) in float shade;
layout(location = 0) out vec4 color;
void main() { color = vec4(vec3(shade), 1.0); })";

constexpr int IN_FLIGHT = 2;   // CPU가 GPU보다 앞서 기록해 둘 수 있는 프레임 수

struct Frame {                 // 겹쳐 그리는 프레임마다 따로 갖는 것
    vkx::Commands cmds;        // 커맨드 버퍼와, 그 GPU 일이 끝났음을 알릴 펜스
    VkSemaphore acquired;      // 스왑체인 이미지를 받았다는 신호
};

VkSemaphore make_semaphore(const vkx::Device& dev) {
    VkSemaphoreCreateInfo sci{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_CREATE_INFO};
    VkSemaphore s;
    vkCreateSemaphore(dev.handle, &sci, nullptr, &s);
    return s;
}

int main() {
    vkx::Instance inst = vkx::make_instance(
        {VK_KHR_SURFACE_EXTENSION_NAME, VK_EXT_HEADLESS_SURFACE_EXTENSION_NAME});
    VkSurfaceKHR surface = vkx::make_headless_surface(inst.handle);
    vkx::Device dev =
        vkx::make_device(inst.handle, {VK_KHR_SWAPCHAIN_EXTENSION_NAME});
    vkx::Swapchain sc{};   // 처음에는 비어 있다
    // 그리기가 끝났다는 신호는 이미지마다 둔다(내보내기가 언제 끝날지 모르므로)
    std::vector<VkSemaphore> rendered;
    auto recreate = [&](VkExtent2D size) {   // 처음, 그리고 창 크기가 바뀔 때마다
        vkDeviceWaitIdle(dev.handle);
        vkx::Swapchain next = vkx::make_swapchain(
            dev, surface, size, VK_PRESENT_MODE_FIFO_KHR, sc.handle);
        vkx::destroy(dev, sc);
        sc = next;
        while (rendered.size() < sc.images.size())
            rendered.push_back(make_semaphore(dev));
        std::printf("스왑체인 %u×%u, 이미지 %zu장\n", sc.extent.width,
                    sc.extent.height, sc.images.size());
    };
    recreate({40, 20});
    auto vs = vkx::make_shader(dev, vkx::compile(VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipelineLayoutCreateInfo lci{
        .sType = VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};
    VkPipelineLayout layout;
    vkCreatePipelineLayout(dev.handle, &lci, nullptr, &layout);
    VkPipeline pipeline = vkx::make_pipeline(dev, {.vertex = vs, .fragment = fs,
                                                   .layout = layout,
                                                   .color = sc.format.format});
    Frame frames[IN_FLIGHT];
    for (Frame& f : frames) f = {vkx::make_commands(dev), make_semaphore(dev)};

    const auto OUT = VK_PIPELINE_STAGE_2_COLOR_ATTACHMENT_OUTPUT_BIT;
    const auto WRITE = VK_ACCESS_2_COLOR_ATTACHMENT_WRITE_BIT;
    const auto ATTACHMENT = VK_IMAGE_LAYOUT_COLOR_ATTACHMENT_OPTIMAL;
    for (int n = 0; n < 6; ++n) {
        if (n == 3) recreate({64, 32});   // 창 크기가 바뀌었다고 치자
        Frame& f = frames[n % IN_FLIGHT];
        // 이 칸을 앞서 쓴 프레임(n - 2)의 GPU 일이 끝나야 다시 기록한다
        if (n >= IN_FLIGHT)
            vkWaitForFences(dev.handle, 1, &f.cmds.done, VK_TRUE, UINT64_MAX);
        uint32_t index = 0;
        VkResult r = vkAcquireNextImageKHR(dev.handle, sc.handle, UINT64_MAX,
                                           f.acquired, VK_NULL_HANDLE, &index);
        if (r == VK_ERROR_OUT_OF_DATE_KHR) {   // 이 스왑체인은 더 못 쓴다
            recreate(sc.extent);
            continue;
        }
        vkResetFences(dev.handle, 1, &f.cmds.done);

        VkCommandBuffer cmd = vkx::begin(f.cmds);
        vkx::Image image{.handle = sc.images[index],
                         .aspect = VK_IMAGE_ASPECT_COLOR_BIT, .extent = sc.extent};
        // 받기 신호를 기다리는 단계(OUT)에 이어서 레이아웃을 바꾼다
        vkx::transition(cmd, image, VK_IMAGE_LAYOUT_UNDEFINED, ATTACHMENT, OUT,
                        VK_ACCESS_2_NONE, OUT, WRITE);
        VkRenderingAttachmentInfo color{
            .sType = VK_STRUCTURE_TYPE_RENDERING_ATTACHMENT_INFO,
            .imageView = sc.views[index], .imageLayout = ATTACHMENT,
            .loadOp = VK_ATTACHMENT_LOAD_OP_CLEAR,
            .storeOp = VK_ATTACHMENT_STORE_OP_STORE};
        VkRenderingInfo info{.sType = VK_STRUCTURE_TYPE_RENDERING_INFO,
                             .renderArea = {{0, 0}, sc.extent}, .layerCount = 1,
                             .colorAttachmentCount = 1,
                             .pColorAttachments = &color};
        vkCmdBeginRendering(cmd, &info);
        vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
        VkViewport viewport{0.0f, 0.0f, float(sc.extent.width),
                            float(sc.extent.height), 0.0f, 1.0f};
        VkRect2D scissor{{0, 0}, sc.extent};
        vkCmdSetViewport(cmd, 0, 1, &viewport);
        vkCmdSetScissor(cmd, 0, 1, &scissor);
        vkCmdDraw(cmd, 3, 1, 0, 0);
        vkCmdEndRendering(cmd);
        vkx::transition(cmd, image, ATTACHMENT, VK_IMAGE_LAYOUT_PRESENT_SRC_KHR,
                        OUT, WRITE, OUT, VK_ACCESS_2_NONE);
        vkEndCommandBuffer(cmd);

        // 받기 신호를 OUT 단계에서 기다리고, 끝나면 이 이미지의 그리기 신호를 낸다
        VkSemaphoreSubmitInfo wait{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_SUBMIT_INFO,
                                   .semaphore = f.acquired, .stageMask = OUT};
        VkSemaphoreSubmitInfo signal = wait;
        signal.semaphore = rendered[index];
        VkCommandBufferSubmitInfo one{
            .sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_SUBMIT_INFO,
            .commandBuffer = cmd};
        VkSubmitInfo2 si{.sType = VK_STRUCTURE_TYPE_SUBMIT_INFO_2,
                         .waitSemaphoreInfoCount = 1, .pWaitSemaphoreInfos = &wait,
                         .commandBufferInfoCount = 1, .pCommandBufferInfos = &one,
                         .signalSemaphoreInfoCount = 1,
                         .pSignalSemaphoreInfos = &signal};
        vkQueueSubmit2(dev.queue, 1, &si, f.cmds.done);
        VkPresentInfoKHR present{.sType = VK_STRUCTURE_TYPE_PRESENT_INFO_KHR,
                                 .waitSemaphoreCount = 1,
                                 .pWaitSemaphores = &rendered[index],
                                 .swapchainCount = 1, .pSwapchains = &sc.handle,
                                 .pImageIndices = &index};
        r = vkQueuePresentKHR(dev.queue, &present);
        std::printf("프레임 %d: 칸 %d, 이미지 %u\n", n, n % IN_FLIGHT, index);
        if (r == VK_ERROR_OUT_OF_DATE_KHR || r == VK_SUBOPTIMAL_KHR)
            recreate(sc.extent);
    }
    vkDeviceWaitIdle(dev.handle);   // 끝내기 전에 모든 일을 기다린다
    for (VkSemaphore s : rendered) vkDestroySemaphore(dev.handle, s, nullptr);
    for (Frame& f : frames) {
        vkDestroySemaphore(dev.handle, f.acquired, nullptr);
        vkx::destroy(dev, f.cmds);
    }
    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    vkx::destroy(dev, sc);
    vkx::destroy(dev);
    vkDestroySurfaceKHR(inst.handle, surface, nullptr);
    vkx::destroy(inst);
}
```

프레임 0: 칸 0, 이미지 0

프레임 2: 칸 0, 이미지 0

프레임 3: 칸 1, 이미지 0

프레임 5: 칸 1, 이미지 0

- **겹쳐 그리는 프레임**: 칸마다 커맨드 버퍼, 펜스, 받기 세마포어를 따로 둡니다. GPU가 프레임 n을 그리는 동안 CPU는 프레임 n+1을 기록하고, 같은 칸을 다시 쓰는 프레임 n+2에서야 펜스를 기다립니다. 칸이 많으면 끊김에 강해지지만 입력에서 화면까지의 지연이 늘어서 보통 두 개를 씁니다
- **세마포어 둘**: 받기 세마포어는 이미지가 비었다는 신호이고, 제출은 이를 색 출력 단계(OUT)에서 기다리므로 정점 처리는 먼저 시작할 수 있습니다. 그리기 세마포어는 내보내기가 기다리는 신호입니다
- **그리기 세마포어를 이미지마다 두는 까닭**: 내보내기가 세마포어 기다림을 언제 마치는지 알 길이 없어서, 그 이미지를 다시 받았을 때 그 이미지의 세마포어를 다시 씁니다. 2025년에 나온 VK_KHR_swapchain_maintenance1은 내보내기 펜스로 이를 알려 줍니다
- **레이아웃**: 받은 이미지는 UNDEFINED에서 첨부물로, 그린 뒤에는 PRESENT_SRC로 바꿉니다. 첫 배리어의 앞 단계를 OUT으로 두어 받기 세마포어의 기다림에 이어지게 했습니다
- **이미지 번호**: 받는 이미지의 순서는 정해져 있지 않습니다. lavapipe의 헤드리스 표면은 내보내자마자 이미지를 돌려줘서 늘 0번이 나오지만, 실제 창에서는 여러 번호가 돕니다. 그래서 칸 번호와 이미지 번호를 따로 다룹니다

## 다시 만들기

창 크기가 바뀌거나 표면이 달라지면 받기나 내보내기가 `VK_ERROR_OUT_OF_DATE_KHR`(더는 못 씀)나 `VK_SUBOPTIMAL_KHR`(쓸 수는 있지만 맞지 않음)를 돌려줍니다. 그러면 GPU를 기다린 뒤 옛 스왑체인을 `oldSwapchain`으로 넘겨 새로 만듭니다. 헤드리스 표면은 이런 값을 내지 않으므로, 프로그램은 3번 프레임에서 크기가 바뀌었다고 치고 다시 만들었습니다. 크기만 바뀌면 뷰포트가 동적 상태라 파이프라인을 그대로 쓰고, 포맷까지 바뀌면 파이프라인도 다시 만듭니다. 창을 최소화해 크기가 0이 되면 스왑체인을 만들 수 없으므로 크기가 돌아올 때까지 기다립니다.

<Note>
겹쳐 그리는 프레임 수(CPU가 앞서 나가는 정도)와 스왑체인 이미지 수(화면 쪽에서 돌리는 버퍼)는 서로 다른 수입니다. 이 프로그램은 칸 두 개와 이미지 다섯 장을 썼습니다. 프레임을 여럿 겹칠 때 생기는 리소스 문제는 [「여러 프레임을 겹쳐 진행하기」](#/read/cpp-vulkan/vsync-frames)에서 봅니다.
</Note>

## 정리

- 프레임 루프는 이미지 받기 → 기록 → 제출 → 내보내기를 되풀이하고, 칸마다 커맨드 버퍼·펜스·받기 세마포어를 두어 CPU와 GPU가 겹쳐 일한다
- 받기 세마포어는 색 출력 단계에서 기다리고, 그리기 세마포어는 이미지마다 두며, 이미지는 첨부물을 거쳐 PRESENT_SRC로 바꿔 내보낸다
- OUT_OF_DATE·SUBOPTIMAL이나 크기 변화가 오면 GPU를 기다린 뒤 옛 스왑체인을 넘겨 다시 만든다
