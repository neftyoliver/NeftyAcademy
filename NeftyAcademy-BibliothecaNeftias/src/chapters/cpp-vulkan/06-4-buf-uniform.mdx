export const summary = '유니폼 버퍼: 매핑해 두고 매 프레임 쓰기, 프레임마다 칸 나누기, 오프셋 정렬, 동적 오프셋';

**유니폼 버퍼**는 셰이더의 `uniform` 블록에 값을 대는 버퍼입니다([「인터페이스 블록과 레이아웃」](#/read/cpp-vulkan/glsl-interface)). 카메라 행렬처럼 프레임마다 바뀌는 값과 물체마다 다른 값을 담으며, CPU가 매 프레임 새로 쓰므로 CPU에 보이는 메모리에 만들어 매핑해 둡니다. 아래 프로그램은 버퍼 하나를 프레임 칸 둘로 나누고, 칸마다 프레임 블록 하나와 물체 블록 셋을 둡니다. 두 프레임을 한꺼번에 기록해 제출하는데(왼쪽 절반이 프레임 0, 오른쪽이 프레임 1), 처음에는 두 프레임이 칸 하나를 같이 쓰고 다음에는 따로 씁니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_pipeline.hpp,vkx_draw.hpp
#include <cstring>
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"

const char* VS = R"(#version 450
layout(binding = 0) uniform Frame { float lift; };               // 프레임마다 하나
layout(binding = 1) uniform Object { vec2 center; float shade; };   // 물체마다 하나
layout(location = 0) out float v_shade;
const vec2 corner[6] = vec2[](vec2(-1, -1), vec2(1, -1), vec2(-1, 1),
                              vec2(-1, 1), vec2(1, -1), vec2(1, 1));
void main() {
    vec2 p = center + corner[gl_VertexIndex] * vec2(0.2, 0.25);
    gl_Position = vec4(p.x, p.y - lift, 0.0, 1.0);    // lift만큼 위로
    v_shade = shade;
})";

const char* FS = R"(#version 450
layout(location = 0) in float v_shade;
layout(location = 0) out float value;
void main() { value = v_shade; })";

struct Frame { float lift; };              // std140으로 4바이트
struct Object { float x, y, shade; };      // vec2(0), float(8): 12바이트

VkDeviceSize align_up(VkDeviceSize n, VkDeviceSize a) {
    return (n + a - 1) / a * a;
}

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const uint32_t W = 48, H = 8, OBJECTS = 3, SLOTS = 2;
    vkx::Image target = vkx::make_image(
        dev, {W, H}, VK_FORMAT_R32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels =
        vkx::make_buffer(dev, W * H * 4, VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);

    // 버퍼 하나를 칸(프레임) 둘로 나누고, 칸마다 Frame 하나와 Object 셋을 둔다.
    // 블록이 시작하는 자리는 장치가 알려 주는 정렬의 배수여야 한다
    const VkDeviceSize A = dev.props.limits.minUniformBufferOffsetAlignment;
    const VkDeviceSize FRAME = align_up(sizeof(Frame), A);
    const VkDeviceSize OBJECT = align_up(sizeof(Object), A);
    const VkDeviceSize SLOT = FRAME + OBJECTS * OBJECT;
    vkx::Buffer ubo = vkx::make_buffer(dev, SLOTS * SLOT,
                                       VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT, true);
    std::printf("정렬 %u바이트: Frame %u, Object %u, 칸 하나 %u, 버퍼 %u바이트\n",
                unsigned(A), unsigned(FRAME), unsigned(OBJECT), unsigned(SLOT),
                unsigned(SLOTS * SLOT));

    // 디스크립터는 버퍼 맨 앞에서 블록 크기만큼만 가리키고, 실제 자리는
    // 세트를 붙일 때 주는 동적 오프셋으로 정한다
    const auto DYNAMIC = VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER_DYNAMIC;
    vkx::Bindings binds = vkx::make_bindings(dev, {DYNAMIC, DYNAMIC});
    VkDescriptorBufferInfo ranges[2] = {{ubo.handle, 0, sizeof(Frame)},
                                        {ubo.handle, 0, sizeof(Object)}};
    for (uint32_t b = 0; b < 2; ++b) {
        VkWriteDescriptorSet w{.sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET,
                               .dstSet = binds.set, .dstBinding = b,
                               .descriptorCount = 1, .descriptorType = DYNAMIC,
                               .pBufferInfo = &ranges[b]};
        vkUpdateDescriptorSets(dev.handle, 1, &w, 0, nullptr);
    }
    VkPipelineLayout layout = vkx::make_layout(dev, {binds.layout});
    auto vs = vkx::make_shader(dev, vkx::compile(VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipeline pipeline = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = fs, .layout = layout,
              .color = target.format});

    for (bool separate : {false, true}) {
        VkCommandBuffer cmd = vkx::begin(cmds);
        vkx::begin_drawing(cmd, target, {{-1.0f}});
        vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
        for (uint32_t f = 0; f < 2; ++f) {     // 프레임 0은 왼쪽, 1은 오른쪽 절반
            const VkDeviceSize base = (separate ? f : 0) * SLOT;
            char* p = static_cast<char*>(ubo.mapped) + base;
            Frame frame{.lift = float(f)};              // 프레임 1은 위로 올린다
            std::memcpy(p, &frame, sizeof frame);
            for (uint32_t i = 0; i < OBJECTS; ++i) {
                Object o{-0.6f + 0.6f * float(i), 0.5f, 0.3f + 0.3f * float(i)};
                std::memcpy(p + FRAME + i * OBJECT, &o, sizeof o);
            }
            VkViewport half{float(f * W / 2), 0.0f, float(W / 2), float(H),
                            0.0f, 1.0f};
            vkCmdSetViewport(cmd, 0, 1, &half);
            for (uint32_t i = 0; i < OBJECTS; ++i) {
                uint32_t offsets[2] = {uint32_t(base),
                                       uint32_t(base + FRAME + i * OBJECT)};
                vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS,
                                        layout, 0, 1, &binds.set, 2, offsets);
                vkCmdDraw(cmd, 6, 1, 0, 0);
            }
        }
        vkx::end_drawing(cmd, target, pixels);
        vkx::submit_and_wait(dev, cmds);

        std::puts(separate ? "프레임마다 칸을 따로 씀:"
                           : "두 프레임이 칸 하나를 같이 씀:");
        auto* v = static_cast<const float*>(pixels.mapped);
        const char* ramp = ".:-=+*#%@";
        for (uint32_t i = 0; i < W * H; ++i) {
            if (i % (W / 2) == 0) std::putchar('|');
            std::putchar(v[i] < 0 ? ' ' : ramp[int(v[i] * 8.999f)]);
            if (i % W == W - 1) std::puts("|");
        }
    }
    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    vkx::destroy(dev, binds);
    for (vkx::Buffer* b : {&ubo, &pixels}) vkx::destroy(dev, *b);
    vkx::destroy(dev, target);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **칸을 같이 쓰면**: 기록된 명령은 실행될 때 버퍼를 읽습니다. 프레임 0을 기록한 뒤 같은 자리에 프레임 1의 값(위로 올리기)을 덮어썼으므로 두 프레임 모두 올라간 그림이 됐습니다. 실제 프레임 루프에서는 GPU가 앞 프레임을 그리는 동안 CPU가 다음 프레임의 값을 쓰게 되어, 한 프레임 안에 옛 값과 새 값이 섞일 수도 있습니다. 그래서 겹쳐 진행하는 프레임 수만큼 칸을 둡니다([「프레임 루프」](#/read/cpp-vulkan/tri-loop))
- **정렬**: 블록이 시작하는 오프셋은 `minUniformBufferOffsetAlignment`의 배수여야 합니다. lavapipe는 16바이트라 4바이트짜리 Frame과 12바이트짜리 Object가 16바이트씩 차지했습니다. 이 값은 장치마다 다르고 명세가 허용하는 가장 큰 값은 256이므로, 오프셋을 상수로 적지 않고 물어서 계산합니다
- **동적 오프셋**: `VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER_DYNAMIC` 디스크립터에는 버퍼와 범위(블록 크기)만 적고, 시작 위치는 `vkCmdBindDescriptorSets`의 마지막 인자로 바인딩 번호 순서대로 줍니다. 그래서 디스크립터 세트 하나로 프레임 둘의 물체 여섯을 모두 그렸습니다. 동적 유니폼 버퍼는 파이프라인 레이아웃을 통틀어 8개까지만 보장됩니다(`maxDescriptorSetUniformBuffersDynamic`)
- **범위의 한도**: 디스크립터 하나가 가리키는 범위는 `maxUniformBufferRange`(보장되는 값은 16384바이트, 1.4부터 65536바이트)를 넘을 수 없습니다. 버퍼 자체는 그보다 커도 됩니다

## 무엇을 어디에 담을까

| 바뀌는 단위 | 예 | 담는 곳 |
| --- | --- | --- |
| 프레임 | 카메라 행렬, 시간, 조명 | 프레임 칸마다 유니폼 블록 하나 |
| 물체 | 모델 행렬, 색 | 동적 오프셋 블록이나 SSBO 배열의 한 칸 |
| 그리기 한 번 | 번호나 플래그 몇 개 | 푸시 상수 |

물체가 수천 개가 되면 블록마다 생기는 정렬 낭비와 세트를 붙이는 호출이 부담이 됩니다. 그래서 물체 데이터를 SSBO 배열 하나에 담고 인스턴스 번호로 고르는 방식을 많이 씁니다([「인스턴싱과 간접 그리기」](#/read/cpp-vulkan/tech-instancing)). 그리기마다 바뀌는 아주 작은 값은 푸시 상수가 맞습니다([「푸시 상수」](#/read/cpp-vulkan/desc-push)).

<Note>
매핑된 유니폼 버퍼에는 쓰기만 합니다. CPU에 보이는 GPU 메모리는 캐시를 거치지 않는 write-combined 메모리일 수 있어서 읽기가 매우 느리고, `p->count += 1`처럼 읽고 고쳐 쓰는 식도 읽기에 들어갑니다. 이 프로그램의 `frame`과 `o`처럼 값을 지역 변수로 만든 뒤 `memcpy`로 한 번에 옮깁니다.
</Note>

## 정리

- 유니폼 버퍼는 CPU에 보이는 메모리에 만들어 매핑해 두고 쓰기만 하며, 겹쳐 진행하는 프레임마다 칸을 따로 둔다
- 블록의 시작 오프셋은 `minUniformBufferOffsetAlignment`의 배수로 맞추고, 동적 오프셋으로 세트 하나에서 블록을 골라 쓴다
- 프레임 단위 값은 유니폼 버퍼, 물체가 많으면 SSBO 배열, 그리기마다 바뀌는 작은 값은 푸시 상수에 담는다
