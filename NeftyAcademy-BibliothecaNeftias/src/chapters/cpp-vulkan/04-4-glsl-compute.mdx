컴퓨트 셰이더는 그림과 상관없이 GPU에 계산을 시킵니다. `vkCmdDispatch(x, y, z)`는 워크그룹을 x×y×z개 띄우고, 워크그룹 하나는 `local_size`로 정한 수만큼의 실행(invocation)으로 이루어집니다. 같은 워크그룹의 실행들은 공유 메모리를 함께 쓰고 `barrier()`로 서로를 기다릴 수 있지만, 다른 워크그룹과는 그럴 수 없습니다. 컴퓨트 파이프라인은 셰이더 하나뿐이라 만드는 도우미도 짧습니다.

```cpp vk file=vkx_compute.hpp with=vkx_instance.hpp,vkx_device.hpp
#pragma once
#include "vkx_device.hpp"

namespace vkx {

// 컴퓨트 파이프라인: 셰이더 하나와 레이아웃뿐이다
inline VkPipeline make_compute(const Device& d, VkShaderModule shader,
                               VkPipelineLayout layout) {
    VkComputePipelineCreateInfo ci{
        .sType = VK_STRUCTURE_TYPE_COMPUTE_PIPELINE_CREATE_INFO,
        .stage = {.sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
                  .stage = VK_SHADER_STAGE_COMPUTE_BIT, .module = shader,
                  .pName = "main"},
        .layout = layout};
    VkPipeline p;
    check(vkCreateComputePipelines(d.handle, VK_NULL_HANDLE, 1, &ci, nullptr, &p),
          "컴퓨트 파이프라인");
    return p;
}

}  // namespace vkx
```

아래 프로그램은 1부터 4096까지의 합을 세 가지 방법으로 구하고, 방법마다 메모리의 한 곳에 `atomicAdd`를 몇 번 했는지 셉니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_compute.hpp,vkx_pipeline.hpp
#include <cstring>
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_compute.hpp"
#include "vkx_pipeline.hpp"

// 1부터 4096까지 더하기를 세 가지로: 모두가 atomicAdd, 워크그룹 공유 메모리로
// 나무 모양 합, 서브그룹 합. 각 방법이 atomicAdd를 몇 번 했는지도 센다
const char* CS = R"(#version 450
#extension GL_KHR_shader_subgroup_arithmetic : require
layout(local_size_x = 64) in;
layout(binding = 0) readonly buffer In { uint v[]; };
layout(binding = 1) buffer Out { uint total[3]; uint atomics[3]; uint subgroup; };
shared uint part[64];

void main() {
    uint x = v[gl_GlobalInvocationID.x], i = gl_LocalInvocationID.x;
    atomicAdd(total[0], x);
    atomicAdd(atomics[0], 1);

    part[i] = x;
    barrier();                        // 그룹의 64개가 모두 여기까지 와야 넘어간다
    for (uint s = 32; s > 0; s >>= 1) {
        if (i < s) part[i] += part[i + s];
        barrier();
    }
    if (i == 0) {
        atomicAdd(total[1], part[0]);
        atomicAdd(atomics[1], 1);
    }

    uint sum = subgroupAdd(x);        // 서브그룹 안의 x를 한 번에 더한다
    if (subgroupElect()) {            // 서브그룹에서 한 명만
        atomicAdd(total[2], sum);
        atomicAdd(atomics[2], 1);
    }
    if (gl_GlobalInvocationID.x == 0) subgroup = gl_SubgroupSize;
})";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const uint32_t N = 4096;
    const auto SSBO = VK_DESCRIPTOR_TYPE_STORAGE_BUFFER;
    const auto USAGE = VK_BUFFER_USAGE_STORAGE_BUFFER_BIT;
    vkx::Buffer in = vkx::make_buffer(dev, N * 4, USAGE, true);
    vkx::Buffer out = vkx::make_buffer(dev, 7 * 4, USAGE, true);
    for (uint32_t k = 0; k < N; ++k) static_cast<uint32_t*>(in.mapped)[k] = k + 1;
    std::memset(out.mapped, 0, 7 * 4);

    vkx::Bindings binds = vkx::make_bindings(dev, {SSBO, SSBO});
    vkx::bind_buffer(dev, binds, 0, SSBO, in);
    vkx::bind_buffer(dev, binds, 1, SSBO, out);
    VkPipelineLayout layout = vkx::make_layout(dev, {binds.layout});
    auto cs = vkx::make_shader(dev, vkx::compile(CS, shaderc_compute_shader));
    VkPipeline pipeline = vkx::make_compute(dev, cs, layout);

    VkCommandBuffer cmd = vkx::begin(cmds);
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_COMPUTE, pipeline);
    vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_COMPUTE, layout, 0, 1,
                            &binds.set, 0, nullptr);
    vkCmdDispatch(cmd, N / 64, 1, 1);   // 워크그룹 64개(각 64개 실행)
    VkMemoryBarrier2 host{.sType = VK_STRUCTURE_TYPE_MEMORY_BARRIER_2,
                          .srcStageMask = VK_PIPELINE_STAGE_2_COMPUTE_SHADER_BIT,
                          .srcAccessMask = VK_ACCESS_2_SHADER_STORAGE_WRITE_BIT,
                          .dstStageMask = VK_PIPELINE_STAGE_2_HOST_BIT,
                          .dstAccessMask = VK_ACCESS_2_HOST_READ_BIT};
    VkDependencyInfo dep{.sType = VK_STRUCTURE_TYPE_DEPENDENCY_INFO,
                         .memoryBarrierCount = 1, .pMemoryBarriers = &host};
    vkCmdPipelineBarrier2(cmd, &dep);
    vkx::submit_and_wait(dev, cmds);

    auto* r = static_cast<const uint32_t*>(out.mapped);
    std::printf("실행 %u개(워크그룹 %u × 64), 서브그룹 크기 %u, 정답 %u\n", N,
                N / 64, r[6], N * (N + 1) / 2);
    const char* names[3] = {"모두 atomicAdd", "공유 메모리 나무", "subgroupAdd"};
    for (int m = 0; m < 3; ++m)
        std::printf("%s: 합 %u, atomicAdd %u번\n", names[m], r[m], r[3 + m]);
    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, cs, nullptr);
    vkx::destroy(dev, binds);
    vkx::destroy(dev, in);
    vkx::destroy(dev, out);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
실행 4096개(워크그룹 64 × 64), 서브그룹 크기 8, 정답 8390656
모두 atomicAdd: 합 8390656, atomicAdd 4096번
공유 메모리 나무: 합 8390656, atomicAdd 64번
subgroupAdd: 합 8390656, atomicAdd 512번
```

- **실행 번호**: `gl_LocalInvocationID`는 워크그룹 안의 번호, `gl_WorkGroupID`는 워크그룹의 번호이고, `gl_GlobalInvocationID`는 워크그룹 번호 × 크기 + 안의 번호입니다. 워크그룹 크기는 장치가 적어도 128개까지 보장하며, 64처럼 서브그룹 크기의 배수로 잡는 것이 무난합니다
- **모두 atomicAdd**: 4096개가 한 주소에 더하므로 하드웨어가 하나씩 줄 세웁니다. 결과는 맞지만 실행이 많을수록 느려집니다
- **공유 메모리 나무**: 워크그룹마다 64칸 배열에 값을 놓고, 절반씩 짝지어 더하기를 6번 하면 0번 칸에 그룹의 합이 남습니다. 단계마다 `barrier()`로 모두를 기다리므로, `barrier()`는 그룹의 모든 실행이 똑같이 지나가는 자리(균일한 흐름)에만 둡니다. 원자 연산은 그룹마다 한 번, 64번으로 줄었습니다
- **서브그룹**: GPU는 실행 몇 개를 한 명령으로 함께 돌리는데, 그 묶음이 서브그룹입니다. `subgroupAdd`는 메모리를 거치지 않고 묶음 안에서 바로 더하고, `subgroupElect`는 그중 한 명만 고릅니다. 크기는 장치마다 달라서(lavapipe 8, NVIDIA 32, AMD 32나 64) 원자 연산 수도 4096 ÷ 8 = 512번입니다

<Note>
워크그룹 사이에는 `barrier()`가 없어서, 다른 워크그룹이 쓴 값을 기다려 읽을 방법이 없습니다. 그룹마다의 결과를 모으려면 이 예처럼 원자 연산으로 합치거나, 디스패치를 둘로 나누고 그 사이에 배리어를 겁니다. 공유 메모리는 워크그룹마다 적어도 16KB(`maxComputeSharedMemorySize`)가 보장됩니다.
</Note>

## 정리

- 컴퓨트 셰이더는 워크그룹을 디스패치해 돌리고, 실행마다 `gl_GlobalInvocationID` 같은 번호로 자기 일을 찾는다
- 같은 워크그룹은 `shared` 메모리와 `barrier()`(균일한 흐름에서만)로 협력하고, 워크그룹 사이의 협력은 원자 연산이나 디스패치 나누기로 한다
- 서브그룹 연산은 메모리 없이 묶음 안에서 계산하므로 가장 싸지만, 서브그룹 크기는 장치마다 다르다
