GPGPU는 GPU를 그림이 아닌 계산에 쓰는 일입니다. 벌칸에서는 창도 스왑체인도 없이 컴퓨트 파이프라인([「컴퓨트 셰이더」](#/read/cpp-vulkan/glsl-compute))과 버퍼, 이미지만으로 됩니다. 아래 프로그램은 이웃 칸을 읽어 새 값을 정하는 격자 계산의 본보기로 생명 게임(산 칸은 이웃이 둘이나 셋이면 남고, 죽은 칸은 이웃이 셋이면 살아납니다)을 GPU에서 24세대 돌리고 CPU의 답과 견줍니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_compute.hpp,vkx_pipeline.hpp
#include <cstring>
#include <string>
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_compute.hpp"
#include "vkx_pipeline.hpp"

// 생명 게임 한 세대: 이웃 여덟 칸을 읽어 새 격자에 쓰고, 산 칸의 수도 센다
const char* STEP = R"(#version 450
layout(local_size_x = 8, local_size_y = 8) in;
layout(set = 0, binding = 0, r32ui) uniform readonly uimage2D before;
layout(set = 0, binding = 1, r32ui) uniform writeonly uimage2D after;
layout(set = 0, binding = 2) buffer Alive { uint alive[]; };
layout(push_constant) uniform Push { uint generation; };
void main() {
    ivec2 size = imageSize(before), p = ivec2(gl_GlobalInvocationID.xy);
    if (p.x >= size.x || p.y >= size.y) return;      // 격자 밖으로 넘친 실행
    uint around = 0u;
    for (int dy = -1; dy <= 1; ++dy)
        for (int dx = -1; dx <= 1; ++dx)             // 가장자리는 반대편과 이어진다
            if (dx != 0 || dy != 0)
                around += imageLoad(before, (p + ivec2(dx, dy) + size) % size).r;
    uint self = imageLoad(before, p).r;
    uint next = uint(around == 3u || (self == 1u && around == 2u));
    imageStore(after, p, uvec4(next));
    if (next == 1u) atomicAdd(alive[generation], 1u);
})";

const int W = 20, H = 10, G = 24;   // 격자 크기와 돌릴 세대 수
using Grid = std::vector<uint32_t>;

Grid step_on_cpu(const Grid& a) {   // 견줄 답: 같은 규칙을 CPU로
    Grid b(W * H);
    for (int y = 0; y < H; ++y)
        for (int x = 0; x < W; ++x) {
            uint32_t around = 0;
            for (int dy = -1; dy <= 1; ++dy)
                for (int dx = -1; dx <= 1; ++dx)
                    if (dx != 0 || dy != 0)
                        around += a[(y + dy + H) % H * W + (x + dx + W) % W];
            b[y * W + x] = around == 3 || (a[y * W + x] == 1 && around == 2);
        }
    return b;
}

void print(const char* title, const uint32_t* cells) {
    std::puts(title);
    for (int y = 0; y < H; ++y) {
        std::string row;
        for (int x = 0; x < W; ++x) row += cells[y * W + x] ? '#' : '.';
        std::puts(row.c_str());
    }
}

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const auto GENERAL = VK_IMAGE_LAYOUT_GENERAL;   // 스토리지 이미지의 레이아웃
    const auto COMPUTE = VK_PIPELINE_STAGE_2_COMPUTE_SHADER_BIT;
    const auto COPY = VK_PIPELINE_STAGE_2_COPY_BIT;
    const auto HOST = VK_PIPELINE_STAGE_2_HOST_BIT;

    // 격자 둘(번갈아 읽고 쓴다), CPU와 주고받을 버퍼, 세대마다의 산 칸 수
    vkx::Image grid[2];
    for (vkx::Image& g : grid)
        g = vkx::make_image(dev, {W, H}, VK_FORMAT_R32_UINT,
                            VK_IMAGE_USAGE_STORAGE_BIT |
                                VK_IMAGE_USAGE_TRANSFER_SRC_BIT |
                                VK_IMAGE_USAGE_TRANSFER_DST_BIT);
    vkx::Buffer cells = vkx::make_buffer(
        dev, W * H * 4,
        VK_BUFFER_USAGE_TRANSFER_SRC_BIT | VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    vkx::Buffer alive = vkx::make_buffer(dev, (G + 1) * 4,
                                         VK_BUFFER_USAGE_STORAGE_BUFFER_BIT, true);
    Grid start(W * H);
    for (int at : {W * 3 + 9, W * 3 + 10, W * 4 + 8, W * 4 + 9, W * 5 + 9})
        start[at] = 1;   // R-펜토미노
    std::memcpy(cells.mapped, start.data(), W * H * 4);
    std::memset(alive.mapped, 0, (G + 1) * 4);

    // 세트 둘: 0번은 grid[0]을 읽어 grid[1]에, 1번은 그 반대로
    const auto IMAGE = VK_DESCRIPTOR_TYPE_STORAGE_IMAGE;
    const auto BUFFER = VK_DESCRIPTOR_TYPE_STORAGE_BUFFER;
    vkx::Bindings sets[2];
    for (int i = 0; i < 2; ++i) {
        sets[i] = vkx::make_bindings(dev, {IMAGE, IMAGE, BUFFER});
        vkx::bind_image(dev, sets[i], 0, IMAGE, grid[i].view, GENERAL);
        vkx::bind_image(dev, sets[i], 1, IMAGE, grid[1 - i].view, GENERAL);
        vkx::bind_buffer(dev, sets[i], 2, BUFFER, alive);
    }
    VkPipelineLayout layout = vkx::make_layout(dev, {sets[0].layout}, 4);
    VkShaderModule shader =
        vkx::make_shader(dev, vkx::compile(STEP, shaderc_compute_shader));
    VkPipeline pipeline = vkx::make_compute(dev, shader, layout);

    VkCommandBuffer cmd = vkx::begin(cmds);
    // 앞 단계가 쓴 것을 뒤 단계가 보게 한다(이미지 레이아웃은 바꾸지 않는다)
    auto then = [&](VkPipelineStageFlags2 src, VkAccessFlags2 wrote,
                    VkPipelineStageFlags2 dst, VkAccessFlags2 reads) {
        VkMemoryBarrier2 b{.sType = VK_STRUCTURE_TYPE_MEMORY_BARRIER_2,
                           .srcStageMask = src, .srcAccessMask = wrote,
                           .dstStageMask = dst, .dstAccessMask = reads};
        VkDependencyInfo dep{.sType = VK_STRUCTURE_TYPE_DEPENDENCY_INFO,
                             .memoryBarrierCount = 1, .pMemoryBarriers = &b};
        vkCmdPipelineBarrier2(cmd, &dep);
    };
    const auto STORAGE_READ = VK_ACCESS_2_SHADER_STORAGE_READ_BIT;
    const auto STORAGE_WRITE = VK_ACCESS_2_SHADER_STORAGE_WRITE_BIT;
    for (vkx::Image& g : grid)
        vkx::transition(cmd, g, VK_IMAGE_LAYOUT_UNDEFINED, GENERAL,
                        VK_PIPELINE_STAGE_2_NONE, VK_ACCESS_2_NONE, COPY | COMPUTE,
                        VK_ACCESS_2_TRANSFER_WRITE_BIT | STORAGE_WRITE);
    VkBufferImageCopy whole{.imageSubresource = {grid[0].aspect, 0, 0, 1},
                            .imageExtent = {W, H, 1}};
    vkCmdCopyBufferToImage(cmd, cells.handle, grid[0].handle, GENERAL, 1, &whole);
    then(COPY, VK_ACCESS_2_TRANSFER_WRITE_BIT, COMPUTE, STORAGE_READ);

    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_COMPUTE, pipeline);
    const uint32_t GX = (W + 7) / 8, GY = (H + 7) / 8;   // 올림 나눗셈
    for (uint32_t g = 1; g <= G; ++g) {   // 세대마다 디스패치 한 번, 제출은 한 번
        vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_COMPUTE, layout, 0, 1,
                                &sets[(g - 1) % 2].set, 0, nullptr);
        vkCmdPushConstants(cmd, layout, VK_SHADER_STAGE_ALL, 0, 4, &g);
        vkCmdDispatch(cmd, GX, GY, 1);
        then(COMPUTE, STORAGE_WRITE, COMPUTE, STORAGE_READ | STORAGE_WRITE);
    }
    // 마지막 격자만 읽어 온다. 산 칸 수는 CPU에 보이는 버퍼에 이미 있다
    then(COMPUTE, STORAGE_WRITE, COPY | HOST,
         VK_ACCESS_2_TRANSFER_READ_BIT | VK_ACCESS_2_HOST_READ_BIT);
    vkCmdCopyImageToBuffer(cmd, grid[G % 2].handle, GENERAL, cells.handle, 1,
                           &whole);
    then(COPY, VK_ACCESS_2_TRANSFER_WRITE_BIT, HOST, VK_ACCESS_2_HOST_READ_BIT);
    vkx::submit_and_wait(dev, cmds);

    std::printf("격자 %d×%d, 워크그룹 %u×%u개(8×8씩): 칸 %d개에 실행 %u개\n", W, H,
                GX, GY, W * H, GX * GY * 64);
    print("0세대", start.data());
    const auto* last = static_cast<const uint32_t*>(cells.mapped);
    const auto* count = static_cast<const uint32_t*>(alive.mapped);
    print((std::to_string(G) + "세대").c_str(), last);
    std::printf("세대마다 산 칸:");
    Grid cpu = start;
    int wrong_counts = 0, wrong_cells = 0;
    for (int g = 1; g <= G; ++g) {
        cpu = step_on_cpu(cpu);
        uint32_t expected = 0;
        for (uint32_t c : cpu) expected += c;
        wrong_counts += count[g] != expected;
        std::printf(" %u", count[g]);
    }
    for (int i = 0; i < W * H; ++i) wrong_cells += last[i] != cpu[i];
    std::printf("\nCPU와 다른 칸 %d개, 산 칸 수가 다른 세대 %d개\n", wrong_cells,
                wrong_counts);

    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, shader, nullptr);
    for (vkx::Bindings& s : sets) vkx::destroy(dev, s);
    for (vkx::Image& g : grid) vkx::destroy(dev, g);
    vkx::destroy(dev, cells);
    vkx::destroy(dev, alive);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **스토리지 이미지**: `imageLoad`와 `imageStore`는 샘플러 없이 정수 좌표의 텍셀을 그대로 읽고 씁니다. 필터도 주소 모드도 없으므로 가장자리 밖의 좌표는 직접 감아 줍니다. 셰이더에서는 `uimage2D`(부호 없는 정수 텍셀)로 선언하고 크기는 `imageSize`로 얻습니다. 이미지는 `VK_IMAGE_USAGE_STORAGE_BIT`로 만들어 `GENERAL` 레이아웃에 두고, 포맷 한정자(`r32ui`)는 뷰의 포맷과 맞춥니다. 스토리지 이미지로 쓸 수 있는지는 포맷마다 달라서 `VK_FORMAT_FEATURE_STORAGE_IMAGE_BIT`로 확인합니다([「압축 포맷」](#/read/cpp-vulkan/tex-compressed))
- **두 이미지를 맞바꾸기**: 새 격자는 옛 격자 전체에서 나와야 하므로 읽는 이미지와 쓰는 이미지를 따로 두고 세대마다 맞바꿉니다. 세트를 둘 만들어 번갈아 붙이면 됩니다. 한 이미지에서 읽으며 쓰게 고쳐 보면 이 기계에서는 24세대 뒤 33칸이 CPU와 달랐습니다
- **제출은 한 번**: 24세대의 디스패치를 커맨드 버퍼 하나에 이어 기록해서 세대 사이에 CPU가 끼지 않습니다. 대신 앞 디스패치가 쓴 것을 다음 디스패치가 읽도록 사이마다 배리어를 둡니다([「파이프라인 배리어」](#/read/cpp-vulkan/vsync-barriers)). 빼면 동기화 검증이 READ_AFTER_WRITE를 알립니다
- **올림 나눗셈과 범위 검사**: 20×10 격자를 8×8 워크그룹으로 덮으려면 3×2개가 필요하고, 그러면 실행 384개 가운데 184개는 격자 밖입니다. 셰이더 첫머리의 검사가 그 실행들을 돌려보냅니다. 검사를 빼면 격자 밖의 실행까지 산 칸을 세어 1세대가 6이 아니라 10으로 나옵니다
- **필요한 것만 읽어 오기**: 세대마다의 산 칸 수는 `atomicAdd`로 작은 버퍼에 모으고, 격자는 마지막 것만 `vkCmdCopyImageToBuffer`로 가져왔습니다. 가져오는 양만큼 복사와 기다림이 생기므로 줄일 수 있는 것은 GPU에서 줄여서 가져옵니다

## 계산기로 쓸 때 따질 것

정수 계산은 CPU와 한 칸도 다르지 않아야 하고, 위에서도 0입니다. 실수는 그렇지 않습니다. 따로 막지 않으면 구현이 곱셈과 덧셈을 하나로 합치거나 차례를 바꿔도 되어서 장치마다 마지막 자리가 다를 수 있습니다. 그래서 실수 결과는 허용 오차를 두고 견주고, 꼭 지켜야 하는 식에는 GLSL의 `precise`를 붙여 재배열을 막습니다. `double`은 `shaderFloat64` 기능이 있는 장치에서만 씁니다.

일감이 크면 나눕니다. 한 차원에 띄울 워크그룹 수에는 한도(`maxComputeWorkGroupCount`, lavapipe는 65535)가 있고, Windows는 GPU가 작업 하나를 2초 안에 끝내지도 내려놓지도 못하면 멈춘 것으로 보고 드라이버를 재설정합니다. 긴 계산은 디스패치와 제출을 여러 번으로 쪼갭니다. 얼마나 돌릴지를 GPU가 정해야 하면 `vkCmdDispatchIndirect`가 워크그룹 수 셋(`VkDispatchIndirectCommand`)을 버퍼에서 읽습니다([「인스턴싱과 간접 그리기」](#/read/cpp-vulkan/tech-instancing)).

<Note>
이 책의 출력은 CPU가 GPU 노릇을 하는 lavapipe에서 얻은 것이라 GPU의 빠르기를 말해 주지 않습니다. 그래서 시간은 재지 않았습니다. 실제 장치에서 걸리는 시간은 타임스탬프 쿼리로 잽니다([「GPU 프로파일링」](#/read/cpp-vulkan/vdbg-profile)). 데이터가 작으면 올리고 내려받는 시간이 계산보다 길 수 있으니, GPU로 옮길지는 재 보고 정합니다.
</Note>

## 정리

- GPGPU는 데이터를 버퍼나 스토리지 이미지에 올려 두고 컴퓨트 디스패치를 이어서 돌린 뒤, 필요한 결과만 읽어 오는 일이다
- 되풀이 계산은 읽는 쪽과 쓰는 쪽을 맞바꾸는 두 리소스로 하고, 디스패치 사이에는 배리어를 둔다
- 정수 결과는 CPU와 똑같아야 하고 실수 결과는 허용 오차로 견주며, 큰 일감은 한도와 시간을 보고 나눈다
