GPU에 일을 넣는 창구가 큐입니다. 같은 능력을 가진 큐들이 한 **패밀리**로 묶이고, 장치는 패밀리마다 무엇을 할 수 있는지(능력 비트), 큐가 몇 개인지, 타임스탬프가 몇 비트인지, 이미지를 얼마 단위로 옮길 수 있는지 알려 줍니다. 앞 쪽의 `make_device`는 그래픽과 컴퓨트를 함께 하는 첫 패밀리의 큐 하나만 받았습니다. 아래 프로그램은 패밀리를 나열하고, 따로 된 컴퓨트·전송 패밀리가 있으면 골라내는 함수를 이 기계의 장치와 따로 꽂는 GPU에서 흔한 표에 각각 적용합니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp
#include "vkx_device.hpp"

struct Pick {
    int graphics = -1, compute = -1, transfer = -1;
};

// 그래픽 패밀리, 그래픽이 없는 컴퓨트 패밀리, 전송만 하는 패밀리를 찾는다.
// 따로 된 것이 없으면 그래픽 패밀리를 함께 쓴다
Pick pick(const std::vector<VkQueueFamilyProperties>& families) {
    Pick p;
    for (int i = 0; i < int(families.size()); ++i) {
        VkQueueFlags q = families[i].queueFlags;
        bool g = q & VK_QUEUE_GRAPHICS_BIT, c = q & VK_QUEUE_COMPUTE_BIT;
        if (g && p.graphics < 0) p.graphics = i;
        if (c && !g && p.compute < 0) p.compute = i;
        bool t = q & VK_QUEUE_TRANSFER_BIT;
        if (t && !g && !c && p.transfer < 0) p.transfer = i;
    }
    if (p.compute < 0) p.compute = p.graphics;
    if (p.transfer < 0) p.transfer = p.graphics;   // 그래픽 큐는 전송도 한다
    return p;
}

void show(const char* title, const std::vector<VkQueueFamilyProperties>& families) {
    std::printf("%s\n", title);
    for (size_t i = 0; i < families.size(); ++i) {
        VkQueueFlags q = families[i].queueFlags;
        std::printf("  패밀리 %zu: 큐 %2u개,%s%s%s%s\n", i, families[i].queueCount,
                    q & VK_QUEUE_GRAPHICS_BIT ? " 그래픽" : "",
                    q & VK_QUEUE_COMPUTE_BIT ? " 컴퓨트" : "",
                    q & VK_QUEUE_TRANSFER_BIT ? " 전송" : "",
                    q & VK_QUEUE_SPARSE_BINDING_BIT ? " 희소" : "");
    }
    Pick p = pick(families);
    std::printf("  고름: 그래픽 %d, 컴퓨트 %d, 전송 %d\n", p.graphics, p.compute,
                p.transfer);
}

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    uint32_t n = 0;
    vkGetPhysicalDeviceQueueFamilyProperties(dev.gpu, &n, nullptr);
    std::vector<VkQueueFamilyProperties> mine(n);
    vkGetPhysicalDeviceQueueFamilyProperties(dev.gpu, &n, mine.data());
    show("이 기계의 장치(lavapipe)", mine);
    VkExtent3D g = mine[0].minImageTransferGranularity;
    std::printf("  타임스탬프 %u비트, 이미지 전송 단위 (%u,%u,%u)\n",
                mine[0].timestampValidBits, g.width, g.height, g.depth);

    const VkQueueFlags G = VK_QUEUE_GRAPHICS_BIT, C = VK_QUEUE_COMPUTE_BIT,
                       T = VK_QUEUE_TRANSFER_BIT, S = VK_QUEUE_SPARSE_BINDING_BIT;
    std::vector<VkQueueFamilyProperties> nvidia = {   // NVIDIA GPU에서 흔한 모양
        {.queueFlags = G | C | T | S, .queueCount = 16},
        {.queueFlags = T | S, .queueCount = 2},
        {.queueFlags = C | T | S, .queueCount = 8}};
    show("따로 꽂는 GPU라면", nvidia);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
  패밀리 0: 큐  1개, 그래픽 컴퓨트 전송 희소
  고름: 그래픽 0, 컴퓨트 0, 전송 0
  타임스탬프 64비트, 이미지 전송 단위 (1,1,1)
따로 꽂는 GPU라면
  패밀리 0: 큐 16개, 그래픽 컴퓨트 전송 희소
  패밀리 1: 큐  2개, 전송 희소
  패밀리 2: 큐  8개, 컴퓨트 전송 희소
  고름: 그래픽 0, 컴퓨트 2, 전송 1
```

- **그래픽 큐는 전송도 한다**: 그래픽이나 컴퓨트를 하는 큐는 복사 같은 전송 명령도 받으므로, 이런 패밀리는 전송 비트를 따로 알리지 않아도 됩니다. 그래서 전송 패밀리는 "전송만 하는 것"으로 찾습니다
- **lavapipe는 패밀리 하나, 큐 하나**: 모든 일을 큐 하나로 받습니다. 그래픽을 지원하는 구현은 그래픽과 컴퓨트를 함께 하는 패밀리를 적어도 하나 두어야 하므로, 그 패밀리 하나만 쓰는 프로그램은 어디서나 돕니다
- **따로 꽂는 GPU**: NVIDIA GPU에서 흔히 보이는 표에서는 그래픽 0번, 컴퓨트만 하는 2번, 전송만 하는 1번이 골라졌습니다. 2번에 넣은 컴퓨트 작업은 그래픽과 겹쳐 돌 수 있고(비동기 컴퓨트), 1번은 복사 엔진(DMA)이라 그리는 동안 다음 텍스처를 옮길 수 있습니다([「스테이징」](#/read/cpp-vulkan/buf-staging)). 비디오 코덱 전용 패밀리를 두는 장치도 있습니다
- **이미지 전송 단위**: 그래픽·컴퓨트 패밀리는 (1,1,1), 곧 아무 영역이나 옮길 수 있습니다. 전송만 하는 패밀리는 (0,0,0)일 수 있는데, 이때는 밉 수준 하나를 통째로만 옮길 수 있습니다
- **타임스탬프 비트**: GPU 시간을 재는 쿼리 값에서 의미 있는 비트 수입니다. 0이면 그 패밀리에서는 시간을 잴 수 없습니다([「GPU 프로파일링」](#/read/cpp-vulkan/vdbg-profile))

## 큐에 일 넣기

명령은 하나씩 보내지 않고 **커맨드 버퍼**에 기록해 한 묶음으로 제출합니다. 커맨드 버퍼는 커맨드 풀에서 받는데, 풀을 만들 때 큐 패밀리를 하나 정하고 그 풀에서 받은 버퍼는 그 패밀리의 큐에만 낼 수 있습니다. 제출은 GPU가 일을 마치기 전에 돌아오므로, 끝났는지는 함께 넘긴 **펜스**로 압니다. 3장부터 쓸 이 흐름을 `vkx_cmd.hpp`로 묶어 둡니다.

```cpp vk file=vkx_cmd.hpp with=vkx_instance.hpp,vkx_device.hpp
#pragma once
#include "vkx_device.hpp"

namespace vkx {

// 커맨드 풀(패밀리 하나에 묶인다), 거기서 받은 커맨드 버퍼 하나, 끝을 알릴 펜스
struct Commands {
    VkCommandPool pool = VK_NULL_HANDLE;
    VkCommandBuffer cmd = VK_NULL_HANDLE;
    VkFence done = VK_NULL_HANDLE;
};

inline Commands make_commands(const Device& d) {
    Commands c;
    // 이 풀의 버퍼는 하나씩 다시 기록할 수 있다
    VkCommandPoolCreateInfo pci{
        .sType = VK_STRUCTURE_TYPE_COMMAND_POOL_CREATE_INFO,
        .flags = VK_COMMAND_POOL_CREATE_RESET_COMMAND_BUFFER_BIT,
        .queueFamilyIndex = d.family};
    check(vkCreateCommandPool(d.handle, &pci, nullptr, &c.pool), "커맨드 풀");
    VkCommandBufferAllocateInfo ai{
        .sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_ALLOCATE_INFO,
        .commandPool = c.pool, .level = VK_COMMAND_BUFFER_LEVEL_PRIMARY,
        .commandBufferCount = 1};
    check(vkAllocateCommandBuffers(d.handle, &ai, &c.cmd), "커맨드 버퍼");
    VkFenceCreateInfo fci{.sType = VK_STRUCTURE_TYPE_FENCE_CREATE_INFO};
    check(vkCreateFence(d.handle, &fci, nullptr, &c.done), "펜스");
    return c;
}

// 기록을 새로 시작한다. 앞에 기록한 내용은 지워진다
inline VkCommandBuffer begin(Commands& c) {
    VkCommandBufferBeginInfo bi{
        .sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_BEGIN_INFO,
        .flags = VK_COMMAND_BUFFER_USAGE_ONE_TIME_SUBMIT_BIT};
    check(vkBeginCommandBuffer(c.cmd, &bi), "기록 시작");
    return c.cmd;
}

// 기록을 마쳐 큐에 넣고, GPU가 끝낼 때까지 CPU가 기다린다
inline void submit_and_wait(const Device& d, Commands& c) {
    check(vkEndCommandBuffer(c.cmd), "기록 끝");
    VkCommandBufferSubmitInfo one{
        .sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_SUBMIT_INFO,
        .commandBuffer = c.cmd};
    VkSubmitInfo2 si{.sType = VK_STRUCTURE_TYPE_SUBMIT_INFO_2,
                     .commandBufferInfoCount = 1, .pCommandBufferInfos = &one};
    check(vkQueueSubmit2(d.queue, 1, &si, c.done), "제출");
    check(vkWaitForFences(d.handle, 1, &c.done, VK_TRUE, UINT64_MAX), "기다리기");
    vkResetFences(d.handle, 1, &c.done);   // 다음 제출에 다시 쓴다
}

inline void destroy(const Device& d, Commands& c) {
    vkDestroyFence(d.handle, c.done, nullptr);
    vkDestroyCommandPool(d.handle, c.pool, nullptr);   // 버퍼도 함께 풀린다
}

}  // namespace vkx
```

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp
#include "vkx_cmd.hpp"

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    for (int i = 1; i <= 2; ++i) {
        vkx::begin(cmds);                  // 기록할 명령은 3장부터 넣는다
        vkx::submit_and_wait(dev, cmds);
        std::printf("제출 %d: 패밀리 %u의 큐가 일을 끝냄\n", i, dev.family);
    }
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
제출 1: 패밀리 0의 큐가 일을 끝냄
제출 2: 패밀리 0의 큐가 일을 끝냄
```

- **`submit_and_wait`**: 기록을 닫아 1.3에서 코어가 된 `vkQueueSubmit2`로 내고, 펜스를 기다린 뒤 되돌려 둡니다. 버퍼를 다시 기록할 수 있는 것은 풀을 `RESET_COMMAND_BUFFER`로 만든 덕입니다
- **제출마다 기다리면**: CPU와 GPU가 번갈아 놉니다. 실험과 초기화에는 충분하지만, 매 프레임 그리는 프로그램은 GPU가 앞 프레임을 그리는 동안 다음 프레임을 기록합니다([「커맨드 버퍼」](#/read/cpp-vulkan/tri-commands), [「프레임 루프」](#/read/cpp-vulkan/tri-loop))

## 큐를 여럿 쓸 때

한 패밀리에서 큐를 여러 개 받으려면 `VkDeviceQueueCreateInfo`의 `queueCount`에 개수를, `pQueuePriorities`에 0.0~1.0의 우선순위를 큐마다 적습니다. 우선순위는 드라이버에게 주는 힌트일 뿐이고, 시스템 전체에서의 순위는 1.4에서 코어가 된 전역 우선순위로 따로 정합니다(높은 순위는 권한이 필요합니다). 큐 16개가 하드웨어 엔진 16개라는 뜻은 아니어서, 같은 엔진을 나눠 쓰는 경우가 많습니다. 큐 하나를 여러 스레드가 함께 쓰면 제출할 때 프로그램이 잠금으로 막아야 합니다.

버퍼와 이미지는 기본(`VK_SHARING_MODE_EXCLUSIVE`)으로 한 패밀리가 가집니다. 전송 패밀리에서 올린 텍스처를 그래픽 패밀리에서 쓰려면, 보내는 큐에서 소유권을 내놓는 배리어와 받는 큐에서 소유권을 받는 배리어를 짝지어 넣고 세마포어로 순서를 잇습니다([「파이프라인 배리어」](#/read/cpp-vulkan/vsync-barriers)). `VK_SHARING_MODE_CONCURRENT`로 만들면 이 절차 없이 여러 패밀리가 함께 쓰지만, 장치에 따라 조금 느릴 수 있습니다.

<Note>
화면에 내보내기(present)는 큐 패밀리의 능력 비트에 들어 있지 않습니다. 어느 패밀리가 어느 표면으로 내보낼 수 있는지는 표면을 만든 뒤 `vkGetPhysicalDeviceSurfaceSupportKHR`로 패밀리마다 묻습니다. 대부분의 장치에서는 그래픽 패밀리가 내보내기도 하지만, 명세가 보장하지는 않습니다([「표면과 스왑체인」](#/read/cpp-vulkan/setup-swapchain)).
</Note>

## 정리

- 큐 패밀리는 같은 능력을 가진 큐의 묶음이고, 그래픽·컴퓨트 패밀리 하나면 어디서나 돌며, 컴퓨트만·전송만 하는 패밀리는 겹쳐 돌리기와 업로드에 쓴다
- 명령은 패밀리 하나에 묶인 풀의 커맨드 버퍼에 기록해 그 패밀리의 큐에 내고, 끝났는지는 펜스로 안다
- 패밀리를 넘나드는 자원은 소유권을 넘기는 배리어 짝이나 CONCURRENT 공유가 필요하고, 내보내기 지원은 표면마다 따로 묻는다
