공장 데이터의 대부분은 **시계열**입니다. 온도, 압력, 속도 같은 태그의 값이 시각과 함께 끝없이 쌓입니다. 이것을 모아 두는 시스템을 이력 저장소(historian)라고 부르며, 추세 화면, 품질 분석, 고장 원인 조사가 모두 여기서 시작합니다. 태그 1000개를 1초마다 기록하면 하루에 8640만 개, 1년이면 300억 개가 넘으므로, **무엇을 남길지**와 **어떻게 빨리 쓰고 읽을지**가 설계의 핵심입니다.

## 압축하고, 쓰고, 묻기

10분 동안 1초마다 잰 온도(올라가고, 머물고, 내려가는)를 두 방식으로 압축하고, 원래 값을 SQLite에 넣은 뒤 분 단위 집계를 물었습니다.

```cpp flags=-lsqlite3
// 이력 저장: 두 가지로 압축하고, SQLite에 한 트랜잭션으로 넣고, 분 단위로 집계한다
#include <sqlite3.h>
#include <algorithm>
#include <cmath>
#include <cstdio>
#include <vector>

struct Point { int t; double v; };
using Series = std::vector<Point>;

Series deadband(const Series& in, double dev) {
    Series out{in.front()};
    for (const Point& p : in)
        if (std::fabs(p.v - out.back().v) > dev) out.push_back(p);
    return out;
}

// 스윙 도어 계열: 남긴 점에서 "지금까지의 모든 점을 ±dev 안으로 지나는 직선"이
// 있는 한 계속 묶는다. 그런 기울기가 없어지면 직전까지 맞던 직선 위의 점을 남긴다
Series swingingDoor(const Series& in, double dev) {
    Series out{in.front()};
    double up = INFINITY, low = -INFINITY;       // 가능한 기울기의 위아래 한계
    for (size_t i = 1; i < in.size(); ++i) {
        const Point a = out.back();
        const double dt = in[i].t - a.t;
        const double newUp = std::min(up, (in[i].v + dev - a.v) / dt);
        const double newLow = std::max(low, (in[i].v - dev - a.v) / dt);
        // 이 점까지 함께 지나는 직선이 없다
        if (newLow > newUp) {
            const double slope = (up + low) / 2; // 직전까지 맞던 기울기의 가운데
            const int t = in[i - 1].t;
            out.push_back({t, a.v + slope * (t - a.t)});
            const double dt2 = in[i].t - t;      // 남긴 점에서 다시 시작
            up = (in[i].v + dev - out.back().v) / dt2;
            low = (in[i].v - dev - out.back().v) / dt2;
        } else {
            up = newUp;
            low = newLow;
        }
    }
    const Point a = out.back();                  // 마지막 점도 직선 위에서
    const int t = in.back().t;
    const double v =
        std::isinf(up) ? in.back().v : a.v + (up + low) / 2 * (t - a.t);
    out.push_back({t, v});
    return out;
}

// 남긴 점으로 되살린 값과 원래 값의 최대 차이(계단 또는 직선 보간)
double maxError(const Series& raw, const Series& kept, bool linear) {
    double worst = 0;
    size_t k = 0;
    for (const Point& p : raw) {
        while (k + 1 < kept.size() && kept[k + 1].t <= p.t) ++k;
        double v = kept[k].v;
        if (linear && k + 1 < kept.size()) {
            const Point &a = kept[k], &b = kept[k + 1];
            v += (b.v - a.v) * (p.t - a.t) / (b.t - a.t);
        }
        worst = std::max(worst, std::fabs(v - p.v));
    }
    return worst;
}

int main() {
    Series raw;                                  // 10분 동안 1초마다 잰 온도
    unsigned rng = 7;
    for (int t = 0; t < 600; ++t) {
        rng = rng * 1103515245u + 12345u;
        const double noise = (int(rng >> 16) % 21 - 10) * 0.005;   // ±0.05
        const double trend = t < 180   ? 20 + t * 0.9              // 오르고
                           : t < 420   ? 182                       // 머물고
                                       : 182 - (t - 420) * 0.3;    // 내려간다
        raw.push_back({t, trend + noise});
    }
    const Series db = deadband(raw, 0.2), sd = swingingDoor(raw, 0.2);
    std::printf("원래 %zu점\n", raw.size());
    std::printf("데드밴드 0.2: %zu점, 최대 오차 %.3f\n", db.size(),
                maxError(raw, db, false));
    std::printf("스윙 도어 0.2: %zu점, 최대 오차 %.3f\n", sd.size(),
                maxError(raw, sd, true));

    sqlite3* conn;
    sqlite3_open(":memory:", &conn);
    sqlite3_exec(conn, "CREATE TABLE hist(tag TEXT, t INTEGER, v REAL)",
                 nullptr, nullptr, nullptr);
    sqlite3_stmt* ins;                           // 준비된 문장 하나를 계속 쓴다
    sqlite3_prepare_v2(conn, "INSERT INTO hist VALUES('oven.temp', ?, ?)", -1,
                       &ins, nullptr);
    sqlite3_exec(conn, "BEGIN", nullptr, nullptr, nullptr);    // 한 트랜잭션으로
    for (const Point& p : raw) {
        sqlite3_bind_int(ins, 1, p.t);
        sqlite3_bind_double(ins, 2, p.v);
        sqlite3_step(ins);
        sqlite3_reset(ins);
    }
    sqlite3_exec(conn, "COMMIT", nullptr, nullptr, nullptr);
    sqlite3_finalize(ins);

    // 분 단위 집계: 추세 화면과 보고서용
    sqlite3_stmt* q;
    sqlite3_prepare_v2(conn, "SELECT t/60, MIN(v), MAX(v), AVG(v) FROM hist "
                             "GROUP BY t/60 ORDER BY 1 LIMIT 4", -1, &q, nullptr);
    while (sqlite3_step(q) == SQLITE_ROW)
        std::printf("%d분: 최소 %.1f, 최대 %.1f, 평균 %.1f\n",
                    sqlite3_column_int(q, 0), sqlite3_column_double(q, 1),
                    sqlite3_column_double(q, 2), sqlite3_column_double(q, 3));
    sqlite3_finalize(q);
    sqlite3_close(conn);
}
```

0분: 최소 19.9, 최대 73.1, 평균 46.6
1분: 최소 74.0, 최대 127.1, 평균 100.6
2분: 최소 128.0, 최대 181.1, 평균 154.5
3분: 최소 182.0, 최대 182.1, 평균 182.0

- **데드밴드**: 마지막으로 남긴 값에서 0.2보다 크게 바뀐 값만 남깁니다. 머무는 구간의 잡음은 모두 버렸지만, 꾸준히 오르내리는 구간에서는 거의 모든 점을 남겨 600점이 360점이 되었습니다
- **스윙 도어 계열**: 남긴 점들을 **직선으로 이었을 때** 원래 값과의 차이가 0.2 안에 들도록, 직선 하나로 덮을 수 있는 동안은 점을 남기지 않습니다. 오르는 구간, 머무는 구간, 내려가는 구간이 모두 몇 개의 직선이 되어 600점이 10점이 되었고, 최대 오차는 약속한 0.2 안입니다. 상용 이력 저장소들이 쓰는 스윙 도어 압축과 같은 생각인데, 이 구현은 오차를 확실히 지키려고 원래 점 대신 직선 위의 점을 남겼습니다
- **일괄 쓰기**: SQLite에서 행마다 따로 커밋하면 커밋마다 디스크 동기화가 일어나 매우 느립니다. 준비된 문장 하나를 다시 쓰고, 한 트랜잭션으로 묶어 넣었습니다. 실제 수집기는 몇 초치를 모아 한 번에 씁니다
- **집계**: 추세 화면과 보고서는 원래 값보다 분 단위의 최소, 최대, 평균을 더 자주 봅니다. 평균만 남기면 짧은 이상 값이 사라지므로 최소와 최대를 함께 남깁니다

<Note>
규모가 커지면 InfluxDB, TimescaleDB(PostgreSQL 확장) 같은 시계열 데이터베이스나 상용 이력 저장소를 씁니다. 어느 쪽이든 오래된 원래 값은 지우고 집계만 남기는 보관 정책, 시각은 UTC로 저장하고 화면에서만 현지 시각으로 바꾸는 규칙, 그리고 값마다 품질([「태그 읽고 쓰기」](#/read/cpp-industrial/plc-tags))을 함께 저장하는 것이 기본입니다. 압축은 저장 공간을 아끼지만 원래 값을 되돌릴 수 없으므로, 품질 검사나 규제 대상인 값에는 쓰지 않거나 아주 작은 폭으로만 씁니다.
</Note>

## 정리

- 이력 저장소는 태그 값을 시각과 함께 쌓으며, 양이 매우 많아 무엇을 남길지와 어떻게 쓰고 읽을지가 핵심이다
- 데드밴드는 바뀐 값만 남기고, 스윙 도어 계열은 직선으로 이었을 때의 오차를 지키며 훨씬 적은 점만 남긴다
- 쓰기는 준비된 문장과 트랜잭션으로 묶고, 화면과 보고서용으로 최소·최대·평균 집계를 함께 남긴다
