연결은 언젠가 끊깁니다. 서버가 다시 시작하거나 다른 서버로 넘어가고, 네트워크 장비가 오래 조용한 연결을 버리고, 관리자가 세션을 끊습니다. 프로그램은 끊김을 알아보고, 잠시 기다렸다 다시 맺고, 하던 트랜잭션을 처음부터 다시 해야 합니다. 그리고 무엇을 기다리든 끝을 정해 둬야 멈추지 않습니다.

## 끊김을 알아보고 다시 하기

아래 예제는 관리자 연결에서 `pg_terminate_backend`로 일하는 연결의 서버 프로세스를 끊습니다. 두 번째 인자 5000은 프로세스가 실제로 끝날 때까지 5초까지 기다리라는 뜻이라서(PostgreSQL 14부터), 뒤의 결과가 매번 같습니다.

```cpp pkg=libpq
#include <libpq-fe.h>
#include <chrono>
#include <cstdio>
#include <string>
#include <thread>

using namespace std::chrono_literals;

const char* kConn = "host=localhost dbname=academy user=academy password=academy "
                    "connect_timeout=5 keepalives_idle=30 keepalives_interval=10 "
                    "keepalives_count=3";

bool exec(PGconn* c, const char* sql) {          // 결과 행이 없는 문장
    PGresult* r = PQexec(c, sql);
    bool ok = PQresultStatus(r) == PGRES_COMMAND_OK;
    PQclear(r);
    return ok;
}

std::string one(PGconn* c, const char* sql) {    // 첫 행의 첫 열
    PGresult* r = PQexec(c, sql);
    std::string v = PQntuples(r) > 0 ? PQgetvalue(r, 0, 0) : "";
    PQclear(r);
    return v;
}

// 연결마다 다시 해야 하는 설정. 새로 맺으면 서버 쪽 세션도 새것이라 사라진다
void setup(PGconn* c) { exec(c, "SET statement_timeout = '2s'"); }

enum class Tx { done, duplicate, failed };

// 이체 하나를 트랜잭션 통째로. 같은 요청 번호는 한 번만 반영된다
Tx transfer(PGconn* c, const std::string& request) {
    if (!exec(c, "BEGIN")) return Tx::failed;
    const char* p[] = {request.c_str()};
    PGresult* r = PQexecParams(c, "INSERT INTO rc_done VALUES ($1) "
                                  "ON CONFLICT DO NOTHING",
                               1, nullptr, p, nullptr, nullptr, 0);
    bool ok = PQresultStatus(r) == PGRES_COMMAND_OK;
    bool fresh = ok && std::string(PQcmdTuples(r)) == "1";
    PQclear(r);
    if (ok && !fresh) {                          // 이미 처리된 요청
        exec(c, "ROLLBACK");
        return Tx::duplicate;
    }
    ok = ok && exec(c, "UPDATE rc_acct SET bal = bal - 10 WHERE id = 1") &&
         exec(c, "UPDATE rc_acct SET bal = bal + 10 WHERE id = 2") &&
         exec(c, "COMMIT");
    return ok ? Tx::done : Tx::failed;
}

// 연결이 끊겨 실패했을 때만 기다렸다가 다시 맺고, 트랜잭션을 처음부터 다시 한다
Tx with_retry(PGconn* c, const std::string& request) {
    auto delay = 50ms;
    for (int attempt = 1;; ++attempt) {
        Tx t = transfer(c, request);
        if (t != Tx::failed) return t;
        if (PQstatus(c) == CONNECTION_OK) {      // 연결은 멀쩡: 다시 해도 같은 오류
            exec(c, "ROLLBACK");
            return t;
        }
        if (attempt == 4) return t;
        std::printf("  시도 %d 실패(연결 끊김), %lldms 뒤 다시 연결\n", attempt,
                    static_cast<long long>(delay.count()));
        std::this_thread::sleep_for(delay);
        PQreset(c);                              // 같은 연결 정보로 새로 맺는다
        if (PQstatus(c) == CONNECTION_OK) setup(c);
        delay *= 2;                              // 지수 백오프
    }
}

void show_error(PGconn* c, const char* label, const char* sql) {
    PGresult* r = PQexec(c, sql);
    std::printf("%s → %s %s\n", label, PQresultErrorField(r, PG_DIAG_SQLSTATE),
                PQresultErrorField(r, PG_DIAG_MESSAGE_PRIMARY));
    PQclear(r);
}

int main() {
    PGconn* admin = PQconnectdb(kConn);
    exec(admin, "SET client_min_messages = warning");
    exec(admin, "DROP TABLE IF EXISTS rc_acct, rc_done");
    exec(admin, "CREATE TABLE rc_acct (id int PRIMARY KEY, bal int)");
    exec(admin, "INSERT INTO rc_acct VALUES (1, 100), (2, 100)");
    exec(admin, "CREATE TABLE rc_done (request_id text PRIMARY KEY)");
    PGconn* w = PQconnectdb(kConn);
    setup(w);
    int old_pid = PQbackendPID(w);

    std::printf("일하는 연결의 서버 프로세스를 끊는다\n");
    std::string kill = "SELECT pg_terminate_backend(" + std::to_string(old_pid) +
                       ", 5000)";                // 끝날 때까지 5초까지 기다린다
    one(admin, kill.c_str());
    Tx t = with_retry(w, "req-1");
    std::printf("  %s, 새 서버 프로세스 %s, statement_timeout %s\n",
                t == Tx::done ? "이체 완료" : "실패",
                PQbackendPID(w) != old_pid ? "예" : "아니오",
                one(w, "SHOW statement_timeout").c_str());

    t = with_retry(w, "req-1");                  // 커밋 응답을 못 받았다고 치고
    std::printf("같은 요청을 다시: %s\n",
                t == Tx::duplicate ? "이미 처리됨" : "또 반영됨");
    std::string bal = one(w, "SELECT string_agg(bal::text, ' ' ORDER BY id) "
                             "FROM rc_acct");
    std::printf("  잔액 %s, 처리 기록 %s행\n", bal.c_str(),
                one(w, "SELECT count(*) FROM rc_done").c_str());

    exec(w, "SET statement_timeout = '100ms'");
    show_error(w, "statement_timeout", "SELECT pg_sleep(1)");
    exec(w, "SET statement_timeout = 0");
    exec(admin, "BEGIN");
    exec(admin, "UPDATE rc_acct SET bal = bal WHERE id = 1");   // 행 잠금을 쥔다
    exec(w, "SET lock_timeout = '100ms'");
    show_error(w, "lock_timeout", "UPDATE rc_acct SET bal = 0 WHERE id = 1");
    exec(admin, "ROLLBACK");
    std::printf("  두 경우 모두 연결은 %s\n",
                PQstatus(w) == CONNECTION_OK ? "그대로" : "끊김");

    bool up = PQping(kConn) == PQPING_OK;
    bool down = PQping("host=localhost port=5499") == PQPING_NO_RESPONSE;
    std::printf("PQping: 5432번 %s, 5499번 %s\n", up ? "응답함" : "?",
                down ? "응답 없음" : "?");
    PQfinish(w);
    PQfinish(admin);
}
```

```text
일하는 연결의 서버 프로세스를 끊는다
  시도 1 실패(연결 끊김), 50ms 뒤 다시 연결
  이체 완료, 새 서버 프로세스 예, statement_timeout 2s
같은 요청을 다시: 이미 처리됨
  잔액 90 110, 처리 기록 1행
statement_timeout → 57014 canceling statement due to statement timeout
lock_timeout → 55P03 canceling statement due to lock timeout
  두 경우 모두 연결은 그대로
PQping: 5432번 응답함, 5499번 응답 없음
```

- **끊김은 PQstatus로**: 끊긴 연결의 실패 결과에는 SQLSTATE가 붙어 있지 않았고, 메시지는 `PQerrorMessage`에 "terminating connection due to administrator command"와 연결이 닫혔다는 글로 남았습니다. 그래서 실패 뒤에 `PQstatus`가 `CONNECTION_BAD`인지를 봅니다. 연결이 멀쩡한데 실패했다면 SQL 쪽 오류라서 다시 연결해도 소용없습니다. 직렬화 실패와 교착은 다른 이유로 다시 하는데, 그것은 [「재시도」](#/read/cpp-database/tx-retry)에서 다룹니다
- **트랜잭션째로 다시**: 연결이 끊기면 서버는 열려 있던 트랜잭션을 되돌립니다. 그래서 실패한 문장부터 이어 갈 수 없고 `BEGIN`부터 다시 합니다. `PQreset`은 같은 연결 정보로 새로 맺지만 서버 쪽 세션은 새것이라, SET, 준비된 문장, 임시 테이블은 `setup`처럼 다시 해야 합니다. 이 일을 라이브러리의 자동 재연결에 맡기면 이런 상태가 소리 없이 사라집니다([「MySQL과 MariaDB」](#/read/cpp-database/cs-mysql))
- **지수 백오프**: 기다리는 시간을 50, 100, 200ms로 두 배씩 늘리고 횟수에 끝을 둡니다. 실제 코드에서는 여기에 무작위 흔들림(jitter)을 더해, 서버가 돌아온 순간 수많은 클라이언트가 한꺼번에 몰려들지 않게 합니다
- **커밋의 응답이 사라지면**: `COMMIT`을 보낸 뒤 응답을 받기 전에 끊기면, 커밋이 되었는지 알 수 없습니다. 이때 그냥 다시 하면 이체가 두 번 될 수 있습니다. 요청마다 번호(멱등 키)를 붙여 `UNIQUE` 열에 함께 넣으면, 두 번째 시도는 이미 처리된 요청임을 알아보고 아무것도 바꾸지 않습니다

## 기다림에 끝 두기

libpq의 `connect_timeout`은 연결을 맺을 때 기다리는 초이고(2보다 작은 값은 2로 봅니다), `keepalives_idle`, `keepalives_interval`, `keepalives_count`는 조용한 연결에 TCP keepalive를 보내 상대가 사라졌는지 알아보게 합니다. 리눅스에서는 `tcp_user_timeout`으로, 보낸 데이터가 확인되지 않은 채 기다리는 시간에도 끝을 둘 수 있습니다. 서버 쪽에서는 `statement_timeout`이 오래 걸리는 문장을 57014로, `lock_timeout`이 잠금을 오래 기다리는 문장을 55P03으로 취소합니다. 두 경우 모두 연결은 그대로이므로, 나중에 다시 할지 포기할지는 프로그램이 정합니다. 이런 값은 연결마다 SET하기보다 `ALTER ROLE ... SET statement_timeout`처럼 계정에 걸어 두면 빠뜨리지 않습니다. `PQping`은 연결을 맺지 않고 서버가 받을 수 있는 상태인지 알려 주어, 재연결을 되풀이하기 전에 살펴보기 좋습니다.

<Note>
MySQL과 MariaDB는 `wait_timeout`(기본 28800초, 8시간)보다 오래 조용한 연결을 서버가 끊습니다. 밤새 쉬던 풀의 연결을 아침에 빌려 쓰면 2006(MySQL server has gone away)이나 2013(Lost connection) 오류가 납니다. 오래 놀던 연결은 빌려 주기 전에 `mysql_ping`이나 가벼운 쿼리로 살아 있는지 보고, 죽었으면 새로 맺어 줍니다.
</Note>

## 정리

- 실패한 뒤 PQstatus가 CONNECTION_BAD이면 끊긴 것이고, 연결이 멀쩡한 실패는 다시 연결해도 고쳐지지 않는다
- 끊기면 서버가 트랜잭션을 되돌리므로 백오프 뒤 다시 맺고 세션 설정을 되살린 다음 트랜잭션을 처음부터 다시 하며, 커밋 여부를 모를 때를 위해 멱등 키를 둔다
- connect_timeout, keepalive, statement_timeout, lock_timeout으로 모든 기다림에 끝을 두고, 타임아웃 오류 뒤의 판단은 프로그램이 한다
