데이터베이스를 쓰는 코드는 실제 데이터베이스에 대고 시험해야 SQL의 오타, 제약의 동작, 열과 멤버의 어긋남([「행을 구조체로」](#/read/cpp-database/map-rows))을 잡습니다. 어려운 점은 둘입니다. 시험이 빨라야 하고, 한 시험이 남긴 데이터가 다른 시험의 결과를 바꾸지 않아야 합니다. 시험마다 똑같은 출발점을 만들어 주는 준비물을 **픽스처**라고 부르며, 여기서는 GoogleTest의 픽스처로 두 가지 방법을 봅니다([「단위 테스트와 퍼징」](#/read/cpp-advanced/tool-testing)).

## 픽스처 두 가지

```cpp pkg=gtest,libpq flags=-lsqlite3
#include <gtest/gtest.h>
#include <libpq-fe.h>
#include <sqlite3.h>
#include <string>
#include <vector>

// ---- 시험할 코드 ----
struct Summary {
    std::string name;
    long long orders, total;
};

std::vector<Summary> summaries(sqlite3* db) {   // 손님마다 주문 수와 합계
    std::vector<Summary> out;
    sqlite3_stmt* st;
    sqlite3_prepare_v2(db, "SELECT c.name, count(o.id), coalesce(sum(o.total), 0) "
                           "FROM customers c LEFT JOIN orders o "
                           "ON o.customer_id = c.id GROUP BY c.id ORDER BY c.id",
                       -1, &st, nullptr);
    while (sqlite3_step(st) == SQLITE_ROW)
        out.push_back({reinterpret_cast<const char*>(sqlite3_column_text(st, 0)),
                       sqlite3_column_int64(st, 1), sqlite3_column_int64(st, 2)});
    sqlite3_finalize(st);
    return out;
}

// ---- SQLite: 시험마다 새 메모리 DB에 스키마를 올린다 ----
class SqliteTest : public testing::Test {
protected:
    void SetUp() override {
        ASSERT_EQ(sqlite3_open(":memory:", &db), SQLITE_OK);
        const char* schema =
            "PRAGMA foreign_keys = ON;"
            "CREATE TABLE customers (id INTEGER PRIMARY KEY, name TEXT NOT NULL);"
            "CREATE TABLE orders (id INTEGER PRIMARY KEY, total INTEGER NOT NULL, "
            "customer_id INTEGER NOT NULL REFERENCES customers)";
        ASSERT_EQ(sqlite3_exec(db, schema, nullptr, nullptr, nullptr), SQLITE_OK);
        auto count = [](unsigned, void* n, void*, void*) {
            ++*static_cast<int*>(n);             // 실행된 문장 수를 센다
            return 0;
        };
        sqlite3_trace_v2(db, SQLITE_TRACE_STMT, count, &statements);
    }
    void TearDown() override { sqlite3_close(db); }

    // 시험에 필요한 데이터만 그 자리에서 만드는 도우미
    long long customer(const std::string& name) {
        sqlite3_stmt* st;
        sqlite3_prepare_v2(db, "INSERT INTO customers (name) VALUES (?)", -1, &st,
                           nullptr);
        sqlite3_bind_text(st, 1, name.c_str(), -1, SQLITE_TRANSIENT);
        sqlite3_step(st);
        sqlite3_finalize(st);
        return sqlite3_last_insert_rowid(db);
    }
    void order(long long who, int total) {
        sqlite3_stmt* st;
        sqlite3_prepare_v2(db, "INSERT INTO orders (customer_id, total) "
                               "VALUES (?, ?)", -1, &st, nullptr);
        sqlite3_bind_int64(st, 1, who);
        sqlite3_bind_int(st, 2, total);
        sqlite3_step(st);
        sqlite3_finalize(st);
    }
    int exec(const std::string& sql) {
        return sqlite3_exec(db, sql.c_str(), nullptr, nullptr, nullptr);
    }

    sqlite3* db = nullptr;
    int statements = 0;
};

TEST_F(SqliteTest, EmptyDatabaseHasNoSummaries) {
    EXPECT_TRUE(summaries(db).empty());
}

TEST_F(SqliteTest, SumsOrdersPerCustomerInOneStatement) {
    long long a = customer("앨리스"), b = customer("밥");
    order(a, 100);
    order(a, 50);
    order(b, 70);
    statements = 0;
    auto s = summaries(db);
    ASSERT_EQ(s.size(), 2u);
    EXPECT_EQ(s[0].name, "앨리스");
    EXPECT_EQ(s[0].orders, 2);
    EXPECT_EQ(s[0].total, 150);
    EXPECT_LE(statements, 1) << "손님마다 쿼리가 나간다 (N+1)";
}

TEST_F(SqliteTest, RejectsOrderOfMissingCustomer) {
    EXPECT_EQ(exec("INSERT INTO orders (customer_id, total) VALUES (99, 1)"),
              SQLITE_CONSTRAINT);
}

// ---- 시험할 코드: 이미 트랜잭션 안이면 저장점을 쓴다 ----
bool ok(PGconn* c, const char* sql, const char* param = nullptr) {
    PGresult* r = PQexecParams(c, sql, param ? 1 : 0, nullptr, &param, nullptr,
                               nullptr, 0);
    bool good = PQresultStatus(r) == PGRES_COMMAND_OK;
    PQclear(r);
    return good;
}

bool transfer(PGconn* c, int amount) {           // 1번 계좌에서 2번으로
    bool nested = PQtransactionStatus(c) != PQTRANS_IDLE;
    ok(c, nested ? "SAVEPOINT tr" : "BEGIN");
    std::string a = std::to_string(amount);
    bool good =
        ok(c, "UPDATE ft_acct SET bal = bal - $1 WHERE id = 1", a.c_str()) &&
        ok(c, "UPDATE ft_acct SET bal = bal + $1 WHERE id = 2", a.c_str());
    if (nested) ok(c, good ? "RELEASE SAVEPOINT tr" : "ROLLBACK TO SAVEPOINT tr");
    else ok(c, good ? "COMMIT" : "ROLLBACK");
    return good;
}

// ---- PostgreSQL: 시험마다 트랜잭션을 열고, 끝나면 되돌린다 ----
class PgTest : public testing::Test {
protected:
    static void SetUpTestSuite() {               // 묶음 전체에서 한 번
        conn = PQconnectdb("host=localhost dbname=academy user=academy "
                           "password=academy");
        ok(conn, "SET client_min_messages = warning");
        ok(conn, "DROP TABLE IF EXISTS ft_acct");
        ok(conn, "CREATE TABLE ft_acct (id int PRIMARY KEY, "
                 "bal int CHECK (bal >= 0))");
        ok(conn, "INSERT INTO ft_acct VALUES (1, 100), (2, 100)");
    }
    static void TearDownTestSuite() { PQfinish(conn); }
    void SetUp() override { ASSERT_TRUE(ok(conn, "BEGIN")); }
    void TearDown() override { ok(conn, "ROLLBACK"); }   // 시험이 바꾼 것을 버린다

    static int balance(int id) {
        std::string s = std::to_string(id);
        const char* p = s.c_str();
        PGresult* r = PQexecParams(conn, "SELECT bal FROM ft_acct WHERE id = $1", 1,
                                   nullptr, &p, nullptr, nullptr, 0);
        int v = std::atoi(PQgetvalue(r, 0, 0));
        PQclear(r);
        return v;
    }
    static PGconn* conn;
};
PGconn* PgTest::conn = nullptr;

TEST_F(PgTest, TransferMovesMoney) {
    ASSERT_TRUE(transfer(conn, 30));
    EXPECT_EQ(balance(1), 70);
    EXPECT_EQ(balance(2), 130);
}

TEST_F(PgTest, NextTestStartsFromOriginalBalances) {
    EXPECT_EQ(balance(1), 100);                  // 앞 시험의 이체는 되돌려졌다
}

TEST_F(PgTest, OverdraftIsRejectedAndTestCanContinue) {
    EXPECT_FALSE(transfer(conn, 500));           // CHECK 위반, 저장점까지만 되돌림
    EXPECT_EQ(balance(1), 100);                  // 바깥 트랜잭션은 계속 쓸 수 있다
}

int main(int argc, char** argv) {
    testing::InitGoogleTest(&argc, argv);
    GTEST_FLAG_SET(print_time, false);           // 출력을 실행마다 같게
    return RUN_ALL_TESTS();
}
```

- **시험마다 새 메모리 DB**: `SetUp`이 `:memory:` 데이터베이스를 열고 스키마를 올리므로 시험마다 빈 데이터베이스에서 시작하고, 몇 밀리초면 끝납니다. 실제 프로젝트라면 스키마를 따로 적지 말고 마이그레이션을 그대로 돌립니다([「스키마와 마이그레이션」](#/read/cpp-database/schema)). 다만 운영이 PostgreSQL이라면 SQLite로만 시험해서는 SQL 방언과 잠금의 차이를 놓칩니다
- **필요한 데이터만 그 자리에서**: `customer`, `order` 같은 도우미로 시험마다 필요한 행만 만듭니다. 큰 공용 데이터 파일을 두면 한 시험을 위해 고친 데이터가 다른 시험을 깨뜨립니다
- **문장 수도 검사한다**: 실행된 문장을 세어 `EXPECT_LE(statements, 1)`로 검사하면, 누가 손님마다 쿼리를 보내도록 고쳤을 때 바로 실패합니다([「N+1 문제」](#/read/cpp-database/dbperf-nplus1))
- **트랜잭션째 되돌리기**: 서버형 데이터베이스는 시험마다 `BEGIN`하고 끝나면 `ROLLBACK`합니다. 그래서 두 번째 시험은 앞 시험의 이체가 사라진 원래 잔액을 보았습니다. 이 방법이 통하려면 시험받는 코드가 스스로 커밋하지 않아야 합니다. `transfer`는 이미 트랜잭션 안이면 저장점을 써서([「RAII 트랜잭션」](#/read/cpp-database/tx-raii)), 잔액 부족으로 실패해도 저장점까지만 되돌리고 시험은 이어졌습니다. 코드가 그냥 `COMMIT`했다면 시험의 트랜잭션까지 커밋되어 데이터가 다음 시험으로 새어 나갑니다

<Note>
되돌리기 픽스처는 커밋되지 않은 데이터를 같은 연결에서만 볼 수 있다는 한계가 있습니다. 시험받는 코드가 풀에서 다른 연결을 빌리거나, 다른 스레드에서 돌거나, 동시성과 커밋 자체를 시험한다면 이 방법으로는 안 됩니다. 그럴 때는 시험을 돌릴 때마다 `CREATE DATABASE 이름 TEMPLATE 원본`으로 미리 준비한 데이터베이스를 복사해 쓰거나, 컨테이너로 새 서버를 띄웁니다([「컨테이너로 띄우기」](#/read/cpp-database/dbtest-container)). 출력을 실행마다 같게 하려고 GoogleTest의 시간 표시(`print_time`)는 껐습니다.
</Note>

## 정리

- 데이터베이스 코드는 실제 데이터베이스에 대고 시험하고, 시험마다 똑같은 출발점을 픽스처로 만들며, 데이터는 시험마다 필요한 만큼만 그 자리에서 만든다
- SQLite는 시험마다 새 메모리 DB로, 서버형은 시험마다 트랜잭션을 열고 되돌리며, 시험받는 코드는 이미 열린 트랜잭션 안에서 저장점을 쓰게 짠다
- 결과뿐 아니라 실행된 문장 수 같은 성질도 검사하고, 여러 연결이나 커밋을 시험할 때는 복사한 데이터베이스나 컨테이너를 쓴다
