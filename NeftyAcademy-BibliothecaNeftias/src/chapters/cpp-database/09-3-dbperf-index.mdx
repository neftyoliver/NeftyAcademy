인덱스는 읽기를 빠르게 하는 대신 쓰기마다 고쳐야 하고 자리를 차지합니다. 그래서 테이블마다 많이 만들기보다, 실제로 자주 도는 쿼리의 WHERE, JOIN, ORDER BY를 보고 거기에 맞춰 짓습니다. [「인덱스」](#/read/cpp-database/db-index)에서 본 기본에 더해, PostgreSQL의 B-tree 인덱스에는 네 가지 도구가 더 있습니다. 열의 순서, 일부 행만 담기, 식을 담기, 읽을 열을 덧붙이기입니다.

## 네 가지 도구

아래 예제는 계획 속 `Buffers` 줄을 읽은 페이지 수(캐시에 있던 것과 디스크에서 읽은 것의 합)로 바꿔 찍어서, 인덱스마다 일의 양을 견줄 수 있게 했습니다.

```cpp pkg=libpq
#include <libpq-fe.h>
#include <cstdio>
#include <cstdlib>
#include <string>

void run(PGconn* c, const char* sql) { PQclear(PQexec(c, sql)); }

std::string size_of(PGconn* c, const char* rel) {   // 테이블이나 인덱스의 크기
    const char* p[] = {rel};
    PGresult* r = PQexecParams(c, "SELECT pg_size_pretty(pg_relation_size($1))", 1,
                               nullptr, p, nullptr, nullptr, 0);
    std::string v = PQntuples(r) > 0 ? PQgetvalue(r, 0, 0) : "";
    PQclear(r);
    return v;
}

// 계획의 줄을 찍되 Buffers 줄은 페이지 수(hit + read)로 바꾸고, 계획 시간 쪽은 뺀다
void plan(PGconn* c, const char* title, const char* sql) {
    std::printf("%s\n", title);
    PGresult* r = PQexec(c, sql);
    for (int i = 0; i < PQntuples(r); ++i) {
        std::string line = PQgetvalue(r, i, 0);
        if (line.starts_with("Planning")) break;
        if (auto p = line.find("Buffers: shared"); p != std::string::npos) {
            long pages = 0;
            for (const char* key : {"hit=", "read="})
                if (auto q = line.find(key); q != std::string::npos)
                    pages += std::atol(line.c_str() + q + std::string(key).size());
            line = line.substr(0, p) + "읽은 페이지 " + std::to_string(pages) +
                   "개";
        }
        if (line.find("Heap Fetches") == std::string::npos)
            std::printf("  %s\n", line.c_str());
    }
    PQclear(r);
}

int main() {
    PGconn* c = PQconnectdb("host=localhost dbname=academy user=academy "
                            "password=academy");
    const char* setup[] = {
        "SET client_min_messages = warning",
        "DROP TABLE IF EXISTS pi_orders",
        "CREATE TABLE pi_orders (id int PRIMARY KEY, customer_id int, status text, "
        "created date, email text, total int)",
        "INSERT INTO pi_orders SELECT i, 1 + (i * 7919) % 1000, "
        "CASE WHEN i % 50 = 0 THEN 'open' ELSE 'done' END, "
        "date '2026-01-01' + (i % 365), 'User' || (i % 5000) || '@Example.com', "
        "(i * 37) % 500 FROM generate_series(1, 20000) i",
        "VACUUM ANALYZE pi_orders"};
    for (const char* sql : setup) run(c, sql);

    // 1. 여러 열 인덱스의 순서: 같음 조건의 열을 앞에, 범위 조건의 열을 뒤에
    const char* q = "EXPLAIN (ANALYZE, BUFFERS, COSTS OFF, TIMING OFF, "
                    "SUMMARY OFF) SELECT count(*) FROM pi_orders "
                    "WHERE status = 'open' AND created >= '2026-07-01'";
    run(c, "CREATE INDEX pi_cs ON pi_orders (created, status)");
    run(c, "ANALYZE pi_orders");
    plan(c, "(created, status) 순서", q);
    run(c, "DROP INDEX pi_cs");
    run(c, "CREATE INDEX pi_sc ON pi_orders (status, created)");
    run(c, "ANALYZE pi_orders");
    plan(c, "(status, created) 순서", q);

    // 2. 부분 인덱스: 자주 찾는 일부 행만 담는다
    run(c, "CREATE INDEX pi_open ON pi_orders (created) WHERE status = 'open'");
    std::printf("크기: 테이블 %s, (status, created) %s, 부분 인덱스 %s\n",
                size_of(c, "pi_orders").c_str(), size_of(c, "pi_sc").c_str(),
                size_of(c, "pi_open").c_str());
    run(c, "DROP INDEX pi_sc");
    plan(c, "부분 인덱스로 찾기",
         "EXPLAIN (COSTS OFF) SELECT id FROM pi_orders "
         "WHERE status = 'open' AND created >= '2026-12-01'");

    // 3. 식 인덱스: 열에 씌운 식과 똑같은 식으로 찾을 때만 쓰인다
    run(c, "CREATE INDEX pi_email ON pi_orders (lower(email))");
    run(c, "ANALYZE pi_orders");                 // 식의 통계도 모은다
    plan(c, "lower(email)로 찾기", "EXPLAIN (COSTS OFF) SELECT id FROM pi_orders "
                                  "WHERE lower(email) = 'user42@example.com'");
    plan(c, "email로 찾기", "EXPLAIN (COSTS OFF) SELECT id FROM pi_orders "
                           "WHERE email = 'user42@example.com'");

    // 4. INCLUDE: 찾는 데는 안 쓰지만 읽을 열을 인덱스에 함께 둔다
    run(c, "CREATE INDEX pi_cust ON pi_orders (customer_id)");
    const char* sum = "EXPLAIN (COSTS OFF) "
                      "SELECT sum(total) FROM pi_orders WHERE customer_id = 42";
    plan(c, "customer_id 인덱스로 합계", sum);
    run(c, "DROP INDEX pi_cust");
    run(c, "CREATE INDEX pi_cust_total ON pi_orders (customer_id) INCLUDE (total)");
    run(c, "VACUUM pi_orders");                  // 가시성 맵을 새로 한다
    plan(c, "INCLUDE (total)을 더한 인덱스로 합계", sum);
    PQfinish(c);
}
```

- **여러 열의 순서**: B-tree는 첫 열로 정렬하고, 같은 값 안에서 둘째 열로 정렬합니다. `(status, created)`에서는 7월 이후의 `open` 행들이 한곳에 모여 있어 4페이지만 읽었고, `(created, status)`에서는 7월 이후의 모든 행을 인덱스 안에서 훑으며 상태를 따져 12페이지를 읽었습니다. 데이터가 커질수록 차이도 커집니다. 같음 조건의 열을 앞에, 범위 조건의 열을 뒤에 둡니다. 여러 열 인덱스는 앞쪽 열만 쓰는 쿼리에도 쓰이지만, 뒤쪽 열만으로 찾는 쿼리에는 잘 쓰이지 않습니다
- **부분 인덱스**: `WHERE status = 'open'`을 붙인 인덱스는 전체의 2%인 열린 주문만 담아 16kB로, 같은 일을 하는 전체 인덱스의 10분의 1이었습니다. 플래너는 쿼리의 조건이 인덱스의 조건을 뜻할 때만 이 인덱스를 씁니다
- **식 인덱스**: `lower(email)`에 만든 인덱스는 쿼리가 똑같은 식으로 찾을 때만 쓰이고, `email = ...`로 찾자 Seq Scan이 되었습니다. 식 인덱스를 만든 뒤 `ANALYZE`하면 식의 값 분포까지 통계에 들어갑니다. SQLite에서도 결정적인 함수로 식 인덱스를 만들 수 있습니다([「확장」](#/read/cpp-database/sqlite-ext))
- **INCLUDE로 덮기**: `customer_id` 인덱스만 있으면 맞는 행을 찾은 뒤 `total`을 읽으러 테이블 페이지를 다녔습니다. `INCLUDE (total)`로 읽을 열을 인덱스에 덧붙이자 테이블을 읽지 않는 Index Only Scan이 되었습니다. INCLUDE한 열은 정렬되지 않아 찾는 데는 쓰이지 않습니다. SQLite에는 INCLUDE가 없지만, 필요한 열을 모두 담은 여러 열 인덱스가 같은 일을 합니다

## 인덱스의 값

INSERT와 DELETE는 모든 인덱스를 고치고, UPDATE도 인덱스에 든 열을 바꾸면 인덱스를 고칩니다. PostgreSQL은 인덱스에 든 열이 그대로이고 같은 페이지에 자리가 있으면 인덱스를 건드리지 않는 HOT 갱신을 하므로, 자주 바뀌는 열은 인덱스에 넣지 않는 편이 낫습니다. 쓰이지 않는 인덱스는 `pg_stat_user_indexes`의 `idx_scan`이 0인 것으로 찾되, 통계를 마지막으로 지운 뒤의 수이고 복제 서버에서 쓰일 수도 있음을 함께 봅니다. `(a)`는 보통 `(a, b)`가 대신하므로 둘 다 둘 필요가 없습니다. 외래 키 열은 저절로 인덱스가 생기지 않으니 조인과 삭제 검사에 쓰인다면 만듭니다. 운영 중인 큰 테이블에는 `CREATE INDEX CONCURRENTLY`로 쓰기를 막지 않고 만듭니다([「스키마와 마이그레이션」](#/read/cpp-database/schema)).

<Note>
부분 인덱스의 조건은 쿼리의 글자를 보고 판단합니다. 그래서 `status = $1`처럼 매개변수로 묶으면, 값과 상관없는 일반 계획에서는 그 값이 `'open'`일지 알 수 없어 부분 인덱스를 쓰지 못합니다. 부분 인덱스를 노리는 조건은 사용자 입력이 아닌 상수이므로 SQL에 그대로 적습니다.
</Note>

## 정리

- 여러 열 인덱스는 같음 조건의 열을 앞에, 범위 조건의 열을 뒤에 두며, 앞쪽 열만 쓰는 쿼리에도 쓰인다
- 자주 찾는 일부 행은 부분 인덱스로, 식으로 찾는 조건은 똑같은 식의 인덱스로, 읽을 열은 INCLUDE로 덧붙여 Index Only Scan을 만든다
- 인덱스는 쓰기마다 고쳐야 하므로 쿼리에 맞춰 적게 만들고, 쓰이지 않거나 겹치는 인덱스는 통계로 찾아 지운다
