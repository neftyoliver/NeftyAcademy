DB의 값은 C++ 변수로 건너오면서 한 번 변환됩니다. 그 규칙은 데이터베이스마다 다르고, 정보를 잃어도 대개 말이 없습니다. NULL은 값과 다른 길로 알려 주므로 그 길을 따로 확인해야 합니다.

## SQLite: 형은 값에 붙는다

SQLite의 값은 INTEGER, REAL, TEXT, BLOB, NULL 다섯 저장 형 가운데 하나이고, 열에 적은 형은 **형 친화도**(type affinity)라는 권고입니다. INTEGER 열은 정수로 바꿀 수 있는 글자는 바꿔 넣고, 못 바꾸면 글자 그대로 넣습니다. 읽는 쪽의 `sqlite3_column_int64`는 무엇이든 정수로 바꿔 주며 실패를 알리지 않습니다.

```cpp flags=-lsqlite3
#include <sqlite3.h>
#include <cstdint>
#include <cstdio>
#include <stdexcept>
#include <string>

const char* kind(int t) {
    switch (t) {
    case SQLITE_INTEGER: return "INTEGER";
    case SQLITE_FLOAT: return "REAL";
    case SQLITE_TEXT: return "TEXT";
    case SQLITE_BLOB: return "BLOB";
    }
    return "NULL";
}

// 저장된 형이 정수가 아니면 바꾸지 않고 멈춘다
long long strict_int(sqlite3_stmt* s, int i) {
    int t = sqlite3_column_type(s, i);
    if (t != SQLITE_INTEGER)
        throw std::runtime_error(std::string("정수 아님 ") + kind(t));
    return sqlite3_column_int64(s, i);
}

// 한 열짜리 문장을 실행해 값과 저장 형, 또는 오류를 보인다
void show(sqlite3* db, const char* label, const char* sql) {
    sqlite3_stmt* st;
    sqlite3_prepare_v2(db, sql, -1, &st, nullptr);
    std::printf("  %s → ", label);
    if (sqlite3_step(st) == SQLITE_ROW) {
        const char* k = kind(sqlite3_column_type(st, 0));    // 형은 변환 전에
        std::printf("%s (%s)\n", sqlite3_column_text(st, 0), k);
    } else {
        std::printf("오류 %d %s\n", sqlite3_extended_errcode(db),
                    sqlite3_errmsg(db));
    }
    sqlite3_finalize(st);
}

int main() {
    sqlite3* db;
    sqlite3_open(":memory:", &db);
    sqlite3_exec(db, "CREATE TABLE loose (src TEXT, qty INTEGER);"
                     "INSERT INTO loose VALUES ('12', 12), ('''12''', '12'),"
                     " ('''12kg''', '12kg'), ('3.9', 3.9), ('3.0', 3.0),"
                     " ('NULL', NULL);"
                     "CREATE TABLE tight (qty INTEGER) STRICT;",
                 nullptr, nullptr, nullptr);
    std::printf("넣은 값   저장된 형  column_int64  strict_int\n");
    sqlite3_stmt* st;
    sqlite3_prepare_v2(db, "SELECT src, qty FROM loose", -1, &st, nullptr);
    while (sqlite3_step(st) == SQLITE_ROW) {
        const char* k = kind(sqlite3_column_type(st, 1));
        std::printf("%-8s  %-9s  %-12lld  ", sqlite3_column_text(st, 0), k,
                    sqlite3_column_int64(st, 1));
        try {
            std::printf("%lld\n", strict_int(st, 1));
        } catch (const std::exception& e) {
            std::printf("%s\n", e.what());
        }
    }
    sqlite3_finalize(st);

    std::printf("STRICT 테이블에 넣기:\n");
    show(db, "'12'", "INSERT INTO tight VALUES ('12') RETURNING qty");
    show(db, "'12kg'", "INSERT INTO tight VALUES ('12kg') RETURNING qty");
    show(db, "3.9", "INSERT INTO tight VALUES (3.9) RETURNING qty");

    std::printf("64비트의 끝:\n");
    show(db, "최댓값 + 1", "SELECT 9223372036854775807 + 1");
    std::uint64_t big = UINT64_MAX;                     // 부호 없는 64비트의 최댓값
    sqlite3_prepare_v2(db, "SELECT ?1", -1, &st, nullptr);
    sqlite3_bind_int64(st, 1, static_cast<long long>(big));
    sqlite3_step(st);
    std::printf("  uint64 %llu → %lld\n", static_cast<unsigned long long>(big),
                sqlite3_column_int64(st, 0));
    sqlite3_finalize(st);

    std::printf("가운데에 0 바이트가 낀 5바이트:\n");
    const std::string raw("ab\0cd", 5);
    sqlite3_prepare_v2(db, "SELECT ?1, ?2, ?3", -1, &st, nullptr);
    sqlite3_bind_text(st, 1, raw.c_str(), -1, SQLITE_TRANSIENT);    // 0에서 끊긴다
    sqlite3_bind_blob(st, 2, raw.data(), int(raw.size()), SQLITE_TRANSIENT);
    sqlite3_bind_zeroblob(st, 3, 0);                                 // 빈 BLOB
    sqlite3_step(st);
    for (int i = 0; i < 3; ++i) {
        const char* k = kind(sqlite3_column_type(st, i));
        const void* p = sqlite3_column_blob(st, i);
        int n = sqlite3_column_bytes(st, i);                 // 길이는 포인터 다음에
        std::printf("  %s %d바이트%s\n", k, n, p ? "" : ", 포인터는 nullptr");
    }
    sqlite3_finalize(st);
    sqlite3_close(db);
}
```

- **조용한 변환**: TEXT로 남은 `'12kg'`에서 `column_int64`는 앞의 숫자만 읽어 12를, REAL 3.9에서는 소수를 버려 3을, NULL에서는 0을 주었습니다. `strict_int`처럼 `sqlite3_column_type`으로 저장 형부터 보고 기대와 다르면 멈추면 이런 값이 섞이지 않습니다. 변환이 한 번 일어난 뒤의 `column_type`은 믿을 수 없으므로 가장 먼저 부릅니다
- **STRICT 테이블**: SQLite 3.37부터 테이블 끝에 `STRICT`를 붙이면 열의 형이 약속이 됩니다. 손실 없이 바뀌는 `'12'`는 정수로 들어가고, `'12kg'`과 3.9는 `SQLITE_CONSTRAINT_DATATYPE`(3091)로 거절됩니다
- **64비트의 끝**: SQLite 정수는 부호 있는 64비트라서 덧셈이 넘치면 REAL로 바뀌어 정밀도를 잃습니다. `uint64_t`를 `bind_int64`로 넣으면 비트 그대로 읽혀 최댓값이 -1이 됩니다. 2⁶³ 이상이 나오는 값(해시, 부호 없는 ID)은 범위를 확인하거나 BLOB으로 둡니다
- **BLOB과 0 바이트**: 길이로 -1을 주면 0 바이트에서 끊겨 2바이트만 들어갔습니다. 앞 쪽 `put`의 -1도 글자에만 맞는 선택입니다. 이진 데이터는 길이를 주어 `bind_blob`으로 넣고, 빈 BLOB에서는 `sqlite3_column_blob`이 널 포인터를 주므로 길이부터 보고 복사합니다

## PostgreSQL: 텍스트 형식과 이진 형식

PostgreSQL은 열의 형을 엄격히 지킵니다. libpq는 결과를 기본으로 사람이 읽는 **텍스트 형식**으로 받고, `PQexecParams`의 마지막 인자를 1로 주면 서버가 정한 바이트 표현인 **이진 형식**으로 받습니다.

```cpp pkg=libpq
#include <libpq-fe.h>
#include <cstdio>
#include <string>

int main() {
    PGconn* c = PQconnectdb("host=localhost dbname=academy user=academy "
                            "password=academy");
    const char* names[] = {"int4", "bool", "bytea", "''", "NULL"};
    for (int fmt = 0; fmt <= 1; ++fmt) {               // 0은 텍스트, 1은 이진
        PGresult* r = PQexecParams(c, "SELECT 42::int4, true, '\\x00ff41'::bytea,"
                                      " ''::text, NULL::text",
                                   0, nullptr, nullptr, nullptr, nullptr, fmt);
        std::printf("%s 형식\n", fmt ? "이진" : "텍스트");
        for (int i = 0; i < PQnfields(r); ++i) {
            const char* v = PQgetvalue(r, 0, i);       // NULL이어도 ""를 준다
            int n = PQgetlength(r, 0, i);
            std::printf("  %-6s %d바이트", names[i], n);
            if (fmt == 0) std::printf(" \"%s\"", v);
            for (int k = 0; fmt == 1 && k < n; ++k)
                std::printf(" %02x", static_cast<unsigned char>(v[k]));
            std::printf("%s\n", PQgetisnull(r, 0, i) ? " ← PQgetisnull 1" : "");
        }
        PQclear(r);
    }

    std::printf("int4 열에 넣기\n");
    PQclear(PQexec(c, "CREATE TEMP TABLE pg_qty (qty int4)"));
    const char* vals[] = {"'12'", "'12kg'", "3.9", "3000000000"};
    for (const char* v : vals) {
        std::string sql = std::string("INSERT INTO pg_qty VALUES (") + v +
                          ") RETURNING qty";
        PGresult* r = PQexec(c, sql.c_str());
        if (PQresultStatus(r) == PGRES_TUPLES_OK)
            std::printf("  %s → %s\n", v, PQgetvalue(r, 0, 0));
        else
            std::printf("  %s → %s %s\n", v,
                        PQresultErrorField(r, PG_DIAG_SQLSTATE),
                        PQresultErrorField(r, PG_DIAG_MESSAGE_PRIMARY));
        PQclear(r);
    }
    PQfinish(c);
}
```

```text
텍스트 형식
  int4   2바이트 "42"
  bool   1바이트 "t"
  bytea  8바이트 "\x00ff41"
  ''     0바이트 ""
  NULL   0바이트 "" ← PQgetisnull 1
이진 형식
  int4   4바이트 00 00 00 2a
  bool   1바이트 01
  bytea  3바이트 00 ff 41
  ''     0바이트
  NULL   0바이트 ← PQgetisnull 1

  '12' → 12
  '12kg' → 22P02 invalid input syntax for type integer: "12kg"
  3.9 → 4
  3000000000 → 22003 integer out of range
```

- **두 형식**: 텍스트 형식에서 bool은 `t`, bytea는 16진 글자 `\x00ff41`로 옵니다. 이진 형식의 int4는 큰 끝(네트워크 순서) 4바이트라 `ntohl`로 풀어야 하고, bytea는 바이트 그대로 옵니다. 이진 형식은 글자를 해석하지 않아도 되는 대신 형마다 서버의 표현을 알아야 합니다
- **NULL과 빈 문자열**: `PQgetvalue`는 NULL에도 빈 문자열을 주고 길이도 0으로 같아서, 둘을 가르는 것은 `PQgetisnull`뿐입니다
- **엄격한 넣기**: 같은 `'12kg'`이 여기서는 22P02 오류이고, int4 범위를 넘는 수는 22003입니다. 3.9는 4로 반올림되어, 3으로 버린 SQLite나 C++의 `static_cast`와 결과가 다릅니다

<Note>
PostgreSQL 안에서도 반올림은 형마다 다릅니다. numeric을 정수로 바꾸면 2.5가 0에서 먼 쪽인 3이 되고, double precision을 바꾸면 가까운 짝수인 2가 됩니다. 반올림이 결과를 바꾸는 값은 어느 형에 담고 어디서 바꿀지 정해 둡니다. 돈과 시간을 담는 형은 [「시간과 소수」](#/read/cpp-database/map-time)에서 다룹니다.
</Note>

## 정리

- SQLite는 값마다 형이 붙고 column 함수는 조용히 변환하므로, 저장 형을 먼저 보는 읽기 함수나 STRICT 테이블로 형을 지킨다
- 64비트를 넘는 정수, 0 바이트가 낀 이진 데이터, 빈 BLOB의 널 포인터처럼 경계에서 값이 바뀌거나 잘리는 자리를 따로 챙긴다
- libpq는 기본으로 텍스트 형식을 주고 이진 형식은 서버의 바이트 표현을 따르며, NULL과 빈 문자열은 PQgetisnull로만 가른다
