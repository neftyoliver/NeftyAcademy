스크립트 언어를 붙이는 기본기(상태 만들기, 스택으로 값 주고받기, 보호 호출)는 게임 엔진 책의 [「스크립트 언어 붙이기」](#/read/cpp-game-engine/scr-lua)에서 다뤘습니다. 문서를 편집하는 프로그램에서는 물음이 더 있습니다. 스크립트가 문서를 고치다 중간에 죽으면 문서는 어떻게 되는가, 스크립트가 들고 있던 물체가 지워지면 어떻게 되는가, 끝나지 않는 스크립트가 창을 얼리면 어떻게 하는가입니다. 예제는 Lua 5.4를 품은 작은 편집기입니다.

```cpp flags=-I/usr/include/lua5.4,-llua5.4
// 스크립트 엔진: 명령 등록부로 들어오고, 조각으로 돌고, 통째로 실행 취소 한 단계
#include <functional>
#include <lua.hpp>
#include <map>
#include <print>
#include <stdexcept>
#include <string>
#include <vector>

using Handle = long long;                     // 칸 번호 * 2^32 + 세대
struct Doc {
    struct Slot {
        std::string name;
        unsigned gen = 0;                     // 칸을 비울 때마다 올린다
        bool alive = false;
    };
    std::vector<Slot> slots;
    Slot& at(Handle h) {
        std::size_t i = static_cast<std::size_t>(h >> 32);
        if (i >= slots.size() || !slots[i].alive || slots[i].gen != unsigned(h))
            throw std::runtime_error("사라진 물체의 핸들");
        return slots[i];
    }
    Handle add(const std::string& name) {     // 빈 칸이 있으면 다시 쓴다
        std::size_t i = 0;
        while (i < slots.size() && slots[i].alive) ++i;
        if (i == slots.size()) slots.emplace_back();
        slots[i].name = name, slots[i].alive = true;
        return static_cast<Handle>(i) << 32 | slots[i].gen;
    }
    std::string text() const {
        std::string out;
        for (const Slot& s : slots)
            if (s.alive) out += (out.empty() ? "" : ", ") + s.name;
        return "[" + out + "]";
    }
};

using Command = std::function<Handle(Doc&, Handle, const std::string&)>;
struct App {
    Doc doc;
    std::vector<Doc> undo;                    // 예제는 문서를 통째로 베껴 둔다
    int calls = 0;
    std::map<std::string, Command> commands{  // 메뉴와 스크립트가 함께 쓰는 등록부
        {"add", [](Doc& d, Handle, const std::string& s) { return d.add(s); }},
        {"remove", [](Doc& d, Handle h, const std::string&) {
             Doc::Slot& s = d.at(h);
             return s.alive = false, ++s.gen, Handle{};
         }},
        {"rename", [](Doc& d, Handle h, const std::string& s) {
             return d.at(h).name = s, Handle{};
         }},
    };
};

bool call(lua_State* L) noexcept {            // C++ 객체는 이 안에서만 산다
    App& app = *static_cast<App*>(lua_touserdata(L, lua_upvalueindex(1)));
    try {
        bool has_handle = lua_isinteger(L, 1);
        const char* text = lua_tostring(L, has_handle ? 2 : 1);
        const Command& fn = app.commands.at(lua_tostring(L, lua_upvalueindex(2)));
        Handle h = has_handle ? lua_tointeger(L, 1) : 0;
        ++app.calls;
        lua_pushinteger(L, fn(app.doc, h, text ? text : ""));
        return true;
    } catch (const std::exception& e) {
        luaL_where(L, 1);                     // "실수:4: " 같은 자리를 앞에 붙인다
        lua_pushstring(L, e.what());
        lua_concat(L, 2);
        return false;
    }
}
int l_command(lua_State* L) { return call(L) ? 1 : lua_error(L); }
void slice(lua_State* L, lua_Debug*) { lua_yield(L, 0); }     // 조각 하나가 끝났다

void run_script(App& app, const std::string& name, const std::string& src) {
    lua_State* L = luaL_newstate();
    luaL_requiref(L, "_G", luaopen_base, 1);  // 표준 라이브러리는 base만 연다
    for (const char* reads_file : {"dofile", "loadfile", "load"})
        lua_pushnil(L), lua_setglobal(L, reads_file);
    for (auto& [id, fn] : app.commands) {     // 명령마다 같은 이름의 함수를 만든다
        lua_pushlightuserdata(L, &app);
        lua_pushstring(L, id.c_str());
        lua_pushcclosure(L, l_command, 2);
        lua_setglobal(L, id.c_str());
    }
    lua_State* co = lua_newthread(L);
    lua_sethook(co, slice, LUA_MASKCOUNT, 100);       // 명령어 100개마다 돌아온다
    Doc before = app.doc;                     // 트랜잭션을 연다
    int n = 0, used = 1;
    app.calls = 0;
    int status = luaL_loadbuffer(co, src.data(), src.size(), ("=" + name).c_str());
    if (status == LUA_OK)                     // 조각 사이에 UI 사건을 처리한다
        while ((status = lua_resume(co, nullptr, 0, &n)) == LUA_YIELD && used < 50)
            ++used;
    std::string result = "끝남";
    if (status == LUA_OK)
        app.undo.push_back(before);           // 스크립트 전체가 한 단계다
    else {
        const char* why = lua_tostring(co, -1);
        result = status == LUA_YIELD ? "조각을 다 써서 멈춤" : why ? why : "?";
        app.doc = before;                     // 한 일을 모두 되돌린다
    }
    std::println("{}: {}", name, result);
    std::println("  조각 {}개, 명령 {}번 -> 문서 {}, 실행 취소 {}단계", used,
                 app.calls, app.doc.text(), app.undo.size());
    lua_close(L);
}

int main() {
    App app;
    run_script(app, "도형", R"(local a = add("원")
local b = add("네모")
rename(a, "큰 원"))");
    run_script(app, "실수", R"(local a = add("세모")
remove(a)
add("별")
rename(a, "작은 세모"))");
    run_script(app, "무한", R"(while true do
  pcall(function() while true do add("점") end end)
end)");
    app.doc = app.undo.back(), app.undo.pop_back();
    std::println("실행 취소 한 번 -> 문서 {}", app.doc.text());
}
```

```text
도형: 끝남
  조각 1개, 명령 3번 -> 문서 [큰 원, 네모], 실행 취소 1단계
실수: 실수:4: 사라진 물체의 핸들
  조각 1개, 명령 4번 -> 문서 [큰 원, 네모], 실행 취소 1단계
무한: 조각을 다 써서 멈춤
  조각 50개, 명령 1249번 -> 문서 [큰 원, 네모], 실행 취소 1단계
실행 취소 한 번 -> 문서 []
```

- **명령 등록부로 들어온다**: 스크립트의 `add`, `remove`, `rename`은 등록부의 명령마다 하나씩 만든 함수이고, 모두 `l_command` 하나를 거칩니다. 메뉴가 부르는 것과 같은 명령이므로([「실제 프로그램들의 구조」](#/read/cpp-complex-app/ana-cases)) 인자를 살피는 곳이 하나입니다. 스크립트만 쓰는 뒷문을 따로 내면 그 문에서만 나는 버그가 생깁니다
- **스크립트 하나가 한 단계**: 도형 스크립트는 명령을 세 번 불렀지만 실행 취소는 1단계이고, 마지막 줄에서 한 번에 되돌아갔습니다. 사용자에게는 스크립트를 돌린 것이 한 번의 일이기 때문입니다
- **실패하면 없던 일로**: 실수 스크립트는 넷째 줄에서 죽었습니다. 앞의 세 명령이 문서를 고쳤지만 문서는 돌리기 전 그대로입니다. 반쯤 고친 문서를 남기지 않는 것은 [「트랜잭션」](#/read/cpp-complex-app/cmd-transaction)과 같고, 예제는 문서를 통째로 베껴 두는 것으로 대신했습니다
- **핸들에 세대를 둔다**: 넷째 줄의 `a`는 지워진 세모를 가리킵니다. 그 칸은 별이 다시 쓰고 있어서, 칸 번호만 보았다면 별의 이름이 바뀌었을 것입니다. 세대가 달라서 오류가 됐고, 오류에는 스크립트의 줄 번호가 붙었습니다
- **조각으로 돌린다**: 명령어 100개마다 훅이 `lua_yield`를 불러 본체로 돌아옵니다. 본체는 그 틈에 창의 사건을 처리하고 계속할지를 정합니다. 무한 스크립트는 조각 50개를 쓰고 멈췄고, 그동안 넣은 점 1249개는 되돌렸습니다. 훅에서 오류를 내서 멈추는 방법은 이 스크립트처럼 `pcall`로 감싸면 삼켜집니다. 다시 깨우지 않는 것은 스크립트가 막을 수 없습니다
- **longjmp와 소멸자**: Lua의 오류는 `longjmp`로 날아가서 C++ 소멸자를 건너뜁니다. 그래서 C++ 일은 `call` 안에서 모두 끝내고, `std::string` 같은 것이 하나도 없는 `l_command`에서 `lua_error`를 부릅니다. Lua를 C++로 빌드하면 오류가 예외로 바뀝니다

## 실제 프로그램

- **GIMP**: 묶는 일을 스크립트가 합니다. `gimp_image_undo_group_start`와 `gimp_image_undo_group_end` 사이의 일들이 실행 취소 한 단계가 되고, 시작한 만큼 끝을 불러야 합니다
- **Blender**: 파이썬 객체가 Blender의 메모리를 직접 가리킵니다. 그래서 지워지거나 다시 할당된 데이터를 들고 있다가 쓰면 죽을 수 있고, 실행 취소 뒤에는 들고 있던 참조가 모두 못 쓰게 됐다고 여기라고 문서가 말합니다. 접근할 때마다 검사하면 스크립트가 아주 느려진다는 것이 까닭입니다. 예제의 `at`이 그 검사이고, 값은 호출마다 치릅니다
- **기록**: Blender의 정보 편집기는 실행한 오퍼레이터를 기록하고, 고른 줄을 복사할 수 있게 합니다. 명령이 id와 인자로 되어 있으면 기록은 그것을 글로 적는 일입니다. 물체를 가리키는 인자는 핸들이 아니라 "지금 고른 것"이나 이름으로 적어야 다른 문서에서 다시 돌릴 수 있습니다

## 더 정할 것

- **훅이 닿지 않는 곳**: 횟수 훅은 Lua 함수가 도는 동안에만 불립니다. 본체의 명령 하나가 오래 걸리면 그 안에서 따로 취소를 봅니다([「백그라운드 작업과 취소」](#/read/cpp-complex-app/rsp-background)). 메모리는 `lua_newstate`에 주는 할당 함수에서 세어 한도를 둡니다
- **조각 사이의 편집**: 조각 사이에 사용자의 입력을 다 받으면 스크립트가 도는 중에 문서가 바뀝니다. 창은 다시 그리되 편집은 막고 취소만 받습니다
- **상태의 수명**: 예제는 스크립트마다 Lua 상태를 새로 만듭니다. 상태를 오래 두면 스크립트가 전역 변수에 핸들을 남깁니다. 그때는 문서를 되돌려도 세대가 뒤로 가지 않게 해야, 남은 핸들이 나중에 생긴 다른 물체를 가리키지 않습니다
- **열어 줄 것**: `base`에도 파일을 읽는 `dofile`과 `loadfile`, 바이트코드를 받는 `load`가 있어서 예제는 셋을 지웠습니다. 무엇을 막을지는 [「샌드박스」](#/read/cpp-complex-app/plg-sandbox)에서 봅니다

<Note>
스크립트에 내보낸 이름은 사용자의 매크로에 적히므로 [「확장 지점」](#/read/cpp-complex-app/plg-points)처럼 바꾸기 어려운 약속이 됩니다. 명령 등록부를 그대로 내보내면 지킬 약속이 따로 늘지 않습니다. 시험도 얻습니다. 스크립트로 문서를 만들고 결과를 견주는 시험은 화면을 눌러 보는 시험보다 훨씬 쌉니다([「테스트 전략」](#/read/cpp-complex-app/qa-tests)).
</Note>

## 정리

- 스크립트는 메뉴와 같은 명령 등록부로 들어오고, 한 번 돌린 것이 실행 취소 한 단계가 되며, 실패하면 모두 되돌린다
- 스크립트가 드는 것은 포인터가 아니라 세대가 붙은 핸들이다. 사라진 물체에 닿으면 죽지 않고 줄 번호가 붙은 오류가 난다
- 횟수 훅에서 양보하게 해 조각으로 돌리면 창이 얼지 않고, 스크립트가 막을 수 없는 방법으로 멈출 수 있다
