export const summary = '메뉴, 도구 모음, 단축키, 리본은 명령 표의 뷰: 상태로 계산하는 켜짐, 조건 붙은 단축키, 리본 줄이기';

같은 명령을 부르는 겉이 여럿입니다. 메뉴, 도구 모음, 단축키, 오른쪽 단추 메뉴, 리본, 명령 찾기 창입니다. 겉마다 이 명령을 지금 쓸 수 있는지를 따로 판단하면 여섯 곳이 서로 어긋납니다. [「실제 프로그램들의 구조」](#/read/cpp-complex-app/ana-cases)에서 만든 명령 표를 가운데에 두고, 겉은 모두 그 표를 읽는 뷰로 만듭니다.

```cpp
// 메뉴, 도구 모음, 단축키, 리본은 모두 명령 표를 보는 뷰다
#include <functional>
#include <map>
#include <print>
#include <ranges>
#include <string>
#include <vector>

struct State {                                   // 명령이 켜질지를 정하는 것들
    bool selection = false, read_only = false, clipboard = false;
    std::string focus = "canvas";
};

using S = const State&;
struct Command {
    std::string title;
    std::function<bool(S)> enabled;
};
const std::map<std::string, Command> commands = {
    {"copy", {"복사", [](S s) { return s.selection; }}},
    {"paste", {"붙이기", [](S s) { return s.clipboard && !s.read_only; }}},
    {"delete", {"지우기", [](S s) { return s.selection && !s.read_only; }}},
    {"save", {"저장", [](S s) { return !s.read_only; }}},
    {"erase_char", {"글자 지우기", [](S) { return true; }}},
};

// 메뉴와 도구 모음은 명령의 이름을 늘어놓은 데이터일 뿐이다
const std::vector<std::string> edit_menu = {"copy", "paste", "delete"};
const std::vector<std::string> toolbar = {"save", "copy", "paste"};

void render(const char* what, const std::vector<std::string>& ids, S s) {
    std::print("{:<10}", what);
    for (const std::string& id : ids) {
        const Command& c = commands.at(id);      // 켜졌는지는 표에 묻는다
        std::print(" {}", c.enabled(s) ? c.title : "(" + c.title + ")");
    }
    std::println("");
}

struct Binding {
    std::string key, command, when;              // when이 비면 어디서나
};
std::string resolve(const std::vector<Binding>& rules, const std::string& key,
                    S s) {
    for (const Binding& b : rules | std::views::reverse)  // 뒤의 규칙이 이긴다
        if (b.key == key && (b.when.empty() || b.when == s.focus))
            return commands.at(b.command).title;
    return "(없음)";
}

struct Group {
    std::string name;
    int width[4];                                // 크게, 중간, 작게, 단추 하나로
    int size = 0;
};
struct Step {
    int group, size;                             // 좁아지면 이 차례로 줄인다
};
void fit(std::vector<Group>& groups, const std::vector<Step>& policy, int width) {
    auto total = [&] {
        int sum = 0;
        for (const Group& g : groups) sum += g.width[g.size];
        return sum;
    };
    for (Group& g : groups) g.size = 0;
    for (const Step& s : policy) {
        if (total() <= width) break;
        groups[s.group].size = s.size;
    }
    std::print("너비 {:>2}:", width);
    for (const Group& g : groups) std::print(" {}[{}]", g.name, "LMSP"[g.size]);
    std::println("  ({}칸 씀)", total());
}

int main() {
    State s;
    render("편집 메뉴", edit_menu, s), render("도구 모음", toolbar, s);
    s.selection = s.clipboard = true;
    std::println("-- 고른 것이 생기고 클립보드가 찼다");
    render("편집 메뉴", edit_menu, s), render("도구 모음", toolbar, s);
    s.read_only = true;
    std::println("-- 읽기 전용 문서로 옮겨 갔다");
    render("편집 메뉴", edit_menu, s), render("도구 모음", toolbar, s);

    std::vector<Binding> keys = {{"Delete", "delete", "canvas"},
                                 {"Delete", "erase_char", "text"},
                                 {"Ctrl+S", "save", ""}};
    std::println("캔버스에서 Delete: {}", resolve(keys, "Delete", s));
    s.focus = "text";
    std::println("글 상자에서 Delete: {}", resolve(keys, "Delete", s));
    keys.push_back({"Ctrl+S", "copy", "text"});  // 사용자가 덧붙인 규칙
    std::println("글 상자에서 Ctrl+S: {}", resolve(keys, "Ctrl+S", s));

    std::vector<Group> ribbon = {{"클립보드", {14, 10, 6, 4}},
                                 {"글꼴", {24, 16, 10, 4}},
                                 {"문단", {20, 14, 8, 4}}};
    std::vector<Step> policy = {{2, 1}, {1, 1}, {0, 1}, {2, 2},
                                {1, 2}, {2, 3}, {1, 3}};
    for (int width : {60, 50, 40, 30, 20}) fit(ribbon, policy, width);
}
```

```text
편집 메뉴  (복사) (붙이기) (지우기)
도구 모음  저장 (복사) (붙이기)
-- 고른 것이 생기고 클립보드가 찼다
편집 메뉴  복사 붙이기 지우기
도구 모음  저장 복사 붙이기
-- 읽기 전용 문서로 옮겨 갔다
편집 메뉴  복사 (붙이기) (지우기)
도구 모음  (저장) 복사 (붙이기)
캔버스에서 Delete: 지우기
글 상자에서 Delete: 글자 지우기
글 상자에서 Ctrl+S: 복사
너비 60: 클립보드[L] 글꼴[L] 문단[L]  (58칸 씀)
너비 50: 클립보드[L] 글꼴[M] 문단[M]  (44칸 씀)
너비 40: 클립보드[M] 글꼴[M] 문단[M]  (40칸 씀)
너비 30: 클립보드[M] 글꼴[S] 문단[S]  (28칸 씀)
너비 20: 클립보드[M] 글꼴[P] 문단[P]  (18칸 씀)
```

- **겉은 이름의 목록이다**: 메뉴와 도구 모음은 명령의 이름을 늘어놓은 데이터입니다. 제목과 켜짐은 표에서 읽으므로, 복사는 두 곳에서 함께 켜지고 함께 꺼집니다
- **켜짐은 상태의 함수다**: "선택이 생기면 복사를 켠다" 같은 코드를 사건마다 흩어 두지 않습니다. 지금 상태에서 켜짐을 계산하는 함수 하나를 명령에 답니다. 읽기 전용 문서로 옮겨 가자 저장, 붙이기, 지우기가 함께 꺼졌습니다
- **단축키는 규칙의 목록이다**: 규칙은 키, 명령, 조건으로 이루어집니다. 같은 Delete가 초점이 어디냐에 따라 다른 명령이 됩니다. 뒤의 규칙이 이기게 해 두면, 사용자의 규칙을 뒤에 덧붙이는 것만으로 기본값을 덮습니다
- **리본은 너비에 맞춰 줄인다**: 묶음마다 크기 단계를 두고, 정해 둔 차례로 한 단계씩 줄여 가다가 들어맞으면 멈춥니다. 50칸에서는 글꼴과 문단이 중간으로, 20칸에서는 둘 다 단추 하나로 접혔습니다

## 겉이 정하는 것과 표에서 읽는 것

| 겉 | 겉이 정하는 것 | 표에서 읽는 것 |
| --- | --- | --- |
| 메뉴 | 차례와 구분선 | 제목, 켜짐, 단축키 표시 |
| 도구 모음 | 차례와 단추 크기 | 아이콘, 켜짐, 풍선 도움말 |
| 단축키 | 키와 조건 | 실행할 명령 |
| 리본 | 탭, 묶음, 줄이는 차례 | 제목, 아이콘, 켜짐 |

겉이 데이터이므로 사용자가 도구 모음을 고치거나 키를 바꾼 것은 파일 하나로 저장되고([「작업 공간 저장」](#/read/cpp-complex-app/win-workspace)), 확장이 더한 명령도 같은 데이터에 끼워 넣으면 됩니다([「확장 지점」](#/read/cpp-complex-app/plg-points)).

## 실제 틀

- **MFC**: 명령 하나에 메뉴 항목과 도구 단추가 함께 있으면 둘 다 같은 갱신 처리기로 갑니다. 처리기는 `ON_UPDATE_COMMAND_UI`로 잇습니다. 메뉴 항목은 메뉴가 펼쳐지기 바로 전에, 도구 모음의 단추는 프로그램이 한가할 때 갱신됩니다
- **VS Code**: 단축키 규칙은 아래에서 위로 살펴, 키와 `when` 조건이 함께 맞는 첫 규칙을 씁니다. 사용자 파일의 규칙은 기본 규칙의 맨 아래에 덧붙고, 명령 이름 앞에 `-`를 붙인 규칙은 그 묶임을 없앱니다
- **Windows 리본**: 기능과 그것을 드러내는 컨트롤을 따로 만들게 합니다. 묶음의 크기는 Large, Medium, Small, Popup이고, 좁아질 때 어느 묶음을 어느 크기로 줄일지는 `ScalingPolicy`에 적습니다. 다 줄여도 모자라면 스크롤이 생깁니다. 그림 같은 물체를 골랐을 때만 나타나는 탭도 둘 수 있습니다
- **Qt**: `QAction` 하나를 메뉴와 도구 모음에 함께 넣습니다([「메인 윈도우, 메뉴, 도구 모음, 도킹」](#/read/cpp-qt/wid-mainwindow))

## 켜짐을 언제 다시 계산하는가

상태가 바뀔 때마다 모든 명령의 켜짐을 다시 계산하면 명령이 수천 개인 프로그램은 느려집니다. MFC가 고른 답은 이렇습니다. 메뉴는 펼쳐질 때 그 안의 항목만 계산하고, 늘 보이는 도구 모음과 리본은 입력이 없는 틈에 계산합니다. 이렇게 하려면 켜짐 함수가 빨라야 하고 부작용이 없어야 합니다. 파일을 읽거나 문서 전체를 훑는 함수를 달면 메뉴가 굼뜨게 열립니다.

<Note>
글 상자에 초점이 있을 때 글자 키가 명령으로 가면 글을 칠 수 없습니다. 단축키 규칙의 조건이 이 일을 맡습니다. 한 글자짜리 단축키에는 글 입력 중이 아니라는 조건을 붙이고, 조합 중인 키는 규칙에 넘기지 않고 입력기에 먼저 줍니다([「입력기」](#/read/cpp-complex-app/txt-ime)).
</Note>

## 정리

- 메뉴, 도구 모음, 단축키, 리본은 명령의 이름을 늘어놓은 데이터이고, 제목과 켜짐은 명령 표에서 읽는다
- 켜짐은 사건마다 켜고 끄지 않고 상태에서 계산하며, 메뉴는 펼칠 때, 늘 보이는 겉은 한가할 때 다시 계산한다
- 단축키는 조건이 붙은 규칙의 목록으로 두어 뒤의 규칙이 이기게 하고, 리본은 정해 둔 차례로 묶음을 줄인다
