CPU로 그릴 때는 그리기 함수가 픽셀을 바로 채웁니다([「QPainter」](#/read/cpp-qt/paint-painter)). GPU는 삼각형과 텍스처밖에 모르고, 삼각형을 그리라는 호출은 하나하나가 비쌉니다. 그래서 GPU 2D 렌더러의 일은 둘입니다. 그리기 명령을 삼각형으로 바꾸는 것, 그리고 그린 순서를 지키면서 호출 수를 줄이는 것입니다.

```cpp
// 2D 그리기를 GPU에 넘기기: 모두 쿼드로 바꾸고 상태가 같은 것끼리 묶는다
#include <algorithm>
#include <print>
#include <string>
#include <vector>

struct Rect { int x0, y0, x1, y1; };             // 끝은 들어가지 않는다

bool overlap(Rect a, Rect b) {
    return a.x0 < b.x1 && b.x0 < a.x1 && a.y0 < b.y1 && b.y0 < a.y1;
}
Rect unite(Rect a, Rect b) {
    return {std::min(a.x0, b.x0), std::min(a.y0, b.y0), std::max(a.x1, b.x1),
            std::max(a.y1, b.y1)};
}

enum { Atlas = -1, Solid, Font, Icon };          // 텍스처 번호. 아이콘은 Icon부터
constexpr int kW = 56, kH = 5;

struct Quad {                                    // 삼각형 둘: 정점 넷, 인덱스 여섯
    Rect r;
    int texture;
    char ink;                                    // 화면 흉내에 찍을 글자
};

struct Batch {                                   // 그리기 호출 하나
    int texture;
    Rect bounds;
    std::vector<Quad> quads;
};

std::vector<Batch> batch(const std::vector<Quad>& quads, bool look_back,
                         bool careful = true) {
    std::vector<Batch> out;
    for (const Quad& q : quads) {
        Batch* home = nullptr;
        for (auto it = out.rbegin(); it != out.rend() && !home; ++it) {
            if (it->texture == q.texture) home = &*it;        // 상태가 같은 묶음
            else if (!look_back) break;                       // 바로 앞만 본다
            else if (careful && overlap(it->bounds, q.r)) break;  // 못 넘는다
        }
        if (!home) out.push_back({q.texture, q.r, {}}), home = &out.back();
        home->bounds = unite(home->bounds, q.r), home->quads.push_back(q);
    }
    return out;
}

std::string paint(const std::vector<Batch>& batches) {       // 묶음 순서대로 그린다
    std::string screen(kW * kH, ' ');
    for (const Batch& b : batches)
        for (const Quad& q : b.quads)
            for (int y = q.r.y0; y < q.r.y1; ++y)
                for (int x = q.r.x0; x < q.r.x1; ++x) screen[y * kW + x] = q.ink;
    return screen;
}

std::string names(const std::vector<Batch>& batches) {
    std::string s;
    for (const Batch& b : batches) {
        int t = b.texture;
        const char* kind = t == Atlas ? " 한 장" : t == Solid ? " 단색" : " 글자";
        s += t >= Icon ? " 아이콘" : kind;
        if (b.quads.size() > 1) s += std::format("x{}", b.quads.size());
    }
    return s;
}

int main() {
    std::vector<Quad> scene = {{{0, 0, kW, kH}, Solid, '.'}};  // 도구 모음의 바탕
    for (int i = 0; i < 6; ++i) {
        int x = 1 + i * 9;
        scene.push_back({{x, 1, x + 9, 4}, Solid, '-'});       // 단추의 바탕
        scene.push_back({{x + 1, 1, x + 4, 4}, Icon + i, char('a' + i)});
        scene.push_back({{x + 4, 2, x + 8, 3}, Font, 'T'});    // 단추의 글자
    }
    scene.push_back({{20, 0, 23, 2}, Solid, '!'});             // 아이콘을 덮는 배지
    scene.push_back({{21, 0, 22, 1}, Font, '3'});              // 배지의 숫자

    std::vector<Batch> each;
    for (const Quad& q : scene) each.push_back({q.texture, q.r, {q}});
    std::string truth = paint(each);
    for (int y = 0; y < kH; ++y) std::println("{}", truth.substr(y * kW, kW));
    std::size_t n = scene.size();
    std::println("쿼드 {}개: 정점 {}, 인덱스 {}", n, n * 4, n * 6);

    auto report = [&](const char* what, const std::vector<Batch>& b) {
        std::println("{}: 호출 {}, 그림 {}", what, b.size(),
                     paint(b) == truth ? "같음" : "다름");
        if (b.size() <= 10) std::println(" {}", names(b));
    };
    report("하나씩", each);
    report("이웃끼리 묶기", batch(scene, false));
    report("겹침을 안 보고 앞당기기", batch(scene, true, false));
    report("겹치지 않을 때만 앞당기기", batch(scene, true));
    for (Quad& q : scene)
        if (q.texture >= Icon) q.texture = Icon;               // 아이콘을 한 장에
    report("아이콘을 한 장에 모으고", batch(scene, true));
    for (Quad& q : scene) q.texture = Atlas;                   // 흰 픽셀과 글자까지
    report("모두 한 장에 모으고", batch(scene, true));
}
```


- **모두 쿼드로**: 바탕, 아이콘, 글자 하나가 저마다 삼각형 두 개입니다. 정점은 자리와 텍스처 좌표와 색을 들고 가고, 쿼드 21개가 정점 84개입니다
- **호출을 가르는 것은 상태**: 그리기 호출 하나 안에서는 텍스처, 셰이더, 섞는 방식, 자르는 영역이 같아야 합니다. 예제에서는 텍스처 하나로 줄였습니다
- **이웃끼리만 묶으면**: 단추마다 바탕, 아이콘, 글자가 번갈아 나오므로 21번이 20번이 될 뿐입니다
- **앞당겨 묶기**: 상태가 같은 묶음을 뒤에서부터 찾아 그리로 보냅니다. 그러면 그 사이의 묶음들보다 먼저 그려지므로, 그것들과 겹치지 않을 때만 됩니다. 겹침을 보지 않은 줄에서는 배지가 아이콘 밑에 깔려 그림이 달라졌습니다. 겹침을 보면 배지와 숫자가 새 묶음이 되어 10번입니다
- **경계 상자로 본다**: 겹침은 묶음의 쿼드를 모두 감싸는 사각형으로 따집니다. 넉넉하게 잡는 것이라 묶을 수 있는 것을 놓칠 수는 있어도 그림이 틀리지는 않습니다
- **아틀라스**: 아이콘 여섯 장을 한 텍스처에 모으면 상태가 같아져 5번이 됩니다. 글리프와 흰 픽셀 하나까지 같은 장에 두면, 단색은 흰 픽셀에 정점의 색을 곱해 그릴 수 있어 한 번입니다. 한 묶음 안에서는 순서가 그대로라 겹쳐도 됩니다

이렇게 묶을 수 있는 것은 그리기 함수가 GPU를 바로 부르지 않기 때문입니다. 명령을 목록에 적어 두었다가 프레임 끝에 추리고 묶어서 보냅니다. 3D 엔진도 그리기 호출을 상태로 정렬합니다([「컬링과 정렬」](#/read/cpp-game-engine/rend-culling)). 2D에는 그린 순서가 곧 결과라는 제약이 하나 더 붙습니다.

## 쿼드가 아닌 것들

- **곡선과 경로**: 삼각형으로 잘게 쪼개 다중 샘플링으로 가장자리를 다듬거나, CPU에서 그려 아틀라스에 올린 뒤 쿼드로 찍습니다. Skia의 새 백엔드는 앞의 것을 먼저 쓰고 안 될 때 뒤의 것으로 갑니다
- **글자**: 글리프를 미리 그려 텍스처에 모아 두고 글자마다 쿼드 하나를 찍습니다. WebRender는 글리프를 CPU에서 그려 GPU의 캐시에 올려 두고 프레임이 바뀌어도 다시 씁니다([「글꼴과 셰이핑」](#/read/cpp-complex-app/txt-shaping))
- **자르기**: 축에 나란한 사각형은 가위 사각형이면 됩니다. Direct2D 문서는 이런 때 레이어 대신 `PushAxisAlignedClip`을 쓰라고 권합니다. 그 밖의 모양은 스텐실이나 깊이 버퍼, 가림 텍스처가 맡습니다
- **묶음의 불투명도**: 겹친 여러 개를 한 덩어리로 반투명하게 하려면 먼저 다른 텍스처에 그리고 그 텍스처를 섞습니다. 텍스처를 새로 잡는 일이 비싸서, 서로 기대지 않는 것들은 한 텍스처에 모아 그립니다

## 실제 렌더러에서

- **Dear ImGui**: 그리기 목록 하나가 정점 버퍼, 인덱스 버퍼, 명령 목록입니다. 명령(`ImDrawCmd`)마다 자를 사각형, 텍스처, 인덱스 수가 있고, 백엔드는 명령마다 가위와 텍스처를 맞추고 인덱스 그리기를 한 번 부릅니다. 정점은 자리, 텍스처 좌표, 색 셋입니다
- **Direct2D**: `BeginDraw`와 `EndDraw` 사이의 명령을 모아 두었다가 `EndDraw` 때, `Flush`를 불렀을 때, 버퍼가 찼을 때 처리합니다. `Flush`는 직접 부르지 말라고 하고, 작은 비트맵 여럿은 큰 비트맵 한 장에 사이를 투명하게 띄워 모으라고 합니다
- **Skia**: 앞선 백엔드 Ganesh는 깊이 검사를 쓰지 않고 그린 순서를 그대로 지켰습니다. 새 백엔드 Graphite는 그리기마다 순서를 나타내는 z 값을 줍니다. 불투명한 그리기는 순서를 바꿔 상태 바꿈을 줄여도 깊이 버퍼가 결과를 지켜 줍니다. 반투명한 것은 여전히 뒤에서 앞으로 그립니다
- **WebRender**: 화면에 나오지 않는 것을 버리고, 셰이더가 같은 것끼리 묶고, 불투명한 것은 앞에서 뒤로, 반투명한 것은 뒤에서 앞으로 그립니다

<Note>
느릴 때는 무엇이 모자란지부터 봅니다. Direct2D 문서의 방법은 그리는 대상의 크기를 바꿔 보는 것입니다. 크기에 따라 빨라지고 느려지면 픽셀을 채우는 양이 병목이고, 그대로면 도형이 복잡한 것이 병목입니다. 호출 수는 이 예제처럼 프레임마다 세어 보면 됩니다.
</Note>

## 정리

- GPU 2D 렌더러는 그리기 명령을 쿼드로 바꾸고, 텍스처와 셰이더 같은 상태가 같은 것끼리 한 호출로 묶는다
- 순서를 바꿔 묶을 수 있는 것은 사이의 것들과 겹치지 않을 때뿐이고, 아틀라스는 상태를 같게 만들어 묶이는 범위를 넓힌다
- 불투명한 것은 깊이 버퍼를 쓰면 순서에서 풀려나고, 곡선과 글자는 삼각형으로 쪼개거나 미리 그려 둔 텍스처로 찍는다
