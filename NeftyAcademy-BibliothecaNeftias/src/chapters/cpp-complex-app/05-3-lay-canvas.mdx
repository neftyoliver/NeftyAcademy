그림, 지도, 도면은 화면보다 훨씬 큽니다. 캔버스는 그 가운데 화면에 걸친 조각만, 지금 배율에 맞는 해상도로 그립니다. [「그림의 자료 구조: 레이어 트리와 타일」](#/read/cpp-complex-app/doc-layers)의 타일은 문서가 픽셀을 간직하는 방식이었고, 여기의 타일은 화면에 내려고 그려 둔 캐시입니다.

```cpp
// 캔버스: 화면과 문서 좌표, 한 점을 붙든 확대, 단계별 타일과 캐시
#include <algorithm>
#include <cmath>
#include <list>
#include <map>
#include <print>
#include <set>
#include <string>
#include <vector>

struct Point { double x, y; };

struct View {
    double w, h;                                 // 화면 크기(픽셀)
    double zoom = 1;                             // 문서의 1이 화면에서 몇 픽셀인가
    Point origin{0, 0};                          // 화면 왼쪽 위에 놓인 문서 좌표
    Point to_doc(Point s) const {
        return {origin.x + s.x / zoom, origin.y + s.y / zoom};
    }
    void zoom_at(Point s, double factor) {       // s 밑의 문서 점은 그대로 둔다
        Point d = to_doc(s);
        zoom *= factor;
        origin = {d.x - s.x / zoom, d.y - s.y / zoom};
    }
    void drag(double dx, double dy) {            // 화면 픽셀만큼 문서를 끈다
        origin.x -= dx / zoom, origin.y -= dy / zoom;
    }
};

constexpr int kTile = 256, kDoc = 4096;          // 타일과 문서의 한 변
constexpr int kTop = 4;                          // 가장 거친 단계

struct Key {                                     // 단계 n은 2^n분의 1로 줄인 그림
    int level, x, y;
    auto operator<=>(const Key&) const = default;
};

int level_for(double zoom) {                     // 늘리지 않고 쓸 가장 거친 단계
    return std::clamp(static_cast<int>(std::floor(std::log2(1 / zoom))), 0, kTop);
}

class Cache {
public:
    explicit Cache(std::size_t limit) : limit_(limit) {}
    bool use(Key k) {                            // 있으면 방금 쓴 것으로 올린다
        if (k.level == kTop) return top_.contains(k);
        auto it = where_.find(k);
        if (it == where_.end()) return false;
        recent_.splice(recent_.begin(), recent_, it->second);
        return true;
    }
    void put(Key k) {
        if (k.level == kTop) { top_.insert(k); return; }  // 이 단계는 버리지 않는다
        recent_.push_front(k), where_[k] = recent_.begin();
        if (recent_.size() > limit_)             // 가장 오래 안 쓴 것을 버린다
            where_.erase(recent_.back()), recent_.pop_back(), ++dropped;
    }
    std::size_t size() const { return recent_.size() + top_.size(); }
    int dropped = 0;

private:
    std::size_t limit_;
    std::list<Key> recent_;
    std::map<Key, std::list<Key>::iterator> where_;
    std::set<Key> top_;
};

struct Frame {
    int sharp = 0, coarse = 0, drawn = 0;
    std::string picture;                         // #은 선명, 숫자는 거친 정도
};

Frame draw(const View& v, Cache& cache, std::size_t budget) {
    int level = level_for(v.zoom);
    int side = kTile << level;                   // 타일 한 장이 덮는 문서 길이
    auto first = [&](double a) { return std::max(0, int(std::floor(a / side))); };
    auto last = [&](double b) {
        return std::min(kDoc / side, int(std::ceil(b / side))) - 1;
    };
    int x0 = first(v.origin.x), x1 = last(v.origin.x + v.w / v.zoom);
    int y0 = first(v.origin.y), y1 = last(v.origin.y + v.h / v.zoom);
    Frame f;
    std::vector<Key> missing;
    for (int y = y0; y <= y1; ++y) {
        for (int x = x0; x <= x1; ++x) {
            Key k{level, x, y};
            if (cache.use(k)) { ++f.sharp, f.picture += '#'; continue; }
            missing.push_back(k);
            Key up = k;                          // 같은 자리를 덮는 더 거친 타일
            do up = {up.level + 1, up.x / 2, up.y / 2}; while (!cache.use(up));
            ++f.coarse, f.picture += static_cast<char>('0' + up.level - level);
        }
        if (y < y1) f.picture += ' ';
    }
    Point c = v.to_doc({v.w / 2, v.h / 2});      // 화면 가운데에서 가까운 것부터
    auto far = [&](Key k) {
        return std::hypot((k.x + 0.5) * side - c.x, (k.y + 0.5) * side - c.y);
    };
    std::ranges::stable_sort(missing, {}, far);
    for (std::size_t i = 0; i < std::min(budget, missing.size()); ++i)
        cache.put(missing[i]), ++f.drawn;        // 한 프레임에 budget장까지만
    return f;
}

void settle(const char* what, const View& v, Cache& cache) {
    std::println("{}: 배율 {:.3f}, 단계 {}", what, v.zoom, level_for(v.zoom));
    for (int n = 1;; ++n) {
        Frame f = draw(v, cache, 6);
        std::println("  프레임 {}: 선명 {:2}, 대신 {:2}, 새로 그림 {} | {}",
                     n, f.sharp, f.coarse, f.drawn, f.picture);
        if (f.drawn == 0) break;
    }
}

int main() {
    Cache cache(26);
    cache.put({kTop, 0, 0});                     // 열 때 전체를 한 장에 그려 둔다
    View v{640, 480};
    v.zoom = std::min(v.w, v.h) / kDoc;          // 문서 전체가 들어오는 배율
    v.origin = {(kDoc - v.w / v.zoom) / 2, (kDoc - v.h / v.zoom) / 2};
    View fit = v;
    settle("한눈에 보기", v, cache);

    Point cursor{180, 200}, before = v.to_doc(cursor);
    v.zoom_at(cursor, 2);
    Point after = v.to_doc(cursor);
    std::println("커서 밑의 문서 좌표 ({:.1f}, {:.1f}) -> ({:.1f}, {:.1f})",
                 before.x, before.y, after.x, after.y);
    settle("커서에서 2배", v, cache);
    v.zoom_at(cursor, 4);
    settle("커서에서 다시 4배", v, cache);
    v.drag(-300, 0);
    settle("왼쪽으로 300픽셀 끌기", v, cache);
    v.drag(300, 0);
    settle("도로 오른쪽으로", v, cache);
    settle("한눈에 보기로 돌아옴", fit, cache);
    std::println("캐시 {}장, 그동안 버린 것 {}장", cache.size(), cache.dropped);
}
```

- **두 좌표계**: 뷰는 배율과 원점, 두 값으로 문서 좌표와 화면 좌표를 오갑니다. 도구와 선택은 문서 좌표로 일하고, 화면 좌표는 그릴 때와 마우스 자리를 받을 때만 씁니다. 고해상도 화면의 배율은 여기에 한 번 더 곱해집니다([「고해상도」](#/read/cpp-complex-app/rnd-hidpi))
- **커서를 붙든 확대**: `zoom_at`은 커서 밑의 문서 점을 먼저 구하고, 배율을 바꾼 뒤 그 점이 같은 화면 자리에 오도록 원점을 다시 셈합니다. 출력의 좌표가 그대로입니다
- **줄여 둔 단계**: 문서 전체가 보이는 배율에서는 8분의 1로 줄인 단계 3의 네 장이면 됩니다. 단계 0뿐이라면 256장을 읽어 줄여야 합니다. 단계를 모두 둬도 타일은 256, 64, 16, 4, 1장이라 3분의 1이 더 들 뿐입니다. `level_for`는 타일을 늘리지 않고 쓸 수 있는 가장 거친 단계를 고르므로, 배율이 1 이하일 때 타일은 제 크기와 절반 사이로만 줄여 그려집니다
- **화면에 걸친 타일만**: 화면 사각형을 문서 좌표로 옮겨 타일의 한 변으로 나누면 그릴 범위가 나옵니다. 끌었을 때는 새로 드러난 세 장만 그렸고, 되돌아갔을 때는 그릴 것이 없었습니다
- **없는 타일은 거친 것으로**: 확대한 첫 프레임은 한두 단계 거친 타일을 늘려 채웁니다. 선명한 타일은 한 프레임에 여섯 장씩, 가운데에서 가까운 것부터 바꿔 끼웁니다. 화면은 잠깐 흐릴 뿐 비거나 멈추지 않습니다
- **캐시의 한도**: 26장을 넘자 가장 오래 안 쓴 것부터 버렸습니다. 한눈에 보기로 돌아왔을 때 두 장이 없었지만, 버리지 않는 가장 거친 한 장이 그 자리를 채웠습니다

## 실제 프로그램에서

- **지도**: Google 지도의 타일은 256×256 픽셀입니다. 확대 단계 0에서 세계가 한 장이고, 단계가 하나 오를 때마다 가로와 세로가 두 배가 됩니다. 픽셀 좌표를 256으로 나눈 몫이 타일 좌표입니다. 번호는 예제와 거꾸로여서 0이 가장 거친 쪽입니다
- **Deep Zoom**: 큰 그림을 단계마다 가로세로 절반으로 줄이고 256×256 타일로 잘라 둔 피라미드입니다. 멀리서 볼 때는 작은 그림 하나만, 확대했을 때는 그 자리의 타일만 받습니다
- **Chromium**: 배율마다 `PictureLayerTiling`이 있고, 타일은 그 배율로 그린 내용의 조각입니다. 화면에 필요한 타일과 화면 가까이의 덜 급한 타일을 모아 작업 스레드에서 그리고, 아직 시작하지 않은 작업은 새 계획에서 빼서 버립니다. 초기 설계 문서는 내용 없는 타일이 드러나면 체커보드를 그리고, 핀치 동안에는 그리는 배율이 알맞은 배율을 뒤따라간다고 적습니다
- **Krita**: Instant Preview는 캔버스의 작은 판에 먼저 그려 보여 주고 진짜 획은 뒤에서 셈합니다. 4K 화면에서 4K 그림을 100%로 볼 때는 빨라지지 않는다고 설명서가 적습니다. 화면이 그림의 픽셀을 다 보여 주고 있어 줄인 판을 쓸 수 없는 경우입니다

<Note>
문서가 바뀌면 그 자리를 덮는 타일을 모든 단계에서 버려야 합니다. 단계 0의 한 장이 바뀌면 그 위로 단계마다 한 장씩, 모두 다섯 장입니다. 무엇을 언제 버리고 다시 그리는지는 [「무효화와 다시 그리기」](#/read/cpp-complex-app/rnd-invalidate)에서, 그리기를 작업 스레드로 옮기고 필요 없어진 작업을 거두는 일은 [「백그라운드 작업과 취소」](#/read/cpp-complex-app/rsp-background)에서 다룹니다.
</Note>

## 정리

- 캔버스는 배율과 원점으로 문서 좌표와 화면 좌표를 오가고, 확대할 때는 커서 밑의 문서 점이 제자리에 있도록 원점을 다시 셈한다
- 문서를 절반씩 줄인 단계를 두고, 배율에 맞는 단계에서 화면에 걸친 타일만 그린다
- 없는 타일은 더 거친 것으로 먼저 채우고 가운데부터 바꿔 끼우며, 캐시는 오래 안 쓴 것부터 버리되 가장 거친 단계는 남긴다
