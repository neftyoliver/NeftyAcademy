SPI와 UART의 신호는 본래 칩 안의 전용 회로가 만듭니다. 그 회로 없이 프로그램이 핀을 하나하나 올리고 내려서 같은 파형을 그리는 것을 비트뱅잉이라고 합니다. libgpiod로도 할 수 있습니다. 되는지 안 되는지는 박자를 누가 쥐고 있느냐에 달렸습니다.

## 박자를 누가 쥐는가

| | 클럭 선이 있는 통신 | 시간이 약속인 통신 |
| --- | --- | --- |
| 보기 | SPI, 시프트 레지스터 | UART, 1-Wire |
| 받는 쪽이 읽는 때 | 클럭이 움직일 때 | 약속한 시간이 지났을 때 |
| 보내는 쪽이 늦으면 | 느려질 뿐이다 | 비트가 틀린다 |

- **클럭이 있으면 기다려 준다**: 74HC595 같은 시프트 레지스터는 클럭이 올라가는 순간에 데이터 선을 읽습니다. 자료에 적힌 것은 가장 짧은 시간뿐입니다. 2V로 쓸 때에도 클럭 펄스는 0.12µs, 데이터는 클럭보다 0.15µs 먼저면 됩니다. 그보다 느린 것은 얼마든지 받습니다
- **시간이 약속이면 기다려 주지 않는다**: UART의 선은 쉴 때 HIGH이고, LOW인 시작 비트가 한 바이트를 엽니다. 받는 쪽은 시작 비트가 내려간 때부터 시간을 세어, 비트마다 한가운데에서 선을 읽습니다. 9600bps면 비트 하나가 104µs입니다

## 같은 바이트를 두 가지로 보내기

아래 예제는 'A'(0x41)를 시프트 레지스터로 한 번, 9600bps UART로 한 번 보내고, 같은 일을 한 번 더 합니다. 받는 쪽은 흉내 칩에 달아 둔 장치들이고, 받은 것을 `장치|`로 시작하는 줄에 적습니다. 두 번째에는 대본이 보내는 도중에 한 번씩 0.15ms 늦게 깨웁니다.

```c gpiosim=bitbang
// 같은 바이트를 두 가지로 보낸다: 클럭 선과 함께(시프트 레지스터), 시간만으로(UART)
#define _POSIX_C_SOURCE 200809L              // clock_nanosleep을 쓰려고
#include <gpiod.h>
#include <stdint.h>
#include <stdio.h>
#include <time.h>

// 시프트 레지스터의 데이터, 클럭, 래치는 핀 15, 16, 18. UART의 TX는 핀 11
enum { DATA = 22, CLOCK = 23, LATCH = 24, TX = 17 };
static const int64_t SECOND = 1'000'000'000;
static const int64_t BIT = SECOND / 9600;    // 9600bps에서 비트 하나는 104µs
static int64_t worst;                        // 가장 늦게 깬 정도

static int64_t now_ns(void) {
    struct timespec now;
    clock_gettime(CLOCK_MONOTONIC, &now);
    return now.tv_sec * SECOND + now.tv_nsec;
}

static void sleep_until(int64_t when) {
    struct timespec at = {.tv_sec = when / SECOND, .tv_nsec = when % SECOND};
    clock_nanosleep(CLOCK_MONOTONIC, TIMER_ABSTIME, &at, nullptr);
    const int64_t late = now_ns() - when;
    if (late > worst) worst = late;
}

static void set(struct gpiod_line_request* lines, unsigned int offset, bool high) {
    gpiod_line_request_set_value(lines, offset, high ? GPIOD_LINE_VALUE_ACTIVE
                                                     : GPIOD_LINE_VALUE_INACTIVE);
}

// 클럭 선과 함께: 데이터를 먼저 놓고 클럭을 올린다. 받는 쪽은 그때 읽는다
static void shift_out(struct gpiod_line_request* lines, uint8_t byte, int64_t* at) {
    for (int i = 7; i >= 0; --i) {           // 높은 비트부터
        set(lines, DATA, byte >> i & 1);
        set(lines, CLOCK, true);
        sleep_until(*at += BIT / 2);
        set(lines, CLOCK, false);
        sleep_until(*at += BIT / 2);
    }
    set(lines, LATCH, true);                 // 여덟 비트를 출력 핀으로 옮긴다
    set(lines, LATCH, false);
}

// 시간만으로: 시작 비트(0), 낮은 비트부터 여덟, 정지 비트(1)를 104µs씩 둔다
static void uart_out(struct gpiod_line_request* lines, uint8_t byte, int64_t* at) {
    const unsigned int frame = (unsigned int)byte << 1 | 1u << 9;
    for (int i = 0; i < 10; ++i) {
        set(lines, TX, frame >> i & 1u);
        sleep_until(*at += BIT);
    }
}

static struct gpiod_line_request* request_lines(struct gpiod_chip* chip) {
    struct gpiod_line_settings* low = gpiod_line_settings_new();
    struct gpiod_line_settings* high = gpiod_line_settings_new();
    struct gpiod_line_config* config = gpiod_line_config_new();
    struct gpiod_line_request* request = nullptr;
    if (low && high && config) {
        gpiod_line_settings_set_direction(low, GPIOD_LINE_DIRECTION_OUTPUT);
        gpiod_line_settings_set_direction(high, GPIOD_LINE_DIRECTION_OUTPUT);
        gpiod_line_settings_set_output_value(high, GPIOD_LINE_VALUE_ACTIVE);
        const unsigned int shift[] = {DATA, CLOCK, LATCH};
        gpiod_line_config_add_line_settings(config, shift, 3, low);
        // UART의 선은 쉴 때 HIGH다
        gpiod_line_config_add_line_settings(config, &(unsigned int){TX}, 1, high);
        request = gpiod_chip_request_lines(chip, nullptr, config);
    }
    gpiod_line_config_free(config);
    gpiod_line_settings_free(high);
    gpiod_line_settings_free(low);
    return request;
}

int main(void) {
    struct gpiod_chip* chip = gpiod_chip_open("/dev/gpiochip0");
    if (!chip) {
        perror("/dev/gpiochip0");
        return 1;
    }
    struct gpiod_line_request* lines = request_lines(chip);
    gpiod_chip_close(chip);
    if (!lines) {
        perror("줄 요청");
        return 1;
    }
    int64_t at = now_ns() + 1'000'000;
    sleep_until(at);
    for (int round = 0; round < 2; ++round) {
        worst = 0;
        shift_out(lines, 'A', &at);
        sleep_until(at += 2 * BIT);
        uart_out(lines, 'A', &at);
        sleep_until(at += 2 * BIT);
        printf("0x41을 두 가지로 보냈다. 가장 늦게 깬 것은 %.3fms\n",
               (double)worst / 1e6);
    }
    gpiod_line_request_release(lines);
}
```

```text
장치|   0.001833s  시프트 레지스터의 출력 01000001 (0x41)
장치|   0.003031s  받은 바이트 0x41 'A'
0x41을 두 가지로 보냈다. 가장 늦게 깬 것은 0.000ms
장치|   0.004125s  시프트 레지스터의 출력 01000001 (0x41)
장치|   0.005323s  받은 바이트 0x43 'C'
0x41을 두 가지로 보냈다. 가장 늦게 깬 것은 0.150ms
```

- **늦지 않으면 둘 다 맞다**: 첫 번째에는 시프트 레지스터의 여덟 출력도, UART가 받은 바이트도 0x41입니다
- **시프트 레지스터는 늦어도 맞다**: 두 번째에도 0x41입니다. 클럭이 0.15ms 늦게 내려갔을 뿐이고, 클럭이 올라갈 때마다 데이터는 이미 놓여 있었습니다
- **UART는 비트 하나가 틀렸다**: 둘째 데이터 비트를 0으로 내리는 일이 0.15ms 늦었습니다. 받는 쪽은 그 비트의 한가운데에서 아직 앞 비트의 1을 읽었고, 0x41이 0x43 'C'가 되었습니다
- **틀린 줄도 모른다**: 정지 비트는 멀쩡해서 받는 쪽은 오류를 알아채지 못합니다. 보내는 쪽도 제 박자를 놓친 것을 깨고 나서야 압니다. 예제는 그것을 `worst`에 적어 두었습니다
- **데이터와 클럭은 따로 쓴다**: `shift_out`은 데이터를 쓴 다음에 클럭을 올립니다. 한 호출로 함께 써도 어느 핀이 먼저 바뀔지는 약속되어 있지 않습니다([「여러 줄 한꺼번에 쓰기」](#/read/libgpiod/out-multi))

## 진짜 시계로 센 틀린 바이트

`uart_out`의 반복문을 값 쓰기 없이 이 책을 쓴 컴퓨터에서 돌리고, 받는 쪽이 위와 같은 방법으로 읽는다고 치고 틀리는 바이트를 셌습니다.

| 속도와 기다리는 방법 | 한가할 때 | CPU가 바쁠 때 |
| --- | --- | --- |
| 300bps, 잠 | 0.4% | 4.6% |
| 1200bps, 잠 | 2.2% | 14% |
| 9600bps, 잠 | 6.9% | 35% |
| 9600bps, 자지 않고 기다림 | 2.6% | 68% |
| 9600bps, 잠, `chrt -f 50` | 4.0% | 3.8% |

- **느릴수록 덜 틀리지만 0이 되지 않는다**: 300bps에서도 500바이트에 둘이 틀렸습니다. 통신에서는 가끔 틀리는 것도 쓸 수 없습니다
- **자지 않고 기다리는 방법**: 시계를 계속 읽으며 때를 기다리면 한가할 때에는 덜 틀립니다. 그러나 CPU를 다투는 프로그램이 있으면 잠보다 나쁩니다. 쉬지 않고 도는 프로그램은 CPU를 많이 쓴 프로그램이어서 차례를 오래 내줍니다
- **실시간 정책으로 쉬지 않고 돌면**: 커널이 1초마다 0.05초씩 그 프로그램을 세웁니다([「사용자 공간에서 얼마나 정확한가」](#/read/libgpiod/tm-jitter)). 이 책이 잰 컴퓨터에서 그 멈춤은 42ms에서 52ms였습니다
- **어느 칸도 0이 아니다**: 실시간 정책으로 잠을 자는 것이 가장 고르지만, 그래도 100바이트에 넷이 틀렸습니다

## 해도 되는 때

- **클럭을 내가 쥐고, 느려도 되는 장치**: 시프트 레지스터로 출력 핀을 늘리는 일, 키패드를 훑는 일이 그렇습니다([「키패드 훑기」](#/read/libgpiod/pj-keypad))
- **잠이 없어도 될 만큼 느리다**: 값을 한 번 쓰는 것은 시스템 호출 한 번이고, 이 책이 잰 컴퓨터에서 가벼운 시스템 호출 하나가 0.2µs쯤이었습니다. 74HC595가 바라는 시간보다 깁니다. 예제의 잠은 UART와 박자를 맞추려고 넣은 것입니다. 보드에서 값 쓰기가 얼마나 걸리는지는 재지 못했습니다
- **시간이 약속인 통신은 하지 않는다**: UART는 헤더의 UART로 보내고([「핀의 대체 기능: I2C, SPI, UART, PWM」](#/read/sbc-start/hdr-functions)), 1-Wire 같은 것은 커널의 드라이버에 맡깁니다([「커널 드라이버에 맡길 것들」](#/read/libgpiod/tm-kernel))
- **받는 쪽도 같다**: 시간이 약속인 신호를 읽으려면 정한 때에 선을 읽어야 하고, 늦게 깨면 다른 비트를 읽습니다. 에지 이벤트의 타임스탬프로 폭을 재는 길은 느린 신호에만 통합니다([「초음파 거리 센서」](#/read/libgpiod/pj-ultrasonic))

<Note>
예제는 이 책의 흉내 칩에서 돌린 것입니다. 받는 장치 둘도 흉내 칩이 흉내 낸 것입니다. 시프트 레지스터는 클럭이 올라갈 때 데이터를 읽고 래치가 올라갈 때 여덟 출력을 적으며, UART는 시작 비트부터 1.5, 2.5, ... 비트 뒤에 선을 한 번씩 읽습니다. 진짜 UART는 한 비트를 여러 번 읽어 가릅니다. 늦게 깨는 것은 대본으로 넣었습니다. 표의 숫자는 라즈베리파이가 아니라 CPU가 둘인 x86-64 가상 머신에서, GPIO 없이 값을 쓸 시각만 적어서 센 것입니다. 진짜 libgpiod 2.2.4의 머리 파일로 x86-64와 aarch64에서 컴파일되는지 확인했고, 보드와 진짜 칩으로 해 보지는 못했습니다.
</Note>

## 정리

- 클럭 선이 있는 통신은 보내는 쪽이 늦어도 느려질 뿐이다. 시간이 약속인 통신은 비트가 틀린다
- 9600bps의 비트 하나는 104µs다. 앞에서 잰 늦음과 같은 크기여서, 잠으로는 지킬 수 없다
- 비트뱅잉은 클럭을 내가 쥐는 느린 장치에만 쓴다. 나머지는 전용 회로와 커널 드라이버의 일이다
