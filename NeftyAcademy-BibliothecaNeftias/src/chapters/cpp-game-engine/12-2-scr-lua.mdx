게임의 규칙과 적의 행동은 끝없이 고쳐집니다. 그때마다 C++를 다시 빌드하고 게임을 다시 켜면 디자이너가 기다리다 지칩니다. 그래서 많은 엔진이 가벼운 **스크립트 언어**를 품고, 게임 쪽 논리는 스크립트로 쓰게 합니다. Lua는 C로 된 작은 인터프리터를 라이브러리처럼 링크해 쓰는 언어라서, 오래전부터 게임에 많이 쓰여 왔습니다.

## Lua 품기

```cpp flags=-I/usr/include/lua5.4,-llua5.4
#include <lua.hpp>
#include <cstring>
#include <print>
#include <string>

struct Enemy { std::string name; float x = 0; int hp = 100; };
Enemy goblin{"고블린", 0, 30};

// ---- 스크립트에 내보내는 C++ 함수: 스택에서 인자를 꺼내고 결과를 올린다 ----
int l_move(lua_State* L) {
    float dx = float(luaL_checknumber(L, 1));     // 숫자가 아니면 오류
    goblin.x += dx;
    lua_pushnumber(L, goblin.x);
    return 1;                                     // 돌려준 값 개수
}
int l_hp(lua_State* L) { lua_pushinteger(L, goblin.hp); return 1; }
int l_say(lua_State* L) {
    std::println("  [{}] {}", goblin.name, luaL_checkstring(L, 1));
    return 0;
}

// 스크립트 오류가 엔진을 죽이지 않게 보호 호출로 부른다
bool call_update(lua_State* L, double dt) {
    lua_getglobal(L, "update");
    if (!lua_isfunction(L, -1)) { lua_pop(L, 1); return false; }
    lua_pushnumber(L, dt);
    if (lua_pcall(L, 1, 0, 0) != LUA_OK) {
        std::println("  스크립트 오류: {}", lua_tostring(L, -1));
        lua_pop(L, 1);
        return false;
    }
    return true;
}

bool load(lua_State* L, const char* src) {
    // "=적.lua"는 오류 메시지에 쓰일 덩어리 이름
    if (luaL_loadbuffer(L, src, std::strlen(src), "=적.lua") != LUA_OK
        || lua_pcall(L, 0, 0, 0) != LUA_OK) {
        std::println("  불러오기 실패: {}", lua_tostring(L, -1));
        lua_pop(L, 1);
        return false;
    }
    return true;
}

int main() {
    lua_State* L = luaL_newstate();
    luaopen_base(L);                              // 필요한 라이브러리만 연다
    lua_pop(L, 1);                                // (io, os는 열지 않는다)
    lua_register(L, "move", l_move);
    lua_register(L, "hp", l_hp);
    lua_register(L, "say", l_say);

    load(L, R"(
        local t = 0
        function update(dt)
            t = t + dt
            local x = move(2 * dt)
            if t >= 0.5 and t - dt < 0.5 then say("반쯤 왔다! x=" .. x) end
        end
    )");
    std::println("-- 첫 스크립트");
    for (int f = 0; f < 60; ++f) call_update(L, 1.0 / 60);
    std::println("  1초 뒤 x={:.2f}", goblin.x);

    std::println("-- 스크립트 고쳐 다시 불러오기(엔진은 그대로)");
    load(L, R"(
        function update(dt)
            if hp() < 50 then move(-5 * dt) else move(2 * dt) end
        end
    )");
    for (int f = 0; f < 30; ++f) call_update(L, 1.0 / 60);
    std::println("  체력 {}이라 도망: x={:.2f}", goblin.hp, goblin.x);

    std::println("-- 실수가 든 스크립트");
    load(L, "function update(dt) move('빨리') end");
    call_update(L, 1.0 / 60);
    load(L, "function update(dt) os.exit(1) end");
    call_update(L, 1.0 / 60);
    load(L, "function update(dt) move(1 end");
    std::println("  엔진은 살아 있다, x={:.2f}", goblin.x);
    lua_close(L);
}
```

## 스택으로 주고받기

C++와 Lua는 **가상 스택**으로 값을 주고받습니다. Lua가 `move(2 * dt)`를 부르면 인자가 스택에 올라온 채로 `l_move`가 불리고, `l_move`는 `luaL_checknumber`로 첫 인자를 꺼낸 뒤 결과를 `lua_pushnumber`로 올리고, 돌려준 값의 개수 1을 반환합니다. 반대로 엔진이 스크립트의 `update`를 부를 때는 함수를 스택에 올리고, 인자를 올리고, 호출합니다.

첫 스크립트는 1초 동안 초당 2만큼 움직여 x=2.00에 도착했습니다. 중간의 "반쯤 왔다"가 31번째 프레임(x = 31 × 2/60 = 1.033…)에 나온 것은, 1/60을 30번 더한 값이 부동소수점으로 0.5에 조금 못 미쳤기 때문입니다. 스크립트도 [「게임 루프와 고정 시간 간격」](#/read/cpp-game-engine/arch-loop)의 시간 누적 문제에서 자유롭지 않습니다. 또 Lua의 숫자를 그대로 문자열로 이으면 14자리까지 찍혀 나옵니다.

## 고치고 다시 불러오기

두 번째 스크립트를 불러오자 `update`가 새 함수로 바뀌었고, 엔진은 멈추지 않았습니다. 체력이 50보다 낮은 고블린은 이제 도망칩니다. 편집기에서 스크립트 파일이 바뀔 때마다 이렇게 다시 불러오면, 게임을 켠 채로 행동을 고칠 수 있습니다. [「동적 라이브러리와 핫 리로드」](#/read/cpp-game-engine/plat-hot-reload)의 C++ 버전보다 훨씬 간단합니다. 다만 스크립트가 들고 있던 상태(첫 스크립트의 `t`)는 함께 사라지므로, 남겨야 할 상태는 엔진 쪽이나 따로 둔 표에 둡니다.

## 오류 가두기와 샌드박스

- 스크립트 오류가 엔진을 죽이면 안 됩니다. `lua_pcall`(보호 호출)은 스크립트 안의 오류를 잡아 오류 코드와 메시지로 돌려줍니다. 숫자 자리에 문자열을 넘긴 실수는 `luaL_checknumber`가 잡아 "bad argument"로 알렸습니다
- 문법 오류는 불러오는 순간 `luaL_loadbuffer`에서 걸립니다. 실패한 불러오기는 기존 `update`를 건드리지 않아서, 고블린은 직전 스크립트대로 계속 움직일 수 있습니다
- `luaL_openlibs`로 모든 표준 라이브러리를 열지 않고 `base`만 열었습니다. 그래서 `os.exit`는 `os`가 없다는 오류가 되었습니다. 모드 제작자나 내려받은 레벨의 스크립트를 돌리는 게임이라면 파일, 프로세스, 동적 로딩 기능은 반드시 막아야 합니다

<Note>
함수를 하나씩 `lua_register`로 내보내는 일은 [「리플렉션 만들기」](#/read/cpp-game-engine/scr-reflection)의 정보로 자동화할 수 있습니다. 필드와 메서드 목록을 보고 Lua 쪽 표를 만들어 주면, 새 컴포넌트를 만들 때 연결 코드를 따로 쓰지 않아도 됩니다. sol2 같은 라이브러리는 템플릿으로 이 연결을 간단히 써 줍니다. 스크립트가 너무 느린 부분은 C++로 옮기는데, 그 경계를 어디에 두느냐는 매 프레임 스택을 오가는 호출 수를 보고 정합니다.
</Note>

## 정리

- 스크립트 언어는 빌드 없이 게임 논리를 고치게 해 주며, Lua는 작은 인터프리터를 라이브러리로 품는다
- C++와 Lua는 가상 스택으로 값을 주고받고, 엔진은 스크립트 함수를 보호 호출로 불러 오류를 가둔다
- 다시 불러오기로 실행 중에 행동을 고치고, 필요한 라이브러리만 열어 스크립트를 샌드박스에 가둔다
