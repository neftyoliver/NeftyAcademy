export const summary = '상태 복제: 서버 상태를 클라이언트에 보내기, 확인된 스냅숏과의 차이와 양자화, 패킷 손실 견디기';

멀티플레이 게임에서 진짜 세계는 **서버**에 하나 있고, 클라이언트들은 그 복사본을 화면에 보여 줍니다. 서버가 매 틱 세계의 상태를 클라이언트에 보내는 것이 **상태 복제**입니다. 모든 엔티티의 모든 값을 매번 보내면 대역폭이 감당되지 않고, 인터넷은 패킷을 잃어버리기도 합니다. 이 장은 보내는 양을 줄이면서도 손실에 버티는 방법을 봅니다.

## 엔티티 100개, 60틱

엔티티 100개 중 매 틱 20개가 움직이고 가끔 체력이 깎입니다. 열 번에 한 번 패킷이 사라집니다.

```cpp
#include <algorithm>
#include <cmath>
#include <cstdint>
#include <cstring>
#include <print>
#include <vector>

using Bytes = std::vector<std::uint8_t>;
struct Ent { float x, y; std::uint16_t hp; };
std::uint32_t rng = 99;
std::uint32_t next() {
    rng ^= rng << 13; rng ^= rng >> 17; rng ^= rng << 5;
    return rng;
}

// 서버 상태 한 틱 진행: 100개 중 20개만 움직이고, 가끔 맞는다
void tick(std::vector<Ent>& w) {
    for (int k = 0; k < 20; ++k) {
        auto& e = w[next() % w.size()];
        e.x += float(int(next() % 2000) - 1000) / 1000;
        e.y += float(int(next() % 2000) - 1000) / 1000;
    }
    if (next() % 4 == 0) w[next() % w.size()].hp -= 10;
}

// 1) 전체 스냅숏: 모든 엔티티를 그대로
Bytes full(const std::vector<Ent>& w) {
    Bytes b(w.size() * sizeof(Ent));
    std::memcpy(b.data(), w.data(), b.size());
    return b;
}

// 2) 기준 스냅숏과의 차이만: [번호][바뀐 필드 비트][필드들]
//    위치는 1/100 단위 정수로 양자화해 2바이트에 담는다
std::int16_t q(float v) { return std::int16_t(std::lround(v * 100)); }
Bytes delta(const std::vector<Ent>& base, const std::vector<Ent>& w) {
    Bytes b;
    auto put16 = [&](std::uint16_t v) {
        b.push_back(v & 0xff);
        b.push_back(v >> 8);
    };
    for (std::size_t i = 0; i < w.size(); ++i) {
        int mask = (q(w[i].x) != q(base[i].x))
                 | (q(w[i].y) != q(base[i].y)) << 1
                 | (w[i].hp != base[i].hp) << 2;
        if (!mask) continue;
        b.push_back(std::uint8_t(i));
        b.push_back(std::uint8_t(mask));
        if (mask & 1) put16(std::uint16_t(q(w[i].x)));
        if (mask & 2) put16(std::uint16_t(q(w[i].y)));
        if (mask & 4) put16(w[i].hp);
    }
    return b;
}
void apply(std::vector<Ent>& w, const Bytes& b) {
    auto get16 = [&](std::size_t& p) {
        std::uint16_t v = b[p] | b[p + 1] << 8;
        p += 2;
        return v;
    };
    for (std::size_t p = 0; p < b.size();) {
        auto& e = w[b[p]]; std::uint8_t mask = b[p + 1]; p += 2;
        if (mask & 1) e.x = std::int16_t(get16(p)) / 100.f;
        if (mask & 2) e.y = std::int16_t(get16(p)) / 100.f;
        if (mask & 4) e.hp = get16(p);
    }
}

struct Result { std::size_t full, delta; int lost; float worst; };

// use_ack: 클라이언트가 확인한 상태를 기준으로 할지,
//          마지막으로 보낸 상태를 기준으로 할지
Result run(bool use_ack) {
    rng = 99;
    std::vector<Ent> server(100);
    for (auto& e : server) e = {float(next() % 100), float(next() % 100), 100};
    std::vector<Ent> client = server, base = server;
    Result r{0, 0, 0, 0};
    for (int t = 1; t <= 60; ++t) {             // 60틱
        tick(server);
        r.full += full(server).size();
        auto pkt = delta(base, server);
        r.delta += pkt.size();
        bool lost = t % 10 == 3;                // 열 번에 한 번 패킷 손실
        if (!use_ack) base = server;            // 보냈으니 받았겠지(?)
        if (lost) { ++r.lost; continue; }       // 확인도 오지 않는다
        apply(client, pkt);
        if (use_ack) base = server;             // 확인이 와야 새 기준
    }
    for (std::size_t i = 0; i < server.size(); ++i)
        r.worst = std::max({r.worst, std::fabs(client[i].x - server[i].x),
                             std::fabs(client[i].y - server[i].y)});
    return r;
}

int main() {
    for (bool ack : {true, false}) {
        auto r = run(ack);
        std::println("{}: 전체 {}B, 차이 {}B ({:.1f}%), "
                     "손실 {}개, 최대 오차 {:.3f}",
                     ack ? "확인된 상태 기준" : "보낸 상태 기준  ",
                     r.full, r.delta, 100.0 * r.delta / r.full, r.lost, r.worst);
    }
}
```

```text
확인된 상태 기준: 전체 72000B, 차이 7114B (9.9%), 손실 6개, 최대 오차 0.005
보낸 상태 기준  : 전체 72000B, 차이 6584B (9.1%), 손실 6개, 최대 오차 0.980
```

## 줄이기: 차이와 양자화

- **차이만 보내기**: 기준 스냅숏과 비교해 바뀐 엔티티의 바뀐 필드만, 번호와 "어느 필드인지" 비트 표시를 붙여 보냅니다. 움직이지 않은 80개는 한 바이트도 쓰지 않습니다
- **양자화**: 위치를 4바이트 `float` 대신 0.01 단위 정수 2바이트로 보냅니다. 화면에서 0.01 차이는 보이지 않으므로 정밀도를 버려도 됩니다. 클라이언트의 오차는 최대 반 칸, 곧 0.005였습니다
- 둘을 합쳐 전체 스냅숏의 약 10%로 줄었습니다. 실제로는 범위를 알고 있는 값(각도, 체력)은 필요한 비트 수만큼만 쓰는 비트 단위 압축을 더하고, 회전은 쿼터니언의 가장 큰 성분을 빼고 나머지 셋만 보내는 식의 기법을 씁니다

## 견디기: 확인된 상태를 기준으로

두 줄의 차이는 **무엇을 기준으로 차이를 계산했느냐**뿐입니다.

- **보낸 상태 기준**은 "보냈으니 받았겠지" 하고 방금 보낸 상태를 다음 기준으로 삼습니다. 그 패킷이 사라지면, 그 틱에 바뀐 값은 다음 차이에 다시 실리지 않아 클라이언트가 영영 모릅니다. 오차가 0.98까지 쌓였고, 시간이 갈수록 커집니다
- **확인된 상태 기준**은 클라이언트가 "이 틱까지 받았다"고 확인을 보낸 스냅숏을 기준으로 삼습니다. 패킷이 사라지면 확인이 오지 않으므로, 다음 차이는 더 오래된 기준과 비교되어 놓친 변화까지 함께 실립니다. 그래서 차이가 조금 더 컸지만(7114 대 6584바이트), 오차는 양자화 오차 0.005에서 멈췄습니다

서버는 클라이언트마다 확인된 틱을 기억하고, 최근 몇십 틱의 스냅숏을 고리 버퍼에 보관해 둡니다. 확인이 너무 오래 오지 않으면 전체 스냅숏을 다시 보냅니다. 재전송을 기다리지 않고 늘 최신 상태를 보내므로, 이 방식은 순서와 도착을 보장하는 TCP 대신 UDP 위에서 돕니다.

<Note>
엔티티가 수천 개인 세계에서는 차이만 보내도 모자랍니다. 그래서 클라이언트마다 **관심 영역**을 두어 가까운 것, 보이는 것, 중요한 것만 보내고, 먼 것은 덜 자주 보냅니다. ## 정리

- 서버가 진짜 상태를 갖고 매 틱 클라이언트에 복제하며, 바뀐 필드만 보내고 위치는 양자화해 전송량을 크게 줄인다
- 차이는 클라이언트가 확인한 스냅숏을 기준으로 계산해야 패킷을 잃어도 놓친 변화가 다음에 함께 실린다
- 서버는 최근 스냅숏을 보관하고 UDP로 늘 최신 상태를 보내며, 엔티티가 많으면 관심 영역으로 거른다
