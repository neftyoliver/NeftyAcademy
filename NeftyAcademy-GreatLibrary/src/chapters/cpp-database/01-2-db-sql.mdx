**SQL**은 데이터베이스에 말하는 언어입니다. 무엇을 원하는지만 적고 어떻게 찾을지는 데이터베이스가 정하는 선언형 언어라서, 같은 질의도 데이터와 인덱스에 따라 전혀 다른 방법으로 실행됩니다([「실행 계획 읽기」](#/read/cpp-database/dbperf-plan)). 테이블을 만들고 바꾸는 문장(`CREATE`, `ALTER`, `DROP`)과 데이터를 읽고 고치는 문장(`SELECT`, `INSERT`, `UPDATE`, `DELETE`)이 있고, 국제 표준이 있지만 데이터베이스마다 사투리가 조금씩 다릅니다.

## 고르고, 잇고, 묶기

`SELECT`는 적는 순서와 생각하는 순서가 다릅니다. 데이터베이스는 `FROM`과 `JOIN`으로 테이블을 이어 붙이고, `WHERE`로 행을 거르고, `GROUP BY`로 묶고, `HAVING`으로 그룹을 거른 뒤, `SELECT`의 열을 계산하고, 마지막에 `ORDER BY`로 줄을 세운다고 생각하면 됩니다. `JOIN`은 양쪽에서 짝이 맞는 행만 남기고, `LEFT JOIN`은 왼쪽 행을 모두 남기며 짝이 없으면 오른쪽 열을 NULL로 채웁니다.

[「테이블과 키」](#/read/cpp-database/db-tables)의 쇼핑몰에 데이터를 조금 넣고 질의해 보았습니다. 김하늘은 다른 두 손님을 추천했고, 자신의 추천인은 비어 있습니다.

```cpp flags=-lsqlite3
#include <sqlite3.h>
#include <cstdio>

// 결과를 표처럼 찍는다
void run(sqlite3* db, const char* title, const char* sql) {
    std::printf("[%s]\n", title);
    bool header = true;
    auto row = [](void* h, int n, char** val, char** col) {
        bool& first = *static_cast<bool*>(h);
        for (int pass = first ? 0 : 1; pass < 2; ++pass) {
            for (int i = 0; i < n; ++i)
                std::printf("%s%s", i ? " | " : "  ",
                            pass == 0 ? col[i] : val[i] ? val[i] : "NULL");
            std::printf("\n");
        }
        first = false;
        return 0;
    };
    char* err = nullptr;
    if (sqlite3_exec(db, sql, row, &header, &err) != SQLITE_OK) {
        std::printf("  오류: %s\n", err);
        sqlite3_free(err);
    }
    if (header && *title) std::printf("  (결과 없음)\n");
}

int main() {
    sqlite3* db;
    sqlite3_open(":memory:", &db);
    sqlite3_exec(db, R"(
        CREATE TABLE customers (id INTEGER PRIMARY KEY, name TEXT,
                                referrer INTEGER);
        CREATE TABLE products (id INTEGER PRIMARY KEY, name TEXT, price INTEGER);
        CREATE TABLE orders (id INTEGER PRIMARY KEY, customer_id INTEGER);
        CREATE TABLE order_items (order_id INTEGER, product_id INTEGER,
                                  qty INTEGER);
        INSERT INTO customers VALUES (1, '김하늘', NULL), (2, '이바다', 1),
                                     (3, '박들', 1);
        INSERT INTO products VALUES (1, '볼트', 120), (2, '너트', 80),
                                    (3, '와셔', 30), (4, '스패너', 9000),
                                    (5, '렌치', 15000);
        INSERT INTO orders VALUES (10, 1), (11, 1), (12, 2);
        INSERT INTO order_items VALUES (10, 1, 100), (10, 2, 100), (11, 4, 1),
                                       (12, 1, 10), (12, 3, 50);
    )", nullptr, nullptr, nullptr);

    run(db, "100원 이상인 상품, 비싼 순",
        "SELECT name, price FROM products WHERE price >= 100 ORDER BY price DESC");
    run(db, "주문마다 손님과 합계 (JOIN, GROUP BY)", R"(
        SELECT o.id AS 주문, c.name AS 손님, sum(i.qty * p.price) AS 합계
        FROM orders o
        JOIN customers c ON c.id = o.customer_id
        JOIN order_items i ON i.order_id = o.id
        JOIN products p ON p.id = i.product_id
        GROUP BY o.id ORDER BY o.id)");
    run(db, "손님마다 주문 수, 주문 없는 손님도 (LEFT JOIN)", R"(
        SELECT c.name, count(o.id) AS 주문수
        FROM customers c LEFT JOIN orders o ON o.customer_id = c.id
        GROUP BY c.id ORDER BY c.id)");
    run(db, "2만 원 넘게 산 손님 (HAVING)", R"(
        SELECT c.name, sum(i.qty * p.price) AS 합계
        FROM customers c JOIN orders o ON o.customer_id = c.id
        JOIN order_items i ON i.order_id = o.id
        JOIN products p ON p.id = i.product_id
        GROUP BY c.id HAVING sum(i.qty * p.price) > 20000)");
    run(db, "한 번도 팔리지 않은 상품 (NOT EXISTS)", R"(
        SELECT name FROM products p
        WHERE NOT EXISTS (SELECT 1 FROM order_items i WHERE i.product_id = p.id))");
    run(db, "추천인이 없는 손님: = NULL",
        "SELECT name FROM customers WHERE referrer = NULL");
    run(db, "추천인이 없는 손님: IS NULL",
        "SELECT name FROM customers WHERE referrer IS NULL");
    run(db, "아무도 추천하지 않은 손님: NOT IN", R"(
        SELECT name FROM customers
        WHERE id NOT IN (SELECT referrer FROM customers))");
    run(db, "아무도 추천하지 않은 손님: NOT EXISTS", R"(
        SELECT name FROM customers c
        WHERE NOT EXISTS (SELECT 1 FROM customers r WHERE r.referrer = c.id))");
    sqlite3_exec(db, "UPDATE products SET price = price + 10 WHERE price < 100",
                 nullptr, nullptr, nullptr);
    std::printf("[100원 미만 상품 10원 올리기] 바뀐 행 %d개\n",
                sqlite3_changes(db));
    sqlite3_close(db);
}
```

```text
  name | price
  렌치 | 15000

  11 | 김하늘 | 9000

  이바다 | 1
  박들 | 0
[2만 원 넘게 산 손님 (HAVING)]

  김하늘 | 29000

  김하늘

```

- **JOIN과 GROUP BY**: 주문 하나에 주문 줄이 여럿이므로 이어 붙이면 줄 수만큼 행이 생깁니다. `GROUP BY o.id`로 주문마다 한 행으로 묶어 합계를 냈습니다
- **LEFT JOIN과 count**: 박들은 주문이 없어도 나왔습니다. `count(o.id)`는 NULL을 세지 않아 0이 되었지만, `count(*)`로 썼다면 LEFT JOIN이 채운 빈 행을 세어 1이 됩니다
- **WHERE와 HAVING**: `WHERE`는 묶기 전의 행을, `HAVING`은 묶은 뒤의 그룹을 거릅니다. 합계로 거르는 조건은 `HAVING`에 씁니다
- **NULL은 값이 아니다**: NULL은 모른다는 뜻이라 `referrer = NULL`은 참도 거짓도 아닌 모름이 되어 아무 행도 고르지 않았습니다. `IS NULL`을 씁니다. 더 무서운 것은 `NOT IN`입니다. 추천인 목록에 김하늘의 NULL이 섞이자 목록에 없다는 판단이 모두 모름이 되어 결과가 통째로 비었습니다. NULL이 있을 수 있는 열에는 `NOT EXISTS`를 씁니다

## 고치기

`INSERT`, `UPDATE`, `DELETE`는 바꾼 행의 수를 돌려줍니다(SQLite는 `sqlite3_changes`). 이 수를 확인하는 습관이 중요합니다. `WHERE`를 빠뜨린 `UPDATE`는 모든 행을 바꾸고, 하나만 바뀌어야 할 곳에서 0이 나오면 찾던 행이 없었다는 뜻입니다. 여러 문장을 한 덩어리로 고칠 때는 [「트랜잭션과 격리 수준」](#/read/cpp-database/db-transactions)으로 묶습니다.

<Note>
사투리의 예를 들면, 앞에서 몇 행만 가져오는 것은 SQLite, PostgreSQL, MySQL에서 `LIMIT 10`, SQL Server에서 `TOP 10`이고, 표준은 `FETCH FIRST 10 ROWS ONLY`입니다. 문자열은 작은따옴표, 이름은 큰따옴표로 감싸는 것이 표준이지만 MySQL은 이름에 역따옴표를 씁니다. 어느 데이터베이스든 사용자 입력을 SQL 문자열에 이어 붙이면 안 됩니다([「SQL 주입과 매개변수 바인딩」](#/read/cpp-database/injection)).
</Note>

## 정리

- SQL은 원하는 것만 적는 선언형 언어이며, SELECT는 FROM과 JOIN, WHERE, GROUP BY, HAVING, SELECT, ORDER BY 순서로 생각한다
- LEFT JOIN은 짝이 없는 왼쪽 행도 NULL로 채워 남기므로 count는 열을 지정해 세고, 묶은 뒤의 조건은 HAVING에 쓴다
- NULL은 모른다는 뜻이라 = NULL은 아무것도 고르지 않고 NOT IN은 목록에 NULL이 섞이면 결과가 비므로, IS NULL과 NOT EXISTS를 쓴다
