SQLite는 작지만 C와 C++에서 넓힐 수 있습니다. SQL에서 부를 함수와 정렬 규칙을 더하고, 프로그램의 자료 구조를 테이블처럼 보이게 하고(가상 테이블), 공유 라이브러리로 된 확장을 불러올 수 있습니다. JSON 함수, 전문 검색(FTS5), 공간 인덱스(R*Tree), 수학 함수 같은 확장은 SQLite에 함께 들어 있지만 빌드에 따라 빠질 수 있으므로 `PRAGMA compile_options`로 확인합니다.

## 함수 더하기

`sqlite3_create_function_v2`로 C 함수를 SQL 함수로 등록합니다. 인자는 `sqlite3_value_*`로 읽고 결과는 `sqlite3_result_*`로 돌려줍니다. 행 하나에 값 하나를 내는 스칼라 함수 말고도, 여러 행을 모으는 집계 함수(단계마다 부르는 함수와 마무리 함수)와 창 함수도 만들 수 있습니다. 정렬 규칙은 `sqlite3_create_collation_v2`로 더하는데, 예를 들어 "부품2"가 "부품10"보다 앞에 오는 자연 정렬을 만들 수 있습니다.

아래는 한글 이름을 초성으로 바꾸는 함수를 C++로 짜서 등록하고, 함께 들어 있는 JSON과 FTS5를 써 본 것입니다.

```cpp flags=-lsqlite3
#include <sqlite3.h>
#include <cstdio>
#include <string>

// 한글 음절을 초성으로 바꾼다: "볼트" -> "ㅂㅌ" (나머지 글자는 그대로)
std::string initials(const std::string& s) {
    static const char32_t cho[] = U"ㄱㄲㄴㄷㄸㄹㅁㅂㅃㅅㅆㅇㅈㅉㅊㅋㅌㅍㅎ";
    std::string out;
    for (size_t i = 0; i < s.size();) {
        unsigned char c = s[i];
        int len = c < 0x80 ? 1 : c < 0xE0 ? 2 : c < 0xF0 ? 3 : 4;
        char32_t cp = len == 1 ? c : c & (0x3F >> (len - 1));
        for (int k = 1; k < len; ++k) cp = (cp << 6) | (s[i + k] & 0x3F);
        if (cp >= 0xAC00 && cp <= 0xD7A3) {    // 가..힣: 초성 = (cp - 가) / 588
            char32_t j = cho[(cp - 0xAC00) / 588];
            out += char(0xE0 | (j >> 12));
            out += char(0x80 | ((j >> 6) & 0x3F));
            out += char(0x80 | (j & 0x3F));
        } else {
            out.append(s, i, len);
        }
        i += len;
    }
    return out;
}

void sql_initials(sqlite3_context* ctx, int, sqlite3_value** argv) {
    auto p = reinterpret_cast<const char*>(sqlite3_value_text(argv[0]));
    if (!p) return sqlite3_result_null(ctx);
    sqlite3_result_text(ctx, initials(p).c_str(), -1, SQLITE_TRANSIENT);
}

void show(sqlite3* db, const char* title, const char* sql, int from = 0) {
    std::printf("[%s]\n", title);
    struct Out { int from, rows = 0; } out{from};
    sqlite3_exec(db, sql, [](void* p, int n, char** v, char**) {
        auto& o = *static_cast<Out*>(p);
        std::printf(" ");
        for (int i = o.from; i < n; ++i) std::printf(" %s", v[i] ? v[i] : "NULL");
        std::printf("\n");
        ++o.rows;
        return 0;
    }, &out, nullptr);
    if (out.rows == 0) std::printf("  (찾지 못함)\n");
}

int main() {
    sqlite3* db;
    sqlite3_open(":memory:", &db);
    sqlite3_create_function_v2(db, "initials", 1,       // 같은 입력엔 늘 같은 결과
                               SQLITE_UTF8 | SQLITE_DETERMINISTIC, nullptr,
                               sql_initials, nullptr, nullptr, nullptr);
    sqlite3_exec(db, R"(
        CREATE TABLE parts (name TEXT, spec TEXT);
        INSERT INTO parts VALUES
          ('육각 볼트', '{"size": "M5", "len": 20}'),
          ('육각 너트', '{"size": "M5"}'),
          ('스프링 와셔', '{"size": "M6"}');
        CREATE INDEX parts_cho ON parts(initials(name));
        CREATE VIRTUAL TABLE notes_word USING fts5(body);
        CREATE VIRTUAL TABLE notes_tri USING fts5(body, tokenize = 'trigram');
    )", nullptr, nullptr, nullptr);
    for (std::string t : {"notes_word", "notes_tri"}) {
        std::string sql = "INSERT INTO " + t + " VALUES "
                          "('볼트의 체결력을 높이려면 와셔를 쓴다'),"
                          "('너트가 풀리면 다시 조인다')";
        sqlite3_exec(db, sql.c_str(), nullptr, nullptr, nullptr);
    }

    show(db, "초성으로 찾기: ㅇㄱ ㅂㅌ",
         "SELECT name FROM parts WHERE initials(name) = 'ㅇㄱ ㅂㅌ'");
    show(db, "그 질의의 계획", "EXPLAIN QUERY PLAN SELECT name FROM parts "
                              "WHERE initials(name) = 'ㅇㄱ ㅂㅌ'", 3);
    show(db, "JSON: 크기가 M5인 부품과 길이",
         "SELECT name, spec->>'$.len' FROM parts "
         "WHERE spec->>'$.size' = 'M5'");
    show(db, "단어 토크나이저로 '체결력' 찾기",
         "SELECT body FROM notes_word WHERE notes_word MATCH '체결력'");
    show(db, "트라이그램 토크나이저로 '체결력' 찾기",
         "SELECT highlight(notes_tri, 0, '[', ']') FROM notes_tri "
         "WHERE notes_tri MATCH '체결력'");
    sqlite3_close(db);
}
```

```text
[초성으로 찾기: ㅇㄱ ㅂㅌ]
  육각 볼트
[그 질의의 계획]
  SEARCH parts USING INDEX parts_cho (<expr>=?)
[JSON: 크기가 M5인 부품과 길이]
  육각 볼트 20
  육각 너트 NULL
[단어 토크나이저로 '체결력' 찾기]
  (찾지 못함)
[트라이그램 토크나이저로 '체결력' 찾기]
  볼트의 [체결력]을 높이려면 와셔를 쓴다
```

- **초성 함수**: 한글 음절은 유니코드에 가(U+AC00)부터 차례로 놓여 있고, 초성 하나에 중성 21개 × 종성 28개 = 588자씩 들어 있어서 (코드 − 0xAC00) / 588이 초성의 번호입니다. 휴대폰 주소록처럼 "ㅇㄱ ㅂㅌ"로 찾을 수 있습니다
- **결정적 함수와 식 인덱스**: 같은 입력에 늘 같은 결과를 내는 함수에 `SQLITE_DETERMINISTIC`을 주면, 그 함수를 씌운 식에 인덱스를 만들 수 있습니다. 계획이 `SEARCH ... USING INDEX parts_cho (<expr>=?)`로 그 인덱스를 썼습니다. 이 표시가 없는 함수로는 인덱스를 만들지 못합니다
- **JSON**: 3.38부터 JSON 함수가 기본으로 들어 있고, `->>`는 JSON 안의 값을 SQL 값으로 꺼냅니다. 키가 없으면 NULL입니다. 자주 찾는 키는 `spec->>'$.size'` 같은 식에 인덱스를 만듭니다
- **한국어 전문 검색**: FTS5의 기본 토크나이저는 띄어쓰기와 문장 부호로 단어를 자르므로 "체결력을"이 한 단어가 되어 '체결력'을 찾지 못했습니다. 한국어는 조사가 붙기 때문에 생기는 일입니다. 트라이그램 토크나이저는 세 글자씩 겹쳐 인덱스에 넣어 문자열 속의 '체결력'을 찾았지만, 세 글자보다 짧은 검색어는 찾지 못하고 인덱스가 커집니다

## 불러오는 확장과 가상 테이블

`sqlite3_load_extension`은 공유 라이브러리로 된 확장을 실행 중에 불러옵니다. 기본으로 꺼져 있으며, `sqlite3_db_config`의 `SQLITE_DBCONFIG_ENABLE_LOAD_EXTENSION`은 C API로 부르는 것만 켜고 SQL의 `load_extension()`은 막아 둡니다. 남이 준 SQL이 아무 라이브러리나 불러오면 안 되기 때문입니다. `sqlite3_create_module`로 등록하는 **가상 테이블**은 C++의 자료 구조나 다른 파일을 테이블처럼 질의하게 해 주며, FTS5와 R*Tree도 가상 테이블로 만들어져 있습니다.

<Note>
남이 만든 데이터베이스 파일을 열 수 있는 프로그램이라면, 그 파일의 스키마(뷰, 트리거, 인덱스)가 내 함수를 부를 수 있다는 점을 생각해야 합니다. 부작용이 있는 함수는 `SQLITE_DIRECTONLY`로 등록해 스키마 안에서는 못 쓰게 하고, 해가 없는 함수만 `SQLITE_INNOCUOUS`로 표시합니다. 한국어 검색의 질을 높이려면 형태소 분석기를 토크나이저로 붙이거나 별도의 검색 엔진을 씁니다.
</Note>

## 정리

- sqlite3_create_function_v2로 C++ 함수를 SQL 함수로 더하고, 결정적 함수는 식 인덱스에 쓸 수 있다
- SQLite에는 JSON, FTS5, R*Tree 같은 확장이 함께 들어 있으며, FTS5의 기본 토크나이저는 조사가 붙는 한국어에 약해 트라이그램이나 형태소 분석이 필요하다
- 공유 라이브러리 확장과 가상 테이블로 더 넓힐 수 있지만, 확장 불러오기는 C API로만 열고 남의 데이터베이스가 부를 수 있는 함수는 권한을 제한한다
