export const summary = '모니터링: application_name, pg_stat_activity, pg_stat_statements, 느린 쿼리, 복제, 클라이언트 지표';

모니터링은 사용자가 알려 주기 전에 문제를 먼저 보는 일입니다. 서버 쪽에서는 지금 누가 무엇을 하며 무엇을 기다리는지, 어떤 쿼리가 시간을 가장 많이 쓰는지, 복제가 얼마나 늦는지를 보고, 프로그램 쪽에서는 쿼리 지연과 오류, 연결 풀의 상태를 잽니다.

## 누가 무엇을 기다리나

아래 프로그램은 `web-1`, `web-2`, `monitor`라는 이름표를 단 연결 셋으로 잠금 대기를 만들어 지켜본 뒤, 같은 데이터를 N+1 꼴과 한 번에 가져오는 꼴로 읽고 pg_stat_statements에 남은 수를 봅니다. pg_stat_statements는 `shared_preload_libraries`에 넣고 서버를 다시 시작한 뒤 `CREATE EXTENSION pg_stat_statements`를 해야 쓸 수 있습니다.

```cpp pkg=libpq
#include <libpq-fe.h>
#include <chrono>
#include <cstdio>
#include <string>
#include <thread>

// application_name으로 이름표를 단 연결
PGconn* open_as(const char* app) {
    std::string s = std::string("host=localhost dbname=academy user=academy "
                                "password=academy application_name=") + app;
    PGconn* c = PQconnectdb(s.c_str());
    PQclear(PQexec(c, "SET client_min_messages = warning"));
    return c;
}

void run(PGconn* c, const char* sql) { PQclear(PQexec(c, sql)); }

std::string one(PGconn* c, const char* sql, const std::string& arg) {
    const char* v[] = {arg.c_str()};
    PGresult* r = PQexecParams(c, sql, 1, nullptr, v, nullptr, nullptr, 0);
    std::string s = PQntuples(r) > 0 ? PQgetvalue(r, 0, 0) : "";
    PQclear(r);
    return s;
}

const char* kActivity = R"(
SELECT a.application_name, a.state,
       coalesce(a.wait_event_type || '/' || a.wait_event, '-'),
       coalesce((SELECT string_agg(b.application_name, ',')
                 FROM pg_stat_activity b
                 WHERE b.pid = ANY (pg_blocking_pids(a.pid))), '-'),
       a.query
FROM pg_stat_activity a
WHERE a.application_name LIKE 'web-%'
ORDER BY a.application_name)";

// 같은 꼴(정규화된 원문)의 쿼리가 지금까지 불린 횟수와 돌려준 행 수
const char* kStatements = "SELECT coalesce(sum(calls), 0) || ' ' || "
                          "coalesce(sum(rows), 0) FROM pg_stat_statements "
                          "WHERE query = $1";

int main() {
    PGconn* w1 = open_as("web-1");
    PGconn* w2 = open_as("web-2");
    PGconn* m = open_as("monitor");
    run(m, "DROP TABLE IF EXISTS acct, mon_parts");
    run(m, "CREATE TABLE acct (id int PRIMARY KEY, bal int)");
    run(m, "INSERT INTO acct VALUES (1, 100)");
    run(m, "CREATE TABLE mon_parts AS "
           "SELECT i AS id, 'p' || i AS name FROM generate_series(1, 100) i");
    run(m, "ALTER TABLE mon_parts ADD PRIMARY KEY (id)");

    run(w1, "BEGIN");
    run(w1, "UPDATE acct SET bal = bal - 10 WHERE id = 1");   // 커밋하지 않고 쉰다
    PQsendQuery(w2, "UPDATE acct SET bal = bal + 10 WHERE id = 1");
    std::string pid = std::to_string(PQbackendPID(w2));
    while (one(m, "SELECT cardinality(pg_blocking_pids($1::int))", pid) == "0")
        std::this_thread::sleep_for(std::chrono::milliseconds(10));

    std::printf("application_name이 web-로 시작하는 연결 (pg_stat_activity)\n");
    PGresult* r = PQexec(m, kActivity);
    for (int i = 0; i < PQntuples(r); ++i)
        std::printf("  %-6s %-20s %-19s 막는 쪽: %s\n         %s\n",
                    PQgetvalue(r, i, 0), PQgetvalue(r, i, 1), PQgetvalue(r, i, 2),
                    PQgetvalue(r, i, 3), PQgetvalue(r, i, 4));
    PQclear(r);

    run(w1, "ROLLBACK");
    PGresult* done = PQgetResult(w2);
    std::printf("web-1이 롤백하자 web-2도 끝남: %s\n", PQcmdStatus(done));
    PQclear(done);
    PQclear(PQgetResult(w2));                     // 끝을 알리는 nullptr

    const char* q1 = "SELECT name FROM mon_parts WHERE id = $1";
    const char* q2 = "SELECT name FROM mon_parts WHERE id = ANY ($1::int[])";
    std::string before1 = one(m, kStatements, q1);
    std::string before2 = one(m, kStatements, q2);
    std::string all = "{";
    for (int id = 1; id <= 100; ++id) {           // N+1 꼴: 한 행에 한 번씩
        std::string s = std::to_string(id);
        one(w1, q1, s);
        all += (id > 1 ? "," : "") + s;
    }
    all += "}";
    one(w1, q2, all);                             // 한 번에 100행
    std::printf("같은 꼴끼리 모은 수 (pg_stat_statements, 이번에 늘어난 몫)\n");
    for (auto [q, before] : {std::pair{q1, before1}, std::pair{q2, before2}}) {
        long c0 = 0, r0 = 0, c1 = 0, r1 = 0;
        std::sscanf(before.c_str(), "%ld %ld", &c0, &r0);
        std::sscanf(one(m, kStatements, q).c_str(), "%ld %ld", &c1, &r1);
        std::printf("  호출 %3ld번, 행 %3ld개: %s\n", c1 - c0, r1 - r0, q);
    }
    run(m, "DROP TABLE acct, mon_parts");
    for (PGconn* c : {w1, w2, m}) PQfinish(c);
}
```

```text
application_name이 web-로 시작하는 연결 (pg_stat_activity)
  web-1  idle in transaction  Client/ClientRead   막는 쪽: -
         UPDATE acct SET bal = bal - 10 WHERE id = 1
  web-2  active               Lock/transactionid  막는 쪽: web-1
         UPDATE acct SET bal = bal + 10 WHERE id = 1
web-1이 롤백하자 web-2도 끝남: UPDATE 1
같은 꼴끼리 모은 수 (pg_stat_statements, 이번에 늘어난 몫)
  호출 100번, 행 100개: SELECT name FROM mon_parts WHERE id = $1
  호출   1번, 행 100개: SELECT name FROM mon_parts WHERE id = ANY ($1::int[])
```

- **이름표**: 연결 문자열의 `application_name`은 `pg_stat_activity`와 서버 로그(`log_line_prefix`의 `%a`)에 나와, 수많은 연결 가운데 어느 프로그램의 것인지 가려 줍니다. 서비스와 역할(웹, 배치, 마이그레이션)마다 다른 이름을 붙입니다
- **상태와 기다림**: `web-1`은 트랜잭션을 연 채 프로그램의 다음 명령을 기다리며(`idle in transaction`, `Client/ClientRead`) 행 잠금을 쥐고 있고, `web-2`는 그 트랜잭션이 끝나기를 기다립니다(`Lock/transactionid`). 막는 쪽은 `pg_blocking_pids`가 바로 알려 주고, 잠금의 자세한 모습은 `pg_locks`에 있습니다([「교착 상태」](#/read/cpp-database/tx-deadlock))
- **query는 마지막 쿼리**: 쉬고 있는 세션의 `query` 열은 지금 도는 쿼리가 아니라 마지막으로 실행한 쿼리입니다. web-1의 UPDATE는 이미 끝났고 지금은 아무 일도 하지 않으면서 잠금만 쥐고 있습니다
- **같은 꼴끼리 모으기**: `pg_stat_statements`는 값을 `$1`처럼 바꾼 원문별로 호출 수, 행 수, 시간(`total_exec_time`, `mean_exec_time`), 버퍼 사용을 모읍니다. N+1 꼴은 호출 100번, 한 번에 가져온 꼴은 1번으로 잡혔습니다([「N+1 문제」](#/read/cpp-database/dbperf-nplus1)). 서버 안의 시간만 재므로 왕복에 든 시간은 보이지 않아서, `total_exec_time`이 큰 순서와 함께 `calls`가 유난히 많은 순서로도 봅니다. 값은 쌓이기만 하므로 전후의 차이를 보거나 `pg_stat_statements_reset()`으로 지웁니다

`idle in transaction`으로 오래 머무는 세션은 다른 쿼리를 막을 뿐 아니라, 그동안 생긴 죽은 행을 VACUUM이 치우지 못하게 해 테이블을 부풀립니다. `xact_start`가 오래된 세션을 찾아 알리고, `idle_in_transaction_session_timeout`으로 쉬는 트랜잭션을, PostgreSQL 17부터는 `transaction_timeout`으로 트랜잭션 전체 길이를 끊습니다. 끊긴 프로그램에게는 연결이 끊긴 것으로 보입니다([「재연결과 타임아웃」](#/read/cpp-database/pool-reconnect)).

## 느린 쿼리, 복제, 테이블

`log_min_duration_statement = '500ms'`로 두면 그보다 오래 걸린 문장이 서버 로그에 남습니다. `auto_explain`을 `session_preload_libraries`에 넣고 `auto_explain.log_min_duration`을 주면 느린 쿼리의 실행 계획까지 남아, 나중에 다시 돌려서는 재현되지 않는 느림을 잡습니다([「실행 계획 읽기」](#/read/cpp-database/dbperf-plan)). 잠금을 `deadlock_timeout`보다 오래 기다린 일은 `log_lock_waits`가 남깁니다(19부터 기본으로 켜짐).

운영 서버의 `pg_stat_replication`은 대기 서버마다 `write_lag`, `flush_lag`, `replay_lag`를 보여 주고, `pg_replication_slots`의 `wal_status`와 `safe_wal_size`는 슬롯이 붙잡은 WAL이 위험한 수준인지 알려 줍니다([「복제」](#/read/cpp-database/dbops-replica)). 테이블별로는 `pg_stat_user_tables`가 순차 스캔 수, 죽은 행 수(`n_dead_tup`), 마지막 autovacuum 시각을, `pg_stat_user_indexes`가 인덱스별 `idx_scan`을 셉니다. 슈퍼유저가 아닌 감시용 역할에는 `pg_monitor`를 주면 다른 사용자의 세션과 통계까지 볼 수 있습니다.

MySQL과 MariaDB에서는 `SHOW PROCESSLIST`가 pg_stat_activity, `performance_schema`의 `events_statements_summary_by_digest`가 pg_stat_statements, `sys.innodb_lock_waits`가 막는 쪽을 보여 주고, 느린 쿼리 로그(`long_query_time`)가 log_min_duration_statement 구실을 합니다.

## 프로그램 쪽에서 잴 것

서버 통계는 서버 안의 시간만 압니다. 사용자가 느끼는 시간과 실패는 프로그램에서 잽니다.

- **쿼리 지연**: 쿼리 종류별로 평균보다 p50, p99 같은 백분위를 봅니다. 평균은 드물게 아주 느린 요청을 가립니다
- **오류**: SQLSTATE별로 셉니다. 40001과 40P01의 재시도, 57014 타임아웃, 08로 시작하는 연결 오류가 갑자기 늘면 알립니다
- **연결 풀**: 쓰는 연결과 쉬는 연결의 수, 빌리려고 기다린 시간과 기다리는 수를 봅니다([「연결 풀」](#/read/cpp-database/pool-pool))

이 값들은 Prometheus나 OpenTelemetry 같은 지표 체계로 내보내고, 느린 쿼리 로그에 요청의 추적 ID를 함께 남겨 서버 쪽 기록과 이어 봅니다.

<Note>
`IN (1, 2, 3)`처럼 값을 늘어놓은 쿼리는 예전 판의 pg_stat_statements에서 목록 길이마다 다른 꼴로 갈라져, 같은 코드가 수십 줄로 흩어졌습니다. PostgreSQL 18부터는 상수 목록을 한 꼴로 묶지만, 위 프로그램처럼 배열 하나를 `= ANY ($1::int[])`로 넘기면 판과 상관없이 늘 한 꼴이고 준비된 문장도 하나로 충분합니다.
</Note>

## 정리

- application_name으로 연결에 이름을 달고, pg_stat_activity의 state와 wait_event, pg_blocking_pids로 누가 무엇을 기다리는지 보며, 쉬는 세션의 query는 마지막 쿼리임을 기억한다
- pg_stat_statements로 같은 꼴의 쿼리를 모아 시간과 호출 수가 많은 쿼리를 찾고, 느린 쿼리 로그와 auto_explain, 복제와 테이블 통계 뷰를 함께 본다
- 사용자가 느끼는 지연과 오류, 연결 풀의 기다림은 프로그램에서 백분위와 SQLSTATE별 개수로 잰다
