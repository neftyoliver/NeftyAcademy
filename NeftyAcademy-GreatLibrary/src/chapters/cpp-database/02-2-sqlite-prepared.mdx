SQLite는 SQL 문장을 받으면 해석해서 내부 가상 기계가 실행할 작은 프로그램(바이트코드)으로 컴파일합니다. **준비된 문장**은 이렇게 컴파일해 둔 문장입니다. 값이 들어갈 자리는 `?` 같은 자리 표시자로 비워 두고, 실행할 때마다 값만 묶어(바인딩) 넣습니다. 해석을 한 번만 하니 빠르고, 값은 SQL로 해석되지 않으니 안전하며([「SQL 주입과 매개변수 바인딩」](#/read/cpp-database/injection)), 정수는 정수로, NULL은 NULL로 정확한 타입을 지킵니다.

## 한 문장의 일생

`sqlite3_prepare_v2`로 준비하고, `sqlite3_bind_*`로 값을 묶고, `sqlite3_step`을 부릅니다. `step`은 결과 행이 있으면 `SQLITE_ROW`를, 끝나면 `SQLITE_DONE`을 돌려주고, 행이 있는 동안 `sqlite3_column_*`으로 열을 읽습니다. 다 쓰면 `sqlite3_reset`으로 처음 상태로 되돌려 다른 값으로 다시 쓰고, 마지막에 `sqlite3_finalize`로 정리합니다. 자리 표시자는 `?`, `?3`처럼 번호를 붙인 것, `:이름`, `@이름`, `$이름`을 쓸 수 있습니다.

```cpp flags=-lsqlite3
#include <sqlite3.h>
#include <cstdio>
#include <stdexcept>
#include <string>
#include <string_view>

class Stmt {                                  // 준비된 문장 하나를 RAII로
public:
    Stmt(sqlite3* db, const char* sql) : db_(db) {
        check(sqlite3_prepare_v2(db, sql, -1, &st_, nullptr));
    }
    ~Stmt() { sqlite3_finalize(st_); }
    Stmt(const Stmt&) = delete;
    Stmt& operator=(const Stmt&) = delete;

    template <class... A> Stmt& bind(const A&... a) {   // ?1, ?2, ...에 차례로
        int i = 1;
        (put(i++, a), ...);
        return *this;
    }
    template <class A> Stmt& set(const char* name, const A& a) {    // :이름에
        put(sqlite3_bind_parameter_index(st_, name), a);
        return *this;
    }
    bool step() {                              // 행이 있으면 true, 끝이면 false
        int rc = sqlite3_step(st_);
        if (rc == SQLITE_ROW) return true;
        if (rc == SQLITE_DONE) return false;
        check(rc);
        return false;
    }
    void reset() { sqlite3_reset(st_); sqlite3_clear_bindings(st_); }
    bool null(int c) const { return sqlite3_column_type(st_, c) == SQLITE_NULL; }
    long long i64(int c) const { return sqlite3_column_int64(st_, c); }
    std::string text(int c) const {            // 포인터는 다음 step까지만 유효하다
        auto p = reinterpret_cast<const char*>(sqlite3_column_text(st_, c));
        return p ? std::string(p, sqlite3_column_bytes(st_, c)) : std::string();
    }
    std::string expanded() const {             // 값을 채운 SQL (기록용)
        char* s = sqlite3_expanded_sql(st_);
        std::string out = s;
        sqlite3_free(s);
        return out;
    }
    int reprepared() const {
        return sqlite3_stmt_status(st_, SQLITE_STMTSTATUS_REPREPARE, 0);
    }

private:
    void put(int i, int v) { check(sqlite3_bind_int(st_, i, v)); }
    void put(int i, long long v) { check(sqlite3_bind_int64(st_, i, v)); }
    void put(int i, std::string_view v) {      // 문자열은 SQLite가 복사하게 한다
        check(sqlite3_bind_text(st_, i, v.data(), int(v.size()), SQLITE_TRANSIENT));
    }
    void put(int i, std::nullptr_t) { check(sqlite3_bind_null(st_, i)); }
    void check(int rc) const {
        if (rc != SQLITE_OK) throw std::runtime_error(sqlite3_errmsg(db_));
    }
    sqlite3* db_;
    sqlite3_stmt* st_ = nullptr;
};

int main() {
    sqlite3* db;
    sqlite3_open(":memory:", &db);
    sqlite3_exec(db, "CREATE TABLE products (id INTEGER PRIMARY KEY, name TEXT, "
                     "price INTEGER)", nullptr, nullptr, nullptr);

    {                                          // 문장은 연결보다 먼저 정리한다
        // 한 번 준비해서 여러 번 쓴다
        Stmt ins(db, "INSERT INTO products (name, price) VALUES (?, ?)");
        struct { const char* name; int price; } rows[] = {
            {"볼트", 120}, {"너트", 80}, {"렌치", 15000}};
        for (auto& r : rows) {                 // 묶고, 실행하고, 되돌린다
            ins.bind(r.name, r.price).step();
            ins.reset();
        }
        ins.bind("M5 'A급' 와셔", nullptr).step();   // 따옴표가 든 이름, 가격 모름
        std::printf("넣은 SQL 예: %s\n", ins.expanded().c_str());
        ins.reset();

        Stmt sel(db, "SELECT name, price FROM products WHERE price >= :min "
                     "OR price IS NULL ORDER BY price");
        sel.set(":min", 100);
        while (sel.step())
            std::printf("  %s: %s\n", sel.text(0).c_str(), sel.null(1)
                        ? "가격 모름" : std::to_string(sel.i64(1)).c_str());

        sqlite3_exec(db, "ALTER TABLE products ADD COLUMN stock INTEGER DEFAULT 0",
                     nullptr, nullptr, nullptr);
        sel.reset();
        sel.set(":min", 10000);
        while (sel.step())
            std::printf("스키마가 바뀐 뒤: %s\n", sel.text(0).c_str());
        std::printf("다시 준비한 횟수: %d\n", sel.reprepared());
    }
    sqlite3_close(db);
}
```

```text
넣은 SQL 예: INSERT INTO products (name, price) VALUES ('M5 ''A급'' 와셔', NULL)
  M5 'A급' 와셔: 가격 모름
  볼트: 120
  렌치: 15000
스키마가 바뀐 뒤: M5 'A급' 와셔
스키마가 바뀐 뒤: 렌치
다시 준비한 횟수: 1
```

- **한 번 준비해 여러 번**: INSERT를 한 번만 준비하고 값만 바꿔 네 번 실행했습니다. `reset`으로 문장을 처음으로 되돌리고, `sqlite3_clear_bindings`로 앞의 값이 다음 실행에 남지 않게 했습니다
- **따옴표와 NULL도 그대로**: 이름 속의 작은따옴표를 따로 처리하지 않았고, `nullptr`은 NULL로 들어갔습니다. `sqlite3_expanded_sql`은 값을 채운 SQL을 보여 주어 기록에 쓸 만하지만, 암호 같은 값이 로그에 남지 않게 조심합니다
- **이름 붙은 매개변수**: `:min`의 번호는 `sqlite3_bind_parameter_index`로 찾습니다. 매개변수 번호는 1부터, 결과 열 번호는 0부터 셉니다
- **저절로 다시 준비**: 문장을 준비한 뒤에 열을 더했는데도 같은 문장이 그대로 동작했습니다. `prepare_v2`로 만든 문장은 스키마가 바뀐 것을 알아채고 스스로 다시 컴파일합니다. 옛 `sqlite3_prepare`는 이때 `SQLITE_SCHEMA` 오류를 냈습니다

## 수명 함정

문자열을 묶을 때 마지막 인자로 `SQLITE_STATIC`을 주면 SQLite는 복사하지 않고 그 포인터를 계속 씁니다. 임시 `std::string`의 내용을 `SQLITE_STATIC`으로 묶으면, 문자열이 사라진 뒤 `step`이 이미 해제된 메모리를 읽는 정의되지 않은 동작이 됩니다. 값이 문장보다 오래 산다는 확신이 없으면 위의 감싸개처럼 `SQLITE_TRANSIENT`로 복사하게 합니다. 반대 방향도 같습니다. `sqlite3_column_text`가 준 포인터는 다음 `step`, `reset`, `finalize` 전까지만 유효하므로 바로 `std::string`으로 복사합니다. 길이는 `sqlite3_column_text`를 부른 다음에 `sqlite3_column_bytes`로 얻는 것이 안전한 순서입니다. 연결과 문장 사이에도 순서가 있습니다. 정리하지 않은 문장이 남은 채로 `sqlite3_close`를 부르면 `SQLITE_BUSY`를 돌려주고 닫지 않아서, 위 예제는 문장들을 블록 안에 두어 연결보다 먼저 정리했습니다. `sqlite3_close_v2`는 남은 문장이 정리될 때까지 닫기를 미룹니다.

<Note>
연결 하나에서 같은 문장을 계속 쓰는 프로그램은 SQL 문자열을 열쇠로 준비된 문장을 모아 두는 캐시를 많이 둡니다. 오래 쓸 문장은 `sqlite3_prepare_v3`에 `SQLITE_PREPARE_PERSISTENT`를 주어 SQLite가 그에 맞게 메모리를 잡게 할 수 있습니다. 문장은 그것을 만든 연결에 속하므로, 연결과 함께 스레드 규칙을 따릅니다([「스레드 모드」](#/read/cpp-database/sqlite-threads)).
</Note>

## 정리

- 준비된 문장은 SQL을 한 번 컴파일해 두고 자리 표시자에 값만 묶어 여러 번 실행하는 것으로, 빠르고 안전하며 타입을 지킨다
- 문장은 준비, 묶기, step, 열 읽기, reset과 clear_bindings, finalize의 순서로 쓰며, prepare_v2 문장은 스키마가 바뀌면 저절로 다시 준비된다
- SQLITE_STATIC으로 묶은 값과 column_text가 준 포인터는 수명이 짧으므로, 확신이 없으면 SQLITE_TRANSIENT로 묶고 읽은 값은 바로 복사한다
