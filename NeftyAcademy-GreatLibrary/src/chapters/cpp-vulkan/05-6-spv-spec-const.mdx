**특수화 상수**는 SPIR-V로 번역할 때가 아니라 파이프라인을 만들 때 값이 정해지는 상수입니다. GLSL에서는 `layout(constant_id = 번호) const` 변수로 선언하고 기본값을 주며, C++에서는 `VkSpecializationInfo`로 번호마다 값을 넘깁니다. 드라이버는 파이프라인을 컴파일할 때 그 값을 진짜 상수로 다루므로, 상수에 따라 갈라지는 코드를 지우거나 반복문을 풀어 쓸 수 있습니다. [「컴퓨트 셰이더」](#/read/cpp-vulkan/glsl-compute)에서는 워크그룹 크기를 셰이더에 적어 두었지만, 아래 프로그램은 워크그룹 크기, 거듭제곱 횟수, 곱할 수를 특수화 상수로 두고, 한 SPIR-V에서 파이프라인 둘을 만들어 차례로 돌립니다.

```cpp vk pkg=SPIRV-Tools with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_compute.hpp,vkx_pipeline.hpp
#include <sstream>
#include <spirv-tools/libspirv.hpp>
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_compute.hpp"
#include "vkx_pipeline.hpp"

const char* CS = R"(#version 450
layout(local_size_x_id = 0) in;                 // 워크그룹 크기도 상수 0번으로
layout(constant_id = 1) const int POWER = 2;    // 몇 제곱할지
layout(constant_id = 2) const float SCALE = 1.0;
layout(binding = 0) buffer Data { float v[]; };
void main() {
    uint i = gl_GlobalInvocationID.x;
    float x = v[i], y = 1.0;
    for (int k = 0; k < POWER; ++k) y *= x;     // 상수가 정해지면 풀어 쓸 수 있다
    v[i] = y * SCALE;
})";

struct Values {   // 특수화 상수에 줄 값들. 상수 번호마다 오프셋과 크기를 알린다
    uint32_t group;
    int32_t power;
    float scale;
};

VkPipeline specialize(const vkx::Device& dev, VkShaderModule cs,
                      VkPipelineLayout layout, const Values& values) {
    VkSpecializationMapEntry map[3] = {
        {0, offsetof(Values, group), sizeof(uint32_t)},
        {1, offsetof(Values, power), sizeof(int32_t)},
        {2, offsetof(Values, scale), sizeof(float)}};
    VkSpecializationInfo info{.mapEntryCount = 3, .pMapEntries = map,
                              .dataSize = sizeof(Values), .pData = &values};
    VkComputePipelineCreateInfo ci{
        .sType = VK_STRUCTURE_TYPE_COMPUTE_PIPELINE_CREATE_INFO,
        .stage = {.sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
                  .stage = VK_SHADER_STAGE_COMPUTE_BIT, .module = cs,
                  .pName = "main", .pSpecializationInfo = &info},
        .layout = layout};
    VkPipeline p;
    vkCreateComputePipelines(dev.handle, VK_NULL_HANDLE, 1, &ci, nullptr, &p);
    return p;
}

int main() {
    std::vector<uint32_t> spirv = vkx::compile(CS, shaderc_compute_shader);
    spvtools::SpirvTools tools(SPV_ENV_VULKAN_1_3);
    std::string text, line;
    tools.Disassemble(spirv, &text, SPV_BINARY_TO_TEXT_OPTION_FRIENDLY_NAMES);
    std::istringstream in(text);   // 특수화 상수가 SPIR-V에 어떻게 적히는지
    while (std::getline(in, line))
        if (line.find("SpecId") != std::string::npos ||
            line.find("OpSpecConstant") != std::string::npos ||
            line.find("LocalSizeId") != std::string::npos)
            std::printf("%s\n", line.c_str() + line.find_first_not_of(' '));

    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const auto SSBO = VK_DESCRIPTOR_TYPE_STORAGE_BUFFER;
    vkx::Buffer data =
        vkx::make_buffer(dev, 64 * 4, VK_BUFFER_USAGE_STORAGE_BUFFER_BIT, true);
    auto* v = static_cast<float*>(data.mapped);
    for (int i = 0; i < 64; ++i) v[i] = 2.0f;
    vkx::Bindings binds = vkx::make_bindings(dev, {SSBO});
    vkx::bind_buffer(dev, binds, 0, SSBO, data);
    VkPipelineLayout layout = vkx::make_layout(dev, {binds.layout});
    VkShaderModule cs = vkx::make_shader(dev, spirv);

    // 같은 SPIR-V에서 파이프라인 둘: 그룹 64개씩 제곱, 그룹 32개씩 세제곱 × 0.5
    Values a{64, 2, 1.0f}, b{32, 3, 0.5f};
    VkPipeline pa = specialize(dev, cs, layout, a);
    VkPipeline pb = specialize(dev, cs, layout, b);
    VkCommandBuffer cmd = vkx::begin(cmds);
    vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_COMPUTE, layout, 0, 1,
                            &binds.set, 0, nullptr);
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_COMPUTE, pa);
    vkCmdDispatch(cmd, 64 / a.group, 1, 1);   // 2 → 4
    VkMemoryBarrier2 mid{.sType = VK_STRUCTURE_TYPE_MEMORY_BARRIER_2,
                         .srcStageMask = VK_PIPELINE_STAGE_2_COMPUTE_SHADER_BIT,
                         .srcAccessMask = VK_ACCESS_2_SHADER_STORAGE_WRITE_BIT,
                         .dstStageMask = VK_PIPELINE_STAGE_2_COMPUTE_SHADER_BIT,
                         .dstAccessMask = VK_ACCESS_2_SHADER_STORAGE_READ_BIT};
    VkDependencyInfo dep{.sType = VK_STRUCTURE_TYPE_DEPENDENCY_INFO,
                         .memoryBarrierCount = 1, .pMemoryBarriers = &mid};
    vkCmdPipelineBarrier2(cmd, &dep);
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_COMPUTE, pb);
    vkCmdDispatch(cmd, 64 / b.group, 1, 1);   // 4 → 64 × 0.5
    mid.dstStageMask = VK_PIPELINE_STAGE_2_HOST_BIT;
    mid.dstAccessMask = VK_ACCESS_2_HOST_READ_BIT;
    vkCmdPipelineBarrier2(cmd, &dep);
    vkx::submit_and_wait(dev, cmds);
    std::printf("2 → 제곱(그룹 64개씩 1번) → 세제곱 × 0.5(그룹 32개씩 2번) = %g, "
                "끝 원소 %g\n", v[0], v[63]);

    vkDestroyPipeline(dev.handle, pa, nullptr);
    vkDestroyPipeline(dev.handle, pb, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, cs, nullptr);
    vkx::destroy(dev, binds);
    vkx::destroy(dev, data);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
OpExecutionModeId %main LocalSizeId %7 %uint_1 %uint_1
OpDecorate %7 SpecId 0
OpDecorate %POWER SpecId 1
OpDecorate %SCALE SpecId 2
OpDecorate %55 SpecId 0
%7 = OpSpecConstant %uint 1
%POWER = OpSpecConstant %int 2
%SCALE = OpSpecConstant %float 1
%55 = OpSpecConstant %uint 1
%56 = OpSpecConstantComposite %v3uint %55 %uint_1 %uint_1
2 → 제곱(그룹 64개씩 1번) → 세제곱 × 0.5(그룹 32개씩 2번) = 32, 끝 원소 32
```

- **SPIR-V에서의 꼴**: 특수화 상수는 `OpSpecConstant`(bool이면 `OpSpecConstantTrue`나 `OpSpecConstantFalse`)로 선언되고 `SpecId` 장식으로 번호를 답니다. 값을 주지 않은 번호는 셰이더의 기본값(`POWER`는 2, `SCALE`은 1)을 씁니다. `local_size_x_id = 0`은 `OpExecutionModeId ... LocalSizeId %7`이 되어 워크그룹 크기 자리에 0번 상수가 들어갑니다(0번이 하나 더 있는 것은 `gl_WorkGroupSize` 몫입니다)
- **값 넘기기**: `VkSpecializationMapEntry`는 상수 번호마다 데이터 덩어리 안의 오프셋과 크기를 알리고, 구조체 하나(`Values`)를 통째로 `pData`로 넘깁니다. 특수화 상수는 스칼라 정수, 부동소수점, bool만 될 수 있고, bool은 4바이트 `VkBool32`로 넘깁니다. 셰이더에 없는 번호의 항목은 아무 영향도 없습니다
- **한 SPIR-V, 파이프라인 둘**: 첫 파이프라인은 그룹 64개씩 제곱하고(2 → 4), 둘째는 그룹 32개씩 세제곱에 0.5를 곱합니다(4 → 32). 워크그룹 크기가 다르므로 디스패치 수도 1과 2로 맞췄습니다
- **maintenance4**: `LocalSizeId`를 쓰는 셰이더는 1.3에서 필수가 된 `maintenance4` 기능을 켜야 쓸 수 있습니다. 켜지 않으면 검증 레이어가 VUID-RuntimeSpirv-LocalSizeId-06434로 알리므로 이 책의 `make_device`가 켜 둡니다. `--target-env=vulkan1.2`로 번역하면 glslc는 `LocalSize 1 1 1`을 적고 `BuiltIn WorkgroupSize` 장식을 단 합성 상수로 덮어쓰는 옛 꼴을 만들어서 이 기능이 필요 없습니다

## 매크로·푸시 상수와 견주기

| 방법 | 값이 정해지는 때 | 결과 |
| --- | --- | --- |
| 매크로(`-D`) | SPIR-V로 번역할 때 | 변형마다 SPIR-V와 파이프라인 |
| 특수화 상수 | 파이프라인을 만들 때 | SPIR-V 하나, 값마다 파이프라인 |
| 푸시 상수·UBO | 그릴 때 | 파이프라인 하나, 드라이버는 값을 모름 |

자주 바뀌는 값은 푸시 상수나 유니폼으로 넘기고, 품질 단계나 커널 크기처럼 드물게 바뀌면서 코드 모양을 바꾸는 값은 특수화 상수로 두면 앞단 번역을 한 번만 하면서 최적화도 얻습니다([「전처리와 include」](#/read/cpp-vulkan/glsl-include)).

<Note>
특수화 상수는 배열 크기에도 쓸 수 있어서, `shared float tile[TILE * TILE];`처럼 워크그룹 공유 메모리의 크기를 파이프라인마다 바꿀 수 있습니다. 다만 값마다 파이프라인이 따로 생기므로, 만들 수 있는 조합을 미리 정해 로딩할 때 만들어 두고 파이프라인 캐시에 담습니다([「파이프라인 캐시」](#/read/cpp-vulkan/ps-cache)).
</Note>

## 정리

- 특수화 상수는 `constant_id`(워크그룹 크기는 `local_size_x_id`)로 선언하고, 파이프라인을 만들 때 `VkSpecializationInfo`로 값을 넘긴다
- SPIR-V에서는 `OpSpecConstant`와 `SpecId` 장식이고, 드라이버가 값을 상수로 접어 최적화하며, `LocalSizeId`에는 maintenance4가 필요하다
- 번역 때 정하는 매크로, 그릴 때 정하는 푸시 상수 사이에서, 드물게 바뀌며 코드 모양을 바꾸는 값에 알맞다
