벌칸 프로그램은 세 겹을 지나 GPU에 닿습니다. 프로그램이 부르는 함수는 먼저 **로더**(리눅스의 `libvulkan.so.1`, 윈도우의 `vulkan-1.dll`)에 들어가고, 로더는 켜 둔 **레이어**를 차례로 거쳐 GPU 회사가 만든 **드라이버**(ICD)에 넘깁니다. 로더는 드라이버와 레이어를 JSON 매니페스트 파일로 찾으므로, 드라이버를 바꾸거나 레이어를 끼우는 데 프로그램을 다시 빌드할 필요가 없습니다.

## 설치할 것

윈도우, 리눅스, 맥에서는 LunarG의 **Vulkan SDK**가 머리 파일, 로더, 검증 레이어, 설정 도구 vkconfig, `vulkaninfo`, glslang·shaderc·Slang 같은 셰이더 컴파일러, SPIRV-Tools, 메모리 할당기 VMA를 한꺼번에 줍니다. 리눅스는 배포판 꾸러미로도 충분해서, 이 책은 우분투의 `libvulkan-dev`, `vulkan-validationlayers`, `vulkan-tools`, `libshaderc-dev`를 썼습니다. 맥에는 벌칸 드라이버가 없어 Metal 위에서 도는 MoltenVK를 씁니다([「MoltenVK」](#/read/cpp-vulkan/port-moltenvk)). GPU가 없는 기계나 CI에서는 CPU로 벌칸을 구현한 Mesa의 lavapipe나 구글의 SwiftShader를 드라이버 자리에 둡니다.

빌드는 `g++ -std=c++20 main.cpp -lvulkan -lshaderc`처럼 로더만 링크하면 됩니다. 이 책처럼 구조체를 지정 초기화로 일부만 채우면 GCC의 `-Wextra`가 빠진 멤버마다 경고를 내므로 `-Wno-missing-field-initializers`를 함께 줍니다. 빠진 멤버는 0으로 채워지는 것이 C++의 규칙이고, 벌칸은 그것을 기본값으로 쓰도록 설계되어 있습니다.

## 검증 레이어 켜기

검증 레이어 `VK_LAYER_KHRONOS_validation`은 모든 호출을 가로채 명세의 규칙을 검사합니다. 인스턴스를 만들 때 이름을 넘겨 켜거나, 코드를 고치지 않고 환경 변수 `VK_LOADER_LAYERS_ENABLE=*validation`(예전 이름은 `VK_INSTANCE_LAYERS`)이나 vkconfig로 켭니다. 메시지를 프로그램에서 받으려면 `VK_EXT_debug_utils` 확장의 메신저에 콜백을 겁니다. 아래 프로그램은 일부러 규칙을 두 번 어기고 콜백이 받은 것을 찍습니다.

```cpp vk
#include <vulkan/vulkan.h>
#include <cstdio>
#include <string_view>

// 검증 레이어가 보내는 메시지를 받는 콜백
VKAPI_ATTR VkBool32 VKAPI_CALL on_message(
        VkDebugUtilsMessageSeverityFlagBitsEXT severity,
        VkDebugUtilsMessageTypeFlagsEXT,
        const VkDebugUtilsMessengerCallbackDataEXT* d, void*) {
    bool error = severity >= VK_DEBUG_UTILS_MESSAGE_SEVERITY_ERROR_BIT_EXT;
    std::printf("   [%s] %s\n", error ? "오류" : "경고", d->pMessageIdName);
    for (uint32_t i = 0; i < d->objectCount; ++i)
        if (d->pObjects[i].pObjectName)
            std::printf("      대상: %s\n", d->pObjects[i].pObjectName);
    // 메시지 끝의 "The Vulkan spec states: ..."가 어긴 규칙의 원문이다
    std::string_view m = d->pMessage;
    if (auto at = m.find("spec states: "); at != m.npos) {
        m = m.substr(at + 13);
        m = m.substr(0, m.find(" (http"));
        while (!m.empty()) {                       // 70자 안쪽에서 낱말로 끊는다
            size_t cut = m.size() <= 70 ? m.size() : m.rfind(' ', 70);
            std::printf("      %.*s\n", int(cut), m.data());
            m.remove_prefix(cut < m.size() ? cut + 1 : cut);
        }
    }
    return VK_FALSE;                               // 호출 자체는 그대로 진행한다
}

int main() {
    const char* layers[] = {"VK_LAYER_KHRONOS_validation"};
    const char* exts[] = {VK_EXT_DEBUG_UTILS_EXTENSION_NAME};
    VkDebugUtilsMessengerCreateInfoEXT mci{
        .sType = VK_STRUCTURE_TYPE_DEBUG_UTILS_MESSENGER_CREATE_INFO_EXT,
        .messageSeverity = VK_DEBUG_UTILS_MESSAGE_SEVERITY_WARNING_BIT_EXT |
                           VK_DEBUG_UTILS_MESSAGE_SEVERITY_ERROR_BIT_EXT,
        .messageType = VK_DEBUG_UTILS_MESSAGE_TYPE_VALIDATION_BIT_EXT,
        .pfnUserCallback = on_message};
    VkApplicationInfo app{.sType = VK_STRUCTURE_TYPE_APPLICATION_INFO,
                          .apiVersion = VK_API_VERSION_1_3};
    VkInstanceCreateInfo ici{.sType = VK_STRUCTURE_TYPE_INSTANCE_CREATE_INFO,
                             .pNext = &mci,   // 인스턴스를 만들고 없앨 때도 받는다
                             .pApplicationInfo = &app,
                             .enabledLayerCount = 1, .ppEnabledLayerNames = layers,
                             .enabledExtensionCount = 1,
                             .ppEnabledExtensionNames = exts};
    VkInstance instance;
    if (vkCreateInstance(&ici, nullptr, &instance) != VK_SUCCESS) return 1;
    auto create_messenger = reinterpret_cast<PFN_vkCreateDebugUtilsMessengerEXT>(
        vkGetInstanceProcAddr(instance, "vkCreateDebugUtilsMessengerEXT"));
    auto destroy_messenger = reinterpret_cast<PFN_vkDestroyDebugUtilsMessengerEXT>(
        vkGetInstanceProcAddr(instance, "vkDestroyDebugUtilsMessengerEXT"));
    auto set_name = reinterpret_cast<PFN_vkSetDebugUtilsObjectNameEXT>(
        vkGetInstanceProcAddr(instance, "vkSetDebugUtilsObjectNameEXT"));
    VkDebugUtilsMessengerEXT messenger;
    create_messenger(instance, &mci, nullptr, &messenger);

    uint32_t n = 1;
    VkPhysicalDevice gpu;
    vkEnumeratePhysicalDevices(instance, &n, &gpu);
    float priority = 1.0f;
    VkDeviceQueueCreateInfo qci{.sType = VK_STRUCTURE_TYPE_DEVICE_QUEUE_CREATE_INFO,
                                .queueFamilyIndex = 0, .queueCount = 1,
                                .pQueuePriorities = &priority};
    VkDeviceCreateInfo dci{.sType = VK_STRUCTURE_TYPE_DEVICE_CREATE_INFO,
                           .queueCreateInfoCount = 1, .pQueueCreateInfos = &qci};
    VkDevice device;
    vkCreateDevice(gpu, &dci, nullptr, &device);

    std::printf("1) 크기가 0인 버퍼\n");
    VkBufferCreateInfo bci{.sType = VK_STRUCTURE_TYPE_BUFFER_CREATE_INFO, .size = 0,
                           .usage = VK_BUFFER_USAGE_VERTEX_BUFFER_BIT};
    VkBuffer empty;
    VkResult r = vkCreateBuffer(device, &bci, nullptr, &empty);
    std::printf("   vkCreateBuffer의 반환값: %d\n", r);
    vkDestroyBuffer(device, empty, nullptr);

    std::printf("2) 이름 붙인 버퍼를 없애지 않고 장치를 없앰\n");
    bci.size = 256;
    VkBuffer vertices;
    vkCreateBuffer(device, &bci, nullptr, &vertices);
    VkDebugUtilsObjectNameInfoEXT name{
        .sType = VK_STRUCTURE_TYPE_DEBUG_UTILS_OBJECT_NAME_INFO_EXT,
        .objectType = VK_OBJECT_TYPE_BUFFER,
        .objectHandle = reinterpret_cast<uint64_t>(vertices),
        .pObjectName = "정점 버퍼"};
    set_name(device, &name);
    vkDestroyDevice(device, nullptr);
    destroy_messenger(instance, messenger, nullptr);
    vkDestroyInstance(instance, nullptr);
}
```

```text
1) 크기가 0인 버퍼
   [오류] VUID-VkBufferCreateInfo-size-00912
      size must be greater than 0
   vkCreateBuffer의 반환값: 0
2) 이름 붙인 버퍼를 없애지 않고 장치를 없앰
   [오류] VUID-vkDestroyDevice-device-05137
      대상: 정점 버퍼
      All child objects created on device must have been destroyed prior to
      destroying device
```

- **드라이버는 성공이라 한다**: 크기가 0인 버퍼에도 `vkCreateBuffer`는 0(`VK_SUCCESS`)을 돌려주었습니다. 잘못을 알아챈 것은 레이어뿐입니다
- **VUID**: 명세의 규칙마다 `VUID-VkBufferCreateInfo-size-00912` 같은 이름이 있습니다. 메시지 끝의 "The Vulkan spec states:" 뒤가 어긴 규칙의 원문이고, 이 이름으로 명세를 찾으면 그 규칙이 놓인 맥락까지 읽을 수 있습니다
- **이름 붙이기**: `vkSetDebugUtilsObjectNameEXT`로 이름을 붙여 두면 메시지에 핸들 대신 "정점 버퍼"가 나옵니다. 같은 이름이 RenderDoc 같은 디버거에도 보입니다([「RenderDoc」](#/read/cpp-vulkan/vdbg-renderdoc))
- **인스턴스의 pNext**: 메신저는 인스턴스가 있어야 만들 수 있으므로, 같은 설정을 `VkInstanceCreateInfo`의 `pNext`에도 걸어 인스턴스를 만들고 없애는 동안의 메시지까지 받습니다
- **확장 함수의 주소**: 로더는 코어 함수와 일부 창 시스템 함수만 바로 내보냅니다. `vkCreateDebugUtilsMessengerEXT` 같은 확장 함수는 `vkGetInstanceProcAddr`로 주소를 얻어 부릅니다. volk 같은 메타 로더는 이 일을 한꺼번에 해 줍니다

## 더 깊은 검사

기본 검사가 보지 못하는 것도 있습니다. **동기화 검증**은 배리어를 빠뜨려 앞 작업과 뒤 작업이 같은 메모리를 겹쳐 읽고 쓰는 곳을 `SYNC-HAZARD-WRITE-AFTER-WRITE` 같은 이름으로 알려 줍니다([「파이프라인 배리어」](#/read/cpp-vulkan/vsync-barriers)). **GPU 지원 검증**은 셰이더가 배열 밖의 디스크립터를 읽는 일처럼 GPU에서만 알 수 있는 잘못을 셰이더에 검사 코드를 끼워 잡고, **모범 사례** 검사는 규칙 위반은 아니지만 느린 사용을 경고합니다. 이런 검사는 vkconfig, `VkValidationFeaturesEXT`, 또는 `VK_KHRONOS_VALIDATION_VALIDATE_SYNC=true` 같은 환경 변수로 켭니다. 레이어는 호출마다 일을 더하므로 개발 빌드에서만 켜고, 배포하는 빌드에서는 끕니다([「검증 레이어 읽기」](#/read/cpp-vulkan/vdbg-validation)).

<Note>
이 책의 예제 출력은 GPU 없이 lavapipe(Mesa 25.2)에서 돌린 결과입니다. 모든 예제를 검증 레이어와 동기화 검증을 켠 채 돌려, 일부러 어긴 예제 말고는 메시지가 하나도 나오지 않는 것을 확인했습니다. 실제 GPU에서는 장치 이름과 일부 수치가 다르게 나옵니다.
</Note>

## 정리

- 호출은 로더 → 레이어 → 드라이버를 지나며, Vulkan SDK나 배포판 꾸러미로 머리 파일·로더·레이어·도구를 갖추고, GPU가 없으면 lavapipe 같은 소프트웨어 구현을 쓴다
- 검증 레이어는 이름이나 환경 변수로 켜고, debug_utils 메신저의 콜백으로 VUID와 규칙 원문을 받으며, 객체에 이름을 붙이면 메시지가 읽기 쉬워진다
- 드라이버는 잘못된 호출에도 성공을 돌려줄 수 있으므로 개발 중에는 동기화 검증까지 켜 두고, 배포 빌드에서는 레이어를 끈다
