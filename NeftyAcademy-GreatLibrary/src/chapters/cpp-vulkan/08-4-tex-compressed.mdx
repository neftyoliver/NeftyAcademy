PNG나 JPEG는 파일을 작게 하지만 GPU는 그대로 읽지 못해서, 풀어 올리면 텍셀마다 4바이트를 그대로 차지합니다. **블록 압축 포맷**은 GPU가 압축된 채로 읽는 포맷입니다. 이미지를 4×4 같은 작은 블록으로 나눠 블록마다 정해진 바이트 수로 줄여 두고, 샘플링할 때 필요한 블록만 그 자리에서 풉니다. 블록 크기가 일정해서 어느 텍셀이든 바로 찾을 수 있고, 비디오 메모리와 메모리 대역폭을 함께 아낍니다. 아래 프로그램은 이 장치가 어떤 포맷을 지원하는지 묻고, BC1 블록 하나를 손으로 만들어 올린 뒤 풀린 텍셀을 읽습니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_pipeline.hpp,vkx_draw.hpp,vkx_texture.hpp
#include "vkx_bind.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"
#include "vkx_texture.hpp"

const char* FS = R"(#version 450
layout(binding = 0) uniform sampler2D tex;
layout(location = 0) out float value;
void main() { value = texelFetch(tex, ivec2(gl_FragCoord.xy), 0).r; })";

struct Kind { const char* name; VkFormat format; uint32_t block, bytes; };
const Kind kinds[] = {
    {"BC1", VK_FORMAT_BC1_RGB_UNORM_BLOCK, 4, 8},
    {"BC5", VK_FORMAT_BC5_UNORM_BLOCK, 4, 16},
    {"BC7", VK_FORMAT_BC7_UNORM_BLOCK, 4, 16},
    {"ETC2 RGB8", VK_FORMAT_ETC2_R8G8B8_UNORM_BLOCK, 4, 8},
    {"ASTC 4x4", VK_FORMAT_ASTC_4x4_UNORM_BLOCK, 4, 16},
    {"ASTC 8x8", VK_FORMAT_ASTC_8x8_UNORM_BLOCK, 8, 16}};

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);

    // 1) 이 장치가 샘플링할 수 있는 압축 포맷과, 1024×1024 한 장의 크기
    std::printf("%-10s %-6s %s\n", "format", "sample", "1024x1024");
    std::printf("%-10s %-6s %4u KiB\n", "RGBA8", "yes", 1024 * 1024 * 4 / 1024);
    for (const Kind& k : kinds) {
        VkFormatProperties p;
        vkGetPhysicalDeviceFormatProperties(dev.gpu, k.format, &p);
        bool ok = p.optimalTilingFeatures & VK_FORMAT_FEATURE_SAMPLED_IMAGE_BIT;
        uint32_t blocks = (1024 / k.block) * (1024 / k.block);
        std::printf("%-10s %-6s %4u KiB\n", k.name, ok ? "yes" : "no",
                    blocks * k.bytes / 1024);
    }

    // 2) BC1 블록 하나를 손으로 만든다: 끝 색 둘(RGB565)과 텍셀마다 2비트 번호.
    //    색0 > 색1이면 번호 0과 1은 두 끝 색, 2와 3은 그 사이를 3등분한 두 색
    auto row = [](int a, int b, int c, int d) {    // 왼쪽 텍셀이 낮은 비트
        return uint8_t(a | b << 2 | c << 4 | d << 6);
    };
    const uint8_t block[8] = {0xFF, 0xFF, 0x00, 0x00,       // 색0 = 0xFFFF, 색1 = 0
                              row(0, 2, 3, 1), row(2, 3, 1, 0),
                              row(3, 1, 0, 2), row(1, 0, 2, 3)};
    vkx::Image tex = vkx::make_texture(
        dev, cmds, {4, 4}, VK_FORMAT_BC1_RGB_UNORM_BLOCK, block, sizeof block);
    VkSampler sampler = vkx::make_sampler(dev, VK_FILTER_NEAREST,
                                          VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE);
    const auto TEXTURE = VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER;
    vkx::Bindings binds = vkx::make_bindings(dev, {TEXTURE});
    vkx::bind_image(dev, binds, 0, TEXTURE, tex.view,
                    VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL, sampler);

    vkx::Image target = vkx::make_image(
        dev, {4, 4}, VK_FORMAT_R32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels =
        vkx::make_buffer(dev, 16 * 4, VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    VkPipelineLayout layout = vkx::make_layout(dev, {binds.layout});
    auto vs = vkx::make_shader(
        dev, vkx::compile(vkx::FULLSCREEN_VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipeline pipeline = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = fs, .layout = layout,
              .color = target.format});
    VkCommandBuffer cmd = vkx::begin(cmds);
    vkx::begin_drawing(cmd, target);
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
    vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, layout, 0, 1,
                            &binds.set, 0, nullptr);
    vkCmdDraw(cmd, 3, 1, 0, 0);
    vkx::end_drawing(cmd, target, pixels);
    vkx::submit_and_wait(dev, cmds);

    std::puts("8바이트 블록 하나가 풀린 텍셀 16개(0~255):");
    auto* v = static_cast<const float*>(pixels.mapped);
    for (int i = 0; i < 16; ++i)
        std::printf("%4d%s", int(v[i] * 255.0f + 0.5f), i % 4 == 3 ? "\n" : "");

    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    vkDestroySampler(dev.handle, sampler, nullptr);
    vkx::destroy(dev, binds);
    for (vkx::Image* i : {&tex, &target}) vkx::destroy(dev, *i);
    vkx::destroy(dev, pixels);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **장치마다 다른 지원**: lavapipe는 데스크톱 GPU처럼 BC 계열만 지원한다고 답했습니다. 모바일 GPU는 대개 반대로 ETC2와 ASTC를 지원합니다. 장치 기능 `textureCompressionBC`, `textureCompressionETC2`, `textureCompressionASTC_LDR`은 그 계열 전체를 지원한다는 뜻이므로, 포맷 하나씩은 `vkGetPhysicalDeviceFormatProperties`로 확인합니다
- **BC1 블록**: 8바이트 가운데 앞 4바이트가 16비트 색(RGB565) 둘이고, 뒤 4바이트는 텍셀 16개가 2비트씩 가진 번호입니다. 번호는 두 끝 색과 그 사이를 3등분한 두 색 가운데 하나를 고릅니다. 흰색과 검정을 끝 색으로 주었더니 255, 170, 85, 0 네 값만 나왔습니다. 블록 하나에 색이 넷뿐이므로 색이 급하게 바뀌는 곳에서는 블록 경계가 보일 수 있습니다
- **올리기**: 압축 텍스처도 `vkCmdCopyBufferToImage`로 올립니다. 영역의 크기는 텍셀 단위(4×4)로 적고, 버퍼에는 블록 데이터 8바이트만 있으면 됩니다. 밉 수준도 수준마다 따로 압축해 둔 데이터를 올립니다([「밉맵」](#/read/cpp-vulkan/tex-mipmaps))
- **크기**: RGBA8의 8분의 1(BC1)에서 4분의 1(BC7)입니다. ASTC는 블록이 늘 16바이트이고 블록의 크기를 4×4에서 12×12까지 고를 수 있어, 화질과 크기를 조절하기 쉽습니다

## 포맷 고르기

| 포맷 | 블록 | 쓰임 |
| --- | --- | --- |
| BC1 | 4×4에 8바이트 | 불투명한 색. 가장 작다 |
| BC3 | 4×4에 16바이트 | 알파가 부드럽게 변하는 색 |
| BC4, BC5 | 8, 16바이트 | 채널 하나, 둘. BC5는 법선 맵에 쓴다 |
| BC6H, BC7 | 16바이트 | HDR 색, 화질이 좋은 색과 알파 |
| ETC2 | 4×4에 8, 16바이트 | 모바일에서 널리 지원한다 |
| ASTC | 4×4~12×12에 16바이트 | 모바일. 블록 크기로 화질을 고른다 |

색 텍스처는 여기서도 `_SRGB_BLOCK` 포맷을 씁니다([「이미지, 뷰, 샘플러」](#/read/cpp-vulkan/tex-image)). 압축은 느리므로 실행 중이 아니라 빌드할 때 도구로 해 두고, DDS나 KTX 같은 파일에 밉 수준과 함께 담습니다.

<Note>
데스크톱과 모바일에 같은 파일을 내보내려면 KTX 2.0 파일에 Basis Universal 방식(ETC1S, UASTC)으로 담습니다. 이 방식은 GPU가 직접 읽는 포맷이 아니라 중간 포맷이어서, 로딩할 때 libktx 같은 라이브러리가 그 장치가 지원하는 BC7, ETC2, ASTC 가운데 하나로 빠르게 바꿉니다. 파일은 하나만 만들고, 포맷은 실행하는 장치에서 고르는 셈입니다.
</Note>

## 정리

- 블록 압축 포맷은 GPU가 압축된 채로 읽어 메모리를 4분의 1에서 8분의 1로 줄이고, 블록 크기가 일정해 어느 텍셀이든 바로 찾는다
- 데스크톱은 BC, 모바일은 ETC2와 ASTC를 주로 지원하므로 포맷 지원을 장치에 물어 고른다
- 압축은 빌드할 때 해 두고 밉과 함께 파일에 담으며, 여러 장치에 내보낼 때는 KTX 2.0과 Basis Universal로 로딩할 때 바꾼다
