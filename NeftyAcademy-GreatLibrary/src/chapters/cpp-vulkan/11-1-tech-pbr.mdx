물리 기반 조명(PBR)은 표면을 아주 작은 거울면(미세면)의 모임으로 보고, 들어온 빛보다 많이 내보내지 않도록 짠 식으로 밝기를 셈합니다. 재질은 바탕색, 금속성, 거칠기 세 값으로 적는 방식이 널리 쓰이고, glTF 2.0의 재질이 이 방식입니다. 아래 셰이더의 식은 glTF 명세 부록 B의 것입니다. 프로그램은 거칠기가 다른 공 셋을 비금속과 금속으로 한 번씩 그려 글자로 찍습니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_pipeline.hpp,vkx_draw.hpp
#include <cmath>
#include <string>
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"

// 가로 20, 세로 10픽셀 칸마다 공 하나. 칸 번호가 커질수록 거칠다(0.2, 0.5, 0.8)
const char* FS = R"(#version 450
layout(push_constant) uniform Push { vec3 light; float metallic; };
layout(location = 0) out vec2 result;   // (난반사, 정반사)
const float PI = 3.14159265;
const float BASE = 0.8;                 // 바탕색(회색 한 가지)

// 미세면의 법선이 H 쪽을 보는 비율: GGX
float D(float NoH, float a) {
    float d = NoH * NoH * (a * a - 1.0) + 1.0;
    return a * a / (PI * d * d);
}
// 미세면끼리 가리는 정도를 4·NoL·NoV로 나눈 것: 스미스
float Vis(float NoL, float NoV, float a) {
    float l = NoL + sqrt(a * a + (1.0 - a * a) * NoL * NoL);
    float v = NoV + sqrt(a * a + (1.0 - a * a) * NoV * NoV);
    return 1.0 / (l * v);
}
// 비스듬히 볼수록 커지는 반사율: 슐릭
float F(float VoH, float f0) { return f0 + (1.0 - f0) * pow(1.0 - VoH, 5.0); }

void main() {
    vec2 cell = vec2(20.0, 10.0);
    vec2 p = mod(gl_FragCoord.xy, cell) / cell * 2.0 - 1.0;
    result = vec2(0.0);
    if (dot(p, p) >= 1.0) return;                        // 공 밖
    vec3 N = vec3(p.x, -p.y, sqrt(1.0 - dot(p, p)));     // 공 위의 법선
    vec3 V = vec3(0.0, 0.0, 1.0);                        // 눈은 정면 멀리
    vec3 L = normalize(light);
    vec3 H = normalize(L + V);
    float NoL = dot(N, L), NoV = N.z;
    if (NoL <= 0.0) return;                              // 빛을 등진 쪽
    float roughness = 0.2 + 0.3 * floor(gl_FragCoord.x / cell.x);
    float a = roughness * roughness;
    float f0 = mix(0.04, BASE, metallic);                // 정면에서의 반사율
    float fresnel = F(dot(V, H), f0);
    float diffuse = (1.0 - fresnel) * mix(BASE, 0.0, metallic) / PI;
    float specular = fresnel * D(dot(N, H), a) * Vis(NoL, NoV, a);
    result = vec2(diffuse, specular) * NoL * PI;         // 빛의 세기는 PI
})";

struct Push { float light[3]; float metallic; };

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const VkExtent2D SIZE{60, 10};
    vkx::Image target = vkx::make_image(
        dev, SIZE, VK_FORMAT_R32G32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels = vkx::make_buffer(dev, SIZE.width * SIZE.height * 8,
                                          VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    VkPipelineLayout layout = vkx::make_layout(dev, {}, sizeof(Push));
    auto vs = vkx::make_shader(
        dev, vkx::compile(vkx::FULLSCREEN_VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipeline pipeline = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = fs, .layout = layout,
              .color = target.format});

    // 하이라이트의 꼭대기(법선 = H)가 칸의 (8, 3)번 픽셀 한가운데에 오게 빛을 둔다
    const float hx = 8.5f / 20 * 2 - 1, hy = -(3.5f / 10 * 2 - 1);
    const float hz = std::sqrt(1 - hx * hx - hy * hy);
    Push push{{2 * hz * hx, 2 * hz * hy, 2 * hz * hz - 1}, 0.0f};   // V를 H로 반사

    for (float metallic : {0.0f, 1.0f}) {
        push.metallic = metallic;
        VkCommandBuffer cmd = vkx::begin(cmds);
        vkx::begin_drawing(cmd, target);
        vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
        vkCmdPushConstants(cmd, layout, VK_SHADER_STAGE_ALL, 0, sizeof push, &push);
        vkCmdDraw(cmd, 3, 1, 0, 0);
        vkx::end_drawing(cmd, target, pixels);
        vkx::submit_and_wait(dev, cmds);

        const auto* px = static_cast<const float*>(pixels.mapped);
        auto at = [&](uint32_t x, uint32_t y) {
            return px + (y * SIZE.width + x) * 2;
        };
        std::puts(metallic == 0 ? "비금속(metallic 0)" : "금속(metallic 1)");
        for (uint32_t y = 0; y < SIZE.height; ++y) {
            std::string row;
            for (uint32_t x = 0; x < SIZE.width; ++x) {
                float sum = at(x, y)[0] + at(x, y)[1];
                // 1을 넘는 밝기도 글자 열 개 안에 들어오게 누른다
                row += " .:-=+*#%@"[int(sum / (1 + sum) * 9.99f)];
            }
            row.erase(row.find_last_not_of(' ') + 1);   // 줄 끝의 빈칸은 뗀다
            std::puts(row.c_str());
        }
        for (uint32_t ball = 0; ball < 3; ++ball) {
            float diffuse = 0, peak = at(ball * 20 + 8, 3)[1];
            int wide = 0;   // 꼭대기의 절반보다 밝은 픽셀 수
            for (uint32_t y = 0; y < SIZE.height; ++y)
                for (uint32_t x = ball * 20; x < ball * 20 + 20; ++x) {
                    diffuse += at(x, y)[0];
                    wide += at(x, y)[1] > peak / 2;
                }
            std::printf("  거칠기 %.1f: 난반사 합 %4.1f, 정반사 꼭대기 %6.2f, "
                        "넓이 %2d픽셀\n", 0.2 + 0.3 * ball, diffuse, peak, wide);
        }
    }

    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    vkx::destroy(dev, pixels);
    vkx::destroy(dev, target);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **D, Vis, F**: 정반사는 `F * D * Vis`입니다. D는 미세면 가운데 H(빛과 눈의 한가운데 방향)를 바라보는 것의 비율로 하이라이트의 모양을 정하고, Vis는 미세면끼리 가리는 정도, F는 비스듬히 볼수록 커지는 반사율입니다. 식에는 거칠기를 제곱한 `a`를 넣는데, 거칠기 값을 고르게 바꿀 때 눈에 보이는 변화도 고르게 하려는 것입니다
- **거칠기**: 거칠수록 하이라이트가 넓고 어둡습니다. 금속 공의 꼭대기는 132.65에서 0.51로 내려가고, 꼭대기의 절반보다 밝은 픽셀은 1개에서 52개로 늘었습니다
- **금속성**: 비금속은 정면에서 4%만 곧바로 반사하고(`f0 = 0.04`), 나머지는 속에 들어갔다 나오는 난반사가 됩니다. 금속은 난반사가 없고 바탕색이 곧 반사율입니다. 그래서 금속 쪽 난반사 합은 0이고 꼭대기는 0.8 / 0.04 = 20배입니다
- **금속 공이 어두운 까닭**: 빛이 하나뿐이면 금속은 하이라이트 밖이 검습니다. 실제 금속은 둘레 전체를 비추므로, 렌더러는 환경 맵을 거칠기별로 미리 흐려 두고 거기서 둘레의 빛을 읽어 더합니다(이미지 기반 조명)
- **1을 넘는 밝기**: 꼭대기 132.65는 화면이 낼 수 있는 1을 훌쩍 넘습니다. 그래서 조명은 부동소수 포맷의 타깃에 그리고, 화면에 맞게 누르는 일은 [「후처리」](#/read/cpp-vulkan/tech-post)에서 합니다

## 재질 값은 어디서 오나

| 값 | 텍스처의 채널 | 포맷 |
| --- | --- | --- |
| 바탕색 | RGB | `_SRGB`(읽을 때 선형 값으로 풀린다) |
| 거칠기, 금속성 | G, B | `_UNORM`(숫자 그대로) |
| 법선 | RGB(압축하면 BC5의 두 채널) | `_UNORM` |

채널 배치는 glTF의 것입니다. 식은 모두 선형 값으로 셈하므로 색을 담은 바탕색 텍스처만 SRGB 포맷으로 만듭니다([「이미지, 뷰, 샘플러」](#/read/cpp-vulkan/tex-image)). 빛이 여럿이면 빛마다 `(diffuse + specular) * NoL`에 그 빛의 세기를 곱해 더하고, 빛의 목록은 유니폼 버퍼나 스토리지 버퍼로 넘깁니다.

<Note>
이 식은 빛이 미세면에 한 번만 튕긴다고 봅니다. 거친 표면에서는 여러 번 튕겨 나오는 빛이 빠져 실제보다 어둡고, glTF 명세도 이 간단한 식이 에너지 보존을 엄밀히 지키지는 못한다고 적습니다. 또 거칠기가 0에 가까우면 D의 분모가 0에 가까워져 값이 터지므로 아래쪽을 막아 둡니다(Filament는 0.089).
</Note>

## 정리

- 정반사는 미세면의 분포 D, 가림 Vis, 반사율 F의 곱이고, 거칠기가 하이라이트의 넓이와 밝기를 정한다
- 금속성은 난반사를 없애고 바탕색을 반사율로 쓰게 하며, 비금속의 정면 반사율은 4%다
- 조명은 선형 값으로 셈하고, 1을 넘는 결과를 부동소수 타깃에 담아 후처리에서 화면에 맞춘다
