정점 셰이더는 정점마다 돌아 클립 좌표(`gl_Position`)와 다음 단계로 넘길 값을 내고, 프래그먼트 셰이더는 래스터라이저가 만든 프래그먼트마다 돌아 첨부물에 쓸 값을 냅니다. 둘 사이의 값은 location 번호로 이어지고, 그 사이에서 어떻게 보간할지는 한정자로 정합니다. 아래 프로그램은 삼각형 하나를 인스턴스 셋으로 나란히 그려, 보간 방식과 버리기를 비교합니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_draw.hpp,vkx_pipeline.hpp
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"

// 인스턴스 셋을 나란히: 0번은 보통 보간, 1번은 flat, 2번은 체스판으로 버리기
const char* VS = R"(#version 450
layout(location = 0) out float shade;              // 기본(smooth) 보간
layout(location = 1) flat out float first;         // 첫 정점의 값 그대로
layout(location = 2) flat out int which;
const vec2 corner[3] = vec2[](vec2(0.5, 0.05), vec2(0.95, 0.95), vec2(0.05, 0.95));
const float tone[3] = float[](0.2, 1.0, 0.5);
void main() {
    vec2 p = corner[gl_VertexIndex];
    float x = (float(gl_InstanceIndex) + p.x) / 3.0;   // 인스턴스마다 3분의 1칸
    gl_Position = vec4(x * 2.0 - 1.0, p.y * 2.0 - 1.0, 0.0, 1.0);
    shade = first = tone[gl_VertexIndex];
    which = gl_InstanceIndex;
})";

const char* FS = R"(#version 450
layout(location = 0) in float shade;
layout(location = 1) flat in float first;
layout(location = 2) flat in int which;
layout(location = 0) out float value;
void main() {
    ivec2 px = ivec2(gl_FragCoord.xy);    // 왼쪽 위가 (0, 0)인 픽셀 좌표
    if (which == 2 && (px.x + px.y) % 2 == 0) discard;
    value = which == 1 ? first : shade;
})";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const uint32_t W = 48, H = 12;
    vkx::Image target = vkx::make_image(
        dev, {W, H}, VK_FORMAT_R32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels =
        vkx::make_buffer(dev, W * H * 4, VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    auto vs = vkx::make_shader(dev, vkx::compile(VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipelineLayoutCreateInfo lci{
        .sType = VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};
    VkPipelineLayout layout;
    vkCreatePipelineLayout(dev.handle, &lci, nullptr, &layout);
    VkPipeline pipeline = vkx::make_pipeline(dev, {.vertex = vs, .fragment = fs,
                                                   .layout = layout,
                                                   .color = target.format});

    VkCommandBuffer cmd = vkx::begin(cmds);
    vkx::begin_drawing(cmd, target, {{-1.0f}});
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
    vkCmdDraw(cmd, 3, 3, 0, 0);   // 정점 3개를 인스턴스 3번
    vkx::end_drawing(cmd, target, pixels);
    vkx::submit_and_wait(dev, cmds);

    auto* v = static_cast<const float*>(pixels.mapped);
    const char* ramp = ".:-=+*#%@";
    for (uint32_t i = 0; i < W * H; ++i) {
        if (i % W == 0) std::putchar('|');
        std::putchar(v[i] < 0 ? ' ' : ramp[int(v[i] * 8.999f)]);
        if (i % W == W - 1) std::puts("|");
    }
    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    vkx::destroy(dev, pixels);
    vkx::destroy(dev, target);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **인스턴스**: `vkCmdDraw(3, 3, 0, 0)`은 정점 셋을 세 번 그리고, 정점 셰이더는 `gl_InstanceIndex`로 몇 번째인지 압니다. 벌칸 GLSL에는 OpenGL의 `gl_VertexID`·`gl_InstanceID` 대신 `gl_VertexIndex`·`gl_InstanceIndex`가 있습니다. 정점 쪽은 값이 같지만, 인스턴스 쪽은 첫 인스턴스 번호가 더해진다는 점이 다릅니다(OpenGL의 `gl_InstanceID`는 늘 0부터)
- **보간 한정자**: 아무것도 안 붙이면 `smooth`로 원근을 고려해 보간합니다(왼쪽). `flat`은 보간하지 않고 유발 정점(벌칸은 기본이 첫 정점)의 값을 그대로 써서 가운데 삼각형은 0.2 하나로 칠해졌습니다. 화면에서 곧게 보간하는 `noperspective`도 있고, 정수 값은 반드시 `flat`이어야 합니다
- **gl_FragCoord와 discard**: `gl_FragCoord`는 왼쪽 위가 원점인 창 좌표이고 픽셀 가운데에서 .5로 끝납니다. 오른쪽 삼각형은 x+y가 짝수인 픽셀을 `discard`로 버려 체스판처럼 비었습니다
- **출력**: `layout(location = n) out`은 n번째 색 첨부물로 갑니다. 여기서는 첨부물이 하나(R32_SFLOAT)라 `float` 하나를 냅니다

## 벌칸의 클립 공간

정점 셰이더가 내는 클립 좌표는 w로 나뉘어 정규화 장치 좌표가 됩니다. 벌칸에서는 x와 y가 -1~1이고 **y가 아래로 커지며**, 깊이 z는 0~1입니다. OpenGL은 y가 위로 커지고 z가 -1~1이라, OpenGL 기준으로 만든 투영 행렬을 그대로 쓰면 그림이 뒤집히고 깊이가 어긋납니다. [「GPU 파이프라인」](#/read/cpp-vulkan/vk-gpu-pipeline)의 삼각형에서 y가 -0.9인 꼭짓점이 위에 그려진 것도 이 때문입니다. 보통은 투영 행렬에서 y를 뒤집고 깊이를 0~1로 맞추거나(GLM은 `GLM_FORCE_DEPTH_ZERO_TO_ONE`), 높이가 음수인 뷰포트로 y를 뒤집습니다(1.1부터 핵심).

<Note>
같은 정점이 인덱스 버퍼에 여러 번 나오면 구현은 앞서 계산한 결과를 다시 쓸 수 있습니다. 그래서 정점 셰이더가 정점마다 꼭 한 번 돈다고 기대하고 부작용(저장 버퍼에 쓰기)을 넣으면 안 됩니다. 프래그먼트 셰이더에서 `gl_FragDepth`에 쓰면 깊이 시험을 셰이더보다 먼저 할 수 없게 되므로 느려질 수 있습니다([「깊이 버퍼」](#/read/cpp-vulkan/tex-depth)).
</Note>

## 정리

- 정점 셰이더는 `gl_VertexIndex`와 `gl_InstanceIndex`(첫 인스턴스 번호가 더해진 값)로 클립 좌표를 내고, 값은 location으로 프래그먼트 셰이더에 이어진다
- 보간은 기본이 `smooth`이고 `flat`(유발 정점의 값, 정수는 필수)과 `noperspective`가 있으며, `gl_FragCoord`는 왼쪽 위가 원점이고 `discard`는 그 프래그먼트를 버린다
- 벌칸의 정규화 장치 좌표는 y가 아래로 커지고 깊이가 0~1이라, OpenGL 기준 행렬은 y와 깊이를 고쳐 쓴다
