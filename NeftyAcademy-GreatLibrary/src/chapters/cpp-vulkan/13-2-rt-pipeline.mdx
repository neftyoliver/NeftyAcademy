레이 트레이싱 파이프라인은 광선이 장면을 지나며 겪는 일마다 다른 셰이더를 부릅니다. **광선 생성** 셰이더가 픽셀마다 돌며 `traceRayEXT`로 광선을 쏘면, 가속 구조를 따라간 끝에 가장 가까이 맞은 물체의 **히트** 셰이더나 아무것도 맞지 않았을 때의 **미스** 셰이더가 불려, **페이로드**에 답을 적어 돌려줍니다. 먼저 이 쪽과 다음 쪽이 함께 그릴 장면을 만듭니다.

```cpp vk file=vkx_scene.hpp with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_accel.hpp
#pragma once
#include "vkx_accel.hpp"

namespace vkx {

// 이 장의 장면: 앞줄에 세모 셋(번호 0~2), 그 뒤에 벽 하나(번호 3)
struct Scene {
    Accel triangle, wall;   // 아래층 둘
    Accel top;              // 위층: 인스턴스 넷
};

// middle은 가운데 세모(번호 1)의 인스턴스에 더할 플래그
inline Scene make_scene(const Device& d, Commands& c,
                        VkGeometryInstanceFlagsKHR middle = 0) {
    Scene s;
    s.triangle = make_blas(d, c, {-0.8f, 0.8f, 0, 0.8f, 0.8f, 0, 0, -0.8f, 0});
    s.wall = make_blas(d, c, {-3.5f, -0.9f, 0, 3.5f, -0.9f, 0, 3.5f, 0.9f, 0,
                              -3.5f, -0.9f, 0, 3.5f, 0.9f, 0, -3.5f, 0.9f, 0});
    std::vector<VkAccelerationStructureInstanceKHR> all(4);
    for (uint32_t i = 0; i < 4; ++i) {
        bool front = i < 3;   // 세모는 z = 0에 나란히, 벽은 z = 2에
        float x = front ? 2.5f * i - 2.5f : 0, z = front ? 0 : 2;
        all[i].transform = {{{1, 0, 0, x}, {0, 1, 0, 0}, {0, 0, 1, z}}};
        all[i].instanceCustomIndex = i;
        all[i].mask = 0xff;
        all[i].instanceShaderBindingTableRecordOffset = front ? 0 : 1;
        all[i].flags = VK_GEOMETRY_INSTANCE_TRIANGLE_FACING_CULL_DISABLE_BIT_KHR |
                       (i == 1 ? middle : 0);
        all[i].accelerationStructureReference =
            front ? s.triangle.address : s.wall.address;
    }
    s.top = make_tlas(d, c, all);
    return s;
}

inline void destroy(const Device& d, Scene& s) {
    destroy(d, s.top);
    destroy(d, s.triangle);
    destroy(d, s.wall);
}

}  // namespace vkx
```

세모는 히트 그룹 0번을, 벽은 1번을 쓰도록 인스턴스에 적어 두었습니다. 아래 프로그램이 이 장면에 광선 48×12개를 쏩니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_pipeline.hpp,vkx_accel.hpp,vkx_scene.hpp
#include <string>
#include "vkx_pipeline.hpp"
#include "vkx_scene.hpp"

#define RT "#version 460\n#extension GL_EXT_ray_tracing : require\n"
// 광선 생성: 픽셀마다 +z로 나란한 광선을 쏘고, 돌아온 값을 이미지에 적는다
const char* RAYGEN = RT R"(
layout(set = 0, binding = 0) uniform accelerationStructureEXT scene;
layout(set = 0, binding = 1, r32ui) uniform uimage2D target;
layout(location = 0) rayPayloadEXT uint seen;
void main() {
    vec2 uv = (vec2(gl_LaunchIDEXT.xy) + 0.5) / vec2(gl_LaunchSizeEXT.xy);
    vec3 origin = vec3(uv.x * 8.0 - 4.0, uv.y * 2.0 - 1.0, -5.0);
    traceRayEXT(scene, gl_RayFlagsOpaqueEXT, 0xff, 0, 0, 0,   // 끝의 0: 미스 0번
                origin, 0.0, vec3(0, 0, 1), 100.0, 0);        // 페이로드 0번
    imageStore(target, ivec2(gl_LaunchIDEXT.xy), uvec4(seen));
})";
// 미스 0: 첫 광선이 아무것도 맞히지 못했다
const char* MISS = RT R"(
layout(location = 0) rayPayloadInEXT uint seen;
void main() { seen = 0u; })";
// 미스 1: 그림자 광선이 빛까지 막히지 않고 갔다
const char* MISS_SHADOW = RT R"(
layout(location = 1) rayPayloadInEXT bool lit;
void main() { lit = true; })";
// 히트 그룹 0(세모): 인스턴스 번호를 돌려준다
const char* HIT_TRIANGLE = RT R"(
layout(location = 0) rayPayloadInEXT uint seen;
void main() { seen = 1u + uint(gl_InstanceCustomIndexEXT); })";
// 히트 그룹 1(벽): 맞은 자리에서 빛 쪽으로 그림자 광선을 한 번 더 쏜다
const char* HIT_WALL = RT R"(
layout(set = 0, binding = 0) uniform accelerationStructureEXT scene;
layout(location = 0) rayPayloadInEXT uint seen;
layout(location = 1) rayPayloadEXT bool lit;
void main() {
    vec3 at = gl_WorldRayOriginEXT + gl_WorldRayDirectionEXT * gl_HitTEXT;
    lit = false;   // 무엇에든 닿으면 그만이고, 히트 셰이더는 부르지 않는다
    traceRayEXT(scene, gl_RayFlagsTerminateOnFirstHitEXT |
                       gl_RayFlagsSkipClosestHitShaderEXT, 0xff, 0, 0, 1,
                at, 0.01, normalize(vec3(-0.6, -0.2, -1.0)), 100.0, 1);
    seen = lit ? 5u : 4u;
})";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_rt_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    vkx::Scene scene = vkx::make_scene(dev, cmds);
    const VkExtent2D SIZE{48, 12};
    vkx::Image target = vkx::make_image(
        dev, SIZE, VK_FORMAT_R32_UINT,
        VK_IMAGE_USAGE_STORAGE_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels = vkx::make_buffer(dev, SIZE.width * SIZE.height * 4,
                                          VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    const auto IMAGE = VK_DESCRIPTOR_TYPE_STORAGE_IMAGE;
    vkx::Bindings set = vkx::make_bindings(
        dev, {VK_DESCRIPTOR_TYPE_ACCELERATION_STRUCTURE_KHR, IMAGE});
    vkx::bind_accel(dev, set, 0, scene.top);
    vkx::bind_image(dev, set, 1, IMAGE, target.view, VK_IMAGE_LAYOUT_GENERAL);
    VkPipelineLayout layout = vkx::make_layout(dev, {set.layout});

    // 단계 다섯과, 단계마다 하나씩인 그룹 다섯. 히트 그룹은 종류가 다르다
    const char* sources[5] = {RAYGEN, MISS, MISS_SHADOW, HIT_TRIANGLE, HIT_WALL};
    VkPipelineShaderStageCreateInfo stages[5];
    VkRayTracingShaderGroupCreateInfoKHR groups[5];
    for (uint32_t i = 0; i < 5; ++i) {
        bool hit = i >= 3;
        auto kind = i == 0 ? shaderc_raygen_shader
                           : hit ? shaderc_closesthit_shader : shaderc_miss_shader;
        stages[i] = {
            .sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
            .stage = i == 0 ? VK_SHADER_STAGE_RAYGEN_BIT_KHR
                            : hit ? VK_SHADER_STAGE_CLOSEST_HIT_BIT_KHR
                                  : VK_SHADER_STAGE_MISS_BIT_KHR,
            .module = vkx::make_shader(dev, vkx::compile(sources[i], kind)),
            .pName = "main"};
        groups[i] = {
            .sType = VK_STRUCTURE_TYPE_RAY_TRACING_SHADER_GROUP_CREATE_INFO_KHR,
            .type = hit ? VK_RAY_TRACING_SHADER_GROUP_TYPE_TRIANGLES_HIT_GROUP_KHR
                        : VK_RAY_TRACING_SHADER_GROUP_TYPE_GENERAL_KHR,
            .generalShader = hit ? VK_SHADER_UNUSED_KHR : i,
            .closestHitShader = hit ? i : VK_SHADER_UNUSED_KHR,
            .anyHitShader = VK_SHADER_UNUSED_KHR,
            .intersectionShader = VK_SHADER_UNUSED_KHR};
    }
    VkRayTracingPipelineCreateInfoKHR ci{
        .sType = VK_STRUCTURE_TYPE_RAY_TRACING_PIPELINE_CREATE_INFO_KHR,
        .stageCount = 5, .pStages = stages, .groupCount = 5, .pGroups = groups,
        .maxPipelineRayRecursionDepth = 2,   // 히트 셰이더가 한 번 더 쏜다
        .layout = layout};
    VkPipeline pipeline;
    VKX_KHR(dev, vkCreateRayTracingPipelinesKHR)(
        dev.handle, VK_NULL_HANDLE, VK_NULL_HANDLE, 1, &ci, nullptr, &pipeline);

    // 셰이더 바인딩 테이블: 그룹의 핸들을 한 칸에 하나씩 옮겨 적은 버퍼
    VkPhysicalDeviceRayTracingPipelinePropertiesKHR rt{
        .sType =
            VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_RAY_TRACING_PIPELINE_PROPERTIES_KHR};
    VkPhysicalDeviceProperties2 props{
        .sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_PROPERTIES_2, .pNext = &rt};
    vkGetPhysicalDeviceProperties2(dev.gpu, &props);
    auto up = [](VkDeviceAddress v, VkDeviceAddress a) {
        return (v + a - 1) / a * a;   // a의 배수로 올림
    };
    const uint32_t HANDLE = rt.shaderGroupHandleSize;
    const VkDeviceSize SLOT = up(HANDLE, rt.shaderGroupBaseAlignment);
    std::vector<uint8_t> handles(5 * HANDLE);
    VKX_KHR(dev, vkGetRayTracingShaderGroupHandlesKHR)(
        dev.handle, pipeline, 0, 5, handles.size(), handles.data());
    vkx::Buffer table = vkx::make_buffer(
        dev, 6 * SLOT, VK_BUFFER_USAGE_SHADER_BINDING_TABLE_BIT_KHR | vkx::ADDRESS,
        true);
    const VkDeviceAddress base = up(vkx::address_of(dev, table), SLOT);
    auto* first = static_cast<uint8_t*>(table.mapped) +
                  (base - vkx::address_of(dev, table));
    for (uint32_t i = 0; i < 5; ++i)
        std::memcpy(first + i * SLOT, &handles[i * HANDLE], HANDLE);
    VkStridedDeviceAddressRegionKHR raygen{base, SLOT, SLOT};           // 칸 0
    VkStridedDeviceAddressRegionKHR miss{base + SLOT, SLOT, 2 * SLOT};  // 칸 1, 2
    VkStridedDeviceAddressRegionKHR hit{base + 3 * SLOT, SLOT, 2 * SLOT};   // 3, 4
    VkStridedDeviceAddressRegionKHR callable{};

    const auto GENERAL = VK_IMAGE_LAYOUT_GENERAL;
    const auto TRACE = VK_PIPELINE_STAGE_2_RAY_TRACING_SHADER_BIT_KHR;
    const auto POINT = VK_PIPELINE_BIND_POINT_RAY_TRACING_KHR;
    VkCommandBuffer cmd = vkx::begin(cmds);
    vkx::transition(cmd, target, VK_IMAGE_LAYOUT_UNDEFINED, GENERAL,
                    VK_PIPELINE_STAGE_2_NONE, VK_ACCESS_2_NONE, TRACE,
                    VK_ACCESS_2_SHADER_STORAGE_WRITE_BIT);
    vkCmdBindPipeline(cmd, POINT, pipeline);
    vkCmdBindDescriptorSets(cmd, POINT, layout, 0, 1, &set.set, 0, nullptr);
    VKX_KHR(dev, vkCmdTraceRaysKHR)(cmd, &raygen, &miss, &hit, &callable,
                                    SIZE.width, SIZE.height, 1);
    vkx::transition(cmd, target, GENERAL, VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL,
                    TRACE, VK_ACCESS_2_SHADER_STORAGE_WRITE_BIT,
                    VK_PIPELINE_STAGE_2_COPY_BIT, VK_ACCESS_2_TRANSFER_READ_BIT);
    vkx::copy_to_buffer(cmd, target, pixels);
    vkx::submit_and_wait(dev, cmds);

    const auto* px = static_cast<const uint32_t*>(pixels.mapped);
    for (uint32_t y = 0; y < SIZE.height; ++y) {
        std::string row;   // 0 미스, 1~3 세모, 4 그늘진 벽, 5 빛 받는 벽
        for (uint32_t x = 0; x < SIZE.width; ++x)
            row += ".123#-"[px[y * SIZE.width + x]];
        std::puts(row.c_str());
    }

    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    for (auto& s : stages) vkDestroyShaderModule(dev.handle, s.module, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkx::destroy(dev, set);
    for (vkx::Buffer* b : {&table, &pixels}) vkx::destroy(dev, *b);
    vkx::destroy(dev, target);
    vkx::destroy(dev, scene);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **단계와 그룹**: 파이프라인에는 셰이더 단계들과 그것을 묶은 그룹을 넘깁니다. 광선 생성 셰이더와 미스 셰이더는 하나씩 일반 그룹이 됩니다. 히트 그룹은 가장 가까운 히트 셰이더에 더해, 교차 후보마다 불려 구멍 난 부분을 걸러 내는 any-hit 셰이더를 함께 담을 수 있습니다(삼각형이 아닌 도형이면 교차 셰이더도 넣습니다)
- **셰이더 바인딩 테이블**: 어느 셰이더를 부를지는 버퍼에 적은 표에서 찾습니다. `vkGetRayTracingShaderGroupHandlesKHR`로 받은 그룹 핸들(lavapipe 32바이트)을 한 칸에 하나씩 옮겨 적고, `vkCmdTraceRaysKHR`에 광선 생성·미스·히트·호출 가능 셰이더의 구역을 주소, 칸 간격, 크기로 알립니다. 구역의 주소는 `shaderGroupBaseAlignment`의 배수여야 합니다
- **칸 고르기**: 히트 구역에서는 (인스턴스에 적은 오프셋 + 기하 번호 × `sbtRecordStride` + `sbtRecordOffset`)번째 칸을 씁니다. 뒤의 둘은 `traceRayEXT`의 다섯째와 넷째 인자입니다. 그래서 벽에 맞은 광선만 1번 그룹으로 갔습니다. 미스 구역에서는 여섯째 인자가 곧 칸 번호입니다
- **페이로드**: 쏘는 쪽은 `rayPayloadEXT`, 받는 쪽은 `rayPayloadInEXT`로 선언하고, `traceRayEXT`의 마지막 인자가 쏘는 쪽의 몇 번 location을 넘길지 고릅니다
- **그림자 광선**: 벽의 히트 셰이더는 맞은 자리에서 빛 쪽으로 한 번 더 쏩니다. 막혔는지만 알면 되므로 첫 교차에서 멈추고 히트 셰이더도 건너뛰게 했고, 끝까지 막히지 않았을 때만 미스 1번이 `lit`을 켭니다. 시작을 0.01 띄운 것은 벽 자신에게 다시 맞지 않게 하려는 것입니다

## 재귀와 표의 쓰임

광선 생성 셰이더의 깊이는 0이고, `traceRayEXT`로 불린 셰이더는 부른 쪽보다 1 깊습니다. 벽의 히트 셰이더(깊이 1)가 쏜 광선의 미스 셰이더는 깊이 2여서 파이프라인을 `maxPipelineRayRecursionDepth` 2로 만들었습니다. 이 값은 장치의 `maxRayRecursionDepth`를 넘을 수 없는데, 어느 장치에나 보장되는 것은 1뿐입니다(lavapipe는 31). 어디서나 돌게 하려면 히트 셰이더는 맞은 자리만 페이로드로 돌려주고 광선 생성 셰이더가 이어서 쏘게 짭니다.

표의 한 칸에는 핸들 뒤에 그 칸만의 데이터(재질 번호, 버퍼 주소)를 더 적을 수 있고, 셰이더는 `layout(shaderRecordEXT)` 블록으로 읽습니다.

<Note>
불투명한 기하에는 `VK_GEOMETRY_OPAQUE_BIT_KHR`를 줍니다. 그러면 교차 후보마다 any-hit 셰이더를 부르지 않아 탐색이 끊기지 않습니다. 잎사귀처럼 텍스처의 알파로 구멍을 내야 하는 기하만 이 표시를 빼고 any-hit 셰이더에서 걸러 냅니다.
</Note>

## 정리

- 광선 생성 셰이더가 `traceRayEXT`로 광선을 쏘면 가장 가까운 히트 셰이더나 미스 셰이더가 페이로드에 답을 적는다
- 셰이더 바인딩 테이블은 그룹 핸들을 칸마다 적은 버퍼이고, 히트 칸은 인스턴스의 오프셋과 `traceRayEXT`의 인자로, 미스 칸은 미스 번호로 고른다
- 셰이더가 광선을 다시 쏘면 깊이가 하나 늘며, 어느 장치에나 보장되는 깊이는 1뿐이다
