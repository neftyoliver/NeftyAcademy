지금까지의 삼각형은 정점 좌표를 셰이더 안의 배열에 적고 `gl_VertexIndex`로 골랐습니다. 실제 메시는 파일에서 읽어 **정점 버퍼**에 담고, 파이프라인의 정점 입력 상태로 그 버퍼를 어떻게 읽을지 알립니다. **바인딩**은 버퍼 하나를 몇 바이트씩 끊어 읽을지(stride)와 정점마다 넘어갈지 인스턴스마다 넘어갈지(input rate)를, **속성**은 끊은 칸의 어디(offset)를 어떤 포맷으로 몇 번 location에 줄지를 정합니다. **인덱스 버퍼**는 삼각형마다 몇 번 정점을 쓸지 적은 목록이어서, 여러 삼각형이 함께 쓰는 정점을 한 번만 저장하게 해 줍니다. 아래 프로그램은 위치와 밝기를 버퍼 둘에 나눠 담고, 인덱스 18개로 육각형 둘을 그립니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_pipeline.hpp,vkx_draw.hpp
#include <cmath>
#include <cstddef>
#include <cstring>
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"

const char* VS = R"(#version 450
layout(location = 0) in vec2 pos;       // 바인딩 0의 버퍼에서
layout(location = 1) in float shade;    // 바인딩 1의 버퍼에서
layout(location = 0) out float v_shade;
void main() {
    gl_Position = vec4(pos, 0.0, 1.0);
    v_shade = shade;
})";

const char* FS = R"(#version 450
layout(location = 0) in float v_shade;
layout(location = 0) out float value;
void main() { value = v_shade; })";

struct Position { float x, y; };

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    const uint32_t W = 48, H = 13;
    vkx::Image target = vkx::make_image(
        dev, {W, H}, VK_FORMAT_R32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels =
        vkx::make_buffer(dev, W * H * 4, VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);

    // 육각형 둘: 가운데 정점 하나와 둘레 여섯. 오른쪽 것은 왼쪽 것 바로 뒤에 둔다
    std::vector<Position> pos;
    std::vector<float> shade;
    for (int s = 0; s < 2; ++s) {
        float cx = s ? 0.5f : -0.5f;
        pos.push_back({cx, 0.0f});
        shade.push_back(s ? 0.5f : 1.0f);
        for (int k = 0; k < 6; ++k) {
            float a = float(k) * 3.14159265f / 3.0f;
            pos.push_back({cx + 0.4f * std::cos(a), 0.85f * std::sin(a)});
            shade.push_back(0.0f);
        }
    }
    std::vector<uint16_t> index;    // 가운데와 둘레의 이웃한 두 점으로 삼각형 여섯
    for (int k = 0; k < 6; ++k)
        index.insert(index.end(), {0, uint16_t(1 + k), uint16_t(1 + (k + 1) % 6)});

    auto upload = [&](const void* data, size_t bytes, VkBufferUsageFlags usage) {
        vkx::Buffer b = vkx::make_buffer(dev, bytes, usage, true);
        std::memcpy(b.mapped, data, bytes);
        return b;
    };
    vkx::Buffer pbuf = upload(pos.data(), pos.size() * sizeof(Position),
                              VK_BUFFER_USAGE_VERTEX_BUFFER_BIT);
    vkx::Buffer sbuf = upload(shade.data(), shade.size() * sizeof(float),
                              VK_BUFFER_USAGE_VERTEX_BUFFER_BIT);
    vkx::Buffer ibuf = upload(index.data(), index.size() * sizeof(uint16_t),
                              VK_BUFFER_USAGE_INDEX_BUFFER_BIT);

    // 바인딩: 버퍼를 몇 바이트씩 끊어 읽을지. 속성: 칸의 어디를 어느 location에
    VkVertexInputBindingDescription bindings[] = {
        {0, sizeof(Position), VK_VERTEX_INPUT_RATE_VERTEX},
        {1, sizeof(float), VK_VERTEX_INPUT_RATE_VERTEX}};
    VkVertexInputAttributeDescription attributes[] = {
        {0, 0, VK_FORMAT_R32G32_SFLOAT, offsetof(Position, x)},
        {1, 1, VK_FORMAT_R32_SFLOAT, 0}};
    VkPipelineVertexInputStateCreateInfo input{
        .sType = VK_STRUCTURE_TYPE_PIPELINE_VERTEX_INPUT_STATE_CREATE_INFO,
        .vertexBindingDescriptionCount = 2, .pVertexBindingDescriptions = bindings,
        .vertexAttributeDescriptionCount = 2,
        .pVertexAttributeDescriptions = attributes};

    auto vs = vkx::make_shader(dev, vkx::compile(VS, shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipelineLayoutCreateInfo lci{
        .sType = VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};
    VkPipelineLayout layout;
    vkCreatePipelineLayout(dev.handle, &lci, nullptr, &layout);
    VkPipeline pipeline = vkx::make_pipeline(
        dev, {.vertex = vs, .fragment = fs, .layout = layout,
              .color = target.format, .vertex_input = &input});

    // 파이프라인 통계 쿼리: 입력 조립이 읽은 정점 수와 정점 셰이더 실행 수를 센다
    VkQueryPoolCreateInfo qci{
        .sType = VK_STRUCTURE_TYPE_QUERY_POOL_CREATE_INFO,
        .queryType = VK_QUERY_TYPE_PIPELINE_STATISTICS, .queryCount = 1,
        .pipelineStatistics =
            VK_QUERY_PIPELINE_STATISTIC_INPUT_ASSEMBLY_VERTICES_BIT |
            VK_QUERY_PIPELINE_STATISTIC_VERTEX_SHADER_INVOCATIONS_BIT};
    VkQueryPool pool;
    vkCreateQueryPool(dev.handle, &qci, nullptr, &pool);

    VkCommandBuffer cmd = vkx::begin(cmds);
    vkCmdResetQueryPool(cmd, pool, 0, 1);
    vkx::begin_drawing(cmd, target, {{-1.0f}});
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
    VkBuffer streams[] = {pbuf.handle, sbuf.handle};
    VkDeviceSize offsets[] = {0, 0};
    vkCmdBindVertexBuffers(cmd, 0, 2, streams, offsets);
    vkCmdBindIndexBuffer(cmd, ibuf.handle, 0, VK_INDEX_TYPE_UINT16);
    vkCmdBeginQuery(cmd, pool, 0, 0);
    vkCmdDrawIndexed(cmd, 18, 1, 0, 0, 0);    // 왼쪽: 인덱스 그대로
    vkCmdEndQuery(cmd, pool, 0);
    vkCmdDrawIndexed(cmd, 18, 1, 0, 7, 0);    // 오른쪽: 인덱스마다 7을 더해서
    vkx::end_drawing(cmd, target, pixels);
    vkx::submit_and_wait(dev, cmds);

    auto* v = static_cast<const float*>(pixels.mapped);
    const char* ramp = ".:-=+*#%@";
    for (uint32_t i = 0; i < W * H; ++i) {
        if (i % W == 0) std::putchar('|');
        std::putchar(v[i] < 0 ? ' ' : ramp[int(v[i] * 8.999f)]);
        if (i % W == W - 1) std::puts("|");
    }
    uint64_t stat[2];
    vkGetQueryPoolResults(dev.handle, pool, 0, 1, sizeof stat, stat, sizeof stat,
                          VK_QUERY_RESULT_64_BIT | VK_QUERY_RESULT_WAIT_BIT);
    std::printf("왼쪽 육각형: 인덱스 %llu개, 정점 셰이더 %llu번\n",
                (unsigned long long)stat[0], (unsigned long long)stat[1]);
    vkDestroyQueryPool(dev.handle, pool, nullptr);
    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    for (vkx::Buffer* b : {&pbuf, &sbuf, &ibuf, &pixels}) vkx::destroy(dev, *b);
    vkx::destroy(dev, target);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

- **바인딩과 속성**: 위치는 바인딩 0(칸마다 8바이트), 밝기는 바인딩 1(4바이트)에서 읽고, `vkCmdBindVertexBuffers`가 바인딩 번호마다 버퍼와 시작 오프셋을 붙입니다. 위치와 밝기를 구조체 하나에 섞어 담으면(interleaved) 바인딩 하나에 속성 둘이 되는데, 그림자처럼 위치만 필요한 패스가 위치 버퍼만 읽도록 일부러 나누기도 합니다. 인스턴스마다 넘어가는 바인딩은 [「인스턴싱과 간접 그리기」](#/read/cpp-vulkan/tech-instancing)에서 씁니다
- **인덱스가 아끼는 것**: 육각형 하나는 삼각형 6개, 꼭짓점 18개지만 서로 다른 정점은 7개입니다. 두 육각형을 인덱스 없이 담으면 432바이트(36 × 12)이고 인덱스로는 204바이트(14 × 12 + 18 × 2)입니다. 더 큰 이득은 셰이더 쪽에 있어서, lavapipe는 인덱스 18개를 처리하며 정점 셰이더를 7번만 돌렸습니다. 같은 인덱스가 다시 나오면 앞서 셰이딩한 결과를 다시 써도 된다고 명세가 허용하기 때문이며, 언제 얼마나 다시 쓸지는 구현마다 다릅니다
- **vertexOffset**: 오른쪽 육각형은 같은 인덱스 18개를 쓰되 `vkCmdDrawIndexed`의 `vertexOffset`에 7을 주어 정점 7~13을 읽었습니다. 그래서 여러 메시를 큰 정점·인덱스 버퍼 하나에 이어 담고 `firstIndex`와 `vertexOffset`으로 골라 그리며, 버퍼 전체가 커도 메시마다 인덱스는 16비트로 둘 수 있습니다
- **인덱스 타입**: 메시의 정점이 65536개를 넘지 않으면 `VK_INDEX_TYPE_UINT16`, 넘으면 `UINT32`를 씁니다. 1.4에서 코어가 된 `UINT8`은 `indexTypeUint8` 기능을 확인하고 씁니다. 띠(strip)나 부채(fan) 토폴로지에서 `primitiveRestartEnable`을 켜면 인덱스 타입의 최댓값(`UINT16`이면 0xFFFF)이 도형을 끊고 새로 시작합니다

<Note>
정점 속성의 포맷은 셰이더 쪽 타입과 달라도 됩니다. 색을 `VK_FORMAT_R8G8B8A8_UNORM`(4바이트)으로 담으면 셰이더에는 0~1의 `vec4`로, 법선을 `VK_FORMAT_R16G16B16A16_SFLOAT`(8바이트)로 담으면 32비트 `vec4`로 들어오므로, 정점 하나를 줄여 메모리 대역폭을 아낄 수 있습니다. 어떤 포맷을 정점 속성에 쓸 수 있는지는 `vkGetPhysicalDeviceFormatProperties`가 알려 주는 `bufferFeatures`에 `VK_FORMAT_FEATURE_VERTEX_BUFFER_BIT`가 있는지로 확인합니다.
</Note>

## 정리

- 정점 입력 상태의 바인딩은 버퍼를 끊는 간격과 넘어가는 단위를, 속성은 location마다 바인딩·포맷·오프셋을 정한다
- 인덱스 버퍼는 함께 쓰는 정점을 한 번만 저장하게 하고, 구현이 정점 셰이딩 결과를 다시 쓸 길을 연다
- `firstIndex`와 `vertexOffset`으로 큰 버퍼 하나에 여러 메시를 담고, 인덱스 타입은 정점 수에 맞춰 `UINT16`이나 `UINT32`를 고른다
