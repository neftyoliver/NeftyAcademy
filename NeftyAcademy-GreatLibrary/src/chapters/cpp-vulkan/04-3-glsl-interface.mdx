셰이더는 바깥의 데이터를 **인터페이스 블록**으로 받습니다. `uniform` 블록(UBO)은 읽기 전용의 작은 데이터, `buffer` 블록(SSBO)은 읽고 쓰는 큰 데이터, `push_constant` 블록은 커맨드 버퍼에 직접 담는 아주 작은 데이터입니다. 앞의 둘은 `set`과 `binding` 번호를 달고, C++ 쪽에서는 그 자리에 버퍼를 **디스크립터**로 겁니다. 디스크립터는 [「세트와 레이아웃」](#/read/cpp-vulkan/desc-sets)에서 자세히 보고, 여기서는 세트 하나를 만드는 도우미만 둡니다.

```cpp vk file=vkx_bind.hpp with=vkx_instance.hpp,vkx_device.hpp,vkx_resource.hpp
#pragma once
#include "vkx_resource.hpp"

namespace vkx {

// 디스크립터 세트 하나: 바인딩마다 종류를 적은 레이아웃, 세트를 담는 풀, 세트
struct Bindings {
    VkDescriptorSetLayout layout = VK_NULL_HANDLE;
    VkDescriptorPool pool = VK_NULL_HANDLE;
    VkDescriptorSet set = VK_NULL_HANDLE;
};

// types[i]가 바인딩 i의 종류다. 어느 셰이더 단계에서나 보이게 한다
inline Bindings make_bindings(const Device& d,
                              std::vector<VkDescriptorType> types) {
    Bindings b;
    std::vector<VkDescriptorSetLayoutBinding> slots;
    std::vector<VkDescriptorPoolSize> sizes;
    for (uint32_t i = 0; i < types.size(); ++i) {
        slots.push_back({.binding = i, .descriptorType = types[i],
                         .descriptorCount = 1, .stageFlags = VK_SHADER_STAGE_ALL});
        sizes.push_back({.type = types[i], .descriptorCount = 1});
    }
    VkDescriptorSetLayoutCreateInfo lci{
        .sType = VK_STRUCTURE_TYPE_DESCRIPTOR_SET_LAYOUT_CREATE_INFO,
        .bindingCount = uint32_t(slots.size()), .pBindings = slots.data()};
    check(vkCreateDescriptorSetLayout(d.handle, &lci, nullptr, &b.layout),
          "세트 레이아웃");
    VkDescriptorPoolCreateInfo pci{
        .sType = VK_STRUCTURE_TYPE_DESCRIPTOR_POOL_CREATE_INFO, .maxSets = 1,
        .poolSizeCount = uint32_t(sizes.size()), .pPoolSizes = sizes.data()};
    check(vkCreateDescriptorPool(d.handle, &pci, nullptr, &b.pool), "풀");
    VkDescriptorSetAllocateInfo ai{
        .sType = VK_STRUCTURE_TYPE_DESCRIPTOR_SET_ALLOCATE_INFO,
        .descriptorPool = b.pool, .descriptorSetCount = 1,
        .pSetLayouts = &b.layout};
    check(vkAllocateDescriptorSets(d.handle, &ai, &b.set), "디스크립터 세트");
    return b;
}

// 바인딩 binding에 버퍼 전체를 건다. type은 레이아웃에 적은 종류와 같아야 한다
inline void bind_buffer(const Device& d, const Bindings& b, uint32_t binding,
                        VkDescriptorType type, const Buffer& buffer) {
    VkDescriptorBufferInfo info{.buffer = buffer.handle, .range = VK_WHOLE_SIZE};
    VkWriteDescriptorSet w{.sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET,
                           .dstSet = b.set, .dstBinding = binding,
                           .descriptorCount = 1, .descriptorType = type,
                           .pBufferInfo = &info};
    vkUpdateDescriptorSets(d.handle, 1, &w, 0, nullptr);
}

// 바인딩 binding에 이미지 뷰(샘플러와 함께일 수도)를 건다
inline void bind_image(const Device& d, const Bindings& b, uint32_t binding,
                       VkDescriptorType type, VkImageView view,
                       VkImageLayout layout, VkSampler sampler = VK_NULL_HANDLE) {
    VkDescriptorImageInfo info{.sampler = sampler, .imageView = view,
                               .imageLayout = layout};
    VkWriteDescriptorSet w{.sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET,
                           .dstSet = b.set, .dstBinding = binding,
                           .descriptorCount = 1, .descriptorType = type,
                           .pImageInfo = &info};
    vkUpdateDescriptorSets(d.handle, 1, &w, 0, nullptr);
}

// 파이프라인 레이아웃: 세트 레이아웃들과 푸시 상수 바이트 수
inline VkPipelineLayout make_layout(const Device& d,
                                    std::vector<VkDescriptorSetLayout> sets = {},
                                    uint32_t push_bytes = 0) {
    VkPushConstantRange push{.stageFlags = VK_SHADER_STAGE_ALL, .size = push_bytes};
    VkPipelineLayoutCreateInfo ci{
        .sType = VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO,
        .setLayoutCount = uint32_t(sets.size()), .pSetLayouts = sets.data(),
        .pushConstantRangeCount = push_bytes ? 1u : 0u,
        .pPushConstantRanges = &push};
    VkPipelineLayout layout;
    check(vkCreatePipelineLayout(d.handle, &ci, nullptr, &layout), "레이아웃");
    return layout;
}

inline void destroy(const Device& d, Bindings& b) {
    vkDestroyDescriptorPool(d.handle, b.pool, nullptr);   // 세트도 함께 풀린다
    vkDestroyDescriptorSetLayout(d.handle, b.layout, nullptr);
}

}  // namespace vkx
```

- **세트, 레이아웃, 풀**: 세트 레이아웃은 바인딩마다 무엇이 오는지(UBO, SSBO, 이미지)를 적은 틀이고, 세트는 그 틀에 실제 버퍼를 끼운 것이며, 세트는 풀에서 받습니다. 그릴 때 `vkCmdBindDescriptorSets`로 세트를 파이프라인 레이아웃의 자리에 붙입니다
- **파이프라인 레이아웃**: 셰이더가 쓸 세트 레이아웃들과 푸시 상수의 크기를 묶은 것입니다. `make_layout`은 [「그래픽 파이프라인」](#/read/cpp-vulkan/tri-pipeline)에서 비워 두었던 그 레이아웃을 만듭니다

블록 안 멤버의 메모리 배치는 **레이아웃 규칙**이 정하고, C++ 쪽 구조체가 그 배치와 한 바이트도 어긋나면 안 됩니다. 아래 프로그램은 같은 구조체를 std140(UBO), std430(SSBO), scalar 세 규칙으로 읽습니다. 버퍼의 k번째 float에 k를 넣어 두었으므로, 멤버를 읽은 값에 4를 곱하면 그 멤버의 바이트 오프셋입니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_draw.hpp,vkx_pipeline.hpp,vkx_bind.hpp
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_draw.hpp"
#include "vkx_pipeline.hpp"

// 같은 구조체를 세 가지 레이아웃의 블록으로 읽는다. 버퍼의 k번째 float에는 k가
// 들어 있으므로, 멤버를 읽은 값에 4를 곱하면 그 멤버의 바이트 오프셋이다
const char* FS = R"(#version 450
#extension GL_EXT_scalar_block_layout : require
struct S { float a; vec3 b; float c; vec2 d[2]; mat3 m; float e; };
layout(std140, binding = 0) uniform U { S s; } u140;
layout(std430, binding = 1) readonly buffer B { S s; } b430;
layout(scalar, binding = 2) readonly buffer C { S s; } bsc;
layout(location = 0) out vec4 o;

float pick(S s, int i) {
    switch (i) {
        case 0: return s.a;       case 1: return s.b.x;     case 2: return s.c;
        case 3: return s.d[0].x;  case 4: return s.d[1].x;  case 5: return s.m[0].x;
        case 6: return s.m[1].x;  default: return s.e;
    }
}
void main() {
    int i = int(gl_FragCoord.x);
    o = vec4(pick(u140.s, i), pick(b430.s, i), pick(bsc.s, i), 0.0) * 4.0;
})";

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    vkx::Buffer data = vkx::make_buffer(dev, 64 * 4,
                                        VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT |
                                            VK_BUFFER_USAGE_STORAGE_BUFFER_BIT,
                                        true);
    for (int k = 0; k < 64; ++k) static_cast<float*>(data.mapped)[k] = float(k);

    const auto UBO = VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER;
    const auto SSBO = VK_DESCRIPTOR_TYPE_STORAGE_BUFFER;
    vkx::Bindings binds = vkx::make_bindings(dev, {UBO, SSBO, SSBO});
    vkx::bind_buffer(dev, binds, 0, UBO, data);    // 같은 버퍼를 세 바인딩에 건다
    vkx::bind_buffer(dev, binds, 1, SSBO, data);
    vkx::bind_buffer(dev, binds, 2, SSBO, data);
    VkPipelineLayout layout = vkx::make_layout(dev, {binds.layout});

    vkx::Image target = vkx::make_image(
        dev, {8, 1}, VK_FORMAT_R32G32B32A32_SFLOAT,
        VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_SRC_BIT);
    vkx::Buffer pixels =
        vkx::make_buffer(dev, 8 * 16, VK_BUFFER_USAGE_TRANSFER_DST_BIT, true);
    auto vs = vkx::make_shader(dev, vkx::compile(vkx::FULLSCREEN_VS,
                                                 shaderc_vertex_shader));
    auto fs = vkx::make_shader(dev, vkx::compile(FS, shaderc_fragment_shader));
    VkPipeline pipeline = vkx::make_pipeline(dev, {.vertex = vs, .fragment = fs,
                                                   .layout = layout,
                                                   .color = target.format});

    VkCommandBuffer cmd = vkx::begin(cmds);
    vkx::begin_drawing(cmd, target);
    vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, pipeline);
    vkCmdBindDescriptorSets(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, layout, 0, 1,
                            &binds.set, 0, nullptr);   // 세트 0번 자리에
    vkCmdDraw(cmd, 3, 1, 0, 0);
    vkx::end_drawing(cmd, target, pixels);
    vkx::submit_and_wait(dev, cmds);

    const char* names[8] = {"a", "b", "c", "d[0]", "d[1]", "m[0]", "m[1]", "e"};
    auto* v = static_cast<const float*>(pixels.mapped);
    std::printf("멤버   std140 std430 scalar\n");
    for (int i = 0; i < 8; ++i)
        std::printf("%-6s %6.0f %6.0f %6.0f\n", names[i], v[4 * i], v[4 * i + 1],
                    v[4 * i + 2]);
    vkDestroyPipeline(dev.handle, pipeline, nullptr);
    vkDestroyPipelineLayout(dev.handle, layout, nullptr);
    vkDestroyShaderModule(dev.handle, vs, nullptr);
    vkDestroyShaderModule(dev.handle, fs, nullptr);
    vkx::destroy(dev, binds);
    vkx::destroy(dev, pixels);
    vkx::destroy(dev, target);
    vkx::destroy(dev, data);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
멤버   std140 std430 scalar
a           0      0      0
b          16     16      4
c          28     28     16
d[0]       32     32     20
d[1]       48     40     28
m[0]       64     48     36
m[1]       80     64     48
e         112     96     72
```

- **std140**: 벌칸 GLSL에서 `uniform` 블록의 기본입니다. `vec3`는 16바이트 경계에 놓이고, 배열 원소와 행렬의 열은 16바이트로 올림해 둡니다. 그래서 `vec2` 배열의 간격이 16(32 → 48)이고 `mat3`는 48바이트를 차지합니다
- **std430**: `buffer` 블록과 푸시 상수의 기본입니다. 배열 간격을 16으로 올리지 않아 `d[1]`이 40에 오지만, `vec3`는 여전히 16바이트 경계라서 `b`는 16, `mat3`의 열 간격도 16입니다
- **scalar**: 장치 기능 `scalarBlockLayout`(1.2부터 코어, `make_device`가 켭니다)과 `GL_EXT_scalar_block_layout`이 있으면 쓸 수 있고, 성분 크기에만 맞춰 C의 구조체처럼 빽빽하게 놓습니다
- **흔한 실수**: C++의 `struct { float a; float b[3]; float c; }`는 `b`를 4에 두지만 std140·std430은 16에 둡니다. `alignas(16)`으로 맞추거나, `vec3` 대신 `vec4`를 쓰거나, scalar 규칙을 씁니다

<Note>
UBO는 장치가 최소 16KB(`maxUniformBufferRange`)까지만 보장하는 대신 많은 GPU에서 빠른 상수 경로로 읽히고, SSBO는 최소 128MB를 보장하며 쓰기와 크기가 정해지지 않은 마지막 배열을 허용합니다. 푸시 상수는 1.4부터 256바이트(그 전에는 128바이트)까지 보장되며, 그릴 때마다 바뀌는 작은 값에 씁니다([「푸시 상수」](#/read/cpp-vulkan/desc-push)).
</Note>

## 정리

- 셰이더는 `uniform`·`buffer`·`push_constant` 블록으로 데이터를 받고, 앞의 둘은 `set`·`binding` 자리에 디스크립터로 버퍼를 건다
- 기본 레이아웃은 `uniform`이 std140(배열·행렬 열을 16바이트로 올림), `buffer`와 푸시 상수가 std430이며, 둘 다 `vec3`를 16바이트 경계에 둔다
- scalar 레이아웃은 C 구조체처럼 빽빽하고, 어느 규칙이든 C++ 구조체의 오프셋을 블록과 똑같이 맞춰야 한다
