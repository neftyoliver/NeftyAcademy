광선 하나가 어느 삼각형에 맞는지를 삼각형마다 따져 볼 수는 없으므로, 레이 트레이싱은 장면을 빨리 훑을 수 있게 정리한 **가속 구조**를 먼저 짓습니다. 속 모양은 구현이 정하고(보통 경계 상자의 나무입니다) 프로그램은 두 층으로 다룹니다. **아래층**(BLAS)은 메시 하나의 삼각형을, **위층**(TLAS)은 아래층을 가리키는 인스턴스들을 담습니다. 이 장의 세 쪽이 함께 쓸 도우미부터 만듭니다.

```cpp vk file=vkx_accel.hpp with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp
#pragma once
#include "vkx_bind.hpp"
#include "vkx_cmd.hpp"
#include "vkx_resource.hpp"

// 확장 함수는 로더가 내보내지 않으므로 장치에서 주소를 받아 부른다
#define VKX_KHR(d, name) \
    reinterpret_cast<PFN_##name>(vkGetDeviceProcAddr((d).handle, #name))

namespace vkx {

// 가속 구조, 레이 트레이싱 파이프라인, 레이 쿼리의 확장과 기능을 켠 장치
inline Device make_rt_device(VkInstance instance) {
    VkPhysicalDeviceRayQueryFeaturesKHR query{
        .sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_RAY_QUERY_FEATURES_KHR,
        .rayQuery = VK_TRUE};
    VkPhysicalDeviceRayTracingPipelineFeaturesKHR pipeline{
        .sType =
            VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_RAY_TRACING_PIPELINE_FEATURES_KHR,
        .pNext = &query, .rayTracingPipeline = VK_TRUE};
    VkPhysicalDeviceAccelerationStructureFeaturesKHR accel{
        .sType =
            VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_ACCELERATION_STRUCTURE_FEATURES_KHR,
        .pNext = &pipeline, .accelerationStructure = VK_TRUE};
    return make_device(instance,
                       {VK_KHR_ACCELERATION_STRUCTURE_EXTENSION_NAME,
                        VK_KHR_DEFERRED_HOST_OPERATIONS_EXTENSION_NAME,
                        VK_KHR_RAY_TRACING_PIPELINE_EXTENSION_NAME,
                        VK_KHR_RAY_QUERY_EXTENSION_NAME},
                       &accel);
}

// 버퍼의 GPU 주소. SHADER_DEVICE_ADDRESS 쓰임새로 만든 버퍼여야 한다
inline VkDeviceAddress address_of(const Device& d, const Buffer& b) {
    VkBufferDeviceAddressInfo info{
        .sType = VK_STRUCTURE_TYPE_BUFFER_DEVICE_ADDRESS_INFO, .buffer = b.handle};
    return vkGetBufferDeviceAddress(d.handle, &info);
}

struct Accel {
    VkAccelerationStructureKHR handle = VK_NULL_HANDLE;
    Buffer storage;                // 가속 구조가 들어 있는 버퍼
    Buffer input;                  // 지을 때 읽힌 정점이나 인스턴스
    VkDeviceAddress address = 0;   // 인스턴스가 이 구조를 가리킬 때 쓰는 주소
    VkAccelerationStructureBuildSizesInfoKHR sizes{};   // 짓기 전에 물은 크기
};

const VkBufferUsageFlags ADDRESS = VK_BUFFER_USAGE_SHADER_DEVICE_ADDRESS_BIT;
const VkBufferUsageFlags BUILD_INPUT =
    VK_BUFFER_USAGE_ACCELERATION_STRUCTURE_BUILD_INPUT_READ_ONLY_BIT_KHR | ADDRESS;

// 기하 하나로 가속 구조를 짓는다: 크기를 묻고, 버퍼와 객체를 만들고, GPU에서 짓는다
inline Accel build(const Device& d, Commands& c,
                   VkAccelerationStructureTypeKHR type,
                   const VkAccelerationStructureGeometryKHR& geometry,
                   uint32_t count, VkBuildAccelerationStructureFlagsKHR flags) {
    VkAccelerationStructureBuildGeometryInfoKHR info{
        .sType = VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_BUILD_GEOMETRY_INFO_KHR,
        .type = type, .flags = flags,
        .mode = VK_BUILD_ACCELERATION_STRUCTURE_MODE_BUILD_KHR,
        .geometryCount = 1, .pGeometries = &geometry};
    Accel a;
    a.sizes.sType = VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_BUILD_SIZES_INFO_KHR;
    VKX_KHR(d, vkGetAccelerationStructureBuildSizesKHR)(
        d.handle, VK_ACCELERATION_STRUCTURE_BUILD_TYPE_DEVICE_KHR, &info, &count,
        &a.sizes);
    a.storage = make_buffer(
        d, a.sizes.accelerationStructureSize,
        VK_BUFFER_USAGE_ACCELERATION_STRUCTURE_STORAGE_BIT_KHR | ADDRESS, false);
    VkAccelerationStructureCreateInfoKHR ci{
        .sType = VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_CREATE_INFO_KHR,
        .buffer = a.storage.handle, .size = a.sizes.accelerationStructureSize,
        .type = type};
    VKX_KHR(d, vkCreateAccelerationStructureKHR)(d.handle, &ci, nullptr, &a.handle);
    // 짓는 동안만 쓰는 임시 버퍼. 주소가 장치가 정한 정렬의 배수여야 한다
    VkPhysicalDeviceAccelerationStructurePropertiesKHR limits{
        .sType =
           VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_ACCELERATION_STRUCTURE_PROPERTIES_KHR};
    VkPhysicalDeviceProperties2 props{
        .sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_PROPERTIES_2, .pNext = &limits};
    vkGetPhysicalDeviceProperties2(d.gpu, &props);
    VkDeviceSize align = limits.minAccelerationStructureScratchOffsetAlignment;
    Buffer scratch =
        make_buffer(d, a.sizes.buildScratchSize + align,
                    VK_BUFFER_USAGE_STORAGE_BUFFER_BIT | ADDRESS, false);
    info.dstAccelerationStructure = a.handle;
    info.scratchData.deviceAddress =
        (address_of(d, scratch) + align - 1) / align * align;
    VkAccelerationStructureBuildRangeInfoKHR range{.primitiveCount = count};
    const VkAccelerationStructureBuildRangeInfoKHR* ranges = &range;
    VkCommandBuffer cmd = begin(c);
    VKX_KHR(d, vkCmdBuildAccelerationStructuresKHR)(cmd, 1, &info, &ranges);
    submit_and_wait(d, c);
    destroy(d, scratch);
    VkAccelerationStructureDeviceAddressInfoKHR ai{
        .sType = VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_DEVICE_ADDRESS_INFO_KHR,
        .accelerationStructure = a.handle};
    a.address =
        VKX_KHR(d, vkGetAccelerationStructureDeviceAddressKHR)(d.handle, &ai);
    return a;
}

const VkBuildAccelerationStructureFlagsKHR FAST_TRACE =
    VK_BUILD_ACCELERATION_STRUCTURE_PREFER_FAST_TRACE_BIT_KHR;

// 아래층: 삼각형들(정점 셋씩, 정점마다 x y z)을 담는다
inline Accel make_blas(const Device& d, Commands& c, const std::vector<float>& xyz,
                       VkBuildAccelerationStructureFlagsKHR flags = FAST_TRACE) {
    Buffer input = make_buffer(d, xyz.size() * 4, BUILD_INPUT, true);
    std::memcpy(input.mapped, xyz.data(), xyz.size() * 4);
    VkAccelerationStructureGeometryKHR g{
        .sType = VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_GEOMETRY_KHR,
        .geometryType = VK_GEOMETRY_TYPE_TRIANGLES_KHR,
        .flags = VK_GEOMETRY_OPAQUE_BIT_KHR};
    g.geometry.triangles = {
        .sType =
            VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_GEOMETRY_TRIANGLES_DATA_KHR,
        .vertexFormat = VK_FORMAT_R32G32B32_SFLOAT,
        .vertexData = {.deviceAddress = address_of(d, input)},
        .vertexStride = 12, .maxVertex = uint32_t(xyz.size() / 3 - 1),
        .indexType = VK_INDEX_TYPE_NONE_KHR};
    Accel a = build(d, c, VK_ACCELERATION_STRUCTURE_TYPE_BOTTOM_LEVEL_KHR, g,
                    uint32_t(xyz.size() / 9), flags);
    a.input = input;
    return a;
}

// 위층: 아래층을 가리키는 인스턴스들을 담는다
inline Accel make_tlas(const Device& d, Commands& c,
                       const std::vector<VkAccelerationStructureInstanceKHR>& all,
                       VkBuildAccelerationStructureFlagsKHR flags = FAST_TRACE) {
    Buffer input = make_buffer(d, all.size() * sizeof all[0], BUILD_INPUT, true);
    std::memcpy(input.mapped, all.data(), all.size() * sizeof all[0]);
    VkAccelerationStructureGeometryKHR g{
        .sType = VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_GEOMETRY_KHR,
        .geometryType = VK_GEOMETRY_TYPE_INSTANCES_KHR};
    g.geometry.instances = {
        .sType =
            VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_GEOMETRY_INSTANCES_DATA_KHR,
        .data = {.deviceAddress = address_of(d, input)}};
    Accel a = build(d, c, VK_ACCELERATION_STRUCTURE_TYPE_TOP_LEVEL_KHR, g,
                    uint32_t(all.size()), flags);
    a.input = input;
    return a;
}

// 위층을 디스크립터에 건다. 가속 구조는 pNext에 매단 구조체로 넘긴다
inline void bind_accel(const Device& d, const Bindings& b, uint32_t binding,
                       const Accel& a) {
    VkWriteDescriptorSetAccelerationStructureKHR as{
        .sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET_ACCELERATION_STRUCTURE_KHR,
        .accelerationStructureCount = 1, .pAccelerationStructures = &a.handle};
    VkWriteDescriptorSet w{
        .sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET, .pNext = &as,
        .dstSet = b.set, .dstBinding = binding, .descriptorCount = 1,
        .descriptorType = VK_DESCRIPTOR_TYPE_ACCELERATION_STRUCTURE_KHR};
    vkUpdateDescriptorSets(d.handle, 1, &w, 0, nullptr);
}

inline void destroy(const Device& d, Accel& a) {
    VKX_KHR(d, vkDestroyAccelerationStructureKHR)(d.handle, a.handle, nullptr);
    destroy(d, a.storage);
    if (a.input.handle) destroy(d, a.input);
}

}  // namespace vkx
```

아래 프로그램은 수를 바꿔 가며 지어 보고 장치가 답한 크기를 찍습니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp,vkx_bind.hpp,vkx_accel.hpp
#include "vkx_accel.hpp"

// 삼각형 n개를 한 줄로 늘어놓은 메시
std::vector<float> strip(int n) {
    std::vector<float> xyz;
    for (int i = 0; i < n; ++i) {
        float x = float(i);
        xyz.insert(xyz.end(), {x, 0, 0, x + 1, 0, 0, x, 1, 0});
    }
    return xyz;
}

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_rt_device(inst.handle);
    vkx::Commands cmds = vkx::make_commands(dev);
    using ull = unsigned long long;

    std::puts("아래층: 삼각형, 정점 데이터, 가속 구조, 임시 버퍼(바이트)");
    vkx::Accel mesh;   // 마지막 것은 남겨 위층이 가리키게 한다
    for (int n : {1, 100, 10000}) {
        if (mesh.handle) vkx::destroy(dev, mesh);
        mesh = vkx::make_blas(dev, cmds, strip(n));
        std::printf("%7d %9d %9llu %9llu\n", n, n * 36,
                    ull(mesh.sizes.accelerationStructureSize),
                    ull(mesh.sizes.buildScratchSize));
    }

    std::puts("위층: 인스턴스, 인스턴스 데이터, 가속 구조, 임시 버퍼(바이트)");
    for (int n : {1, 3, 1000}) {
        std::vector<VkAccelerationStructureInstanceKHR> all(n);
        for (int i = 0; i < n; ++i) {
            // 3×4 행렬(행마다 넷): 여기서는 y로 2씩 옮기기만 한다
            all[i].transform = {{{1, 0, 0, 0}, {0, 1, 0, 2.0f * i}, {0, 0, 1, 0}}};
            all[i].instanceCustomIndex = i;   // 셰이더가 읽을 번호(24비트)
            all[i].mask = 0xff;               // 광선의 마스크와 겹쳐야 맞는다
            all[i].accelerationStructureReference = mesh.address;
        }
        vkx::Accel scene = vkx::make_tlas(dev, cmds, all);
        std::printf("%7d %9zu %9llu %9llu\n", n, n * sizeof all[0],
                    ull(scene.sizes.accelerationStructureSize),
                    ull(scene.sizes.buildScratchSize));
        vkx::destroy(dev, scene);
    }

    vkx::destroy(dev, mesh);
    vkx::destroy(dev, cmds);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
아래층: 삼각형, 정점 데이터, 가속 구조, 임시 버퍼(바이트)
      1        36       152     41184
    100      3600     10400     52232
  10000    360000   1040016   1301320
위층: 인스턴스, 인스턴스 데이터, 가속 구조, 임시 버퍼(바이트)
      1        64       264     41208
      3       192       544     41440
   1000     64000    175992    177032
```

- **두 층인 까닭**: 삼각형 10,000개짜리 아래층은 한 번만 짓고, 위층의 인스턴스 1,000개가 모두 그것을 가리켰습니다. 인스턴스 하나는 64바이트(3×4 변환 행렬, 24비트 번호와 8비트 마스크, 24비트 SBT 오프셋과 8비트 플래그, 아래층의 주소)여서, 물체를 옮기거나 늘려도 삼각형 쪽은 다시 짓지 않습니다
- **크기는 묻는다**: 속 모양이 구현마다 달라 크기도 장치가 정합니다. lavapipe는 삼각형 하나에 104바이트쯤(정점 데이터의 세 배 가까이)을 달라고 했고, 임시 버퍼는 삼각형이 하나여도 41KB였습니다. `vkGetAccelerationStructureBuildSizesKHR`의 답대로 버퍼를 만들고 그 위에 객체를 올립니다
- **입력은 주소로**: 정점과 인스턴스는 버퍼의 GPU 주소로 넘깁니다([「푸시 상수」](#/read/cpp-vulkan/desc-push)). 그 버퍼는 빌드 입력과 `SHADER_DEVICE_ADDRESS` 쓰임새로 만들고, 임시 버퍼의 주소는 `minAccelerationStructureScratchOffsetAlignment`의 배수로 맞춥니다
- **짓는 차례**: `vkCmdBuildAccelerationStructuresKHR`는 커맨드 버퍼에 기록되어 GPU에서 돕니다. 한 호출 안에서는 차례가 보장되지 않으므로 위층은 아래층을 다 지은 뒤에 따로 짓고, 위층을 쓰는 동안에는 아래층을 살려 둡니다
- **셰이더에 넘기기**: 위층은 `ACCELERATION_STRUCTURE_KHR` 종류의 디스크립터로 겁니다([「세트와 레이아웃」](#/read/cpp-vulkan/desc-sets)). 이 장의 함수들은 확장 함수여서 `vkGetDeviceProcAddr`로 주소를 받아 부릅니다

## 움직이는 장면과 메모리

지을 때 플래그를 고릅니다. `PREFER_FAST_TRACE`는 광선이 빨리 지나가게, `PREFER_FAST_BUILD`는 빨리 짓게 해 달라는 부탁이고 함께 줄 수는 없습니다. `ALLOW_UPDATE`로 지으면 정점 위치나 인스턴스의 변환만 바뀌었을 때 다시 짓지 않고 고쳐 쓸 수 있지만, 많이 움직일수록 추적이 느려집니다. NVIDIA는 위층은 고쳐 쓰지 말고 새로 짓고, 아래층은 조금 변할 때만 고쳐 쓰라고 권합니다.

`ALLOW_COMPACTION`으로 지은 구조는 실제로 필요한 크기를 쿼리로 물어, 그 크기로 만든 새 구조에 압축 모드로 복사할 수 있습니다. NVIDIA는 절반까지 줄기도 한다고 하는데, lavapipe에서는 그대로였습니다.

<Note>
CPU에서 바로 짓는 명령(`vkBuildAccelerationStructuresKHR`)도 있지만 Khronos는 2026년 7월에 이 길을 더는 권하지 않기로 했습니다. 새 코드는 이 쪽처럼 커맨드 버퍼에 기록해 GPU에서 짓습니다.
</Note>

## 정리

- 가속 구조는 삼각형을 담는 아래층과, 아래층을 변환·번호·마스크와 함께 가리키는 인스턴스를 담는 위층으로 이루어진다
- 크기를 묻고, 버퍼와 객체를 만들어 커맨드 버퍼에서 짓는다. 아래층이 먼저이고, 위층이 쓰이는 동안 살려 둔다
- 플래그로 추적 속도와 짓는 속도, 고쳐 쓰기, 압축을 고르며, 움직이는 물체는 위층을 다시 지어 옮긴다
