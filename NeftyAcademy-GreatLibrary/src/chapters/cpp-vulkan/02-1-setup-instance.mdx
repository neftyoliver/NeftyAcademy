인스턴스는 프로그램과 벌칸 로더를 잇는 첫 객체입니다. 만들 때 세 가지를 정합니다. 프로그램이 쓸 벌칸의 판(`apiVersion`), 켤 인스턴스 확장(창 시스템의 표면, 디버그 메시지), 켤 레이어입니다. 이 쪽에서 인스턴스를 만드는 도우미 `vkx_instance.hpp`를 쓰고, 다음 쪽부터는 이 머리 파일을 불러 씁니다.

## 인스턴스를 만드는 도우미

```cpp vk file=vkx_instance.hpp
#pragma once
#include <vulkan/vulkan.h>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <vector>

namespace vkx {

// 실패하면 무엇이 실패했는지 알리고 끝낸다
inline void check(VkResult r, const char* what) {
    if (r == VK_SUCCESS) return;
    std::fprintf(stderr, "%s 실패 (VkResult %d)\n", what, r);
    std::exit(1);
}

inline bool has_layer(const char* name) {
    uint32_t n = 0;
    vkEnumerateInstanceLayerProperties(&n, nullptr);    // 개수를 먼저 묻고
    std::vector<VkLayerProperties> all(n);
    vkEnumerateInstanceLayerProperties(&n, all.data());  // 그만큼 받는다
    for (auto& l : all)
        if (std::strcmp(l.layerName, name) == 0) return true;
    return false;
}

inline bool has_instance_extension(const char* name) {
    uint32_t n = 0;
    vkEnumerateInstanceExtensionProperties(nullptr, &n, nullptr);
    std::vector<VkExtensionProperties> all(n);
    vkEnumerateInstanceExtensionProperties(nullptr, &n, all.data());
    for (auto& e : all)
        if (std::strcmp(e.extensionName, name) == 0) return true;
    return false;
}

// 검증 레이어의 오류와 경고를 표준 출력으로 내보낸다
inline VKAPI_ATTR VkBool32 VKAPI_CALL report(
        VkDebugUtilsMessageSeverityFlagBitsEXT, VkDebugUtilsMessageTypeFlagsEXT,
        const VkDebugUtilsMessengerCallbackDataEXT* d, void*) {
    std::printf("[검증] %s\n", d->pMessage);
    return VK_FALSE;
}

struct Instance {
    VkInstance handle = VK_NULL_HANDLE;
    VkDebugUtilsMessengerEXT messenger = VK_NULL_HANDLE;   // 레이어가 없으면 빈 값
};

// 벌칸 1.3 인스턴스. 검증 레이어가 깔려 있으면 켜고 메시지를 받는다
inline Instance make_instance(std::vector<const char*> extensions = {}) {
    const char* layer = "VK_LAYER_KHRONOS_validation";
    const bool validation = has_layer(layer);
    if (validation) extensions.push_back(VK_EXT_DEBUG_UTILS_EXTENSION_NAME);
    VkInstanceCreateFlags flags = 0;
    if (has_instance_extension(VK_KHR_PORTABILITY_ENUMERATION_EXTENSION_NAME)) {
        extensions.push_back(VK_KHR_PORTABILITY_ENUMERATION_EXTENSION_NAME);
        flags |= VK_INSTANCE_CREATE_ENUMERATE_PORTABILITY_BIT_KHR;  // MoltenVK도
    }
    VkDebugUtilsMessengerCreateInfoEXT mci{
        .sType = VK_STRUCTURE_TYPE_DEBUG_UTILS_MESSENGER_CREATE_INFO_EXT,
        .messageSeverity = VK_DEBUG_UTILS_MESSAGE_SEVERITY_WARNING_BIT_EXT |
                           VK_DEBUG_UTILS_MESSAGE_SEVERITY_ERROR_BIT_EXT,
        .messageType = VK_DEBUG_UTILS_MESSAGE_TYPE_VALIDATION_BIT_EXT,
        .pfnUserCallback = report};
    VkApplicationInfo app{.sType = VK_STRUCTURE_TYPE_APPLICATION_INFO,
                          .pApplicationName = "vkx",
                          .apiVersion = VK_API_VERSION_1_3};
    VkInstanceCreateInfo ci{.sType = VK_STRUCTURE_TYPE_INSTANCE_CREATE_INFO,
                            .pNext = validation ? &mci : nullptr,
                            .flags = flags,
                            .pApplicationInfo = &app,
                            .enabledLayerCount = validation ? 1u : 0u,
                            .ppEnabledLayerNames = &layer,
                            .enabledExtensionCount = uint32_t(extensions.size()),
                            .ppEnabledExtensionNames = extensions.data()};
    Instance inst;
    check(vkCreateInstance(&ci, nullptr, &inst.handle), "vkCreateInstance");
    if (validation) {
        auto create = reinterpret_cast<PFN_vkCreateDebugUtilsMessengerEXT>(
            vkGetInstanceProcAddr(inst.handle, "vkCreateDebugUtilsMessengerEXT"));
        check(create(inst.handle, &mci, nullptr, &inst.messenger), "메신저 만들기");
    }
    return inst;
}

inline void destroy(Instance& inst) {
    if (inst.messenger) {
        auto destroy = reinterpret_cast<PFN_vkDestroyDebugUtilsMessengerEXT>(
            vkGetInstanceProcAddr(inst.handle, "vkDestroyDebugUtilsMessengerEXT"));
        destroy(inst.handle, inst.messenger, nullptr);
    }
    vkDestroyInstance(inst.handle, nullptr);
}

}  // namespace vkx
```

- **두 번 부르는 나열**: 목록을 돌려주는 벌칸 함수는 먼저 배열 없이 불러 개수를 받고, 그만큼 배열을 마련해 다시 부릅니다. 배열이 모자라면 채울 수 있는 만큼만 채우고 `VK_INCOMPLETE`를 돌려줍니다
- **apiVersion**: 프로그램이 쓸 가장 높은 판입니다. 장치가 1.4를 지원해도 여기에 1.3을 적었으면 1.4의 기능은 쓸 수 없고, 1.0을 적고 1.3 함수를 부르는 것도 잘못입니다
- **레이어는 있으면 켠다**: 검증 레이어는 개발 도구라 사용자의 기계에는 없을 수 있습니다. 있는지 보고 켜야 레이어가 없는 곳에서도 프로그램이 돕니다. 레이어가 있으면 [「개발 환경과 검증 레이어」](#/read/cpp-vulkan/vk-environment)에서 본 메신저를 걸어, 규칙을 어긴 호출을 `[검증]`으로 찍습니다
- **portability**: 맥의 MoltenVK처럼 벌칸을 다 구현하지 못한 드라이버는 `VK_KHR_portability_enumeration`을 켜고 `ENUMERATE_PORTABILITY` 플래그를 주어야 장치 목록에 나옵니다. 확장이 있으면 켜 두고, 장치 쪽의 짝은 [「물리 장치와 논리 장치」](#/read/cpp-vulkan/setup-device)에서 맞춥니다
- **check**: 실패하면 무엇이 실패했는지 찍고 끝냅니다. 예제를 짧게 하려는 것이고, 실제 프로그램은 오류를 호출한 쪽으로 돌려보내 처리합니다

## 무엇이 있는지 보기

```cpp vk with=vkx_instance.hpp
#include "vkx_instance.hpp"

int main() {
    uint32_t v = 0;
    vkEnumerateInstanceVersion(&v);                // 1.1부터 있는 함수
    std::printf("로더가 아는 벌칸: %u.%u.%u\n", VK_API_VERSION_MAJOR(v),
                VK_API_VERSION_MINOR(v), VK_API_VERSION_PATCH(v));
    const char* wanted[] = {
        "VK_KHR_surface",         "VK_KHR_xcb_surface",   "VK_KHR_wayland_surface",
        "VK_KHR_win32_surface",   "VK_EXT_metal_surface", "VK_EXT_headless_surface",
        "VK_EXT_debug_utils",     "VK_KHR_portability_enumeration"};
    std::printf("인스턴스 확장\n");
    for (const char* e : wanted)
        std::printf("  %-32s %s\n", e,
                    vkx::has_instance_extension(e) ? "있음" : "없음");
    std::printf("검증 레이어: %s\n",
                vkx::has_layer("VK_LAYER_KHRONOS_validation") ? "있음" : "없음");

    // 창 없이 쓸 표면 확장 둘을 켜고 만든다
    vkx::Instance inst = vkx::make_instance(
        {VK_KHR_SURFACE_EXTENSION_NAME, VK_EXT_HEADLESS_SURFACE_EXTENSION_NAME});
    std::printf("인스턴스를 만듦, 메신저 %s\n", inst.messenger ? "있음" : "없음");
    vkx::destroy(inst);
}
```

```text
로더가 아는 벌칸: 1.3.275
인스턴스 확장
  VK_KHR_surface                   있음
  VK_KHR_xcb_surface               있음
  VK_KHR_wayland_surface           있음
  VK_KHR_win32_surface             없음
  VK_EXT_metal_surface             없음
  VK_EXT_headless_surface          있음
  VK_EXT_debug_utils               있음
  VK_KHR_portability_enumeration   있음
검증 레이어: 있음
인스턴스를 만듦, 메신저 있음
```

- **창 시스템마다 다른 표면**: 화면에 그리려면 `VK_KHR_surface`와 함께 창 시스템의 확장(리눅스의 `xcb`·`xlib`·`wayland`, 윈도우의 `win32`, 맥의 `metal`, 안드로이드의 `android`)을 켭니다. GLFW의 `glfwGetRequiredInstanceExtensions`나 SDL3의 `SDL_Vulkan_GetInstanceExtensions`가 이 목록을 알려 줍니다. 이 책의 예제는 창이 없는 곳에서도 돌도록 `VK_EXT_headless_surface`를 씁니다([「표면과 스왑체인」](#/read/cpp-vulkan/setup-swapchain))
- **로더의 판과 장치의 판**: 로더는 1.3.275까지 알고, 장치는 1.4를 지원합니다([「벌칸의 설계」](#/read/cpp-vulkan/vk-design)). 프로그램은 둘 중 낮은 쪽과 자기가 적은 `apiVersion` 안에서만 씁니다
- **나머지는 기계마다**: 출력은 이 책을 만든 리눅스 기계의 것입니다. 윈도우라면 `win32`가 있고 `xcb`가 없습니다

<Note>
로더에도 환경 변수가 있습니다. `VK_LOADER_DEBUG=all`은 로더가 어떤 드라이버와 레이어를 찾아 어떤 순서로 쌓았는지 찍고, `VK_LOADER_LAYERS_ENABLE`은 레이어를 켭니다(예전 이름 `VK_INSTANCE_LAYERS`도 아직 받습니다). 스팀 오버레이나 화면 녹화 프로그램은 **암묵 레이어**를 깔아 모든 벌칸 프로그램에 저절로 끼어드는데, 원인 모를 충돌이 나면 `VK_LOADER_LAYERS_DISABLE=~implicit~`로 모두 빼고 시험해 봅니다.
</Note>

## 정리

- 인스턴스는 apiVersion, 인스턴스 확장, 레이어를 정해 만들고, 벌칸의 목록 함수는 개수를 먼저 묻고 다시 부르는 두 번 호출로 쓴다
- 화면에 그리려면 VK_KHR_surface와 창 시스템별 표면 확장을, 창이 없으면 VK_EXT_headless_surface를 켜고, 검증 레이어와 portability 확장은 있을 때만 켠다
- 이 쪽의 vkx_instance.hpp가 검증 레이어와 메신저까지 맡으므로, 앞으로의 예제는 make_instance로 시작해 destroy로 끝난다
