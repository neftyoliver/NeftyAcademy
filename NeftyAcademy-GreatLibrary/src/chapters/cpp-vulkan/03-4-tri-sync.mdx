벌칸은 순서를 저절로 지켜 주지 않습니다. 제출은 바로 돌아오고, 같은 큐에 낸 일도 앞의 것이 끝나기 전에 뒤의 것이 시작될 수 있습니다. 순서를 거는 도구는 누가 누구를 기다리는지에 따라 다릅니다.

| 도구 | 누가 누구를 기다리나 | 주로 쓰는 곳 |
| --- | --- | --- |
| 배리어 | 한 큐 안에서 뒤 명령이 앞 명령을 | 명령 사이의 순서, 레이아웃 |
| 펜스 | CPU가 GPU를 | 제출 하나가 끝났는지 |
| 바이너리 세마포어 | GPU가 GPU를 | 제출 사이, 스왑체인 |
| 타임라인 세마포어 | GPU와 CPU가 서로 | 카운터 하나로 여러 단계 |

타임라인 세마포어는 1.2에서 핵심이 된, 커지기만 하는 64비트 카운터입니다. 제출마다 "카운터가 몇 이상이 되면 시작하고, 끝나면 몇으로 올린다"를 적고, CPU도 값을 올리거나(`vkSignalSemaphore`) 기다릴(`vkWaitSemaphores`) 수 있습니다. 아래 프로그램은 복사 두 개를 먼저 내 두고, CPU가 원본을 채운 뒤에야 1을 알려 GPU를 풀어 줍니다.

```cpp vk with=vkx_instance.hpp,vkx_device.hpp,vkx_cmd.hpp,vkx_resource.hpp
#include "vkx_cmd.hpp"
#include "vkx_resource.hpp"

// 앞 단계의 쓰기가 끝나고 보인 뒤에 뒤 단계가 시작되게 하는 메모리 배리어
void barrier(VkCommandBuffer cmd, VkPipelineStageFlags2 src,
             VkAccessFlags2 src_access, VkPipelineStageFlags2 dst,
             VkAccessFlags2 dst_access) {
    VkMemoryBarrier2 b{.sType = VK_STRUCTURE_TYPE_MEMORY_BARRIER_2,
                       .srcStageMask = src, .srcAccessMask = src_access,
                       .dstStageMask = dst, .dstAccessMask = dst_access};
    VkDependencyInfo dep{.sType = VK_STRUCTURE_TYPE_DEPENDENCY_INFO,
                         .memoryBarrierCount = 1, .pMemoryBarriers = &b};
    vkCmdPipelineBarrier2(cmd, &dep);
}

int main() {
    vkx::Instance inst = vkx::make_instance();
    vkx::Device dev = vkx::make_device(inst.handle);
    const auto SRC = VK_BUFFER_USAGE_TRANSFER_SRC_BIT;
    const auto DST = VK_BUFFER_USAGE_TRANSFER_DST_BIT;
    vkx::Buffer in = vkx::make_buffer(dev, 16, SRC, true);
    vkx::Buffer mid = vkx::make_buffer(dev, 16, SRC | DST, false);   // GPU 전용
    vkx::Buffer out = vkx::make_buffer(dev, 16, DST, true);

    // 타임라인 세마포어: 0에서 시작해 커지기만 하는 64비트 카운터
    VkSemaphoreTypeCreateInfo type{
        .sType = VK_STRUCTURE_TYPE_SEMAPHORE_TYPE_CREATE_INFO,
        .semaphoreType = VK_SEMAPHORE_TYPE_TIMELINE};
    VkSemaphoreCreateInfo sci{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_CREATE_INFO,
                              .pNext = &type};
    VkSemaphore timeline;
    vkCreateSemaphore(dev.handle, &sci, nullptr, &timeline);

    const auto COPY = VK_PIPELINE_STAGE_2_COPY_BIT;
    VkBufferCopy all{.size = 16};
    vkx::Commands a = vkx::make_commands(dev), b = vkx::make_commands(dev);
    VkCommandBuffer cmd = vkx::begin(a);   // A: in → mid (CPU가 쓴 값을 본다)
    barrier(cmd, VK_PIPELINE_STAGE_2_HOST_BIT, VK_ACCESS_2_HOST_WRITE_BIT, COPY,
            VK_ACCESS_2_TRANSFER_READ_BIT);
    vkCmdCopyBuffer(cmd, in.handle, mid.handle, 1, &all);
    vkEndCommandBuffer(cmd);
    cmd = vkx::begin(b);                   // B: mid → out (CPU가 읽는다)
    vkCmdCopyBuffer(cmd, mid.handle, out.handle, 1, &all);
    barrier(cmd, COPY, VK_ACCESS_2_TRANSFER_WRITE_BIT, VK_PIPELINE_STAGE_2_HOST_BIT,
            VK_ACCESS_2_HOST_READ_BIT);
    vkEndCommandBuffer(cmd);

    // 카운터가 wait 이상이 되면 복사를 시작하고, 끝나면 signal로 올린다
    auto submit = [&](VkCommandBuffer cb, uint64_t wait, uint64_t signal) {
        VkSemaphoreSubmitInfo w{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_SUBMIT_INFO,
                                .semaphore = timeline, .value = wait,
                                .stageMask = COPY};
        VkSemaphoreSubmitInfo s = w;
        s.value = signal;
        VkCommandBufferSubmitInfo one{
            .sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_SUBMIT_INFO,
            .commandBuffer = cb};
        VkSubmitInfo2 si{.sType = VK_STRUCTURE_TYPE_SUBMIT_INFO_2,
                         .waitSemaphoreInfoCount = 1, .pWaitSemaphoreInfos = &w,
                         .commandBufferInfoCount = 1, .pCommandBufferInfos = &one,
                         .signalSemaphoreInfoCount = 1,
                         .pSignalSemaphoreInfos = &s};
        vkQueueSubmit2(dev.queue, 1, &si, VK_NULL_HANDLE);
    };
    submit(a.cmd, 1, 2);   // 1은 아직 아무도 알리지 않았다
    submit(b.cmd, 2, 3);
    uint64_t now = 0;
    vkGetSemaphoreCounterValue(dev.handle, timeline, &now);
    std::printf("두 번 제출한 뒤 카운터 %llu: GPU가 1을 기다린다\n",
                (unsigned long long)now);

    auto* src = static_cast<uint32_t*>(in.mapped);
    for (uint32_t i = 0; i < 4; ++i) src[i] = 40 + i;   // 그동안 원본을 채운다
    VkSemaphoreSignalInfo one{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_SIGNAL_INFO,
                              .semaphore = timeline, .value = 1};
    vkSignalSemaphore(dev.handle, &one);                 // CPU가 1을 알린다
    std::puts("CPU가 원본을 채우고 1을 알림");

    const uint64_t done = 3;
    VkSemaphoreWaitInfo wait{.sType = VK_STRUCTURE_TYPE_SEMAPHORE_WAIT_INFO,
                             .semaphoreCount = 1, .pSemaphores = &timeline,
                             .pValues = &done};
    vkWaitSemaphores(dev.handle, &wait, UINT64_MAX);     // 3이 될 때까지 기다린다
    vkGetSemaphoreCounterValue(dev.handle, timeline, &now);
    auto* dst = static_cast<const uint32_t*>(out.mapped);
    std::printf("카운터 %llu, 결과 %u %u %u %u\n", (unsigned long long)now, dst[0],
                dst[1], dst[2], dst[3]);
    vkDestroySemaphore(dev.handle, timeline, nullptr);
    vkx::destroy(dev, a);
    vkx::destroy(dev, b);
    vkx::destroy(dev, in);
    vkx::destroy(dev, mid);
    vkx::destroy(dev, out);
    vkx::destroy(dev);
    vkx::destroy(inst);
}
```

```text
두 번 제출한 뒤 카운터 0: GPU가 1을 기다린다
CPU가 원본을 채우고 1을 알림
카운터 3, 결과 40 41 42 43
```

- **먼저 내고 나중에 알리기**: 타임라인 세마포어는 아직 아무도 알리지 않은 값을 기다리는 제출도 받습니다. 두 번 제출한 뒤의 카운터가 0인 것은 GPU가 CPU의 1을 기다리고 있다는 뜻입니다. 바이너리 세마포어는 알릴 쪽이 먼저 제출되어 있어야 기다릴 수 있습니다
- **단계 마스크**: 기다림의 `stageMask`는 어느 단계부터 기다릴지, 알림의 `stageMask`는 어느 단계까지 끝나야 알릴지를 정합니다. 세마포어는 메모리 의존성도 함께 걸므로 A가 쓴 `mid`를 B가 그대로 읽습니다
- **CPU가 쓴 값**: 제출하기 전에 쓴 값은 제출이 GPU에 보이게 해 주지만, 제출한 뒤에 쓴 값은 그렇지 않습니다. 그래서 A의 처음에 HOST 쓰기 배리어를 넣었습니다. lavapipe에서는 빼도 결과가 같지만 명세가 요구합니다

## 펜스와 바이너리 세마포어

펜스는 제출 하나(또는 `vkAcquireNextImageKHR`)에 붙어 GPU가 끝났음을 CPU에 알리는 깃발입니다. 신호를 받은 펜스는 `vkResetFences`로 되돌려야 다시 쓸 수 있고, 첫 기다림을 건너뛰도록 신호를 받은 상태(`SIGNALED`)로 만들 수도 있습니다. 바이너리 세마포어는 한 번 알리고 한 번 기다리는 GPU 사이의 신호입니다. 스왑체인의 이미지 받기와 내보내기는 바이너리 세마포어만 받으므로, 화면에 그리는 프로그램은 두 세마포어를 함께 씁니다([「프레임 루프」](#/read/cpp-vulkan/tri-loop)).

`vkQueueWaitIdle`과 `vkDeviceWaitIdle`은 큐나 장치의 모든 일이 끝날 때까지 기다리는 큰 망치입니다. 프로그램을 끝내거나 리소스를 통째로 바꿀 때 말고는 쓰지 않습니다.

<Note>
한 큐 안의 순서는 배리어 대신 이벤트(`vkCmdSetEvent2`, `vkCmdWaitEvents2`)로도 겁니다. 알리는 자리와 기다리는 자리를 떼어 놓을 수 있어서, 그 사이의 명령은 막히지 않습니다([「파이프라인 배리어」](#/read/cpp-vulkan/vsync-barriers)).
</Note>

## 정리

- 한 큐 안의 순서는 배리어, CPU가 GPU를 기다리는 데는 펜스, 제출 사이에는 세마포어를 쓴다
- 타임라인 세마포어는 커지기만 하는 카운터라서 CPU와 GPU가 서로 알리고 기다릴 수 있고, 알리기 전에 기다리는 제출을 먼저 낼 수 있다
- 제출한 뒤에 CPU가 쓴 값은 HOST 배리어로 보이게 하고, 펜스는 되돌려 다시 쓰며, 스왑체인에는 바이너리 세마포어를 쓴다
