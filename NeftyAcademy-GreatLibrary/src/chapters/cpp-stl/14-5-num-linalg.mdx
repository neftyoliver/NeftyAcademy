[「vector」](#/read/cpp-stl/seq-vector)에서 2차원 격자를 `vector` 하나에 담고 `y * width + x`로 칸을 찾았습니다. 그 곱셈과 덧셈을 대신해 주는 것이 C++23의 `std::mdspan`입니다. [`span`](#/read/cpp-stl/seq-span)이 "연속한 메모리의 한 구간"이라면 `mdspan`은 "그 구간을 몇 행 몇 열로 본 것"입니다. 메모리를 갖지 않고 보기만 합니다.

<Note kind="warn">
GCC 14의 libstdc++에는 `<mdspan>`이 없습니다. 이 페이지의 `mdspan` 예제는 **Clang 20과 libc++ 20**(`-stdlib=libc++`)으로 돌린 것입니다. GCC만 쓸 수 있을 때의 길은 아래 「mdspan이 없을 때」에 있습니다.
</Note>

## 같은 메모리를 여러 모양으로

```cpp libcxx
#include <cstddef>
#include <mdspan>
#include <print>
#include <vector>

// 행마다 합을 찍는다. 행과 열의 수는 실행 중에 정해진다
void print_row_sums(std::mdspan<const double, std::dextents<std::size_t, 2>> m)
{
    for (std::size_t r = 0; r < m.extent(0); ++r) {
        double sum = 0;
        for (std::size_t c = 0; c < m.extent(1); ++c)
            sum += m[r, c];
        std::println("행 {} 합 {}", r, sum);
    }
}

int main()
{
    std::vector<double> storage(12);
    for (std::size_t i = 0; i < storage.size(); ++i)
        storage[i] = static_cast<double>(i);

    std::mdspan grid(storage.data(), 3, 4);     // 3행 4열로 본다
    std::println("{}행 {}열, [1, 2] = {}", grid.extent(0), grid.extent(1),
                 grid[1, 2]);

    grid[2, 3] = 99.0;                          // 보기를 통해 쓰면 원본이 바뀐다
    std::println("{}", storage[11]);
    print_row_sums(grid);

    // 같은 12칸을 다른 모양으로
    std::mdspan wide(storage.data(), 2, 6);
    std::mdspan cube(storage.data(), 2, 3, 2);
    std::println("{} {}", wide[1, 0], cube[1, 2, 1]);
}
```

```text
3행 4열, [1, 2] = 6
99
행 0 합 6
행 1 합 22
행 2 합 126
6 99
```

`grid[1, 2]`는 C++23에서 [`[]`가 인자를 여럿 받게](#/read/cpp-syntax/op-index) 되어 가능한 문법입니다. `grid[1][2]`가 아닙니다. 차원마다의 크기는 `extent(차원)`으로 묻습니다.

`mdspan`은 [`span`처럼](#/read/cpp-stl/seq-span) 원본의 수명에 기댑니다. `storage`가 사라지거나 다시 할당되면 보기는 죽은 메모리를 가리킵니다. 함수에는 값으로 넘깁니다.

## 크기를 타입에 넣기

차원의 크기는 실행 중에 정할 수도, 타입에 못 박을 수도 있습니다.

```cpp libcxx
#include <array>
#include <cstddef>
#include <mdspan>
#include <print>
#include <vector>

int main()
{
    std::vector<double> storage(12);
    std::array<float, 9> identity{1, 0, 0, 0, 1, 0, 0, 0, 1};

    // 둘 다 실행 중에: 포인터 + 크기 둘
    using Dynamic2 = std::dextents<std::size_t, 2>;
    std::mdspan<double, Dynamic2> dynamic(storage.data(), 3, 4);
    // 둘 다 타입에: 포인터뿐
    std::mdspan<float, std::extents<std::size_t, 3, 3>> fixed(identity.data());
    // 열만 타입에
    using Rows4 = std::extents<std::size_t, std::dynamic_extent, 4>;
    std::mdspan<double, Rows4> mixed(storage.data(), 3);

    std::println("{} {} {}", sizeof dynamic, sizeof fixed, sizeof mixed);
    std::println("{} {} {}", fixed[1, 1], fixed[1, 2], mixed.extent(0));
}
```

```text
24 8 16
1 0 3
```

3×3 회전 행렬이나 4×4 변환 행렬처럼 크기가 정해진 것은 타입에 넣습니다. 보기가 포인터 하나 크기가 되고, 컴파일러가 루프 횟수를 알므로 풀어서 최적화합니다. 이미지처럼 크기가 실행 중에 정해지는 것은 `std::dextents<std::size_t, 2>`입니다.

## 행 우선과 열 우선

같은 메모리를 2차원으로 읽는 순서가 둘 있습니다.

| 배치 | 메모리에 놓이는 순서 | 쓰는 곳 |
| --- | --- | --- |
| `std::layout_right` (기본) | 한 행이 끝나면 다음 행 | C와 C++의 배열, 이미지 |
| `std::layout_left` | 한 열이 끝나면 다음 열 | 포트란, BLAS와 LAPACK, OpenGL의 행렬 |

```cpp libcxx
#include <chrono>
#include <cstddef>
#include <mdspan>
#include <print>
#include <vector>

int main()
{
    std::vector<double> small(12);
    for (std::size_t i = 0; i < small.size(); ++i)
        small[i] = static_cast<double>(i);
    using Extents = std::dextents<std::size_t, 2>;
    std::mdspan<double, Extents> by_rows(small.data(), 3, 4);
    std::mdspan<double, Extents, std::layout_left> by_columns(small.data(), 3, 4);
    std::println("{} {}", by_rows[1, 2], by_columns[1, 2]);

    // 4096 x 4096 을 두 방향으로 훑는다
    const std::size_t n = 4096;
    std::vector<double> storage(n * n, 1.0);
    std::mdspan grid(storage.data(), n, n);
    using clock = std::chrono::steady_clock;
    using ms = std::chrono::duration<double, std::milli>;

    double sum = 0;
    const auto t0 = clock::now();
    for (std::size_t r = 0; r < n; ++r)
        for (std::size_t c = 0; c < n; ++c)
            sum += grid[r, c];
    const auto t1 = clock::now();
    for (std::size_t c = 0; c < n; ++c)
        for (std::size_t r = 0; r < n; ++r)
            sum += grid[r, c];
    const auto t2 = clock::now();
    std::println("행 먼저 {:.0f} ms, 열 먼저 {:.0f} ms", ms(t1 - t0).count(),
                 ms(t2 - t1).count());
    std::println("{}", sum);
}
```

```text nocheck
6 7
행 먼저 18 ms, 열 먼저 192 ms
33554432
```

첫 줄은 같은 12칸에서 `[1, 2]`가 가리키는 칸이 배치에 따라 다르다는 것입니다. 다른 라이브러리가 만든 행렬을 볼 때는 그쪽 배치를 템플릿 인자로 밝혀야 하고, 틀리면 전치된 행렬을 읽게 됩니다.

둘째 줄이 더 중요합니다. 1,600만 칸을 **메모리에 놓인 순서대로** 훑으면 18ms, 가로질러 훑으면 192ms입니다(x86-64 서버, Clang 20 `-O2`). 계산은 똑같고 [캐시](#/read/cpp-stl/seq-vector)를 쓰는 방식만 다릅니다. 이중 루프에서는 **바깥 루프가 메모리에서 멀리 뛰는 차원**을 돌게 합니다. 기본 배치에서는 행이 바깥입니다.

## mdspan이 없을 때

GCC 14에서는 여러 인자를 받는 `operator[]`로 작은 보기를 만들어 씁니다. 20줄이면 됩니다.

```cpp
#include <cstddef>
#include <print>
#include <span>
#include <vector>

template <typename T>
class Grid2D {
public:
    Grid2D(std::span<T> cells, std::size_t rows, std::size_t columns)
        : cells_(cells), rows_(rows), columns_(columns)
    {
    }

    T& operator[](std::size_t r, std::size_t c) const
    {
        return cells_[r * columns_ + c];
    }
    std::size_t rows() const { return rows_; }
    std::size_t columns() const { return columns_; }
    // 한 행을 span 으로
    std::span<T> row(std::size_t r) const
    {
        return cells_.subspan(r * columns_, columns_);
    }

private:
    std::span<T> cells_;
    std::size_t rows_;
    std::size_t columns_;
};

int main()
{
    std::vector<int> storage(12);
    const Grid2D<int> grid(storage, 3, 4);
    for (std::size_t r = 0; r < grid.rows(); ++r)
        for (std::size_t c = 0; c < grid.columns(); ++c)
            grid[r, c] = static_cast<int>(r * 10 + c);

    for (const int value : grid.row(2))
        std::print("[{}]", value);
    std::println(" / storage[6] = {}", storage[6]);
}
```

```text
[20][21][22][23] / storage[6] = 12
```

필요한 것이 2차원 하나뿐이면 이쪽이 오히려 읽기 쉽습니다. 나중에 `std::mdspan`으로 바꿀 때 부르는 쪽의 `grid[r, c]`는 그대로 둘 수 있습니다.

## linalg

C++26에는 `mdspan` 위에서 도는 선형대수 함수 모음 `<linalg>`가 들어옵니다. 행렬과 벡터의 곱, 내적, 삼각 행렬 풀기 같은 BLAS의 연산을 표준 함수로 옮긴 것입니다.

```cpp nocheck
#include <linalg>

std::mdspan a(a_storage.data(), 3, 3);
std::mdspan x(x_storage.data(), 3);
std::mdspan y(y_storage.data(), 3);
std::linalg::matrix_vector_product(a, x, y);        // y = A x
const double d = std::linalg::dot(x, y);
```

<Note kind="warn">
위 조각은 **돌려 보지 못했습니다.** 이 책을 쓰는 환경의 GCC 14와 Clang 20(libc++ 20) 어느 쪽에도 `<linalg>`가 없습니다. 인터페이스는 표준안(P1673)에 적힌 대로이고, 쓸 수 있게 되면 예제를 돌려 보고 고칠 자리입니다. 그때까지 행렬 계산은 Eigen이나 BLAS 구현을 씁니다. `mdspan`의 일부만 잘라 보는 `std::submdspan`(C++26)도 libc++ 20에는 아직 없습니다.
</Note>

## 정리

- `mdspan`은 1차원 메모리를 여러 차원으로 보는, 갖지 않는 보기다. `m[r, c]`로 읽고 쓰고, 크기를 타입에 넣으면 포인터 하나 크기가 된다
- 기본 배치는 행 우선이다. 다른 라이브러리의 행렬은 배치를 맞춰서 보고, 루프는 메모리에 놓인 순서대로 돈다. 재어 보니 열 배 차이였다
- GCC 14에는 `<mdspan>`이 없어 작은 보기를 직접 만들고, `<linalg>`는 아직 어느 쪽에도 없다
