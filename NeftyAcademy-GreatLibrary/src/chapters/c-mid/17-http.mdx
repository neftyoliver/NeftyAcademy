**HTTP**는 웹의 언어이고, 요즘은 앱과 기기가 서버와 이야기하는 표준 방법(REST API)이기도 합니다. HTTP/1.1은 [「TCP」](#/read/c-mid/net-tcp) 위에 **사람이 읽을 수 있는 글자**로 주고받는 약속이라, 소켓만으로도 직접 짜 볼 수 있습니다.

## 요청과 응답의 생김새

요청은 이렇게 생겼습니다. 줄 끝은 반드시 `\r\n`(CR LF)입니다.

```text nocheck
GET /cat HTTP/1.1          ← 요청 줄: 방법, 경로, 판
Host: localhost            ← 머리(header): 이름: 값
Connection: close
                           ← 빈 줄: 머리 끝
(몸통이 있으면 여기부터)
```

응답은 첫 줄이 **상태 줄**(판, 상태 코드, 설명)이고 나머지 모양은 같습니다.

| 상태 코드 | 뜻 | 예 |
| --- | --- | --- |
| 2xx | 성공 | 200 OK, 201 Created |
| 3xx | 다른 곳으로 가라 | 301, 302, 304 Not Modified |
| 4xx | 요청이 잘못됨 | 400, 401, 403, 404 Not Found |
| 5xx | 서버가 잘못됨 | 500, 503 |

방법(method)은 읽기 `GET`, 새로 만들기 `POST`, 바꾸기 `PUT`·`PATCH`, 지우기 `DELETE`가 주로 쓰입니다.

## 소켓으로 직접 짜기

서버는 경로가 `/cat`이면 JSON을, 아니면 404를 돌려줍니다.

```c flags=-D_POSIX_C_SOURCE=200809L
#include <arpa/inet.h>
#include <stdio.h>
#include <string.h>
#include <sys/socket.h>
#include <threads.h>
#include <unistd.h>

static int server(void *arg)
{
    int c = accept(*(int *)arg, nullptr, nullptr);
    char req[1024] = "";
    size_t used = 0;
    // 빈 줄(\r\n\r\n)이 나올 때까지가 요청의 머리
    while (used < sizeof req - 1 && !strstr(req, "\r\n\r\n")) {
        ssize_t r = recv(c, req + used, sizeof req - 1 - used, 0);
        if (r <= 0)
            break;
        used += (size_t)r;
        req[used] = '\0';
    }
    char method[8], path[64];
    int ok = sscanf(req, "%7s %63s", method, path) == 2;
    const char *body = "{\"name\":\"minerva\",\"tail\":\"up\"}";
    char resp[512];
    int n;
    if (ok && strcmp(method, "GET") == 0 && strcmp(path, "/cat") == 0)
        n = snprintf(resp, sizeof resp,
                     "HTTP/1.1 200 OK\r\n"
                     "Content-Type: application/json\r\n"
                     "Content-Length: %zu\r\n"
                     "Connection: close\r\n\r\n%s", strlen(body), body);
    else
        n = snprintf(resp, sizeof resp,
                     "HTTP/1.1 404 Not Found\r\n"
                     "Content-Length: 0\r\nConnection: close\r\n\r\n");
    send(c, resp, (size_t)n, 0);
    close(c);
    return 0;
}

static void get(struct sockaddr_in *a, const char *path)
{
    int lfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in s = *a;
    socklen_t len = sizeof s;
    bind(lfd, (struct sockaddr *)&s, sizeof s);
    listen(lfd, 1);
    getsockname(lfd, (struct sockaddr *)&s, &len);
    thrd_t t;
    thrd_create(&t, server, &lfd);

    int fd = socket(AF_INET, SOCK_STREAM, 0);
    connect(fd, (struct sockaddr *)&s, sizeof s);
    char req[256];
    int n = snprintf(req, sizeof req,
                     "GET %s HTTP/1.1\r\n"
                     "Host: localhost\r\n"
                     "Connection: close\r\n\r\n", path);
    send(fd, req, (size_t)n, 0);

    char resp[1024];
    size_t used = 0;
    ssize_t r;
    while ((r = recv(fd, resp + used, sizeof resp - 1 - used, 0)) > 0)
        used += (size_t)r;          // 서버가 닫을 때까지 읽는다
    resp[used] = '\0';
    printf("--- GET %s\n", path);
    for (char *p = resp; *p; p++)   // \r은 빼고 찍는다
        if (*p != '\r')
            putchar(*p);
    putchar('\n');
    close(fd);
    thrd_join(t, nullptr);
    close(lfd);
}

int main(void)
{
    struct sockaddr_in a = {.sin_family = AF_INET,
                            .sin_addr.s_addr = htonl(INADDR_LOOPBACK)};
    get(&a, "/cat");
    get(&a, "/dog");
    return 0;
}
```

```text
--- GET /cat
HTTP/1.1 200 OK
Content-Type: application/json
Content-Length: 30
Connection: close

{"name":"minerva","tail":"up"}
--- GET /dog
HTTP/1.1 404 Not Found
Content-Length: 0
Connection: close

```

- 서버는 **빈 줄**(`\r\n\r\n`)을 볼 때까지 읽어야 요청 머리가 끝났음을 압니다. TCP에는 경계가 없으니, 이것이 HTTP의 경계 표시입니다
- 몸통의 끝은 `Content-Length`로 알립니다. 클라이언트는 그만큼 읽으면 됩니다. 이 예제는 `Connection: close`로 "다 보내면 연결을 닫겠다"고 약속해서, 클라이언트가 단순히 닫힐 때까지 읽습니다
- 처음 만든 판에서는 서버의 `req` 버퍼를 비우지 않고 `strstr`을 불렀다가, gcc와 clang에서 결과가 달라졌습니다. 초기화하지 않은 배열을 읽은 것입니다. 받는 버퍼는 처음부터 빈 문자열로 시작합니다

<Note kind="warn">
이 서버는 요청을 이해하는 모양만 보여 주는 장난감입니다. 실제 HTTP 파싱은 머리의 대소문자, 줄 접기, 한 연결에 요청 여러 개, 덩어리 전송(`Transfer-Encoding: chunked`), 너무 긴 머리 같은 함정이 많고, 잘못 짜면 보안 구멍이 됩니다. 실제 서버는 검증된 라이브러리나 웹 서버(nginx 등) 뒤에서 만듭니다.
</Note>

## 실제로는 libcurl

바깥 서버에 요청할 때 C에서는 **libcurl**을 가장 많이 씁니다. HTTPS(암호화), 리다이렉트, 압축, HTTP/2를 다 처리해 줍니다.

```c nocheck
#include <stdio.h>
#include <curl/curl.h>

// 받은 조각마다 불린다. 여기서는 그대로 화면에 쓴다
static size_t on_data(char *p, size_t size, size_t n, void *user)
{
    (void)user;
    return fwrite(p, size, n, stdout);
}

int main(void)
{
    curl_global_init(CURL_GLOBAL_DEFAULT);
    CURL *h = curl_easy_init();
    if (!h)
        return 1;
    curl_easy_setopt(h, CURLOPT_URL, "https://example.com/");
    curl_easy_setopt(h, CURLOPT_WRITEFUNCTION, on_data);
    curl_easy_setopt(h, CURLOPT_FOLLOWLOCATION, 1L);   // 3xx면 따라가기
    curl_easy_setopt(h, CURLOPT_TIMEOUT, 10L);

    CURLcode rc = curl_easy_perform(h);
    if (rc != CURLE_OK) {
        fprintf(stderr, "실패: %s\n", curl_easy_strerror(rc));
    } else {
        long status;
        curl_easy_getinfo(h, CURLINFO_RESPONSE_CODE, &status);
        fprintf(stderr, "상태 %ld\n", status);
    }
    curl_easy_cleanup(h);
    curl_global_cleanup();
    return rc == CURLE_OK ? 0 : 1;
}
```

이 검사용 컴퓨터(libcurl 8.5.0)는 바깥 인터넷이 막혀 있어 "Failure when receiving data from the peer"로 실패했습니다. 컴파일과 실행까지는 확인했고, 인터넷이 되는 컴퓨터에서는 example.com의 HTML이 나온 뒤 `상태 200`이 찍힙니다. `-lcurl`로 링크하고, 우분투에서는 `libcurl4-openssl-dev` 패키지가 필요합니다. 받은 데이터는 `CURLOPT_WRITEFUNCTION`에 넘긴 함수로 조각조각 들어오며, 이 함수는 처리한 바이트 수를 돌려줘야 합니다.

## HTTP의 판

| 판 | 실려 가는 곳 | 특징 |
| --- | --- | --- |
| HTTP/1.1 | TCP | 글자로 된 머리, 위에서 직접 짠 것 |
| HTTP/2 | TCP + TLS | 이진 형식, 한 연결에 요청 여러 개를 섞어 보냄 |
| HTTP/3 | QUIC(UDP) | TCP의 기다림 없이 여러 흐름을 따로 |

판이 바뀌어도 방법, 경로, 머리, 상태 코드의 뜻은 같습니다. 그래서 1.1의 글자 모양을 알면 나머지도 읽힙니다. 지금의 웹은 거의 모두 **HTTPS**, 곧 [「TLS로 암호화하기」](#/read/c-mid/iot-tls)의 TLS 위의 HTTP입니다.

## 정리

- HTTP/1.1 요청은 요청 줄, 머리, 빈 줄, 몸통 순서이고, 응답은 상태 줄로 시작하며 줄 끝은 `\r\n`이다
- 머리의 끝은 빈 줄, 몸통의 끝은 `Content-Length`나 연결 닫기로 알리고, 상태 코드의 첫 자리로 결과의 종류를 안다
- 장난감은 소켓으로 짤 수 있지만 실제로는 libcurl 같은 검증된 라이브러리를 쓰며, 지금의 웹은 TLS 위의 HTTPS다
