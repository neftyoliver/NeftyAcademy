글 상자의 높이는 줄 수에서 나오고, 줄 수는 줄을 어디서 끊느냐에서 나옵니다. [「제약과 박스 모델」](#/read/cpp-complex-app/lay-box)에서 미뤄 둔 물음입니다. 조판은 두 번 담는 일입니다. 낱말을 줄에 담고, 줄을 쪽에 담습니다. 두 번 모두 끊어도 되는 자리 가운데 몇 곳을 고르는 문제이고, 고르는 기준에 따라 결과가 달라집니다.

```cpp
// 줄 나눔 두 가지(채우는 대로, 문단 전체를 보고)와 쪽 나눔의 규칙
#include <algorithm>
#include <format>
#include <print>
#include <sstream>
#include <string>
#include <vector>

int cols(const std::string& s) {                 // 화면 칸 수: 한글은 두 칸
    int n = 0;
    for (unsigned char c : s) n += c < 0x80 ? 1 : (c & 0xC0) == 0xC0 ? 2 : 0;
    return n;
}

using Words = std::vector<std::string>;
using Breaks = std::vector<std::size_t>;         // 줄마다 첫 낱말의 번호

int length(const Words& w, std::size_t from, std::size_t to) {  // 한 줄에 놓은 너비
    int len = static_cast<int>(to - from) - 1;   // 낱말 사이의 빈칸
    for (std::size_t i = from; i < to; ++i) len += cols(w[i]);
    return len;
}

Breaks greedy(const Words& w, int width) {       // 들어가는 데까지 채운다
    Breaks starts{0};
    for (std::size_t i = 1; i < w.size(); ++i)
        if (length(w, starts.back(), i + 1) > width) starts.push_back(i);
    return starts;
}

Breaks optimal(const Words& w, int width) {      // 남는 칸의 제곱 합이 가장 작게
    std::size_t n = w.size();
    std::vector<long> best(n + 1, 0);            // i번 낱말부터 끝까지의 가장 싼 값
    std::vector<std::size_t> next(n + 1, n);     // 그때 다음 줄이 시작하는 낱말
    for (std::size_t i = n; i-- > 0;) {
        best[i] = -1;
        for (std::size_t j = i + 1; j <= n; ++j) {
            int len = length(w, i, j);
            if (len > width && j > i + 1) break; // 낱말 하나는 넘쳐도 놓는다
            long slack = j == n ? 0 : width - len;   // 끝줄의 빈칸은 세지 않는다
            long cost = slack * slack + best[j];
            if (best[i] < 0 || cost < best[i]) best[i] = cost, next[i] = j;
        }
    }
    Breaks starts;
    for (std::size_t i = 0; i < n; i = next[i]) starts.push_back(i);
    return starts;
}

void show(const char* what, const Words& w, Breaks starts, int width) {
    starts.push_back(w.size());
    long cost = 0;
    std::string out;
    for (std::size_t k = 0; k + 1 < starts.size(); ++k) {
        std::string line;
        for (std::size_t i = starts[k]; i < starts[k + 1]; ++i)
            line += (line.empty() ? "" : " ") + w[i];
        int slack = std::max(0, width - cols(line));
        if (k + 2 < starts.size()) cost += slack * slack;
        out += "  " + line + std::string(slack, ' ') + "|";
        out += std::string(slack, '<') + "\n";
    }
    std::print("{} (남는 칸의 제곱 합 {})\n{}", what, cost, out);
}

struct Para {
    std::string name;
    int lines;
    bool keep_next = false;                      // 제목: 다음 문단과 붙어 다닌다
};

void paginate(const std::vector<Para>& paras, int height, bool rules) {
    int page = 1, room = height;
    std::string out = "  1쪽:";
    auto turn = [&] {
        if (room > 0) out += std::format(" [빈 줄 {}]", room);
        out += std::format("\n  {}쪽:", ++page), room = height;
    };
    for (const Para& para : paras) {
        int left = para.lines, first = 1;
        bool lonely = room < left + 2 && room < height;  // 다음 문단이 못 따라온다
        if (rules && para.keep_next && lonely) turn();
        while (left > 0) {
            int take = std::min(left, room);
            if (rules && take < left) {          // 문단이 쪽을 넘는다
                if (left - take < 2) take = left - 2;  // 다음 쪽에 한 줄만 안 가게
                if (take < 2) take = 0;                // 이 쪽에 한 줄만 안 남게
                if (take == 0 && room == height) take = room;  // 새 쪽이면 푼다
            }
            if (take > 0) {
                int last = first + take - 1;
                out += std::format(" {}({}~{})", para.name, first, last);
                left -= take, first += take, room -= take;
            }
            if (left > 0) turn();
        }
    }
    std::println("{}", out);
}

int main() {
    std::istringstream text(
        "편집기는 글자를 줄에 담고 줄을 쪽에 담는다 가득 채우기만 하면 어떤 줄에는 "
        "큰 빈칸이 남으므로 문단 전체를 보고 빈칸을 고르게 나누는 방법도 쓴다");
    Words w;
    for (std::string word; text >> word;) w.push_back(word);
    show("채우는 대로", w, greedy(w, 24), 24);
    show("문단 전체를 보고", w, optimal(w, 24), 24);
    Words typing(w.begin(), w.begin() + 14);     // "남으므로"를 치기 전
    show("문단 전체를 보고, 치는 도중", typing, optimal(typing, 24), 24);

    std::vector<Para> paras = {
        {"가", 9}, {"제목", 1, true}, {"나", 8}, {"다", 3}, {"라", 8}, {"마", 10}};
    std::println("규칙 없이 한 쪽에 10줄");
    paginate(paras, 10, false);
    std::println("외톨이 줄과 홀로 남는 제목을 막고");
    paginate(paras, 10, true);
}
```

- **채우는 대로**: `greedy`는 들어가는 데까지 담고 넘치는 낱말부터 다음 줄로 보냅니다. 한 번 훑으면 끝나지만, 셋째 줄을 한 칸만 남기고 채운 값으로 넷째 줄이 여덟 칸 비었습니다
- **문단 전체를 보고**: `optimal`은 줄마다 남는 칸을 제곱해 더한 값이 가장 작은 나눔을 찾습니다. "어떤"을 일부러 내려보내 두 줄을 6칸과 3칸으로 고르게 했습니다. 제곱이라서 큰 빈칸 하나가 작은 빈칸 여럿보다 비쌉니다
- **뒤에서부터 푼다**: `best[i]`는 i번 낱말에서 시작하는 나머지의 가장 싼 값입니다. 앞을 어떻게 끊었든 나머지는 같은 문제이므로 한 번만 풀어 적어 둡니다. 끝줄의 빈칸은 흠이 아니어서 세지 않습니다
- **치는 동안 앞줄이 움직인다**: 열네 낱말까지 쳤을 때 "어떤"은 셋째 줄에 있습니다. 다음 낱말을 치면 다섯째 줄이 생기면서 두 줄 위의 "어떤"이 내려갑니다. 채우는 대로는 앞에서부터 정하므로 고친 줄의 앞줄보다 위는 바뀌지 않습니다
- **외톨이 줄**: 규칙 없이 담으면 3쪽 머리에 "다"의 끝줄이, 3쪽 끝에 "마"의 첫 줄이 홀로 떨어집니다. 규칙은 쪽을 넘는 문단이 양쪽에 두 줄씩은 두게 합니다. 세 줄인 "다"는 그렇게 나눌 수 없어 통째로 넘어갔고, "라"는 한 줄을 덜 담아 두 줄을 넘겼습니다
- **제목은 다음 문단과 함께**: 1쪽 끝의 제목처럼 뒤에 본문 두 줄이 따라올 자리가 없으면 제목부터 다음 쪽으로 넘깁니다
- **규칙의 값은 빈 줄**: 세 쪽이 한 줄씩 비었고 네 쪽이던 글이 다섯 쪽이 됐습니다. 새 쪽에서도 지킬 수 없는 규칙은 풉니다. 풀지 않으면 담기가 끝나지 않습니다

예제는 너비를 칸으로 세지만 실제 너비는 글꼴에서 나옵니다([「글꼴과 셰이핑」](#/read/cpp-complex-app/txt-shaping)). 나눈 결과는 문서가 아니라 문서에서 셈한 캐시이고, 너비나 글꼴이 바뀌면 버리고 다시 만듭니다.

## 조판기와 편집기에서

- **끊어도 되는 자리**: 예제는 빈칸에서만 끊습니다. 유니코드의 줄 나눔 알고리즘(UAX #14)은 글자의 속성으로 끊어도 되는 자리를 내놓고, 그 가운데 어디를 고를지는 너비를 아는 위층에 맡깁니다
- **한국어는 두 가지**: 같은 문서가 한국어는 두 방식을 모두 쓴다고 적습니다. 양쪽을 맞출 때는 음절 사이 어디서나, 오른쪽을 흘릴 때는 빈칸에서 끊는 쪽이 흔합니다. 기본값은 앞의 것이고, CSS에서는 `word-break: keep-all`이 뒤의 것입니다
- **TeX**: 문단 전체를 보는 줄 나눔은 크누스와 플래스가 TeX에 쓰려고 만든 방법입니다. 글을 상자, 늘고 줄어드는 빈칸, 벌점의 줄로 보고 동적 계획법으로 손실이 가장 작은 나눔을 찾습니다. `optimal`은 여기서 늘고 줄어드는 빈칸과 벌점을 뺀 뼈대입니다
- **Typst**: 문단의 `linebreaks`로 `"simple"`과 `"optimized"`를 고릅니다. 정하지 않으면 양쪽 맞춤인 문단에 뒤의 것을 씁니다
- **Android**: `LineBreaker`의 `BREAK_STRATEGY_SIMPLE`이 채우는 대로이고, 문서는 가장 빠르며 편집기에 알맞다고 적습니다. `BREAK_STRATEGY_HIGH_QUALITY`와 `BREAK_STRATEGY_BALANCED`는 문단 전체를 봅니다
- **CSS**: 쪽 끝에 남길 줄 수가 `orphans`, 쪽 머리로 넘길 줄 수가 `widows`이고 처음 값은 둘 다 2입니다. 끊을 자리가 모자라면 이 규칙부터 풉니다. 제목 뒤에서 끊지 말라는 뜻은 `break-after: avoid`로 적습니다
- **Word**: 문단 서식에 Widow/Orphan control, Keep with next, Keep lines together, Page break before가 있고 첫째 것은 처음부터 켜져 있습니다. 쪽을 배경에서 다시 나누는 옵션도 있습니다([「백그라운드 작업과 취소」](#/read/cpp-complex-app/rsp-background))

<Note>
다시 나누는 범위는 좁힐 수 있습니다. 문단은 서로의 줄 나눔에 끼어들지 않으므로 고친 문단만 다시 나눕니다. 줄 수가 그대로면 쪽 나눔은 건드리지 않습니다. 줄 수가 바뀌었으면 그 쪽부터 다시 담다가, 전과 같은 줄에서 시작하는 쪽을 만나면 멈춥니다. 그 뒤는 전과 같기 때문입니다.
</Note>

## 정리

- 줄 나눔은 끊어도 되는 자리를 찾는 일과 그 가운데 고르는 일로 나뉘고, 고르는 방법에는 채우는 대로와 문단 전체를 보는 것이 있다
- 문단 전체를 보는 방법은 남는 칸의 손실을 더한 값이 가장 작은 나눔을 동적 계획법으로 찾는다. 줄은 고르지만 치는 동안 앞줄이 움직인다
- 쪽 나눔은 외톨이 줄과 홀로 남는 제목을 막는 규칙을 지키며 줄을 쪽에 담고, 그 값으로 빈 줄을 낸다
