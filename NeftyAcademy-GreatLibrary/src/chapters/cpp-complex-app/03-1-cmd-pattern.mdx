[「문서, 명령, 뷰의 층」](#/read/cpp-complex-app/ana-layers)에서 문서를 고치는 길을 명령 하나로 모았습니다. 명령은 무엇을 어디에 할지 담은 값이고, 실행하는 법과 되돌리는 법을 함께 가집니다. 문서를 고치는 유일한 길이므로, 명령 하나가 약속을 어기면 실행 취소가 문서를 망가뜨립니다. 이 쪽은 그 약속을 다룹니다.

```cpp
// 명령 객체: 대상은 ID로 가리키고, 되돌리기는 역연산이나 적어 둔 전 값으로 한다
#include <algorithm>
#include <format>
#include <memory>
#include <print>
#include <string>
#include <vector>

struct Shape {
    int id, x;
    std::string name;
};

struct Doc {
    std::vector<Shape> shapes;
    int next_id = 1;
    Shape* find(int id) {
        auto it = std::ranges::find(shapes, id, &Shape::id);
        return it == shapes.end() ? nullptr : &*it;
    }
    std::string str() const {
        std::string s;
        for (const Shape& sh : shapes)
            s += std::format(" {}:{}@{}", sh.id, sh.name, sh.x);
        return s.empty() ? " (빔)" : s;
    }
};

struct Command {
    virtual ~Command() = default;
    virtual bool apply(Doc&) = 0;                // 못 하면 false, 문서는 그대로
    virtual void revert(Doc&) = 0;               // apply 바로 뒤의 문서에만 쓴다
    virtual std::string name() const = 0;
};

struct Add : Command {
    std::string what;
    int x, id = 0;
    bool fresh;                                  // 참이면 매번 새 ID(잘못된 예)
    Add(std::string w, int x, bool fresh = false)
        : what(std::move(w)), x(x), fresh(fresh) {}
    bool apply(Doc& d) override {
        if (id == 0 || fresh) id = d.next_id++;  // 처음 받은 ID를 계속 쓴다
        d.shapes.push_back({id, x, what});
        return true;
    }
    void revert(Doc& d) override { d.shapes.pop_back(); }
    std::string name() const override { return what + " 넣기"; }
};

struct Move : Command {                          // 역연산으로 되돌린다
    int id, dx;
    Move(int id, int dx) : id(id), dx(dx) {}
    bool apply(Doc& d) override {
        Shape* s = d.find(id);
        if (!s) return false;
        s->x += dx;
        return true;
    }
    void revert(Doc& d) override { d.find(id)->x -= dx; }
    std::string name() const override { return std::format("{}번 옮기기", id); }
};

struct AlignLeft : Command {                     // 역연산이 없다: 전 값을 적어 둔다
    std::vector<int> ids, before;
    explicit AlignLeft(std::vector<int> ids) : ids(std::move(ids)) {}
    bool apply(Doc& d) override {
        before.clear();
        for (int id : ids) {
            Shape* s = d.find(id);
            if (!s) return false;                // 아직 아무것도 고치지 않았다
            before.push_back(s->x);
        }
        if (before.empty()) return false;
        for (int id : ids) d.find(id)->x = std::ranges::min(before);
        return true;
    }
    void revert(Doc& d) override {
        for (std::size_t i = 0; i < ids.size(); ++i) d.find(ids[i])->x = before[i];
    }
    std::string name() const override { return "왼쪽 맞추기"; }
};

struct History {
    Doc& doc;
    std::vector<std::unique_ptr<Command>> done;
    std::size_t at = 0;                          // 이 앞까지가 실행된 명령
    explicit History(Doc& d) : doc(d) {}

    void say(const char* what, const Command& c) {
        std::println("{} {:<12} ->{}", what, c.name(), doc.str());
    }
    void run(std::unique_ptr<Command> c) {
        bool ok = c->apply(doc);
        say(ok ? "실행" : "실패", *c);
        if (!ok) return;                         // 실패한 명령은 쌓지 않는다
        done.resize(at);
        done.push_back(std::move(c));
        ++at;
    }
    void undo() {
        done[--at]->revert(doc);
        say("취소", *done[at]);
    }
    void redo() {
        bool ok = done[at]->apply(doc);
        say(ok ? "다시" : "실패", *done[at]);
        if (ok) ++at;
    }
};

int main() {
    Doc doc;
    History h{doc};
    h.run(std::make_unique<Add>("별", 3));
    h.run(std::make_unique<Move>(1, 5));
    h.run(std::make_unique<Add>("달", 20));
    h.run(std::make_unique<AlignLeft>(std::vector{1, 2}));
    h.run(std::make_unique<Move>(9, 1));         // 없는 도형
    for (int i = 0; i < 4; ++i) h.undo();
    for (int i = 0; i < 4; ++i) h.redo();

    std::println("-- Add가 다시 실행할 때마다 새 ID를 받는다면");
    Doc doc2;
    History h2{doc2};
    h2.run(std::make_unique<Add>("별", 3, true));
    h2.run(std::make_unique<Move>(1, 5));
    h2.undo();
    h2.undo();
    h2.redo();
    h2.redo();
}
```

```text
실행 별 넣기      -> 1:별@3
실행 1번 옮기기   -> 1:별@8
실행 달 넣기      -> 1:별@8 2:달@20
실행 왼쪽 맞추기  -> 1:별@8 2:달@8
실패 9번 옮기기   -> 1:별@8 2:달@8
취소 왼쪽 맞추기  -> 1:별@8 2:달@20
취소 달 넣기      -> 1:별@8
취소 1번 옮기기   -> 1:별@3
취소 별 넣기      -> (빔)
다시 별 넣기      -> 1:별@3
다시 1번 옮기기   -> 1:별@8
다시 달 넣기      -> 1:별@8 2:달@20
다시 왼쪽 맞추기  -> 1:별@8 2:달@8
-- Add가 다시 실행할 때마다 새 ID를 받는다면
실행 별 넣기      -> 1:별@3
실행 1번 옮기기   -> 1:별@8
취소 1번 옮기기   -> 1:별@3
취소 별 넣기      -> (빔)
다시 별 넣기      -> 2:별@3
실패 1번 옮기기   -> 2:별@3
```

- **대상은 ID로**: 명령은 도형을 포인터나 번호로 기억하지 않습니다. 명령과 명령 사이에 벡터가 다시 잡히고 차례가 바뀌기 때문입니다([「선택 모델」](#/read/cpp-complex-app/doc-selection))
- **다시 만들어도 같은 ID**: 넣기 명령은 처음 받은 ID를 기억했다가 다시 실행할 때 그 ID로 넣습니다. 매번 새 ID를 받으면 마지막 두 줄처럼 됩니다. 별이 2번으로 돌아오고, 1번을 옮기려던 명령은 대상을 찾지 못합니다
- **되돌리기는 실행 바로 뒤의 문서에만**: 기록이 차례를 지켜 주므로, `revert`는 자기 `apply`가 막 끝난 문서만 상대합니다. 넣기를 되돌릴 때 맨 뒤를 빼기만 해도 되는 까닭입니다. 명령 밖에서 문서를 고치는 코드가 하나라도 있으면 이 약속이 깨집니다
- **역연산이 있으면 역연산으로**: 옮기기는 반대로 옮기면 되고, 적어 둘 것은 명령의 인자뿐입니다
- **없으면 전 값을 적는다**: 왼쪽으로 맞춘 결과만 보고는 어디서 왔는지 알 수 없습니다. 고칠 곳의 전 값을 실행할 때 적어 둡니다. 지우기도 이쪽입니다
- **못 하면 손대지 않는다**: 검사를 모두 마친 뒤에 고치기 시작합니다. 실패한 명령은 기록에 넣지 않습니다. 반쯤 고치고 실패하는 명령은 되돌릴 수도 없는 문서를 남깁니다([「트랜잭션」](#/read/cpp-complex-app/cmd-transaction))

## 되돌리는 세 가지 방법

| 방법 | 적어 두는 것 | 맞는 일 | 조심할 점 |
| --- | --- | --- | --- |
| 역연산 | 명령의 인자 | 옮기기, 돌리기 | 정확히 돌아오지 않는 연산 |
| 전 값 | 고친 곳의 옛 값 | 맞추기, 지우기, 속성 바꾸기 | 고친 곳이 크면 메모리 |
| 옛 판 | 문서의 판 전체 | 불변 모델의 모든 명령 | 구조 공유가 있어야 한다 |

역연산은 실수에서 어긋납니다. `double`로 0.3에 0.1을 더했다가 빼면 0.30000000000000004가 됩니다. 좌표가 실수인 문서에서는 옮기기도 전 값을 적어 두는 편이 안전합니다. 옛 판을 통째로 간직하는 방법은 [「불변 모델과 변경 추적」](#/read/cpp-complex-app/doc-immutable)에서 보았습니다.

## 실제 틀

- **wxWidgets**: `wxCommand`를 물려받아 `Do`와 `Undo`를 채우고, 메뉴에 보일 이름을 돌려줍니다. 명령을 처리기에 맡기면 처리기가 `Do`를 부르는데, `Do`가 거짓을 돌려주면 그 명령은 기록에 들어가지 않습니다. 문서는 `Undo`를 짜는 흔한 길로 역연산과 문서 전체 되돌리기를 들고, 역연산은 되돌리는 차례가 정해져 있다는 데 기대는 방법이라고 적습니다
- **Qt**: `QUndoCommand`의 `redo`와 `undo`를 채워 스택에 넣습니다. Qt 책의 [「명령과 실행 취소」](#/read/cpp-qt/qarch-undo)에서 다뤘습니다
- **Cocoa**: 명령 클래스가 없습니다. 값을 고치는 메서드가 `NSUndoManager`에 되돌릴 때 부를 메서드와 인자를 등록합니다. 되돌리는 중에 불린 메서드가 다시 등록한 것은 다시 실행 쪽에 쌓이므로, 고치는 메서드 하나가 실행, 취소, 다시 실행을 모두 맡습니다

게임 엔진 책의 [「실행 취소」](#/read/cpp-game-engine/ed-undo)는 같은 구조를 에디터에 넣었습니다.

<Note>
메뉴와 단축키가 부르는 이름 붙은 명령([「실제 프로그램들의 구조」](#/read/cpp-complex-app/ana-cases))과 이 쪽의 명령 객체는 층이 다릅니다. 메뉴의 "지우기"는 지금의 선택을 읽어 ID 목록을 담은 명령 객체를 만들고, 명령 객체는 선택도 화면도 다시 읽지 않습니다. 실행할 때마다 화면을 읽는 명령은 다시 실행했을 때 다른 것을 지울 수 있습니다.
</Note>

## 정리

- 명령은 대상을 ID로 기억하고, 다시 실행해도 같은 ID의 물체를 만들어 뒤의 명령이 대상을 잃지 않게 한다
- 되돌리기는 실행 바로 뒤의 문서에만 쓰이며, 역연산이 정확하면 역연산으로, 아니면 실행할 때 적어 둔 전 값으로 한다
- 명령은 검사를 마친 뒤에 고치고, 실패하면 문서를 그대로 둔 채 기록에 들어가지 않는다
