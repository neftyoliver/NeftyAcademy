맞춤법 검사는 글을 낱말로 가르고, 낱말마다 사전에 묻고, 틀린 곳에 물결 밑줄을 긋는 일입니다. 사전과 고칠 말 찾기는 Hunspell 같은 라이브러리나 운영체제가 맡아 줍니다. 편집기에 남는 일은 무엇을 언제 다시 검사하고 그 결과를 어디에 두느냐입니다. 예제는 사전의 작은 모형과 편집기 쪽의 일을 함께 보입니다.

```cpp
// 맞춤법 검사: 접사 규칙이 든 사전, 고칠 말 찾기, 고친 낱말만 다시 검사하기
#include <algorithm>
#include <map>
#include <print>
#include <set>
#include <string>
#include <vector>

struct Affix { bool prefix; char flag; std::string add; };

struct Dictionary {                          // Hunspell 사전을 본뜬 작은 모형
    std::vector<Affix> affixes{              // .aff 파일의 "PFX A 0 re ." 같은 줄
        {true, 'A', "re"}, {true, 'U', "un"},
        {false, 'S', "s"}, {false, 'D', "ed"}, {false, 'G', "ing"}};
    std::map<std::string, std::string> stems{    // .dic 파일의 "lock/AUSDG" 같은 줄
        {"lock", "AUSDG"}, {"load", "AUSDG"}, {"paint", "ASDG"}, {"scroll", "SDG"},
        {"door", "S"}, {"the", ""}, {"was", ""}, {"and", ""}};
    int lookups = 0;

    bool has(const std::string& stem, const std::string& flags) const {
        auto it = stems.find(stem);          // 낱말이 있고 그 표시를 다 가졌는가
        return it != stems.end() && std::ranges::all_of(flags, [&](char f) {
            return it->second.contains(f);
        });
    }
    bool known(const std::string& word) {    // 접사를 떼어 가며 사전에서 찾는다
        ++lookups;
        std::vector<std::pair<std::string, std::string>> tries{{word, ""}};
        for (const Affix& p : affixes)
            if (p.prefix && word.starts_with(p.add))
                tries.push_back({word.substr(p.add.size()), {p.flag}});
        for (const auto& [rest, flags] : tries) {
            if (has(rest, flags)) return true;
            for (const Affix& s : affixes)
                if (!s.prefix && rest.ends_with(s.add) &&
                    has(rest.substr(0, rest.size() - s.add.size()), flags + s.flag))
                    return true;
        }
        return false;
    }
};

std::vector<std::string> suggest(Dictionary& dict, const std::string& word) {
    std::vector<std::string> out;            // 한 번 고쳐서 닿는 말 가운데 아는 것
    auto offer = [&](const std::string& s) {
        if (dict.known(s) && !std::ranges::contains(out, s)) out.push_back(s);
    };
    for (std::size_t i = 0; i <= word.size(); ++i) {
        if (i + 1 < word.size()) {           // 이웃한 두 글자의 자리 바꾸기
            std::string s = word;
            std::swap(s[i], s[i + 1]);
            offer(s);
        }
        if (i < word.size()) offer(std::string(word).erase(i, 1));  // 한 글자 빼기
        for (char c = 'a'; c <= 'z'; ++c) {  // 한 글자 넣기, 한 글자 바꾸기
            offer(std::string(word).insert(i, 1, c));
            if (i < word.size()) offer(std::string(word).replace(i, 1, 1, c));
        }
    }
    return out;
}

struct Document {
    Dictionary& dict;
    std::string text;
    std::set<std::size_t> wrong;             // 틀린 낱말이 시작하는 자리

    static bool letter(char c) { return c >= 'a' && c <= 'z'; }
    std::size_t word_end(std::size_t i) const {
        while (i < text.size() && letter(text[i])) ++i;
        return i;
    }
    void edit(std::size_t at, std::size_t n, const std::string& s, bool typing) {
        std::set<std::size_t> moved;         // 고친 곳 뒤의 표시는 자리가 밀린다
        for (std::size_t w : wrong)
            if (w < at || w >= at + n) moved.insert(w < at ? w : w - n + s.size());
        wrong = std::move(moved);
        text.replace(at, n, s);

        std::size_t caret = at + s.size(), from = at, to = word_end(caret);
        while (from > 0 && letter(text[from - 1])) --from;
        wrong.erase(wrong.lower_bound(from), wrong.upper_bound(to));
        for (std::size_t i = from; i < to; ++i) {    // 닿은 낱말만 다시 본다
            if (!letter(text[i])) continue;
            std::size_t end = word_end(i);   // 캐럿이 끝에 붙은 낱말은 아직 치는 중
            if (!(typing && end == caret) && !dict.known(text.substr(i, end - i)))
                wrong.insert(i);
            i = end;
        }
    }
    void show(const char* what) {
        std::string marks(text.size(), ' ');
        for (std::size_t w : wrong)
            std::fill(marks.begin() + w, marks.begin() + word_end(w), '~');
        marks.erase(marks.find_last_not_of(' ') + 1);
        std::println("{}: 조회 {}번\n  {}\n  {}", what, dict.lookups, text, marks);
        dict.lookups = 0;
    }
};

int main() {
    Dictionary dict;
    std::print("사전:");
    for (const char* w : {"unlocked", "reloading", "repaints", "unpaint", "doored"})
        std::print(" {}({})", w, dict.known(w) ? "맞음" : "틀림");
    dict.lookups = 0;

    Document doc{dict, {}, {}};
    doc.edit(0, 0, "the door was unlockd and the paint was reloded", false);
    doc.show("\n처음 검사");
    for (std::size_t at : doc.wrong) {
        std::string word = doc.text.substr(at, doc.word_end(at) - at);
        std::print("  {} ->", word);
        for (const std::string& s : suggest(dict, word)) std::print(" {}", s);
        std::println(" (조회 {}번)", dict.lookups);
        dict.lookups = 0;
    }
    doc.edit(13, 7, "unlocked", false);
    doc.show("unlockd를 고친 뒤");
    for (char c : std::string(" scrol")) doc.edit(doc.text.size(), 0, {c}, true);
    doc.show("끝에 scrol을 치는 동안");
    doc.edit(doc.text.size(), 0, ".", true);
    doc.show("마침표를 치자");
}
```

```text
사전: unlocked(맞음) reloading(맞음) repaints(맞음) unpaint(틀림) doored(틀림)
처음 검사: 조회 9번
  the door was unlockd and the paint was reloded
               ~~~~~~~                   ~~~~~~~
  unlockd -> unlock unlocked unlocks (조회 403번)
  reloded -> reloaded (조회 403번)
unlockd를 고친 뒤: 조회 1번
  the door was unlocked and the paint was reloded
                                          ~~~~~~~
끝에 scrol을 치는 동안: 조회 1번
  the door was unlocked and the paint was reloded scrol
                                          ~~~~~~~
마침표를 치자: 조회 1번
  the door was unlocked and the paint was reloded scrol.
                                          ~~~~~~~ ~~~~~
```

- **사전은 낱말꼴을 다 적지 않는다**: Hunspell의 사전은 낱말 목록(.dic)과 접사 규칙(.aff) 두 파일입니다. `lock/AUSDG`는 lock에 A, U, S, D, G 규칙을 붙여도 된다는 뜻입니다. 접두사와 접미사는 함께 붙을 수 있어서 이 한 줄이 열두 가지 꼴을 받아들입니다. paint에는 U가 없으므로 unpaint는 틀린 말입니다. 실제 규칙에는 조건이 붙습니다. `SFX D y ied [^aeiou]y`는 모음이 아닌 글자와 y로 끝나는 낱말에서 y를 떼고 ied를 붙입니다
- **고칠 말은 틀린 말을 조금씩 바꿔 찾는다**: 글자 하나를 빼거나 넣거나 바꾸거나 이웃끼리 자리를 바꾼 말 가운데 사전이 아는 것을 고릅니다. 검사는 조회 한 번인데 고칠 말에는 403번이 듭니다. 그래서 미리 찾아 두지 않고 사용자가 메뉴를 열 때 찾습니다. Hunspell은 .aff 파일에 적힌 자주 틀리는 바꿔치기(`REP`)를 먼저 해 보고, 자판에서 이웃한 글자(`KEY`)로도 바꿔 봅니다
- **고친 곳만 다시 검사한다**: 낱말 하나를 고친 뒤의 조회는 한 번입니다. 틀린 곳은 글자의 위치로 적어 두므로, 앞쪽이 바뀌면 [「선택 모델」](#/read/cpp-complex-app/doc-selection)의 책갈피처럼 밀어 줍니다. reloded 밑의 표시가 한 칸 밀린 것이 그것입니다
- **치는 중인 낱말은 미룬다**: scrol을 치는 동안에는 표시가 없고, 마침표를 쳐서 낱말을 벗어나자 검사합니다. 치는 내내 빨간 줄이 따라다니지 않게 하려는 것입니다. Firefox의 편집기도 캐럿이 놓인 낱말은 검사하지 않습니다. 캐럿을 다른 곳으로 옮길 때에도 떠난 낱말을 검사해야 합니다. 조합 중인 글([「입력기」](#/read/cpp-complex-app/txt-ime))도 확정되기 전에는 검사하지 않습니다
- **표시는 문서가 아니다**: 틀린 곳의 목록은 문서 모델 밖에 둡니다. 저장하지 않고 실행 취소에도 넣지 않습니다. 목록이 바뀌면 그 낱말이 놓인 곳만 다시 그립니다([「무효화와 다시 그리기」](#/read/cpp-complex-app/rnd-invalidate))

## 큰 문서와 실제 검사기

- **검사는 한가할 때 조금씩**: 긴 문서를 열자마자 전부 검사하지 않습니다. 화면에 보이는 문단부터, 입력이 없는 틈에 조금씩 합니다. Chromium은 편집할 때마다 곧바로 돌던 맞춤법 검사를 한가한 시간에 돌도록 바꿨습니다. 입력이 늦어지는 것을 막으려는 것이었습니다. 다른 스레드에서 돌릴 때는 문서의 한 판을 넘기고([「불변 모델과 변경 추적」](#/read/cpp-complex-app/doc-immutable)), 결과가 돌아왔을 때 그 사이에 바뀐 문단의 것은 버립니다
- **낱말 가르기**: 빈칸만으로는 가를 수 없습니다. 문장 부호를 떼어 내고 숫자와 주소, 코드는 건너뜁니다. ICU의 낱말 경계 찾기는 유니코드의 규칙을 따르고, 빈칸 없이 쓰는 중국어, 일본어, 타이어는 사전으로 가릅니다
- **목록은 셋이다**: 언어의 사전, 사용자가 추가한 낱말, 이 문서에서만 무시하기로 한 낱말입니다. macOS는 셋 가운데 어느 것도 받아 주지 않는 낱말을 틀린 것으로 보고, 무시 목록을 문서와 함께 저장할 수 있게 해 줍니다
- **언어는 구간마다**: 한 문서에 여러 언어가 섞입니다. 검사기는 언어마다 하나이므로(Windows의 `ISpellChecker`가 그렇습니다), 글의 구간에 달린 언어 속성을 보고 검사기를 고릅니다

| 검사기 | 검사 | 고칠 말 | 추가와 무시 |
|---|---|---|---|
| Hunspell | `spell` | `suggest` | `add` |
| Windows | `ISpellChecker`의 `Check` | `Suggest` | `Add`, `Ignore` |
| macOS | `NSSpellChecker`의 `requestChecking` | `guesses` | `learnWord`, `ignoreWord` |

<Note>
사전을 직접 실을지 운영체제의 검사기를 쓸지는 무엇을 맞출지에 달렸습니다. Hunspell을 실으면 모든 플랫폼에서 같은 결과가 나옵니다. LibreOffice와 Firefox, Chrome이 Hunspell을 씁니다. 운영체제의 검사기를 쓰면 사용자가 다른 앱에서 추가한 낱말과 언어 설정을 그대로 물려받습니다. Enchant는 Hunspell, Nuspell, Aspell 같은 여러 검사기를 한 인터페이스로 감싸서 앱이 검사기를 몰라도 되게 해 줍니다.
</Note>

## 정리

- 사전은 낱말 목록과 접사 규칙으로 되어 있고, 검사는 접사를 떼어 가며 찾는 일이다. 고칠 말 찾기는 검사보다 훨씬 비싸므로 필요할 때만 한다
- 편집기는 고친 낱말만 다시 검사하고, 치는 중인 낱말은 미루고, 틀린 곳의 목록을 편집에 맞춰 옮긴다
- 틀린 곳의 표시는 문서 밖의 장식이다. 큰 문서에서는 보이는 곳부터 한가할 때 검사한다
