한글, 일본어, 중국어는 키 하나가 글자 하나가 아닙니다. 키는 먼저 입력기(IME)로 가고, 입력기는 조합 중인 글과 확정된 글을 따로 알려 줍니다. 툴킷의 글 상자는 이 일을 대신 받아 주지만, 캔버스에 글을 직접 그리는 편집기는 입력기의 상대 역을 스스로 맡아야 합니다. 예제는 두벌식 한글 입력기의 작은 모형과 편집기를 한 프로그램에 넣은 것입니다.

```cpp
// 입력기: 조합 중인 글과 확정된 글을 따로 받는 편집기. 입력기는 두벌식 한글의 모형
#include <print>
#include <string>
#include <string_view>

using Text = std::u32string;
std::string utf8(const Text& t) {            // 이 예제의 글자는 ASCII 아니면 한글
    std::string out;
    for (char32_t c : t)
        if (c < 0x80) out += char(c);
        else out += {char(0xE0 | c >> 12), char(0x80 | (c >> 6 & 0x3F)),
                     char(0x80 | (c & 0x3F))};
    return out;
}

const std::string kConsonantKeys = "rRseEfaqQtTdwWczxvg";
const std::string kVowelKeys = "koiOjpuPhynbml"; // 겹모음과 겹받침은 뺐다
const Text kCho = U"ㄱㄲㄴㄷㄸㄹㅁㅂㅃㅅㅆㅇㅈㅉㅊㅋㅌㅍㅎ";
const Text kJong = U"_ㄱㄲㄳㄴㄵㄶㄷㄹㄺㄻㄼㄽㄾㄿㅀㅁㅂㅄㅅㅆㅇㅈㅊㅋㅌㅍㅎ";
const Text kVowels = U"ㅏㅐㅑㅒㅓㅔㅕㅖㅗㅛㅜㅠㅡㅣ";

struct Ime {                                 // 입력기: 키를 받아 조합하고 확정한다
    char32_t cho = 0, jung = 0, jong = 0;    // 조합 중인 낱자. 없으면 0
    Text commit;                             // 이번 키로 확정된 글

    Text preedit() const {                   // 조합 중인 글
        if (!cho || !jung) return cho || jung ? Text(1, cho ? cho : jung) : Text();
        std::size_t l = kCho.find(cho), v = jung - U'ㅏ';
        std::size_t t = jong ? kJong.find(jong) : 0;
        return Text(1, char32_t(0xAC00 + (l * 21 + v) * 28 + t));
    }
    Text finish() {                          // 조합을 끝내고 그 글을 돌려준다
        Text done = preedit();
        cho = jung = jong = 0;
        return done;
    }
    bool key(char k) {                       // 돌려주는 값: 입력기가 이 키를 썼는가
        commit.clear();
        if (auto i = kConsonantKeys.find(k); i != std::string::npos) {
            char32_t c = kCho[i];
            if (cho && jung && !jong && kJong.find(c) != Text::npos) jong = c;
            else commit = finish(), cho = c; // 받침이 못 되면 새 글자의 첫소리
        } else if (auto j = kVowelKeys.find(k); j != std::string::npos) {
            char32_t next = jong;            // 받침이 있었으면 다음 글자로 넘긴다
            if (cho && !jung) jung = kVowels[j];
            else jong = 0, commit = finish(), cho = next, jung = kVowels[j];
        } else if (k == '\b' && (cho || jung)) {
            if (jong) jong = 0;              // 조합 중에는 낱자 하나만 지운다
            else if (cho && jung) jung = 0;
            else cho = jung = 0;
        } else {
            commit = finish();               // 그 밖의 키: 조합을 끝내고 넘긴다
            return false;
        }
        return true;
    }
};

struct Editor {                              // 편집기: 확정된 글만 문서에 넣는다
    Text doc, preedit;                       // preedit는 캐럿 자리에 겹쳐 그린다
    std::size_t caret = 0;
    int undo_steps = 0;

    void insert(const Text& s) {
        if (s.empty()) return;
        doc.insert(caret, s);
        caret += s.size(), ++undo_steps;
    }
    void key(char k) {                       // 입력기가 쓰지 않은 키
        if (k != '\b') insert(Text(1, char32_t(k)));
        else if (caret) doc.erase(--caret, 1), ++undo_steps;
    }
    int caret_x() const {                    // 입력기에 알려 줄 캐럿의 자리(칸)
        int x = 0;
        for (char32_t c : doc.substr(0, caret)) x += c < 0x80 ? 1 : 2;
        return x;
    }
    std::string view() const {               // 캐럿은 |, 조합 중인 글은 [ ] 안에
        Text t = doc, mark = preedit.empty() ? U"|" : U"[" + preedit + U"]";
        return utf8(t.insert(caret, mark));
    }
};

void type(Ime& ime, Editor& ed, std::string_view keys, bool show = true) {
    for (char k : keys) {
        bool used = ime.key(k);
        ed.insert(ime.commit);               // 확정된 글을 먼저 넣고
        ed.preedit = ime.preedit();          // 조합 중인 글을 바꿔 그린다
        if (!used) ed.key(k);
        if (!show) continue;
        auto text = [](const Text& t) { return t.empty() ? "-" : utf8(t); };
        std::string name = k == '\b' ? "BS" : std::string(1, k);
        std::println("  {:<3} {:<4} {:<4} {:<4} {}", name, text(ime.commit),
                     text(ed.preedit), ed.caret_x(), ed.view());
    }
}

int main() {
    Ime ime;
    Editor ed;
    std::println("  키  확정 조합 자리 화면");
    type(ime, ed, "gksrmf.");
    std::println("키 7번에 실행 취소 단계는 {}", ed.undo_steps);
    std::println("받침이 다음 글자로 넘어간다");
    type(ime, ed, "rksk");
    std::println("조합 중에 지우면 낱자부터");
    type(ime, ed, "\b\b\b");

    for (bool finish : {false, true}) {      // 조합 중에 맨 앞을 눌렀다
        Ime i = ime;
        Editor e = ed;
        type(i, e, "dlq", false);
        std::print("{}: {}", finish ? "끝내고 옮김" : "캐럿만 옮김", e.view());
        if (finish) e.insert(i.finish());
        e.caret = 0;
        e.preedit = i.preedit();
        std::print(" -> {}", e.view());
        type(i, e, "l", false);
        std::println(" -> {}", e.view());
    }
}
```

```text
  키  확정 조합 자리 화면
  g   -    ㅎ   0    [ㅎ]
  k   -    하   0    [하]
  s   -    한   0    [한]
  r   한   ㄱ   2    한[ㄱ]
  m   -    그   2    한[그]
  f   -    글   2    한[글]
  .   글   -    5    한글.|
키 7번에 실행 취소 단계는 3
받침이 다음 글자로 넘어간다
  r   -    ㄱ   5    한글.[ㄱ]
  k   -    가   5    한글.[가]
  s   -    간   5    한글.[간]
  k   가   나   7    한글.가[나]
조합 중에 지우면 낱자부터
  BS  -    ㄴ   7    한글.가[ㄴ]
  BS  -    -    7    한글.가|
  BS  -    -    5    한글.|
캐럿만 옮김: 한글.[입] -> [입]한글. -> 이[비]한글.
끝내고 옮김: 한글.[입] -> |한글.입 -> [ㅣ]한글.입
```

- **조합 중인 글은 문서에 넣지 않았다**: `ㅎ`, `하`, `한`으로 바뀌는 동안 문서는 비어 있습니다. 편집기는 이 글을 캐럿 자리에 겹쳐 그릴 뿐입니다(화면의 `[ ]`, 실제로는 밑줄). 문서에 넣고 범위만 표시해 두는 구현도 있습니다(macOS의 marked text). 그때는 저장, 검색, 실행 취소에서 그 범위를 빼야 합니다
- **확정은 다음 키가 정한다**: `r`을 쳐야 `한`이 확정되고, 새 조합 `ㄱ`이 함께 옵니다. 확정된 글을 먼저 넣고 조합을 그 뒤에 둡니다. Wayland의 규칙도 이 순서를 정해 둡니다
- **조합은 되돌아가기도 한다**: `간`의 받침은 모음이 오자 다음 글자의 첫소리가 됩니다. 두벌식의 도깨비불 현상입니다. 화면에 보였던 `간`은 어디에도 확정되지 않았습니다. 조합 중인 글은 덧붙이는 것이 아니라 매번 통째로 바꿉니다
- **실행 취소는 확정 단위로**: 키는 일곱 번이지만 단계는 셋입니다. 조합 중의 변화는 실행 취소 스택에 넣지 않습니다. 이어 친 글자를 더 합치는 일은 [「명령 묶기와 합치기」](#/read/cpp-complex-app/cmd-merge)에서 다뤘습니다
- **키는 입력기가 먼저 본다**: 조합 중의 지우기 키는 입력기가 받아 낱자를 지웁니다. 조합이 없을 때에야 편집기로 와서 문서의 글자를 지웁니다. Windows에서는 입력기가 쓴 키가 `VK_PROCESSKEY`라는 값으로 바뀌어 옵니다. 이런 키를 단축키로 처리하면 안 됩니다
- **캐럿을 옮기기 전에 조합을 끝낸다**: 조합 중에 다른 곳을 눌렀을 때 캐럿만 옮기면, 조합 중인 글이 따라가서 엉뚱한 자리에 확정됩니다. 입력기에 조합을 끝내라고 알린 뒤에 옮깁니다. Windows에서는 `ImmNotifyIME`로 확정이나 취소를 시키고, macOS에서는 입력 컨텍스트에 `discardMarkedText`를 보냅니다
- **자리와 둘레의 글을 알려 준다**: `자리` 열은 편집기가 입력기에 알리는 캐럿의 자리입니다. 한자나 일본어, 중국어의 후보 창이 그 옆에 뜹니다. 스크롤이나 확대로 자리가 바뀌면 다시 알립니다. 입력기는 캐럿 둘레의 글과 선택 범위도 받아 갑니다

## 플랫폼마다의 이름

같은 일을 플랫폼마다 다르게 부릅니다. Qt는 이것들을 `QInputMethodEvent` 하나로 묶어 위젯에 줍니다.

| 일 | Windows(IMM) | macOS | Wayland |
|---|---|---|---|
| 조합 중인 글 | `WM_IME_COMPOSITION`의 `GCS_COMPSTR` | `setMarkedText` | `preedit_string` |
| 확정된 글 | 같은 메시지의 `GCS_RESULTSTR` | `insertText` | `commit_string` |
| 캐럿 자리 | `ImmSetCandidateWindow` | `firstRectForCharacterRange`에 답한다 | `set_cursor_rectangle` |

- **아무것도 안 해도 글자는 온다**: Windows에서 `WM_IME_COMPOSITION`을 `DefWindowProc`에 넘기면 입력기가 제 창에 조합을 그리고, 확정된 글은 `WM_CHAR`로 옵니다. 조합을 문서 안에 그리려면 메시지를 직접 처리합니다. macOS에서 글 뷰를 처음부터 만들 때는 `NSTextInputClient`를 구현하고, `keyDown`에서 받은 키를 입력 컨텍스트에 넘깁니다. 글이 되지 않는 키는 `doCommandBySelector`로 명령의 이름이 되어 돌아옵니다
- **글 입력을 켜고 끈다**: SDL은 `SDL_StartTextInput`을 부른 뒤에만 글 입력 이벤트를 줍니다. Wayland에서도 글을 받을 곳이 바뀔 때마다 `enable`을 보냅니다. 글을 치는 동안에만 켜 두면, 한글 상태에서도 그림 도구의 한 글자 단축키가 조합에 먹히지 않습니다
- **위치의 단위가 다르다**: Wayland는 글 안의 위치를 UTF-8 바이트로 셉니다. 편집기 안의 단위와 오갈 때마다 바꿔 줍니다

<Note>
입력기는 동아시아 글자만의 일이 아닙니다. macOS에서 Option-E 다음에 E를 눌러 é를 칠 때도 같은 길을 지납니다. 영문 자판으로만 시험하면 이 코드는 한 번도 실행되지 않습니다. 한글로는 받침이 넘어가는 조합과 조합 중의 클릭을, 일본어나 중국어 입력기로는 후보 창의 자리와 여러 글자에 걸친 조합을 시험합니다. 조합 중인 글도 셰이핑을 거쳐 줄 안에 놓이므로([「글꼴과 셰이핑」](#/read/cpp-complex-app/txt-shaping)), 조합이 길어지면 줄 나눔도 다시 합니다.
</Note>

## 정리

- 입력기는 조합 중인 글과 확정된 글을 따로 알린다. 조합 중인 글은 매번 통째로 바뀌고, 실행 취소에는 확정된 글만 들어간다
- 키는 입력기가 먼저 보고, 쓰지 않은 키만 편집기와 단축키로 온다. 캐럿이나 초점을 옮기기 전에는 조합을 끝내게 한다
- 편집기는 받기만 하지 않는다. 캐럿의 자리와 둘레의 글을 알려 줘야 후보 창이 제자리에 뜬다
