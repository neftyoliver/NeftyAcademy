큰 프로그램은 코드도 실행도 나눕니다. 컴파일하고 배포하는 단위가 **모듈**(라이브러리)이고, 실행하는 단위가 **프로세스**입니다. 무엇을 같은 프로세스에 두고 무엇을 떼어 내느냐가 안정성과 빠르기를 가릅니다. 아래 프로그램에는 망가진 파일을 만나면 죽는 가져오기 모듈이 있고, 그것을 두 가지 자리에 두어 봅니다.

```cpp
#include <chrono>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <string>
#include <sys/socket.h>
#include <sys/wait.h>
#include <unistd.h>

// 파일을 읽어 쪽 수를 돌려주는 가져오기 모듈. 망가진 파일을 만나면 죽는 벌레가 있다
int import_pages(const std::string& file) {
    if (file == "broken.doc") std::abort();
    return int(file.size());
}

// 1) 같은 프로세스: 앱이 모듈의 함수를 직접 부른다
void app_in_one_process() {
    int pages = 3;                                 // 편집 중이던 문서
    for (const char* file : {"report.doc", "broken.doc", "notes.doc"}) {
        pages += import_pages(file);
        std::printf("  %s 가져옴, 문서는 %d쪽\n", file, pages);
        std::fflush(stdout);
    }
}

// 2) 다른 프로세스: 모듈은 작업자 프로세스에서 돌고, 앱과는 메시지만 주고받는다
struct Worker { pid_t pid; int fd; };

Worker start_worker() {
    int pair[2];
    socketpair(AF_UNIX, SOCK_SEQPACKET, 0, pair);   // 메시지 단위로 오가는 한 쌍
    pid_t pid = fork();
    if (pid == 0) {                                 // 작업자: 요청을 받아 답한다
        close(pair[0]);
        char name[64];
        for (ssize_t n; (n = recv(pair[1], name, sizeof name, 0)) > 0;) {
            int pages = import_pages(std::string(name, size_t(n)));
            send(pair[1], &pages, sizeof pages, 0);
        }
        _exit(0);
    }
    close(pair[1]);
    return {pid, pair[0]};
}

// 요청 하나를 보내고 답을 기다린다. 작업자가 죽었으면 -1
int ask(Worker& w, const std::string& file) {
    send(w.fd, file.data(), file.size(), MSG_NOSIGNAL);
    int pages = 0;
    if (recv(w.fd, &pages, sizeof pages, 0) == sizeof pages) return pages;
    int status = 0;
    waitpid(w.pid, &status, 0);
    std::printf("  %s: 작업자가 죽음(신호 %d). 앱은 살아서 작업자를 다시 띄움\n",
                file.c_str(), WTERMSIG(status));
    close(w.fd);
    w = start_worker();
    return -1;
}

int main() {
    std::puts("[한 프로세스]");
    std::fflush(stdout);
    pid_t app = fork();                    // 앱이 죽는 것을 밖에서 지켜본다
    if (app == 0) {
        app_in_one_process();
        _exit(0);
    }
    int status = 0;
    waitpid(app, &status, 0);
    std::printf("  앱이 죽음(신호 %d). 편집 중이던 3쪽도 함께 사라짐\n",
                WTERMSIG(status));

    std::puts("[두 프로세스]");
    std::fflush(stdout);
    Worker worker = start_worker();
    int pages = 3;
    for (const char* file : {"report.doc", "broken.doc", "notes.doc"})
        if (int more = ask(worker, file); more >= 0) {
            pages += more;
            std::printf("  %s 가져옴, 문서는 %d쪽\n", file, pages);
        }

    // 값: 함수 호출 한 번과 프로세스 사이 왕복 한 번
    using clock = std::chrono::steady_clock;
    const int rounds = 2000;
    volatile int sink = 0;
    auto t0 = clock::now();
    for (int i = 0; i < rounds; ++i) sink = sink + import_pages("a.doc");
    auto t1 = clock::now();
    for (int i = 0; i < rounds; ++i) sink = sink + ask(worker, "a.doc");
    auto t2 = clock::now();
    auto ns = [&](auto d) {
        using std::chrono::nanoseconds;
        return std::chrono::duration_cast<nanoseconds>(d).count() / rounds;
    };
    std::fprintf(stderr, "한 번에: 함수 호출 %lld ns, 프로세스 왕복 %lld ns\n",
                 static_cast<long long>(ns(t1 - t0)),
                 static_cast<long long>(ns(t2 - t1)));
    std::printf("프로세스 사이 왕복이 함수 호출보다 %s\n",
                t2 - t1 > (t1 - t0) * 10 ? "열 배 넘게 느림" : "비슷함");

    close(worker.fd);                      // 연결을 닫으면 작업자가 스스로 끝난다
    waitpid(worker.pid, nullptr, 0);
}
```

```text
[한 프로세스]
  report.doc 가져옴, 문서는 13쪽
  앱이 죽음(신호 6). 편집 중이던 3쪽도 함께 사라짐
[두 프로세스]
  report.doc 가져옴, 문서는 13쪽
  broken.doc: 작업자가 죽음(신호 6). 앱은 살아서 작업자를 다시 띄움
  notes.doc 가져옴, 문서는 22쪽
프로세스 사이 왕복이 함수 호출보다 열 배 넘게 느림
```

- **같은 프로세스**: 앱이 모듈의 함수를 직접 부릅니다. 빠르고 데이터를 그대로 주고받지만, 모듈 하나가 죽으면 앱 전체가 죽습니다. 편집 중이던 문서 3쪽도 함께 사라졌습니다
- **작업자 프로세스**: 모듈은 따로 띄운 프로세스에서 돌고, 앱과는 메시지만 주고받습니다. 작업자가 죽으면 연결이 끊겨 `recv`가 답 대신 끝을 돌려주므로 앱이 알아채고, `waitpid`로 까닭을 읽은 뒤 새 작업자를 띄웁니다. 문서는 그대로이고 다음 파일도 가져왔습니다
- **왕복의 값**: 이 기계에서 한 번 돌려 보니 함수 호출은 수백 ns, 프로세스 사이 왕복은 수만 ns였습니다(시간은 표준 오류로 내보냈습니다). 그래서 글자 하나를 넣을 때마다 부르는 일은 떼어 낼 수 없고, 파일 하나나 그림 한 장처럼 덩어리가 큰 일을 떼어 냅니다. 큰 데이터는 메시지에 복사하지 않고 공유 메모리로 넘깁니다
- **돌봐야 할 것**: 작업자를 띄우고, 죽었는지와 멈췄는지를 알아채고, 다시 띄우고, 하던 요청을 다시 보낼지 정해야 합니다. 여기서는 망가진 파일을 포기했습니다. 같은 요청을 다시 보내면 새 작업자도 죽기 때문입니다
- **fork만 한 것**: 이 예제의 작업자는 같은 프로그램의 사본입니다. 실제 프로그램은 작업자를 다른 실행 파일로 띄우고, 권한을 낮춰 파일과 네트워크에 닿지 못하게도 합니다([「샌드박스」](#/read/cpp-complex-app/plg-sandbox))

## 실제 프로그램이 나눈 자리

- **Chromium**: 창과 UI를 맡고 다른 프로세스를 관리하는 브라우저 프로세스가 있고, 웹 내용은 렌더러 프로세스들이 맡습니다. 렌더러가 죽으면 그 탭에만 죽었다는 그림이 뜨고, 렌더러는 샌드박스 안에서 닿을 수 있는 것이 제한됩니다
- **VS Code**: 확장은 확장 호스트에서 돕니다. 확장이 시작을 늦추거나 UI 동작을 느리게 하거나 UI를 고치지 못하게 하려는 것이고, 확장은 필요해질 때에야 불러옵니다
- **GIMP**: 플러그인은 GIMP 본체가 띄우는 독립된 프로세스이고, 본체와는 프로세스 사이 통신으로 정보를 주고받습니다. 다만 본체와 같은 권한으로 돌므로, 충돌은 막아도 나쁜 플러그인이 하는 일까지 막지는 못합니다

## 한 프로세스 안의 모듈

한 프로세스 안에서도 코드는 라이브러리로 나눕니다. LibreOffice의 소스는 150개쯤 되는 모듈로 이루어져 있어서, 시스템의 차이를 감추는 `sal`, 그래픽 툴킷 `vcl`, 글 편집 엔진 `editeng`, 워드 프로세서 `sw`, 스프레드시트 `sc`, 가져오기와 내보내기 필터 `filter`가 따로 있습니다.

- **의존 방향을 빌드로 지킨다**: 문서와 명령의 모듈이 UI 툴킷을 링크하지 않으면, 그 안에서 창을 부르는 코드는 컴파일되지 않습니다([「문서, 명령, 뷰의 층」](#/read/cpp-complex-app/ana-layers)). 규칙을 글로 적는 것보다 확실합니다
- **늦게 불러온다**: 모듈을 시작할 때 모두 올리면 창이 늦게 뜹니다. 드물게 쓰는 기능은 처음 쓸 때 불러옵니다
- **경계에서는 타입을 조심한다**: 따로 빌드한 모듈끼리는 컴파일러와 표준 라이브러리의 판이 다를 수 있어서, C++ 타입을 그대로 넘기지 못합니다([「ABI가 안정된 인터페이스」](#/read/cpp-complex-app/plg-abi))

| 둘 곳 | 부르는 값 | 죽으면 | 맞는 일 |
| --- | --- | --- | --- |
| 같은 프로세스의 모듈 | 함수 호출 | 앱 전체가 죽음 | 문서, 명령, 그리기처럼 자주 부르는 코드 |
| 작업자 프로세스 | 메시지 왕복 | 그 프로세스만 죽음 | 파일 가져오기, 플러그인, 미리보기처럼 남의 데이터나 코드를 다루는 일 |

<Note>
시간은 ASan을 켠 채 한 번 잰 값이라 크기의 차례만 보면 됩니다. 프로세스를 나눠서 얻는 것은 충돌 격리만이 아닙니다. 작업자가 끝나면 그 메모리가 통째로 돌아오고, 작업자가 오래 걸려도 창은 멈추지 않으며, 권한을 낮춘 작업자는 뚫려도 할 수 있는 일이 적습니다.
</Note>

## 정리

- 같은 프로세스의 모듈은 함수 호출로 빠르게 부르지만 하나가 죽으면 앱 전체가 죽고, 작업자 프로세스는 죽어도 앱과 문서가 남는다
- 프로세스 사이 왕복은 함수 호출보다 훨씬 비싸므로, 덩어리가 크고 믿기 어려운 일(파일 가져오기, 플러그인)을 떼어 낸다
- 한 프로세스 안에서도 모듈로 나눠 의존 방향을 빌드로 지키고, 드물게 쓰는 모듈은 늦게 불러온다
