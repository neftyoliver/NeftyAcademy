플러그인은 본체와 따로 빌드됩니다. 컴파일러도, 표준 라이브러리의 판도, 빌드한 해도 다를 수 있습니다. C++ 타입이 그 경계를 왜 넘지 못하는지와 C 함수로 내놓는 법은 C++ 고급 책의 [「C ABI로 내놓기」](#/read/cpp-advanced/iop-c-abi)와 [「ABI 안정성」](#/read/cpp-advanced/iop-abi-stable)에서 다뤘습니다. 플러그인에서는 두 가지가 더 붙습니다. 본체가 플러그인을 부르고 플러그인도 본체를 부르며, 한쪽만 새로 빌드된 채로 여러 해를 함께 돌아야 합니다.

```cpp
// 플러그인 경계: 크기를 적은 함수 표와 이름으로 묻는 확장. 판이 달라도 함께 돈다
#include <algorithm>
#include <cstring>
#include <print>
#include <stdint.h>
#include <string>

extern "C" {
struct HostV1 {                      // 본체가 플러그인에게 주는 표, 첫 판
    uint32_t size;                   // 이 표의 바이트 수. 표는 뒤로만 자란다
    void* data;                      // 본체의 것. 플러그인은 열어 보지 않는다
    void (*log)(const HostV1*, const char* text);
};
struct HostV2 {                      // 둘째 판: 첫 판은 그대로 두고 뒤에 더했다
    HostV1 v1;
    void (*request_redraw)(const HostV1*);
};
struct Preview {                     // 이름으로 묻는 확장 "app.preview/1"
    int (*render)(void* self, char* out, uint32_t cap);
};
struct Plugin {                      // 플러그인이 본체에게 주는 표
    uint32_t abi;                    // 큰 판. 다르면 id 말고는 읽지 않는다
    const char* id;
    void* (*create)(const HostV1*);
    void (*destroy)(void* self);     // 만든 쪽이 없앤다
    int (*apply)(void* self, const char* in, char* out, uint32_t cap);
    const void* (*extension)(void* self, const char* id);    // 모르면 null
};
}
constexpr uint32_t kAbi = 1;

int put(const std::string& s, char* out, uint32_t cap) {    // 필요한 길이를 돌려줌
    if (cap) out[s.copy(out, cap - 1)] = '\0';
    return static_cast<int>(s.size());
}

namespace upper {                    // 첫 판의 헤더로 빌드한 플러그인
void* create(const HostV1* h) { return h->log(h, "upper: 준비됨"), new int{}; }
void destroy(void* self) { delete static_cast<int*>(self); }
int apply(void*, const char* in, char* out, uint32_t cap) {
    std::string s = in;
    for (char& c : s)
        if (c >= 'a' && c <= 'z') c -= 'a' - 'A';
    return put(s, out, cap);
}
const void* extension(void*, const char*) { return nullptr; }
constexpr Plugin table{kAbi, "upper", create, destroy, apply, extension};
}

namespace mirror {                   // 둘째 판의 헤더로 빌드한 플러그인
struct Self {
    const HostV1* host;
    std::string last;
};
void* create(const HostV1* h) { return new Self{h, ""}; }
void destroy(void* self) { delete static_cast<Self*>(self); }
int apply(void* self, const char* in, char* out, uint32_t cap) {
    auto* me = static_cast<Self*>(self);
    me->last = in;                   // 받은 글은 호출 동안만 유효하다. 베껴 둔다
    std::ranges::reverse(me->last);
    const HostV1* h = me->host;
    if (h->size >= sizeof(HostV2))   // 크기를 보고 나서야 뒤쪽을 읽는다
        reinterpret_cast<const HostV2*>(h)->request_redraw(h);
    else
        h->log(h, "mirror: 본체가 옛 판이라 다시 그리기를 청하지 못함");
    return put(me->last, out, cap);
}
int render(void* self, char* out, uint32_t cap) {
    return put("[" + static_cast<Self*>(self)->last + "]", out, cap);
}
constexpr Preview preview{render};
const void* extension(void*, const char* id) {
    return std::strcmp(id, "app.preview/1") == 0 ? &preview : nullptr;
}
constexpr Plugin table{kAbi, "mirror", create, destroy, apply, extension};
}

constexpr Plugin alien{2, "alien", {}, {}, {}, {}};     // 다음 큰 판의 플러그인

void host_log(const HostV1*, const char* s) { std::println("    [log] {}", s); }
void host_redraw(const HostV1* h) { ++*static_cast<int*>(h->data); }

void run(const char* name, const HostV1* host) {
    std::println("{}(표 {}바이트)", name, host->size);
    for (const Plugin* p : {&upper::table, &mirror::table, &alien}) {
        if (p->abi != kAbi) {
            std::println("  {}: ABI {} -> 본체는 {}이라 쓰지 않음", p->id, p->abi,
                         kAbi);
            continue;
        }
        char out[64];
        void* self = p->create(host);
        p->apply(self, "Hello", out, sizeof out);
        std::print("  {}: Hello -> {}", p->id, out);
        auto* pv = static_cast<const Preview*>(p->extension(self, "app.preview/1"));
        if (pv) {
            pv->render(self, out, sizeof out);
            std::println(", 미리 보기 {}", out);
        } else
            std::println(", 미리 보기 없음(단추를 감춘다)");
        p->destroy(self);
    }
    std::println("  다시 그리기 요청 {}번", *static_cast<int*>(host->data));
}

int main() {
    int redraws_old = 0, redraws_new = 0;
    const HostV1 old_host{sizeof(HostV1), &redraws_old, host_log};
    const HostV2 new_host{{sizeof(HostV2), &redraws_new, host_log}, host_redraw};
    run("옛 본체", &old_host);
    run("새 본체", &new_host.v1);
}
```

```text
옛 본체(표 24바이트)
    [log] upper: 준비됨
  upper: Hello -> HELLO, 미리 보기 없음(단추를 감춘다)
    [log] mirror: 본체가 옛 판이라 다시 그리기를 청하지 못함
  mirror: Hello -> olleH, 미리 보기 [olleH]
  alien: ABI 2 -> 본체는 1이라 쓰지 않음
  다시 그리기 요청 0번
새 본체(표 32바이트)
    [log] upper: 준비됨
  upper: Hello -> HELLO, 미리 보기 없음(단추를 감춘다)
  mirror: Hello -> olleH, 미리 보기 [olleH]
  alien: ABI 2 -> 본체는 1이라 쓰지 않음
  다시 그리기 요청 1번
```

- **표가 둘이다**: 본체는 `Plugin` 표로 플러그인을 부르고, 플러그인은 `create`에서 받은 표로 본체를 부릅니다. 본체가 이름으로 찾는 기호는 플러그인의 표 하나뿐이고 나머지는 모두 함수 포인터입니다. 플러그인이 본체의 함수를 링크하지 않으므로 본체의 실행 파일이 달라져도 다시 빌드하지 않습니다
- **표는 뒤로만 자란다**: 한 파일에 옛 헤더와 새 헤더를 함께 보이려고 `HostV1`과 `HostV2`로 나눴습니다. 실제로는 구조체 하나의 뒤에 필드를 더하고, 옛 플러그인은 옛 헤더로 빌드된 채 남습니다. `upper`는 새 본체에서도 그대로 돌았습니다. 앞쪽 24바이트의 뜻이 그대로이기 때문입니다
- **크기를 보고 읽는다**: 새 헤더로 빌드한 `mirror`는 `size`가 32 이상일 때에만 `request_redraw`를 읽습니다. 옛 본체에서는 그 기능 없이 돌았습니다. 확인하지 않고 읽으면 표 바깥의 메모리를 함수 포인터로 부르게 됩니다
- **확장은 이름으로 묻는다**: 본체는 `"app.preview/1"`을 물어 표를 받거나 null을 받고, null이면 그 기능의 단추를 감춥니다. 판 번호로 기능을 짐작하지 않습니다. 플러그인마다 하는 일이 다르기 때문입니다. 이름에 판이 들어 있으니 확장을 깨서 고칠 때에는 새 이름을 냅니다
- **큰 판은 마지막 수단이다**: `alien`은 큰 판이 달라서 본체가 `abi`와 `id`만 읽었습니다. 그 둘의 자리는 영원히 바꾸지 않습니다. 큰 판을 올리면 그때까지의 플러그인을 모두 잃으므로, 앞의 세 가지로 안 될 때에만 올립니다
- **수명을 적는다**: 객체는 만든 쪽의 `destroy`로 없앱니다. 양쪽의 할당자가 다를 수 있기 때문입니다([「DLL과 공유 라이브러리의 경계」](#/read/cpp-advanced/iop-shared)). 인자로 받은 글은 호출 동안만 유효해서 `mirror`는 베껴 두었습니다. 확장의 표는 준 쪽의 것이고, 플러그인 객체가 없어질 때까지 유효합니다

## 실제 인터페이스

- **CLAP**: 오디오 플러그인 규격입니다. 라이브러리가 내보이는 것은 `clap_plugin_entry` 하나이고, 그 안에 판, `init`, `deinit`, 이름으로 공장을 얻는 `get_factory`가 있습니다. 플러그인의 표와 본체의 표에 저마다 `get_extension`이 있어 서로에게 확장을 이름으로 묻고, 없으면 null을 받습니다. 1.x로 빌드한 플러그인은 어느 1.y 본체에서든 올라갑니다
- **COM**: 인터페이스는 IID를 받아 공개된 뒤에는 어느 것도 바꿀 수 없습니다. 다른 인터페이스는 `QueryInterface`로 묻고, 한 번 된다고 답한 IID는 그 객체에서 계속 되어야 합니다. 받은 쪽이 풀 메모리는 COM이 정한 할당자로 잡습니다
- **Node-API**: Node.js의 네이티브 애드온용 인터페이스입니다. `node_api.h`와 `node_api_types.h`에 선언된 것만 쓰면 Node.js의 큰 판이 바뀌어도 다시 빌드하지 않습니다. 판은 쌓이기만 해서, n판은 그 뒤의 모든 Node.js에 들어 있습니다
- **벌칸**: 구조체 앞의 `sType`과 `pNext`, 이름으로 켜는 확장이 같은 생각입니다([「벌칸의 설계」](#/read/cpp-vulkan/vk-design))
- **C++ 인터페이스를 그대로**: Qt 플러그인은 가상 함수 인터페이스를 넘깁니다. 그 대신 본체와 같은 큰 판의 Qt와 맞는 컴파일러로 빌드해야 합니다([「플러그인」](#/read/cpp-qt/qarch-plugins)). 플러그인을 본체와 함께 빌드해 내보낸다면 이것으로 충분합니다

## 표에 함께 적을 것

- **스레드**: 함수마다 어느 스레드에서 불러도 되는지 적습니다. CLAP은 `[main-thread]`, `[audio-thread]`, `[thread-safe]` 같은 표시를 함수마다 달았습니다. 플러그인이 다른 스레드에서 본체의 `request_callback`을 부르면, 본체가 나중에 메인 스레드에서 플러그인의 `on_main_thread`를 불러 줍니다([「UI 스레드의 규칙」](#/read/cpp-complex-app/rsp-ui-thread))
- **실패**: 표의 함수는 예외를 내보내지 않습니다. 예제에서는 줄였지만, 플러그인은 함수마다 예외를 잡아 반환값으로 바꿉니다
- **내리지 않는다**: 플러그인이 만든 객체, 등록한 콜백, 실행 취소 스택의 명령이 그 코드를 가리킵니다. 하나라도 남은 채 라이브러리를 내리면 다음 호출에서 죽습니다. 끈 플러그인은 기여만 빼고([「확장 지점」](#/read/cpp-complex-app/plg-points)) 코드는 프로세스가 끝날 때까지 둡니다
- **C++ 껍데기**: 표를 감싸는 C++ 클래스를 헤더로만 줍니다. 그 코드는 플러그인 안에서 컴파일되므로 경계를 넘는 것은 여전히 C의 표뿐입니다

<Note>
다시 빌드해서 통과하는 시험은 API가 그대로라는 것만 알려 줍니다. ABI는 다시 빌드하지 않은 바이너리로만 확인됩니다. 옛 헤더로 빌드한 플러그인을 저장해 두고, 본체를 고칠 때마다 올려서 돌려 봅니다. 구조체의 크기와 필드의 자리를 `static_assert`로 적어 두면 실수로 가운데에 필드를 끼울 때 컴파일이 멈춥니다.
</Note>

## 정리

- 플러그인 경계는 C의 함수 표 둘이다. 표는 뒤로만 자라고, 뒤쪽 필드는 크기를 확인하고 읽는다
- 기능이 있는지는 판 번호가 아니라 이름으로 묻는다. 큰 판은 모든 플러그인을 버릴 때에만 올린다
- 누가 없애는지, 언제까지 유효한지, 어느 스레드에서 부르는지를 함수마다 적고, 올린 코드는 내리지 않는다
